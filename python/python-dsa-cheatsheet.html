11. Trees & Recursion

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val, self.left, self.right = val, left, right

# DFS recursive (inorder)
def inorder(n, out):
    if not n: return
    inorder(n.left, out); out.append(n.val); inorder(n.right, out)

# DFS iterative (inorder)
def inorder_iter(root):
    out, stk, cur = [], [], root
    while cur or stk:
        while cur:
            stk.append(cur); cur = cur.left
        cur = stk.pop(); out.append(cur.val); cur = cur.right
    return out

# BFS / level-order
from collections import deque
def bfs(root):
    if not root: return []
    q, out = deque([root]), []
    while q:
        level = []
        for _ in range(len(q)):
            n = q.popleft(); level.append(n.val)
            if n.left:  q.append(n.left)
            if n.right: q.append(n.right)
        out.append(level)
    return out

# Lowest common ancestor (binary tree)
def lca(root, p, q):
    if not root or root is p or root is q: return root
    L = lca(root.left, p, q); R = lca(root.right, p, q)
    return root if L and R else L or R

12. Graphs

# Adjacency list
from collections import defaultdict
g = defaultdict(list)
for u, v, w in edges:
    g[u].append((v, w)); g[v].append((u, w))   # undirected

# BFS shortest path (unweighted)
def bfs(src, n):
    dist = [-1]*n; dist[src] = 0
    q = deque([src])
    while q:
        u = q.popleft()
        for v, _ in g[u]:
            if dist[v] == -1:
                dist[v] = dist[u] + 1
                q.append(v)
    return dist

# DFS iterative (avoid recursion limit)
def dfs(src):
    seen = {src}; stk = [src]
    while stk:
        u = stk.pop()
        for v, _ in g[u]:
            if v not in seen:
                seen.add(v); stk.append(v)

# Dijkstra (non-negative weights)
import heapq
def dijkstra(src, n):
    INF = float('inf')
    dist = [INF]*n; dist[src] = 0
    h = [(0, src)]
    while h:
        d, u = heapq.heappop(h)
        if d > dist[u]: continue
        for v, w in g[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd; heapq.heappush(h, (nd, v))
    return dist

# Topological sort (Kahn's)
def topo(n):
    indeg = [0]*n
    for u in range(n):
        for v, _ in g[u]: indeg[v] += 1
    q = deque(i for i in range(n) if indeg[i] == 0)
    order = []
    while q:
        u = q.popleft(); order.append(u)
        for v, _ in g[u]:
            indeg[v] -= 1
            if indeg[v] == 0: q.append(v)
    return order if len(order) == n else []   # [] if cycle

# Union-Find (DSU) with path compression + union-by-rank
class DSU:
    def __init__(self, n):
        self.p = list(range(n)); self.r = [0]*n
    def find(self, x):
        while self.p[x] != x:
            self.p[x] = self.p[self.p[x]]; x = self.p[x]
        return x
    def union(self, a, b):
        a, b = self.find(a), self.find(b)
        if a == b: return False
        if self.r[a] < self.r[b]: a, b = b, a
        self.p[b] = a
        if self.r[a] == self.r[b]: self.r[a] += 1
        return True

13. Dynamic Programming Templates

Top-down memoization (cleanest)

from functools import cache

class Solution:
    def rob(self, nums):
        @cache
        def dp(i):
            if i >= len(nums): return 0
            return max(dp(i+1), nums[i] + dp(i+2))
        return dp(0)
⚠ @cache on instance methods leaks! Use a free def dp(...) inside the method, or functools.cache after wrapping self as needed.

Bottom-up 1-D & 2-D

# LIS O(n log n)
import bisect
def lis(a):
    tails = []
    for x in a:
        i = bisect.bisect_left(tails, x)
        if i == len(tails): tails.append(x)
        else: tails[i] = x
    return len(tails)

# 0/1 knapsack
def knap(W, w, v):
    dp = [0]*(W+1)
    for i in range(len(w)):
        for c in range(W, w[i]-1, -1):     # reverse!
            dp[c] = max(dp[c], dp[c-w[i]] + v[i])
    return dp[W]

# Edit distance
def edit(a, b):
    m, n = len(a), len(b)
    dp = [[0]*(n+1) for _ in range(m+1)]
    for i in range(m+1): dp[i][0] = i
    for j in range(n+1): dp[0][j] = j
    for i in range(1, m+1):
        for j in range(1, n+1):
            if a[i-1] == b[j-1]:
                dp[i][j] = dp[i-1][j-1]
            else:
                dp[i][j] = 1 + min(dp[i-1][j], dp[i][j-1], dp[i-1][j-1])
    return dp[m][n]

Sliding window — variable size (longest subarray with sum ≤ K)

def longest(nums, K):
    l = s = best = 0
    for r, x in enumerate(nums):
        s += x
        while s > K:
            s -= nums[l]; l += 1
        best = max(best, r - l + 1)
    return best

16. Trie / Segment Tree / BIT

Trie

class Trie:
    def __init__(self): self.root = {}
    def insert(self, w):
        n = self.root
        for c in w: n = n.setdefault(c, {})
        n['$'] = True
    def search(self, w):
        n = self.root
        for c in w:
            if c not in n: return False
            n = n[c]
        return '$' in n

Fenwick tree (BIT) — point update + prefix sum

class BIT:
    def __init__(self, n): self.n=n; self.t=[0]*(n+1)
    def update(self, i, v):
        i += 1
        while i <= self.n: self.t[i] += v; i += i & -i
    def query(self, i):
        i += 1; s = 0
        while i > 0: s += self.t[i]; i -= i & -i
        return s

17. Intervals & Sweepline

# Merge overlapping
def merge(iv):
    iv.sort()
    out = [iv[0]]
    for s, e in iv[1:]:
        if s <= out[-1][1]: out[-1][1] = max(out[-1][1], e)
        else: out.append([s, e])
    return out

# Meeting rooms II — sweepline / heap
def rooms(iv):
    iv.sort()
    h = []
    for s, e in iv:
        if h and h[0] <= s: heapq.heappop(h)
        heapq.heappush(h, e)
    return len(h)

# Sieve of Eratosthenes
def sieve(n):
    p = [True]*(n+1); p[:2] = [False]*2
    for i in range(2, int(n**.5)+1):
        if p[i]:
            for j in range(i*i, n+1, i): p[j] = False
    return [i for i, ok in enumerate(p) if ok]

20. Top 20 Python Gotchas in Interviews

  1. Mutable default arg — def f(a=[]) shares state across calls.
  2. [[0]*n]*m creates m references to one row. Use list comp.
  3. Integer division floors toward −∞: -7 // 2 == -4 (not −3).
  4. Modulo sign follows divisor: (-7) % 3 == 2.
  5. String concat in loop = O(n²). Use "".join(parts).
  6. list.pop(0) = O(n). Use deque.popleft().
  7. x in list = O(n). Convert to set for membership.
  8. Recursion limit 1000 — bump it; iteratively for n > 10⁴.
  9. copy() is shallow. Nested? Use copy.deepcopy or list comprehension.
  10. Sets / dicts of mutable — list/dict not hashable; use tuple/frozenset.
  11. Float equality — never ==; use math.isclose.
  12. 0.1 + 0.2 ≠ 0.3 — convert to int (mult 10ⁿ) or use Decimal.
  13. range(n) doesn't materialize — fine; but list(range(n)) allocates.
  14. Variable capture in lambdas inside loops: fns = [lambda: i for i in range(3)] — all return 2. Default-arg trick: lambda i=i: i.
  15. and/or short-circuit and return values, not bools: 0 or 'x' == 'x'.
  16. is vs == — only use is for None, True, False.
  17. Iterating & mutating a list/dict throws / skips. Copy: for k in list(d): ....
  18. Generator exhausted once — re-create or convert with list().
  19. sort() returns None; sorted() returns list.
  20. True == 1 & False == 0; True + True == 2. Sometimes useful, often a bug.

    else: lo = mid + 1
