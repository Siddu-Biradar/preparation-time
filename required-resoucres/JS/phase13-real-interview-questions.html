function allSettled(promises) {
  // Handle empty array
  if (!promises.length) return Promise.resolve([]);
  
  return new Promise((resolve) => {
    const results = new Array(promises.length);
    let completed = 0;
    
    promises.forEach((promise, index) => {
      // Wrap non-promise values
      Promise.resolve(promise)
        .then(
          (value) => {
            results[index] = { status: "fulfilled", value };
          },
          (reason) => {
            results[index] = { status: "rejected", reason };
          }
        )
        .finally(() => {
          completed++;
          if (completed === promises.length) {
            resolve(results);
          }
        });
    });
  });
}

// 1. Empty array → resolve with []
// 2. Non-promise values (wrapped with Promise.resolve)
// 3. Maintaining order (results[index], not push)
// 4. Never rejects (always resolves with array)
// 5. Each result has status + value/reason

Google Hard Implement a debounce with leading & trailing options

Round: Onsite Coding | What they look for: Closure mastery, timer management, options handling

function debounce(fn, delay, options = {}) {
  const { leading = false, trailing = true } = options;
  let timerId = null;
  let lastArgs = null;
  let lastThis = null;
  let lastCallTime = 0;
  
  function invoke() {
    const args = lastArgs;
    const thisArg = lastThis;
    lastArgs = lastThis = null;
    fn.apply(thisArg, args);
  }
  
  function startTimer(pendingFunc, wait) {
    timerId = setTimeout(() => {
      timerId = null;
      pendingFunc();
    }, wait);
  }
  
  function debounced(...args) {
    const now = Date.now();
    const isInvoking = shouldInvoke(now);
    
    lastArgs = args;
    lastThis = this;
    lastCallTime = now;
    
    if (isInvoking) {
      if (timerId === null) {
        // Leading edge — first call
        if (leading) {
          invoke();
        }
        startTimer(() => {
          if (trailing && lastArgs) invoke();
        }, delay);
        return;
      }
    }
    
    // Reset timer on each call
    if (timerId !== null) {
      clearTimeout(timerId);
    }
    startTimer(() => {
      if (trailing && lastArgs) invoke();
    }, delay);
  }
  
  function shouldInvoke(time) {
    const timeSinceLastCall = time - lastCallTime;
    return timerId === null;
  }
  
  // Cancel method
  debounced.cancel = function() {
    if (timerId !== null) clearTimeout(timerId);
    timerId = null;
    lastArgs = lastThis = null;
    lastCallTime = 0;
  };
  
  // Flush method (invoke immediately)
  debounced.flush = function() {
    if (timerId !== null) {
      clearTimeout(timerId);
      timerId = null;
      if (lastArgs) invoke();
    }
  };
  
  return debounced;
}

// Usage:

// First keystroke → immediate call (leading)
// Subsequent keystrokes → wait 300ms after last (trailing)

// Recursive approach (clean but has call stack limit)
function flattenRecursive(arr, depth = Infinity) {
  if (depth < 1) return arr.slice();
  
  return arr.reduce((result, item) => {
    if (Array.isArray(item)) {
      result.push(...flattenRecursive(item, depth - 1));
    } else {
      result.push(item);
    }
    return result;
  }, []);
}

// Iterative approach (no stack overflow — Google prefers this!)
function flattenIterative(arr, depth = Infinity) {
  const result = [];
  const stack = arr.map(item => [item, depth]); // [value, remainingDepth]
  
  while (stack.length > 0) {
    const [item, d] = stack.pop();
    
    if (Array.isArray(item) && d > 0) {
      // Push children in reverse order (so first child is processed first)
      for (let i = item.length - 1; i >= 0; i--) {
        stack.push([item[i], d - 1]);
      }
    } else {
      result.push(item);
    }
  }
  
  return result;
}

// Generator approach (memory efficient for huge arrays)
function* flattenGen(arr, depth = Infinity) {
  for (const item of arr) {
    if (Array.isArray(item) && depth > 0) {
      yield* flattenGen(item, depth - 1);
    } else {
      yield item;
    }
  }
}

Google Expert Implement event emitter with once, off, and wildcard support

class EventEmitter {
  #events = new Map();
  
  on(event, listener) {
    if (!this.#events.has(event)) {
      this.#events.set(event, []);
    }
    this.#events.get(event).push({ fn: listener, once: false });
    return this; // Chainable
  }
  
  once(event, listener) {
    if (!this.#events.has(event)) {
      this.#events.set(event, []);
    }
    this.#events.get(event).push({ fn: listener, once: true });
    return this;
  }
  
  off(event, listener) {
    if (!listener) {
      this.#events.delete(event); // Remove all listeners for event
      return this;
    }
    
    const listeners = this.#events.get(event);
    if (listeners) {
      const index = listeners.findIndex(l => l.fn === listener);
      if (index !== -1) listeners.splice(index, 1);
      if (listeners.length === 0) this.#events.delete(event);
    }
    return this;
  }
  
  emit(event, ...args) {
    const listeners = this.#events.get(event);
    if (!listeners) return false;
    
    // Copy array to avoid issues if listener modifies the list
    const toCall = [...listeners];
    
    // Remove once listeners
    this.#events.set(event, listeners.filter(l => !l.once));
    
    toCall.forEach(({ fn }) => {
      try {
        fn.apply(this, args);
      } catch (error) {
        console.error(`Error in listener for "${event}":`, error);
      }
    });
    
    // Also trigger wildcard listeners
    const wildcardListeners = this.#events.get("*");
    if (wildcardListeners && event !== "*") {
      wildcardListeners.forEach(({ fn }) => fn.call(this, event, ...args));
    }
    
    return true;
  }
  
  listenerCount(event) {
    return this.#events.get(event)?.length || 0;
  }
  
  removeAllListeners() {
    this.#events.clear();
    return this;
  }
}

function throttle(fn, interval) {
  let lastRunTime = 0;
  let timerId = null;
  let lastArgs = null;
  let lastThis = null;
  
  return function(...args) {
    const now = Date.now();
    const remaining = interval - (now - lastRunTime);
    
    lastArgs = args;
    lastThis = this;
    
    if (remaining <= 0) {
      // Enough time has passed — invoke immediately
      if (timerId) {
        clearTimeout(timerId);
        timerId = null;
      }
      lastRunTime = now;
      fn.apply(lastThis, lastArgs);
      lastArgs = lastThis = null;
    } else if (!timerId) {
      // Schedule trailing call
      timerId = setTimeout(() => {
        lastRunTime = Date.now();
        timerId = null;
        fn.apply(lastThis, lastArgs);
        lastArgs = lastThis = null;
      }, remaining);
    }
  };
}

// Basic curry (fixed arity):
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    }
    return function(...moreArgs) {
      return curried.apply(this, [...args, ...moreArgs]);
    };
  };
}

// sum(1)(2)(3)() === 6   ← calling with no args triggers evaluation
function infiniteCurry(fn) {
  return function curried(...args) {
    if (args.length === 0) {
      return fn.call(this);
    }
    return function(...moreArgs) {
      if (moreArgs.length === 0) {
        return fn.apply(this, args);
      }
      return curried.apply(this, [...args, ...moreArgs]);
    };
  };
}

function sum(...initial) {

    return inner;
  }
  
  inner.valueOf = () => total;
  inner[Symbol.toPrimitive] = () => total;
  inner.toString = () => String(total);
  
  return inner;
}

Meta Expert Implement a function to deep compare two objects

Round: Onsite | Asked frequently

function deepEqual(a, b) {
  // Same reference or both primitive and equal
  if (a === b) return true;
  
  // Handle null/undefined
  if (a == null || b == null) return false;
  
  // Handle NaN
  if (Number.isNaN(a) && Number.isNaN(b)) return true;
  
  // Different types
  if (typeof a !== typeof b) return false;
  
  // Not objects → primitive inequality
  if (typeof a !== "object") return false;
  
  // Handle Date
  if (a instanceof Date && b instanceof Date) {
    return a.getTime() === b.getTime();
  }
  
  // Handle RegExp
  if (a instanceof RegExp && b instanceof RegExp) {
    return a.source === b.source && a.flags === b.flags;
  }
  
  // Handle Array
  if (Array.isArray(a) !== Array.isArray(b)) return false;
  
  // Handle Map
  if (a instanceof Map && b instanceof Map) {
    if (a.size !== b.size) return false;
    for (const [key, val] of a) {
      if (!b.has(key) || !deepEqual(val, b.get(key))) return false;
    }
    return true;
  }
  
  // Handle Set
  if (a instanceof Set && b instanceof Set) {
    if (a.size !== b.size) return false;
    for (const val of a) {
      if (!b.has(val)) return false;
    }
    return true;
  }
  
  // Handle plain objects
  const keysA = Object.keys(a);
  const keysB = Object.keys(b);
  
  if (keysA.length !== keysB.length) return false;
  
  return keysA.every(key => 
    Object.prototype.hasOwnProperty.call(b, key) && deepEqual(a[key], b[key])
  );
}

// Tests:

console.log(deepEqual([1, [2, 3]], [1, [2, 3]]));                       // true
console.log(deepEqual(NaN, NaN));                                        // true
console.log(deepEqual(new Date("2024-01-01"), new Date("2024-01-01"))); // true
console.log(deepEqual({ a: 1 }, { a: 1, b: undefined }));              // false

function retry(fn, options = {}) {
  const {
    maxRetries = 3,
    initialDelay = 1000,
    maxDelay = 30000,
    backoffFactor = 2,
    retryOn = () => true, // Function to decide if error is retryable
    onRetry = () => {},   // Callback on each retry
  } = options;
  
  return async function(...args) {
    let lastError;
    
    for (let attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        return await fn.apply(this, args);
      } catch (error) {
        lastError = error;
        
        if (attempt === maxRetries || !retryOn(error)) {
          throw error;
        }
        
        const delay = Math.min(
          initialDelay * Math.pow(backoffFactor, attempt),
          maxDelay
        );
        
        // Add jitter to prevent thundering herd
        const jitter = delay * (0.5 + Math.random() * 0.5);
        
        onRetry({ error, attempt: attempt + 1, delay: jitter });
        
        await new Promise(resolve => setTimeout(resolve, jitter));
      }
    }
    
    throw lastError;
  };
}

// Usage:

  maxRetries: 3,
  initialDelay: 1000,
  retryOn: (err) => err?.status >= 500, // Only retry server errors
  onRetry: ({ attempt, delay }) => {
    console.log(`Retry #${attempt} in ${delay}ms`);
  }
});

class AsyncQueue {
  #queue = [];
  #running = 0;
  #concurrency;
  
  constructor(concurrency = 3) {
    this.#concurrency = concurrency;
  }
  
  async add(asyncFn) {
    return new Promise((resolve, reject) => {
      this.#queue.push({ fn: asyncFn, resolve, reject });
      this.#run();
    });
  }
  
  async #run() {
    while (this.#running < this.#concurrency && this.#queue.length > 0) {
      const { fn, resolve, reject } = this.#queue.shift();
      this.#running++;
      
      try {
        const result = await fn();
        resolve(result);
      } catch (error) {
        reject(error);
      } finally {
        this.#running--;
        this.#run(); // Process next in queue
      }
    }
  }
  
  get pending() { return this.#queue.length; }
  get active() { return this.#running; }
}

function memoize(fn, options = {}) {
  const { maxSize = 100, ttl = Infinity, keyFn = JSON.stringify } = options;
  const cache = new Map();
  
  function get(key) {
    const entry = cache.get(key);
    if (!entry) return undefined;
    
    // Check TTL
    if (Date.now() - entry.timestamp > ttl) {
      cache.delete(key);
      return undefined;
    }
    
    // Move to end (LRU: most recently used goes to end)
    cache.delete(key);
    cache.set(key, entry);
    
    return entry;
  }
  
  function memoized(...args) {
    const key = keyFn(args);
    const cached = get(key);
    
    if (cached) return cached.value;
    
    const result = fn.apply(this, args);
    
    // Evict oldest if at capacity
    if (cache.size >= maxSize) {
      const oldestKey = cache.keys().next().value;
      cache.delete(oldestKey);
    }
    
    cache.set(key, { value: result, timestamp: Date.now() });
    return result;
  }
  
  memoized.cache = cache;
  memoized.clear = () => cache.clear();
  memoized.delete = (key) => cache.delete(key);
  
  return memoized;
}

// Shallow (like native Object.assign):
function shallowAssign(target, ...sources) {
  if (target == null) throw new TypeError("Cannot convert null/undefined to object");
  
  const to = Object(target);
  
  for (const source of sources) {
    if (source == null) continue;
    
    // Own enumerable properties (including Symbols!)
    for (const key of [...Object.keys(source), ...Object.getOwnPropertySymbols(source)]) {
      if (Object.prototype.hasOwnProperty.call(source, key)) {
        to[key] = source[key];
      }
    }
  }
  
  return to;
}

// Deep merge (Microsoft follow-up):
function deepMerge(target, ...sources) {
  if (!sources.length) return target;
  
  const source = sources.shift();
  
  if (isPlainObject(target) && isPlainObject(source)) {
    for (const key of Object.keys(source)) {
      if (isPlainObject(source[key])) {
        if (!target[key]) target[key] = {};
        deepMerge(target[key], source[key]);
      } else if (Array.isArray(source[key])) {
        target[key] = [...source[key]]; // Clone arrays
      } else {
        target[key] = source[key];
      }
    }
  }
  
  return deepMerge(target, ...sources);

  /** @type {Map<string, Function[]>} */
  #listeners = new Map();
  
  /**
   * @template {keyof Events} K
   * @param {K} event
   * @param {(...args: Events[K]) => void} listener
   */
  on(event, listener) {
    if (!this.#listeners.has(event)) {
      this.#listeners.set(event, []);
    }
    this.#listeners.get(event).push(listener);
    return () => this.off(event, listener);
  }
  
  /**
   * @template {keyof Events} K
   * @param {K} event
   * @param {...Events[K]} args
   */
  emit(event, ...args) {
    const listeners = this.#listeners.get(event) || [];
    listeners.forEach(fn => fn(...args));
  }
  
  off(event, listener) {
    const arr = this.#listeners.get(event);
    if (arr) {
      const idx = arr.indexOf(listener);
      if (idx > -1) arr.splice(idx, 1);
    }
  }
}

        } else {

        }
      });

      result[newKey] = value;
    }
  }
  return result;
}

class ExpiringStorage {
  #prefix;
  
  constructor(prefix = "app_") {
    this.#prefix = prefix;
    this.#cleanup(); // Clean expired on init
  }
  
  set(key, value, ttlMs) {
    const item = {
      value,
      expiry: ttlMs ? Date.now() + ttlMs : null,
      created: Date.now()
    };
    
    try {
      localStorage.setItem(this.#prefix + key, JSON.stringify(item));
    } catch (e) {
      // QuotaExceeded — evict oldest items
      if (e.name === "QuotaExceededError") {
        this.#evictOldest();
        localStorage.setItem(this.#prefix + key, JSON.stringify(item));
      }
    }
  }
  
  get(key) {
    const raw = localStorage.getItem(this.#prefix + key);
    if (!raw) return null;
    
    try {
      const item = JSON.parse(raw);
      
      // Check expiry
      if (item.expiry && Date.now() > item.expiry) {
        this.remove(key);
        return null;
      }
      
      return item.value;
    } catch {
      return null;
    }
  }
  
  remove(key) {
    localStorage.removeItem(this.#prefix + key);
  }
  
  has(key) {
    return this.get(key) !== null;
  }
  
  #cleanup() {
    const keys = Object.keys(localStorage).filter(k => k.startsWith(this.#prefix));
    
    for (const fullKey of keys) {
      const raw = localStorage.getItem(fullKey);
      try {
        const item = JSON.parse(raw);
        if (item.expiry && Date.now() > item.expiry) {
          localStorage.removeItem(fullKey);
        }
      } catch {
        // Invalid JSON — remove corrupt entries
        localStorage.removeItem(fullKey);
      }
    }
  }
  
  #evictOldest() {
    const keys = Object.keys(localStorage)
      .filter(k => k.startsWith(this.#prefix));
    
    const items = keys.map(k => {
      try {
        return { key: k, created: JSON.parse(localStorage.getItem(k)).created };
      } catch {
        return { key: k, created: 0 };
      }
    }).sort((a, b) => a.created - b.created);
    
    // Remove oldest 10%
    const toRemove = Math.max(1, Math.floor(items.length * 0.1));
    items.slice(0, toRemove).forEach(item => localStorage.removeItem(item.key));
  }
  
  clear() {
    Object.keys(localStorage)
      .filter(k => k.startsWith(this.#prefix))
      .forEach(k => localStorage.removeItem(k));
  }
}

// Series execution: one after another
function series(tasks) {
  return new Promise((resolve, reject) => {
    const results = [];
    let index = 0;
    
    function next() {
      if (index >= tasks.length) {
        resolve(results);
        return;
      }
      
      const task = tasks[index];
      index++;
      
      Promise.resolve(task())
        .then(result => {
          results.push(result);
          next();
        })
        .catch(reject);
    }
    
    next();
  });
}

// Parallel with limit:
function parallel(tasks, limit = Infinity) {
  return new Promise((resolve, reject) => {
    const results = new Array(tasks.length);
    let completed = 0;
    let started = 0;
    let hasRejected = false;
    
    function startNext() {
      if (hasRejected || started >= tasks.length) return;
      
      const index = started++;
      
      Promise.resolve(tasks[index]())
        .then(result => {
          results[index] = result;
          completed++;
          
          if (completed === tasks.length) {
            resolve(results);
          } else {
            startNext();
          }
        })
        .catch(error => {
          hasRejected = true;
          reject(error);
        });
    }
    
    // Start initial batch
    const initialBatch = Math.min(limit, tasks.length);
    for (let i = 0; i < initialBatch; i++) {
      startNext();
    }
  });
}

function promiseRace(promises) {
  return new Promise((resolve, reject) => {
    const arr = Array.from(promises);

function deepClone(obj, visited = new WeakMap()) {
  // Primitives & null
  if (obj === null || typeof obj !== "object") return obj;
  
  // Handle circular references
  if (visited.has(obj)) return visited.get(obj);
  
  // Handle Date
  if (obj instanceof Date) return new Date(obj.getTime());
  
  // Handle RegExp
  if (obj instanceof RegExp) return new RegExp(obj.source, obj.flags);
  
  // Handle Map
  if (obj instanceof Map) {
    const map = new Map();
    visited.set(obj, map);
    obj.forEach((value, key) => {
      map.set(deepClone(key, visited), deepClone(value, visited));
    });
    return map;
  }
  
  // Handle Set
  if (obj instanceof Set) {
    const set = new Set();
    visited.set(obj, set);
    obj.forEach(value => {
      set.add(deepClone(value, visited));
    });
    return set;
  }
  
  // Handle Array
  if (Array.isArray(obj)) {
    const arr = [];
    visited.set(obj, arr);
    obj.forEach((item, index) => {
      arr[index] = deepClone(item, visited);
    });
    return arr;
  }
  
  // Handle plain objects (preserve prototype)
  const clone = Object.create(Object.getPrototypeOf(obj));
  visited.set(obj, clone);
  
  // Copy all own properties (including symbols)
  const allKeys = [...Object.keys(obj), ...Object.getOwnPropertySymbols(obj)];
  for (const key of allKeys) {
    const descriptor = Object.getOwnPropertyDescriptor(obj, key);
    if (descriptor.value !== undefined) {
      descriptor.value = deepClone(descriptor.value, visited);
    }
    Object.defineProperty(clone, key, descriptor);
  }
  
  return clone;
}

class LRUCache {
  #capacity;
  #cache; // Map maintains insertion order!
  
  constructor(capacity) {
    this.#capacity = capacity;
    this.#cache = new Map();
  }
  
  get(key) {
    if (!this.#cache.has(key)) return -1;
    
    // Move to end (most recently used)
    const value = this.#cache.get(key);
    this.#cache.delete(key);
    this.#cache.set(key, value);
    
    return value;
  }

    // If key exists, delete it (will re-add at end)
    if (this.#cache.has(key)) {
      this.#cache.delete(key);
    }
    
    // If at capacity, evict LEAST recently used (first item in Map)

  #onTick;
  #onComplete;
  #timerId = null;
  #paused = false;

  constructor(options) {
    this.#targetTime = options.duration; // ms

  }

    if (this.#timerId) return;
    
    const startTime = Date.now();
    const endTime = startTime + this.#remaining;
    
    this.#timerId = setInterval(() => {
      const now = Date.now();
      this.#remaining = Math.max(0, endTime - now);
      
      this.#onTick(this.#formatTime(this.#remaining));
      
      if (this.#remaining <= 0) {
        this.stop();
        this.#onComplete();
      }
    }, 100); // Update every 100ms for smooth display
    
    this.#paused = false;
  }

    if (this.#timerId) {
      clearInterval(this.#timerId);
      this.#timerId = null;
      this.#paused = true;
    }
  }
  
  resume() {
    if (this.#paused) this.start();
  }
  
  stop() {
    clearInterval(this.#timerId);
    this.#timerId = null;
  }
  
  reset(duration) {
    this.stop();
    this.#remaining = duration || this.#targetTime;
    this.#onTick(this.#formatTime(this.#remaining));
  }

    const totalSeconds = Math.ceil(ms / 1000);
    const hours = Math.floor(totalSeconds / 3600);
    const minutes = Math.floor((totalSeconds % 3600) / 60);

      hours: String(hours).padStart(2, "0"),
      minutes: String(minutes).padStart(2, "0"),

      display: `${String(hours).padStart(2, "0")}:${String(minutes).padStart(2, "0")}:${String(seconds).padStart(2, "0")}`
    };
  }
  
  get isRunning() { return this.#timerId !== null; }
  get isPaused() { return this.#paused; }

class LazyImageLoader {

  #options;
  #loadedCount = 0;
  
  constructor(options = {}) {

      rootMargin: options.rootMargin || "200px 0px", // Start loading 200px before visible

      onError: options.onError || (() => {})
    };
    
    this.#observer = new IntersectionObserver(
      this.#handleIntersection.bind(this),
      {
        rootMargin: this.#options.rootMargin,
        threshold: this.#options.threshold
      }
    );
  }
  
  observe(selector = "img[data-src]") {
    const images = document.querySelectorAll(selector);
    images.forEach(img => {
      // Set placeholder
      if (!img.src) img.src = this.#options.placeholder;
      img.style.transition = "opacity 0.3s ease";
      this.#observer.observe(img);
    });
    return this;
  }
  
  #handleIntersection(entries) {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        this.#loadImage(entry.target);
        this.#observer.unobserve(entry.target);
      }
    });
  }
  
  #loadImage(img) {
    const src = img.dataset.src;
    const srcset = img.dataset.srcset;
    
    // Preload in memory first (avoid layout shift)
    const tempImg = new Image();
    
    tempImg.onload = () => {
      img.src = src;
      if (srcset) img.srcset = srcset;
      img.style.opacity = "1";
      img.removeAttribute("data-src");
      img.removeAttribute("data-srcset");
      this.#loadedCount++;
      this.#options.onLoad(img, this.#loadedCount);
    };
    
    tempImg.onerror = () => {
      img.alt = "Failed to load image";
      this.#options.onError(img, src);
    };
    
    img.style.opacity = "0.5";
    tempImg.src = src;
  }
  
  disconnect() {
    this.#observer.disconnect();
  }
  
  get loadedCount() { return this.#loadedCount; }
}

class RateLimiter {
  #tokens;
  #maxTokens;
  #refillRate; // tokens per second
  #lastRefill;
  
  constructor(options = {}) {
    this.#maxTokens = options.maxTokens || 10;
    this.#tokens = this.#maxTokens;
    this.#refillRate = options.refillRate || 1; // 1 token/sec
    this.#lastRefill = Date.now();
  }
  
  #refill() {
    const now = Date.now();
    const elapsed = (now - this.#lastRefill) / 1000;
    const tokensToAdd = elapsed * this.#refillRate;
    
    this.#tokens = Math.min(this.#maxTokens, this.#tokens + tokensToAdd);
    this.#lastRefill = now;
  }
  
  tryConsume(tokens = 1) {
    this.#refill();
    
    if (this.#tokens >= tokens) {
      this.#tokens -= tokens;
      return { allowed: true, remaining: Math.floor(this.#tokens) };
    }
    
    const waitTime = (tokens - this.#tokens) / this.#refillRate * 1000;
    return { allowed: false, retryAfter: Math.ceil(waitTime) };
  }
  
  // Async version — waits for token availability
  async consume(tokens = 1) {
    const result = this.tryConsume(tokens);
    
    if (result.allowed) return result;
    
    await new Promise(r => setTimeout(r, result.retryAfter));
    return this.consume(tokens); // Retry after wait
  }
  
  get availableTokens() {
    this.#refill();
    return Math.floor(this.#tokens);
  }
}

class Observable {
  #subscribeFn;
  
  constructor(subscribeFn) {
    this.#subscribeFn = subscribeFn;
  }
  
  subscribe(observer) {
    // Normalize observer (function → object)
    const normalizedObserver = typeof observer === "function"
      ? { next: observer, error: () => {}, complete: () => {} }
      : { next: observer.next || (() => {}), 
          error: observer.error || (() => {}), 
          complete: observer.complete || (() => {}) };
    
    let isUnsubscribed = false;
    
    const safeObserver = {
      next: (val) => { if (!isUnsubscribed) normalizedObserver.next(val); },
      error: (err) => { if (!isUnsubscribed) { normalizedObserver.error(err); isUnsubscribed = true; } },
      complete: () => { if (!isUnsubscribed) { normalizedObserver.complete(); isUnsubscribed = true; } }
    };
    
    const cleanup = this.#subscribeFn(safeObserver);
    
    return {
      unsubscribe() {
        isUnsubscribed = true;
        if (typeof cleanup === "function") cleanup();
      }
    };
  }
  
  // Operators:
  map(transformFn) {
    return new Observable((observer) => {
      const sub = this.subscribe({
        next: (val) => observer.next(transformFn(val)),
        error: (err) => observer.error(err),
        complete: () => observer.complete()
      });
      return () => sub.unsubscribe();
    });
  }
  
  filter(predicateFn) {
    return new Observable((observer) => {
      const sub = this.subscribe({
        next: (val) => { if (predicateFn(val)) observer.next(val); },
        error: (err) => observer.error(err),
        complete: () => observer.complete()
      });
      return () => sub.unsubscribe();
    });
  }
  
  take(count) {
    return new Observable((observer) => {
      let taken = 0;
      const sub = this.subscribe({
        next: (val) => {
          if (taken < count) { observer.next(val); taken++; }
          if (taken >= count) { observer.complete(); sub.unsubscribe(); }
        },
        error: (err) => observer.error(err),
        complete: () => observer.complete()
      });
      return () => sub.unsubscribe();
    });
  }
  
  // Static creators:
  static fromEvent(element, eventName) {
    return new Observable((observer) => {
      const handler = (e) => observer.next(e);
      element.addEventListener(eventName, handler);
      return () => element.removeEventListener(eventName, handler);
    });
  }
  
  static interval(ms) {
    return new Observable((observer) => {
      let count = 0;
      const id = setInterval(() => observer.next(count++), ms);
      return () => clearInterval(id);
    });
  }
}

class ReconnectingWebSocket {
  #url;
  #ws = null;
  #options;
  #retryCount = 0;
  #listeners = new Map();
  #forceClosed = false;
  #messageQueue = [];
  
  constructor(url, options = {}) {
    this.#url = url;
    this.#options = {
      maxRetries: options.maxRetries ?? 10,
      baseDelay: options.baseDelay ?? 1000,
      maxDelay: options.maxDelay ?? 30000,
      onStatusChange: options.onStatusChange || (() => {})
    };
    this.#connect();
  }
  
  #connect() {
    this.#options.onStatusChange("connecting");
    this.#ws = new WebSocket(this.#url);
    
    this.#ws.onopen = () => {
      this.#retryCount = 0;
      this.#options.onStatusChange("connected");
      // Flush queued messages
      while (this.#messageQueue.length > 0) {
        this.#ws.send(this.#messageQueue.shift());
      }
    };
    
    this.#ws.onclose = (event) => {
      if (this.#forceClosed) return;
      this.#options.onStatusChange("disconnected");
      this.#scheduleReconnect();
    };
    
    this.#ws.onerror = () => {
      this.#ws.close();
    };
    
    this.#ws.onmessage = (event) => {
      const handlers = this.#listeners.get("message") || [];
      handlers.forEach(fn => fn(JSON.parse(event.data)));
    };
  }
  
  #scheduleReconnect() {
    if (this.#retryCount >= this.#options.maxRetries) {
      this.#options.onStatusChange("failed");
      return;
    }
    
    const delay = Math.min(
      this.#options.baseDelay * Math.pow(2, this.#retryCount),
      this.#options.maxDelay
    );
    
    this.#retryCount++;
    this.#options.onStatusChange(`reconnecting (attempt ${this.#retryCount})`);
    setTimeout(() => this.#connect(), delay);
  }
  
  send(data) {
    const msg = typeof data === "string" ? data : JSON.stringify(data);
    
    if (this.#ws?.readyState === WebSocket.OPEN) {
      this.#ws.send(msg);
    } else {
      this.#messageQueue.push(msg); // Queue for when connection restores
    }
  }
  
  on(event, handler) {
    if (!this.#listeners.has(event)) this.#listeners.set(event, []);
    this.#listeners.get(event).push(handler);
    return () => {
      const handlers = this.#listeners.get(event);
      const idx = handlers.indexOf(handler);
      if (idx > -1) handlers.splice(idx, 1);
    };
  }
  
  close() {
    this.#forceClosed = true;
    this.#ws?.close();
    this.#options.onStatusChange("closed");
  }
  
  get state() {
    return ["CONNECTING", "OPEN", "CLOSING", "CLOSED"][this.#ws?.readyState ?? 3];
  }
}

1Difference between == and ===?=== checks value AND type (strict). == coerces types first then compares (abstract). Always use === except for null checks (x == null covers both null and undefined).
2What is closure?A function that remembers its lexical scope even when executed outside that scope. The inner function maintains a reference to the outer function's variables. Used for data privacy, factories, and maintaining state.
3Explain event delegation.Instead of attaching listeners to each child, attach ONE listener to the parent. Events bubble up → check e.target to identify which child was clicked. Benefits: fewer listeners, works for dynamically added elements.
4What is the event loop?JS is single-threaded. The event loop continuously checks: (1) execute call stack until empty, (2) drain all microtasks (promises), (3) execute one macrotask (setTimeout). Repeat.
5var vs let vs const?var: function-scoped, hoisted (initialized undefined), re-declarable. let: block-scoped, hoisted (TDZ), re-assignable. const: block-scoped, hoisted (TDZ), NOT re-assignable (but objects/arrays are mutable).
6What is prototypal inheritance?Objects inherit from other objects via __proto__ chain. When property isn't found on object, JS walks up the prototype chain until null. Classes are syntactic sugar over prototype-based inheritance.
7What is 'this' in JS?Determined by HOW function is called (not where defined): new → new object, call/apply/bind → specified, obj.method() → obj, standalone → globalThis/undefined(strict). Arrow functions inherit from enclosing lexical scope.
8Explain Promise states.3 states: pending (initial), fulfilled (resolved with value), rejected (failed with reason). Once settled (fulfilled/rejected), state is IMMUTABLE — can't change back. .then() returns a new promise (chaining).
9Difference between null and undefined?undefined: variable declared but not assigned, or missing function params, or missing object property. null: intentional absence of value, explicitly assigned. typeof undefined = "undefined", typeof null = "object" (bug).
10What is hoisting?During compilation, declarations (var, function, class) are processed before execution. var → initialized to undefined. let/const → hoisted but NOT initialized (TDZ). Function declarations → fully hoisted (available before declaration line).
11Deep copy vs shallow copy?Shallow: copies first level only (spread, Object.assign). Nested objects share references. Deep: copies ALL levels recursively. Use structuredClone (modern), JSON.parse(JSON.stringify()) (limitations: no functions/dates/undefined), or custom recursive function.
12What is debounce vs throttle?Debounce: delays execution until N ms after LAST call (search input). Throttle: ensures execution at most once per N ms (scroll handler). Debounce waits for silence; throttle limits frequency.
13Explain async/await.Syntactic sugar over promises. async function always returns a promise. await pauses execution until promise settles (doesn't block main thread — other code runs). Error handling with try/catch. Internally uses generators + promises.
14What is CORS?Cross-Origin Resource Sharing. Browser security policy that blocks requests to different origins. Server must send Access-Control-Allow-Origin header. Preflight (OPTIONS) sent for non-simple requests. Not enforced by server — it's browser-side enforcement.
15Explain Web Workers.Run JavaScript in background threads. Can't access DOM (postMessage for communication). Use for CPU-intensive tasks (image processing, data parsing). Types: Dedicated (one page), Shared (multiple tabs), Service Worker (proxy/offline).
16What is tree shaking?Dead code elimination at build time. Bundler removes unused exports. ONLY works with ES Modules (static import/export), NOT CommonJS (dynamic require). Mark package.json "sideEffects": false for better results.
17WeakMap vs Map?Map: any key type, keys prevent GC, has size, iterable. WeakMap: only object keys, keys DON'T prevent GC (weak reference), no size/iteration. Use WeakMap for metadata attached to DOM elements or objects that may be GC'd.
18What is XSS and how to prevent?Cross-Site Scripting: injecting malicious scripts via user input. Prevent: escape/sanitize all user input before rendering, use textContent (not innerHTML), CSP headers, HttpOnly cookies, trusted types API.
19Explain generator functions.function* that can pause/resume execution with yield. Returns an iterator. yield produces a value AND pauses. next() resumes and can send a value back in. Used for: lazy sequences, async flow control, infinite iterables.
20Difference between call, apply, bind?All set 'this' explicitly. call(thisArg, arg1, arg2): invokes immediately with individual args. apply(thisArg, [args]): invokes immediately with array of args. bind(thisArg, ...args): returns NEW function with bound 'this' (doesn't invoke).

21What are symbols used for?Unique identifiers that never collide. Use for: private-ish properties, preventing name collisions in libraries, well-known symbols (Symbol.iterator, Symbol.toPrimitive) to customize built-in behavior. Not truly private (Reflect.ownKeys reveals them).
22CommonJS vs ES Modules?CJS: require()/module.exports, synchronous loading, copies of values, dynamic (can conditionally require). ESM: import/export, async loading, live bindings (references), static (analyzable by tools → tree shaking). ESM is the standard going forward.
23What is a Service Worker?A proxy between browser and network. Runs in background thread, intercepts fetch requests. Use for: offline support (Cache API), push notifications, background sync. Lifecycle: install → activate → fetch. Must be HTTPS.
24Explain the render pipeline.JS → Style → Layout → Paint → Composite. Layout calculates positions (expensive!). Paint fills pixels. Composite combines layers (GPU-accelerated, cheap). Animate with transform/opacity to skip layout+paint. Avoid layout thrashing (read-write-read patterns).
25What is a memory leak in JS?When objects are no longer needed but still referenced (can't be GC'd). Common causes: forgotten event listeners, closures holding large data, growing caches without bounds, detached DOM references, uncleared timers. Detect with Chrome DevTools Memory tab.
