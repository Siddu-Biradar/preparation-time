1. Function Declarations, Expressions & Arrow Functions

📺 YouTube Tutorials
  • Functions in JS — Namaste JavaScript (Akshay Saini)
  • Arrow Functions — Web Dev Simplified
  • Function Expressions vs Declarations — Traversy Media

Theory

Function Declaration (Statement)

// Fully hoisted — can be called BEFORE declaration
greet("Sid"); // Works!

function greet(name) {
  return `Hello, ${name}!`;
}

// Characteristics:
// - Hoisted completely (both name and body)
// - Creates its own 'this' binding
// - Has 'arguments' object
// - Can be named (useful for recursion)
// - Has .name property: greet.name === "greet"

Function Expression

// NOT hoisted — variable is hoisted but as undefined
// greet("Sid"); // TypeError: greet is not a function

const greet = function(name) {
  return `Hello, ${name}!`;
};

// Named Function Expression (NFE) — name only accessible inside
const factorial = function fact(n) {
  return n <= 1 ? 1 : n * fact(n - 1); // 'fact' available inside
};
// fact(5); // ReferenceError — not accessible outside
factorial.name; // "fact" (not "factorial")

Arrow Functions (ES6)

// Concise syntax
const add = (a, b) => a + b;       // implicit return (expression)
const square = x => x * x;         // single param: no parentheses needed
const getRandom = () => Math.random(); // no params: () required
const getObj = () => ({ key: "value" }); // return object: wrap in ()

// Block body (explicit return needed)
const greet = (name) => {
  const message = `Hello, ${name}!`;
  return message;
};

// KEY DIFFERENCES from regular functions:
// 1. NO own 'this' — inherits from enclosing lexical scope
// 2. NO 'arguments' object (use ...rest instead)
// 3. Cannot be used as constructor (no 'new')
// 4. NO 'prototype' property
// 5. Cannot be used as generator (no yield)
// 6. No own 'super' binding

// 'this' difference example:
const obj = {
  name: "Sid",
  regularMethod: function() {
    console.log(this.name); // "Sid" — this = obj
    setTimeout(function() {
      console.log(this.name); // undefined — this = window!
    }, 100);
  },
  arrowMethod: function() {
    console.log(this.name); // "Sid"
    setTimeout(() => {
      console.log(this.name); // "Sid" — arrow inherits 'this'!
    }, 100);
  }
};

Comparison Table

Hoisted✅ Fully❌ Only var name❌ Only var name
Own this✅✅❌ (lexical)
arguments✅✅❌

Angular Note: In Angular components, always use arrow functions for callbacks (setTimeout, subscribe, etc.) to preserve component's this. But define class methods as regular methods (TypeScript handles this via the class context).

🎯 Interview Questions

Easy What are the differences between arrow functions and regular functions?

Answer: Arrow functions: (1) no own 'this' (lexical), (2) no 'arguments' object, (3) can't be constructors, (4) no prototype, (5) can't be generators. They're best for callbacks and short operations; not for object methods or constructors.

Hard When should you NOT use arrow functions?

Answer: (1) Object methods that need 'this', (2) Event handlers that need 'this' to be the element, (3) Prototype methods, (4) Constructors, (5) Functions that need 'arguments' object, (6) Dynamic context functions used with call/apply/bind (won't work — arrow's this can't be overridden).

2. Parameters, Arguments & Rest/Spread

📺 YouTube Tutorials
  • Rest & Spread Operators — Web Dev Simplified
  • Default Parameters — Traversy Media

Theory

Default Parameters (ES6)

// Old way (ES5)
function greet(name) {
  name = name || "World"; // Bug: "" would also use default!
}

// Modern way (ES6)
function greet(name = "World") {
  return `Hello, ${name}!`;
}
greet();         // "Hello, World!"
greet("Sid");    // "Hello, Sid!"
greet(undefined); // "Hello, World!" (undefined triggers default)
greet(null);     // "Hello, null!" (null does NOT trigger default)
greet("");       // "Hello, !" (empty string does NOT trigger default)

// Default can use previous params
function createUser(name, role = "user", id = `${name}_${Date.now()}`) {
  return { name, role, id };
}

// Default with destructuring
function config({ host = "localhost", port = 3000 } = {}) {
  return `${host}:${port}`;
}
config();                    // "localhost:3000"
config({ port: 8080 });     // "localhost:8080"

arguments Object

// 'arguments' is array-LIKE (not a real array)
function sum() {
  console.log(arguments);        // { 0: 1, 1: 2, 2: 3, length: 3 }
  console.log(arguments.length); // 3
  // No array methods! (no map, filter, reduce)
  
  // Convert to real array:
  const args = Array.from(arguments);
  // or: const args = [...arguments];
  // or: const args = Array.prototype.slice.call(arguments);
  
  return args.reduce((sum, n) => sum + n, 0);
}
sum(1, 2, 3); // 6

// ⚠️ arguments is NOT available in arrow functions
const arrowSum = () => {
  console.log(arguments); // ReferenceError (or refers to outer scope's arguments)
};

Rest Parameters (...rest)

// Modern replacement for arguments — gives REAL array
function sum(...numbers) {
  return numbers.reduce((total, n) => total + n, 0);
}
sum(1, 2, 3, 4); // 10

// Must be LAST parameter
function logActivity(action, ...details) {
  console.log(`Action: ${action}`);
  console.log(`Details: ${details.join(", ")}`);
}
logActivity("click", "button", "header", "nav");

// Destructuring with rest
function first([head, ...tail]) {
  return { head, tail };
}
first([1, 2, 3, 4]); // { head: 1, tail: [2, 3, 4] }

Spread Operator (...spread)

// Spread in function calls
const nums = [3, 1, 4, 1, 5];
Math.max(...nums); // 5 (same as Math.max(3,1,4,1,5))

// Spread in arrays (shallow copy)
const original = [1, 2, 3];
const copy = [...original];       // new array
const combined = [...original, 4, 5]; // [1,2,3,4,5]
const merged = [...arr1, ...arr2];

// Spread in objects (shallow copy)
const user = { name: "Sid", age: 25 };
const updated = { ...user, age: 26 }; // override age
const full = { ...user, role: "dev" }; // add property

// ⚠️ Spread is SHALLOW — nested objects still share reference
const nested = { a: 1, b: { c: 2 } };
const clone = { ...nested };
clone.b.c = 99;
console.log(nested.b.c); // 99 — shared reference!

🎯 Interview Questions

Medium What is the difference between rest and spread?

Answer: Same syntax (...) but opposite purposes. Rest COLLECTS multiple elements into an array (function params, destructuring). Spread EXPANDS an iterable into individual elements (function calls, array/object literals). Rest is always at receiving side, Spread at giving side.

Medium What happens if you pass undefined vs null for a default parameter?

Answer: undefined triggers the default value. null does NOT — it's treated as an intentional value. This is because default params check argument === undefined.

3. Scope (Global, Function, Block, Lexical)

📺 YouTube Tutorials
  • Scope & Scope Chain — Namaste JavaScript (Akshay Saini)
  • JS Scope Explained — Web Dev Simplified
  • Lexical Scope — Fun Fun Function

Theory

Types of Scope

// 1. GLOBAL SCOPE — accessible everywhere
var globalVar = "I'm global";     // window.globalVar (var only)
let globalLet = "I'm global too"; // NOT on window
// Variables declared without keyword (sloppy mode) are global

// 2. FUNCTION SCOPE — accessible within the function
function myFunc() {
  var funcScoped = "only here";
  let alsoHere = "only here too";
  // Both accessible within this function and nested functions
}
// funcScoped — ReferenceError outside

// 3. BLOCK SCOPE — {} creates scope for let/const (NOT var)
{
  var x = 1;   // LEAKS out! (function-scoped)
  let y = 2;   // Block-scoped
  const z = 3; // Block-scoped
}
console.log(x); // 1
// console.log(y); // ReferenceError
// console.log(z); // ReferenceError

// if, for, while, switch all create block scopes for let/const
if (true) {
  let blockVar = "trapped";
}
// blockVar — ReferenceError

// 4. MODULE SCOPE — each ES module has its own scope
// Variables in a module are NOT global (unlike script tags)

Lexical Scope (Static Scope)

// Scope is determined by WHERE code is WRITTEN, not where it's called
function outer() {
  let secret = "lexical!";
  
  function inner() {
    console.log(secret); // Can access — inner is WRITTEN inside outer
  }
  
  return inner;
}

const fn = outer();
fn(); // "lexical!" — still has access! (this is closure)

// Scope Chain: inner → outer → global
// Engine looks up the chain until it finds the variable or reaches global

Scope Chain Visualization

let a = "global";

function first() {
  let b = "first";
  
  function second() {
    let c = "second";
    
    function third() {
      let d = "third";
      console.log(a); // "global" — found in global scope
      console.log(b); // "first" — found in first's scope
      console.log(c); // "second" — found in second's scope
      console.log(d); // "third" — found in own scope
    }
    third();
  }
  second();
}
first();

/*
Scope Chain for third():
  third's scope { d } 
    → second's scope { c }
      → first's scope { b }
        → global scope { a }
*/

Variable Shadowing

let x = 10;

function shadow() {
  let x = 20; // Shadows outer x
  console.log(x); // 20
  
  if (true) {
    let x = 30; // Shadows function x
    console.log(x); // 30
  }
  
  console.log(x); // 20 (block x is gone)
}

shadow();
console.log(x); // 10 (global x unchanged)

// ILLEGAL shadowing: let cannot be shadowed by var in same scope
let y = 10;
{
  var y = 20; // SyntaxError! var would escape block and conflict
}

🎯 Interview Questions

Medium What is lexical scope?

Answer: Lexical (static) scope means a function's scope is determined by where it's defined in the source code, not where it's called. Inner functions can access variables from outer functions because the scope chain is established at write-time. This is the foundation of closures.

Hard How does the scope chain resolution work?

Answer: When a variable is accessed, the engine first checks the current scope's Environment Record. If not found, it follows the [[OuterEnv]] reference to the parent scope, continuing up the chain until reaching global scope. If still not found → ReferenceError. This chain is created at function definition time (lexically).

4. Hoisting Deep Dive

📺 YouTube Tutorials
  • Hoisting in JavaScript — Namaste JavaScript (Akshay Saini)
  • Hoisting Explained — Web Dev Simplified
  • Hoisting, TDZ & Scope — Lydia Hallie

Theory

Hoisting is JavaScript's behavior of moving declarations to the top of their scope during the creation phase. It's not actually "moving" — it's the engine registering identifiers during the creation phase before executing code.

What Gets Hoisted and How

// 1. var — hoisted with initialization to 'undefined'
console.log(x); // undefined
var x = 5;
// Behaves as if:
// var x = undefined;  (creation phase)
// console.log(x);     (execution phase)
// x = 5;             (execution phase)

// 2. let/const — hoisted but UNINITIALIZED (TDZ)
// console.log(y); // ReferenceError: Cannot access 'y' before initialization
let y = 10;
// They ARE hoisted (engine knows about them), just not initialized

// 3. Function Declaration — FULLY hoisted (name + body)
sayHello(); // "Hello!" — works before declaration!
function sayHello() {
  console.log("Hello!");
}

// 4. Function Expression — only variable is hoisted
// greet(); // TypeError: greet is not a function
var greet = function() { console.log("Hi"); };
// With var: greet = undefined initially
// With let/const: TDZ error

// 5. Class — hoisted but NOT initialized (like let/const)
// new Dog(); // ReferenceError
class Dog {
  bark() { console.log("Woof"); }
}

// 6. import — hoisted to the top (entire module loaded first)
// This is why you can use imports before they appear in code

Hoisting Priority (within same scope)

// Function declarations are hoisted ABOVE var declarations
console.log(typeof foo); // "function" (not "undefined")
var foo = "hello";
function foo() { return "world"; }
// After hoisting:
// function foo() {...}  ← function wins
// var foo;             ← var ignored (already declared)
// console.log(typeof foo); → "function"
// foo = "hello";       ← reassignment

Tricky Hoisting Scenarios

// Scenario 1: Function inside block (non-strict mode behavior varies!)
// In strict mode / modules: function declarations are block-scoped
console.log(foo); // undefined (in non-strict) or ReferenceError (strict)
if (true) {
  function foo() { return 1; }
}

// Scenario 2: Hoisting in loops
for (var i = 0; i < 3; i++) {
  // var i is hoisted to function/global scope
}
console.log(i); // 3 (accessible!)

// Scenario 3: Multiple declarations
var a = 1;
var a = 2; // Re-declaration is fine for var
console.log(a); // 2

// Scenario 4: Function expression with var
console.log(bar); // undefined
console.log(bar()); // TypeError: bar is not a function
var bar = function() { return "bar"; };
Mental Model: Don't think of code as "moving." Think of two passes: Pass 1 (creation) registers all identifiers in memory. Pass 2 (execution) runs code line by line. var gets undefined in Pass 1. let/const get "uninitialized" in Pass 1. function declarations get their full body in Pass 1.

🏋️ Hands-on Exercise

Exercise 2: Predict the output (very common interview question):

var a = 1;
function b() {
  a = 10;
  return;
  function a() {}
}
b();
console.log(a); // ?

Answer: 1. Inside b(), function a(){} is hoisted, creating a LOCAL 'a'. Then a = 10 modifies the local 'a', not the global one. Global 'a' remains 1.

🎯 Interview Questions

Hard What is the output of: var a = 1; function b() { a = 10; return; function a() {} } b(); console.log(a);

Answer: Output is 1. function a() {} inside b() is hoisted, creating a local variable 'a' in b's scope. The assignment a = 10 modifies this local 'a', not the global one. The global a remains 1.

Expert Are let and const hoisted? Explain with proof.

Answer: Yes! Proof: let x = 'outer'; { console.log(x); let x = 'inner'; } gives ReferenceError (not 'outer'). If let wasn't hoisted, it would print 'outer' from the outer scope. The error proves the engine KNOWS about the inner x (it's hoisted) but won't let you access it until initialization (TDZ).

5. Closures (Complete Mastery)

📺 YouTube Tutorials
  • Closures — Namaste JavaScript (Akshay Saini)
  • Closures Explained — Fireship
  • Closures — Fun Fun Function
  • Practical Closures — Web Dev Simplified

Theory

Definition: A closure is a function that remembers and has access to variables from its lexical scope (where it was defined), even when executing outside that scope.

Every function in JavaScript creates a closure. The function maintains a reference to its outer environment (not a copy of variables — the live reference).

Basic Closure

function outer() {
  let count = 0; // This variable is "closed over"
  
  function inner() {
    count++; // Accesses outer's variable
    return count;
  }
  
  return inner; // Return the function (not the result)
}

const counter = outer(); // outer() executes and returns inner
console.log(counter()); // 1
console.log(counter()); // 2
console.log(counter()); // 3
// 'count' persists between calls because inner has a closure over it!
// outer's execution context is gone, but count lives on via closure.

Closure Mental Model

/*
When outer() returns inner(), the inner function is "bundled" with 
its lexical environment (a reference to outer's scope).

This bundle (function + environment) IS the closure.

┌─────────────────────────────────┐
│ Closure (returned from outer)   │
│  ┌────────────────────────────┐ │
│  │ Function: inner()          │ │
│  └────────────────────────────┘ │
│  ┌────────────────────────────┐ │
│  │ Environment:               │ │
│  │  { count: 0 }              │ │
│  │  (reference to             │ │
│  │   outer's scope)           │ │
│  └────────────────────────────┘ │
└─────────────────────────────────┘
*/

Practical Closure Patterns

// 1. DATA PRIVACY / ENCAPSULATION
function createBankAccount(initialBalance) {
  let balance = initialBalance; // Private variable
  
  return {
    deposit(amount) {
      if (amount > 0) balance += amount;
      return balance;
    },
    withdraw(amount) {
      if (amount > 0 && amount <= balance) balance -= amount;
      return balance;
    },
    getBalance() {
      return balance;
    }
  };
}

const account = createBankAccount(1000);
account.deposit(500);     // 1500
account.withdraw(200);    // 1300
account.getBalance();     // 1300
// account.balance → undefined! Truly private!

// 2. FUNCTION FACTORIES
function multiplier(factor) {
  return (number) => number * factor;
}
const double = multiplier(2);
const triple = multiplier(3);
double(5);  // 10
triple(5);  // 15

// 3. MEMOIZATION
function memoize(fn) {
  const cache = {}; // Closed over
  return function(...args) {
    const key = JSON.stringify(args);
    if (key in cache) {
      console.log("From cache");
      return cache[key];
    }
    const result = fn(...args);
    cache[key] = result;
    return result;
  };
}

const expensiveCalc = memoize((n) => {
  console.log("Computing...");
  return n * n;
});
expensiveCalc(5); // "Computing..." → 25
expensiveCalc(5); // "From cache" → 25

// 4. EVENT HANDLERS (Angular-relevant)
function setupClickCounter(buttonId) {
  let clicks = 0; // Private to this setup
  document.getElementById(buttonId).addEventListener('click', () => {
    clicks++;
    console.log(`Button clicked ${clicks} times`);
  });
}

// 5. MODULE PATTERN (pre-ES6 modules)
const Calculator = (function() {
  let history = []; // Private
  
  return {
    add(a, b) { 
      const result = a + b;
      history.push(`${a}+${b}=${result}`);
      return result;
    },
    getHistory() { return [...history]; }
  };
})();

Common Closure Pitfalls

// PITFALL 1: Loop with var (CLASSIC interview question)
for (var i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 100);
}
// Output: 3, 3, 3 (all share same 'i' which is 3 after loop)

// FIX 1: Use let (creates new binding per iteration)
for (let i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 100);
}
// Output: 0, 1, 2

// FIX 2: IIFE (creates new scope per iteration)
for (var i = 0; i < 3; i++) {
  (function(j) {
    setTimeout(() => console.log(j), 100);
  })(i);
}
// Output: 0, 1, 2

// FIX 3: setTimeout 3rd argument
for (var i = 0; i < 3; i++) {
  setTimeout((j) => console.log(j), 100, i);
}

// PITFALL 2: Closures hold references, not values
function buggy() {
  let result = [];
  for (var i = 0; i < 3; i++) {
    result.push(function() { return i; });
  }
  return result;
}
const fns = buggy();
fns[0](); // 3 (not 0!)
fns[1](); // 3 (not 1!)
fns[2](); // 3 (not 2!)
// All functions close over the SAME 'i' variable

// PITFALL 3: Memory leaks
function createLeak() {
  const hugeData = new Array(1000000).fill("*");
  return function() {
    // Even if we don't USE hugeData, it's still in the closure!
    console.log("I exist");
  };
}
// hugeData can't be garbage collected while the returned function lives

🏋️ Hands-on Exercise

Exercise 3: Implement a once(fn) function — the returned function only executes fn once:

function once(fn) {
  let called = false;
  let result;
  return function(...args) {
    if (!called) {
      called = true;
      result = fn(...args);
    }
    return result;
  };
}

const initialize = once(() => {
  console.log("Initializing...");
  return { status: "ready" };
});
initialize(); // "Initializing..." → { status: "ready" }
initialize(); // → { status: "ready" } (no log, returns cached)

Exercise 4: Implement a createCounter() with increment, decrement, reset, and getValue:

function createCounter(initial = 0) {
  let count = initial;
  return {
    increment: () => ++count,
    decrement: () => --count,
    reset: () => (count = initial),
    getValue: () => count
  };
}

🎯 Interview Questions

Medium What is a closure? Give a practical example.

Answer: A closure is a function bundled with its lexical environment — it retains access to outer scope variables even after the outer function has returned. Practical: data privacy (module pattern), function factories (multiplier), memoization, event handlers that need persistent state.

Hard Explain the classic for-loop setTimeout problem and 3 solutions.

Answer: With var, all iterations share one 'i' variable. By the time callbacks run, loop is done and i equals the final value. Solutions: (1) let — creates new binding per iteration, (2) IIFE — creates new scope capturing current i, (3) setTimeout's third argument — passes current i as callback param.

Expert How can closures cause memory leaks?

Answer: A closure retains a reference to its entire outer scope, not just used variables (though engines may optimize). If a closure lives long (event handlers, timers), large objects in its scope can't be GC'd. Patterns: DOM event listeners holding component references, setInterval callbacks, unsubscribed observables. Fix: nullify references, remove listeners, use WeakRef.

6. IIFE (Immediately Invoked Function Expressions)

📺 YouTube Tutorials
  • IIFE Explained — Akshay Saini
  • IIFE Pattern — Traversy Media

Theory

// IIFE: function that executes immediately after creation
(function() {
  console.log("I run immediately!");
})();

// Arrow IIFE
(() => {
  console.log("Arrow IIFE!");
})();

// Named IIFE (useful for stack traces)
(function init() {
  console.log("Named IIFE");
})();

// With parameters
(function(global) {
  global.myLib = { version: "1.0" };
})(window);

// IIFE returning a value
const result = (function() {
  const x = 10;
  const y = 20;
  return x + y;
})(); // result = 30

// Classic use: creating private scope (pre-ES6 modules)
const module = (function() {
  let private = 0;
  
  return {
    increment() { private++; },
    getValue() { return private; }
  };
})();

// Modern use: top-level await alternative (before ES2022)
(async function() {
  const data = await fetch('/api/data');
  // ... process data
})();
Modern Relevance: IIFEs are less needed now because ES6 modules provide file-level scope, and let/const provide block scope. However, they still appear in legacy code, library patterns, and interview questions. Angular projects rarely need IIFEs since TypeScript/modules handle scoping.

🎯 Interview Questions

Easy What is an IIFE and why was it used?

Answer: IIFE is a function that's defined and executed immediately: (function(){})();. Used to create private scope (avoid polluting global), initialize modules, create closures for loops (var problem), and execute async operations at top level. Less needed with ES6 modules and let/const.

7. Higher-Order Functions & Callbacks

📺 YouTube Tutorials
  • Higher-Order Functions — Namaste JavaScript
  • Higher-Order Functions — Fun Fun Function
  • Callbacks Explained — Web Dev Simplified

Theory

A Higher-Order Function (HOF) is a function that: (1) takes one or more functions as arguments, OR (2) returns a function.

// HOF that takes a function (map, filter, reduce, forEach, etc.)
const numbers = [1, 2, 3, 4, 5];

// map — transform each element
const doubled = numbers.map(n => n * 2); // [2, 4, 6, 8, 10]

// filter — keep elements passing test
const evens = numbers.filter(n => n % 2 === 0); // [2, 4]

// reduce — accumulate to single value
const sum = numbers.reduce((acc, n) => acc + n, 0); // 15

// find — first element passing test
const found = numbers.find(n => n > 3); // 4

// every — all pass test?
const allPositive = numbers.every(n => n > 0); // true

// some — any pass test?
const hasEven = numbers.some(n => n % 2 === 0); // true

// HOF that returns a function
function withLogging(fn) {
  return function(...args) {
    console.log(`Calling ${fn.name} with`, args);
    const result = fn(...args);
    console.log(`Result:`, result);
    return result;
  };
}

const loggedAdd = withLogging(function add(a, b) { return a + b; });
loggedAdd(3, 4); // Logs: "Calling add with [3, 4]" → "Result: 7"

Building Custom HOFs

// Custom map implementation
function myMap(arr, callback) {
  const result = [];
  for (let i = 0; i < arr.length; i++) {
    result.push(callback(arr[i], i, arr));
  }
  return result;
}

// Custom filter implementation
function myFilter(arr, callback) {
  const result = [];
  for (let i = 0; i < arr.length; i++) {
    if (callback(arr[i], i, arr)) {
      result.push(arr[i]);
    }
  }
  return result;
}

// Custom reduce implementation
function myReduce(arr, callback, initialValue) {
  let acc = initialValue !== undefined ? initialValue : arr[0];
  let startIdx = initialValue !== undefined ? 0 : 1;
  for (let i = startIdx; i < arr.length; i++) {
    acc = callback(acc, arr[i], i, arr);
  }
  return acc;
}

8. The 'this' Keyword (All Rules)

📺 YouTube Tutorials
  • this keyword — Namaste JavaScript (Akshay Saini)
  • this Explained — Fireship
  • this keyword — Programming with Mosh
  • this in JS — Fun Fun Function

Theory

this is NOT the function itself. It's NOT the function's scope. It's a runtime binding determined by how the function is called (call-site), not where it's defined (except arrow functions).

The 4 Rules of 'this' (in priority order)

// RULE 1: new Binding (highest priority)
// When function called with 'new', this = newly created object
function Person(name) {
  // this = {} (new empty object)
  this.name = name;
  // return this (implicit)
}
const p = new Person("Sid"); // this = p

// RULE 2: Explicit Binding (call, apply, bind)
function greet() {
  console.log(`Hello, ${this.name}`);
}
const user = { name: "Sid" };
greet.call(user);   // "Hello, Sid" — this = user
greet.apply(user);  // "Hello, Sid" — this = user
const bound = greet.bind(user);
bound();            // "Hello, Sid" — this = user (permanently)

// RULE 3: Implicit Binding
// When function called as object method, this = the object
const obj = {
  name: "Sid",
  greet() {
    console.log(this.name); // "Sid" — this = obj
  }
};
obj.greet(); // Implicit: obj is the call-site context

// Implicit binding LOST:
const fn = obj.greet; // Extracting method
fn(); // undefined — no object context, falls to default rule!

// RULE 4: Default Binding (lowest priority)
// Standalone function call: this = window (non-strict) or undefined (strict)
function standalone() {
  console.log(this); // window (non-strict) / undefined (strict)
}
standalone();

Arrow Function Exception

// Arrow functions IGNORE all 4 rules above
// They inherit 'this' from enclosing LEXICAL scope
const obj = {
  name: "Sid",
  regular() {
    console.log(this.name); // "Sid" (implicit binding)
    
    const arrow = () => {
      console.log(this.name); // "Sid" (inherited from regular's this)
    };
    arrow();
    
    function nested() {
      console.log(this.name); // undefined! (default binding)
    }
    nested();
  }
};

// Arrow's this CANNOT be overridden
const arrowFn = () => console.log(this);
arrowFn.call({ x: 1 });  // still window/global (call has no effect!)
arrowFn.bind({ x: 1 })(); // still window/global (bind has no effect!)

'this' in Different Contexts

// Global context
console.log(this); // window (browser) / {} (Node module)

// Inside object method
const car = {
  brand: "Tesla",
  getBrand() { return this.brand; }  // "Tesla"
};

// Inside constructor
class Dog {
  constructor(name) {
    this.name = name; // this = new instance
  }
}

// Inside event handler (DOM)
button.addEventListener('click', function() {
  console.log(this); // the button element (implicit binding)
});
button.addEventListener('click', () => {
  console.log(this); // window! (arrow inherits from enclosing scope)
});

// Inside setTimeout
const obj2 = {
  name: "Timer",
  start() {
    setTimeout(function() {
      console.log(this.name); // undefined (default binding)
    }, 100);
    setTimeout(() => {
      console.log(this.name); // "Timer" (arrow inherits)
    }, 100);
  }
};

// Inside class methods (Angular components)
class MyComponent {
  name = "Component";
  
  handleClick = () => {
    // Arrow property: 'this' always = instance (safe for callbacks)
    console.log(this.name);
  };
  
  handleEvent() {
    // Regular method: 'this' depends on call-site
    console.log(this.name); // may be undefined if called as callback
  }
}

🎯 Interview Questions

Hard Explain all 4 rules of 'this' binding with examples.

Answer: Priority: new > explicit > implicit > default. (1) new: this = new object, (2) call/apply/bind: this = specified object, (3) obj.method(): this = obj, (4) standalone(): this = window/undefined. Arrow functions bypass all rules — they lexically inherit this.

Expert How does 'this' work in Angular components? Why do arrow function class properties solve callback issues?

Answer: In Angular, component methods use implicit binding (this = component instance when called as this.method()). But when passing methods as callbacks (to addEventListener, subscribe, setTimeout), implicit binding is lost. Arrow function class properties (handleClick = () => {}) solve this because they capture 'this' at instantiation time (in constructor), so they always refer to the component instance regardless of how they're called.

9. call(), apply(), bind()

📺 YouTube Tutorials
  • call, apply, bind — Namaste JavaScript (Akshay Saini)
  • call & apply — Fun Fun Function
  • bind Explained — Web Dev Simplified

Theory

// All three set 'this' explicitly
function introduce(greeting, punctuation) {
  return `${greeting}, I'm ${this.name}${punctuation}`;
}
const person = { name: "Sid" };

// call — invokes immediately, args passed individually
introduce.call(person, "Hello", "!");    // "Hello, I'm Sid!"

// apply — invokes immediately, args passed as array
introduce.apply(person, ["Hi", "."]);    // "Hi, I'm Sid."

// bind — returns NEW function with 'this' permanently set
const sidIntro = introduce.bind(person);
sidIntro("Hey", "!");                    // "Hey, I'm Sid!"
// bind can also partially apply arguments:
const sidHello = introduce.bind(person, "Hello");
sidHello("...");                         // "Hello, I'm Sid..."

Practical Use Cases

// 1. Borrowing methods
const nums = { 0: "a", 1: "b", 2: "c", length: 3 };
// Array-like object — can't use array methods directly
const arr = Array.prototype.slice.call(nums); // ["a", "b", "c"]
// Modern: Array.from(nums)

// 2. Finding max/min of array
const values = [5, 2, 8, 1, 9];
Math.max.apply(null, values); // 9
// Modern: Math.max(...values)

// 3. Preserving context in callbacks
class Timer {
  constructor() {
    this.seconds = 0;
  }
  start() {
    // Without bind: 'this' would be lost in setInterval
    setInterval(this.tick.bind(this), 1000);
    // Or use arrow: setInterval(() => this.tick(), 1000);
  }
  tick() {
    this.seconds++;
    console.log(this.seconds);
  }
}

// 4. Function currying with bind
function multiply(a, b) { return a * b; }
const double = multiply.bind(null, 2); // first arg fixed to 2
double(5);  // 10
double(10); // 20

Implementing call, apply, bind (Polyfills)

// Polyfill for call
Function.prototype.myCall = function(context, ...args) {
  context = context || globalThis;
  const sym = Symbol(); // unique key to avoid conflicts
  context[sym] = this;  // attach function to context
  const result = context[sym](...args);
  delete context[sym];
  return result;
};

// Polyfill for apply
Function.prototype.myApply = function(context, args = []) {
  context = context || globalThis;
  const sym = Symbol();
  context[sym] = this;
  const result = context[sym](...args);
  delete context[sym];
  return result;
};

// Polyfill for bind
Function.prototype.myBind = function(context, ...boundArgs) {
  const fn = this;
  return function(...callArgs) {
    return fn.apply(context, [...boundArgs, ...callArgs]);
  };
};

🎯 Interview Questions

Medium What is the difference between call, apply, and bind?

Answer: call/apply invoke immediately; bind returns a new function. call passes args individually (fn.call(obj, a, b)), apply passes as array (fn.apply(obj, [a, b])), bind returns function with fixed this and optionally pre-filled args. Mnemonic: call = commas, apply = array.

Hard Implement Function.prototype.bind polyfill.

Answer: Function.prototype.myBind = function(ctx, ...bound) { const fn = this; return function(...args) { return fn.apply(ctx, [...bound, ...args]); }; }; Key points: returns new function, merges bound + call-time args, uses apply to set context.

10. Currying & Partial Application

📺 YouTube Tutorials
  • Currying in JavaScript — Fun Fun Function
  • Currying — Akshay Saini
  • Partial Application — Fireship

Theory

Currying: Transforming a function with multiple arguments into a sequence of functions each taking a single argument. f(a, b, c) → f(a)(b)(c)

Partial Application: Fixing a number of arguments to a function, producing a function with fewer arguments. f(a, b, c) with a fixed → g(b, c)

// CURRYING
// Non-curried
function add(a, b, c) { return a + b + c; }
add(1, 2, 3); // 6

// Curried version
function curriedAdd(a) {
  return function(b) {
    return function(c) {
      return a + b + c;
    };
  };
}
curriedAdd(1)(2)(3); // 6

// Arrow syntax (cleaner)
const curriedAdd2 = a => b => c => a + b + c;
curriedAdd2(1)(2)(3); // 6

// Reusable partially applied functions
const add5 = curriedAdd(5);
const add5and3 = add5(3);
add5and3(2); // 10

// PRACTICAL CURRYING EXAMPLES
// 1. Logger
const log = (level) => (module) => (message) =>
  console.log(`[${level}] [${module}] ${message}`);

const errorLog = log("ERROR");
const authError = errorLog("Auth");
authError("Invalid token"); // "[ERROR] [Auth] Invalid token"

// 2. API URL builder
const apiUrl = (base) => (version) => (endpoint) =>
  `${base}/api/${version}/${endpoint}`;

const myApi = apiUrl("https://example.com")("v2");
myApi("users");  // "https://example.com/api/v2/users"
myApi("posts");  // "https://example.com/api/v2/posts"

// 3. Event handler (Angular-relevant)
const handleEvent = (eventType) => (handler) => (element) => {
  element.addEventListener(eventType, handler);
  return () => element.removeEventListener(eventType, handler);
};
const onClick = handleEvent("click");
const onClickLog = onClick(() => console.log("clicked"));

Generic Curry Utility

// Converts any function into curried form
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn(...args);  // All args received — execute
    }
    return function(...more) {
      return curried(...args, ...more); // Collect more args
    };
  };
}

// Usage:
const multiply = (a, b, c) => a * b * c;
const curriedMultiply = curry(multiply);

curriedMultiply(2)(3)(4);    // 24
curriedMultiply(2, 3)(4);    // 24 (also works with multiple args)
curriedMultiply(2)(3, 4);    // 24

// Infinite currying (interview favorite)
function infiniteSum(a) {
  return function(b) {
    if (b === undefined) return a;
    return infiniteSum(a + b);
  };
}
infiniteSum(1)(2)(3)(4)(); // 10

11. Recursion & Tail Call Optimization

📺 YouTube Tutorials
  • Recursion in JS — Web Dev Simplified
  • Recursion — freeCodeCamp
  • Tail Call Optimization — Fun Fun Function

Theory

// Recursion: function that calls itself with a base case
function factorial(n) {
  if (n <= 1) return 1;     // Base case (stops recursion)
  return n * factorial(n - 1); // Recursive case
}
factorial(5); // 5 * 4 * 3 * 2 * 1 = 120

// Call stack for factorial(4):
// factorial(4) waiting for factorial(3) × 4
//   factorial(3) waiting for factorial(2) × 3
//     factorial(2) waiting for factorial(1) × 2
//       factorial(1) → returns 1 (base case)
//     factorial(2) → 1 × 2 = 2
//   factorial(3) → 2 × 3 = 6
// factorial(4) → 6 × 4 = 24

// Common recursive patterns:
// 1. Fibonacci
function fib(n) {
  if (n <= 1) return n;
  return fib(n - 1) + fib(n - 2);
}
// ⚠️ O(2^n) — exponential! Use memoization or iteration

// 2. Deep flatten array
function flatten(arr) {
  return arr.reduce((flat, item) => {
    return flat.concat(Array.isArray(item) ? flatten(item) : item);
  }, []);
}
flatten([1, [2, [3, [4]]]]); // [1, 2, 3, 4]

// 3. Deep clone object
function deepClone(obj) {
  if (obj === null || typeof obj !== "object") return obj;
  if (obj instanceof Date) return new Date(obj);
  if (obj instanceof RegExp) return new RegExp(obj);
  
  const clone = Array.isArray(obj) ? [] : {};
  for (const key in obj) {
    if (obj.hasOwnProperty(key)) {
      clone[key] = deepClone(obj[key]);
    }
  }
  return clone;
}

// 4. Tree traversal (DOM, file systems)
function walkDOM(node, callback) {
  callback(node);
  node = node.firstElementChild;
  while (node) {
    walkDOM(node, callback);
    node = node.nextElementSibling;
  }
}

// 5. Power function
function power(base, exp) {
  if (exp === 0) return 1;
  if (exp < 0) return 1 / power(base, -exp);
  return base * power(base, exp - 1);
}

Tail Call Optimization (TCO)

// TCO: when recursive call is the LAST operation, engine can reuse stack frame
// Only supported in Safari (Strict mode). V8/SpiderMonkey do NOT support TCO.

// NOT tail-recursive (must multiply AFTER the call returns)
function factorial(n) {
  if (n <= 1) return 1;
  return n * factorial(n - 1); // has pending operation (multiply)
}

// TAIL-recursive version (accumulator pattern)
function factorialTCO(n, acc = 1) {
  if (n <= 1) return acc;
  return factorialTCO(n - 1, n * acc); // last operation IS the call
}

// Converting to iterative (manual TCO — works everywhere)
function factorialIterative(n) {
  let result = 1;
  for (let i = 2; i <= n; i++) {
    result *= i;
  }
  return result;
}

// Trampoline pattern (simulate TCO in any engine)
function trampoline(fn) {
  return function(...args) {
    let result = fn(...args);
    while (typeof result === "function") {
      result = result();
    }
    return result;
  };
}

const trampolinedFactorial = trampoline(function fact(n, acc = 1) {
  if (n <= 1) return acc;
  return () => fact(n - 1, n * acc); // return function, not call
});
trampolinedFactorial(100000); // Won't stack overflow!
