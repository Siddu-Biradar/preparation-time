// insertAdjacentHTML — FAST, no serialize/reparse of existing DOM

Before
");
element.insertAdjacentHTML("afterbegin", "
First child
");

Last child
");

After
");

// ─── PERFORMANCE ───────────────────────────────────────────

// Problem: Each DOM write triggers potential reflow/repaint
// Solution 1: DocumentFragment (batch inserts)
const fragment = document.createDocumentFragment();
for (let i = 0; i < 1000; i++) {
  const li = document.createElement("li");
  li.textContent = `Item ${i}`;
  fragment.appendChild(li); // Append to fragment (no reflows!)
}
document.querySelector("ul").appendChild(fragment); // ONE reflow

// Solution 2: Clone and replace
const list = document.querySelector("ul");
const clone = list.cloneNode(false); // Shallow clone (empty)
for (let i = 0; i < 1000; i++) {
  const li = document.createElement("li");
  li.textContent = `Item ${i}`;
  clone.appendChild(li);
}
list.replaceWith(clone); // ONE operation

// Solution 3: Batch reads and writes separately
// BAD (thrashing: read → write → read → write → ...)
elements.forEach(el => {
  const height = el.offsetHeight; // READ (forces layout)
  el.style.height = height + 10 + "px"; // WRITE (invalidates layout)
});
// GOOD (batch reads, then batch writes)
const heights = elements.map(el => el.offsetHeight); // All reads
elements.forEach((el, i) => {
  el.style.height = heights[i] + 10 + "px"; // All writes
});
XSS Warning: NEVER use innerHTML with user-provided data. Use textContent or sanitize with DOMPurify. In Angular, the framework auto-sanitizes bindings — but direct DOM manipulation via ElementRef bypasses it.

4. Event Model (Capturing, Bubbling, Delegation)

📺 YouTube Tutorials
  • Event Bubbling & Capturing — Web Dev Simplified
  • Event Delegation — Akshay Saini
  • JavaScript Events Deep Dive — Traversy Media

Theory

/*
Event Propagation (3 phases):

Phase 1: CAPTURING (top → target)
window → document → html → body → div → button

Phase 2: TARGET
button (the element that was clicked)

Phase 3: BUBBLING (target → top)
button → div → body → html → document → window
*/

// addEventListener with capture option:
element.addEventListener("click", handler, { capture: true }); // Capture phase
element.addEventListener("click", handler, { capture: false }); // Bubble phase (default)
element.addEventListener("click", handler, true); // Shorthand for capture: true

// Event object properties:
element.addEventListener("click", (e) => {
  e.target;          // Element that triggered the event (deepest)
  e.currentTarget;   // Element the listener is attached to (=this)
  e.type;            // "click"
  e.timeStamp;       // When event occurred
  e.bubbles;         // Does this event bubble?
  e.eventPhase;      // 1=capturing, 2=target, 3=bubbling
  
  e.stopPropagation();      // Stop propagation (current handler runs)
  e.stopImmediatePropagation(); // Stop ALL handlers on this element too
  e.preventDefault();       // Prevent default browser behavior
});

// ─── EVENT DELEGATION ─────────────────────────────────────
// Instead of adding listeners to EVERY child, listen on parent

// ❌ Bad: 1000 listeners for 1000 items
document.querySelectorAll("li").forEach(li => {
  li.addEventListener("click", handleClick);
});

// ✅ Good: 1 listener for unlimited items
document.querySelector("ul").addEventListener("click", (e) => {
  const li = e.target.closest("li"); // Find the li (handles nested elements)
  if (!li) return; // Click wasn't on an li
  console.log("Clicked:", li.textContent);
});

// Delegation advantages:
// 1. Works for dynamically added elements
// 2. Uses less memory (1 handler vs N)
// 3. Less setup/teardown code

// Angular equivalent: (click) binding handles this via zones
// But for Renderer2 or native DOM in directives, delegation is key

// Common events and best practices:
// Mouse: click, dblclick, mouseenter (no bubble), mouseleave, mouseover (bubbles)
// Keyboard: keydown (preferred), keyup, keypress (deprecated)
// Form: submit, input (real-time), change (on blur), focus, blur
// focus/blur DON'T bubble — use focusin/focusout instead!

// Passive listeners (performance for scroll/touch):
window.addEventListener("scroll", handler, { passive: true });
// Tells browser: "I won't call preventDefault()" → enables optimizations

🎯 Interview Questions

Hard Explain event bubbling vs capturing. When would you use capture phase?

Answer: Events propagate in 3 phases: capture (window→target), target, bubble (target→window). Use capture for: intercepting events BEFORE they reach children (e.g., analytics, global error catching), handling focus/blur events that don't bubble, or stopping propagation before delegation handlers fire.

Medium What's the difference between e.target and e.currentTarget?

Answer: e.target is the element that ORIGINATED the event (deepest element clicked). e.currentTarget is the element the handler is ATTACHED to. In delegation, parent has the handler (currentTarget=parent), but the click was on a child (target=child). Inside the handler, this === e.currentTarget (unless using arrow functions).

5. Custom Events & Event Emitter

📺 YouTube Tutorials
  • Custom Events — Web Dev Simplified
  • Event Emitter Pattern — Academind

Theory

// Custom DOM Events
const event = new CustomEvent("user:login", {
  detail: { userId: 1, name: "Sid" },  // Custom data
  bubbles: true,                        // Can bubble up
  cancelable: true                      // Can be prevented
});

// Dispatch (fire) the event on any DOM element
document.dispatchEvent(event);

// Listen for it
document.addEventListener("user:login", (e) => {
  console.log("User logged in:", e.detail.name);
});

// Event Emitter (pub/sub pattern — framework independent)
class EventEmitter {
  #listeners = new Map();
  
  on(event, callback) {
    if (!this.#listeners.has(event)) {
      this.#listeners.set(event, new Set());
    }
    this.#listeners.get(event).add(callback);
    return () => this.off(event, callback); // Return unsubscribe fn
  }
  
  off(event, callback) {
    this.#listeners.get(event)?.delete(callback);
  }
  
  emit(event, ...args) {
    this.#listeners.get(event)?.forEach(cb => cb(...args));
  }
  
  once(event, callback) {
    const wrapper = (...args) => {
      callback(...args);
      this.off(event, wrapper);
    };
    this.on(event, wrapper);
  }
}

// Usage:
const bus = new EventEmitter();
const unsub = bus.on("data", (payload) => console.log(payload));
bus.emit("data", { id: 1 }); // Logs: { id: 1 }
unsub(); // Unsubscribe

// Angular comparison:
// - @Output() EventEmitter — same concept for component communication
// - RxJS Subject — more powerful (async, operators, backpressure)

7. IntersectionObserver & MutationObserver

📺 YouTube Tutorials

Theory

// IntersectionObserver — detect when elements enter/leave viewport
// Use cases: Lazy loading, infinite scroll, analytics (impressions), animations

const observer = new IntersectionObserver((entries) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      // Element is visible!
      entry.target.classList.add("visible");
      
      // Lazy load image:
      const img = entry.target;
      img.src = img.dataset.src;
      observer.unobserve(img); // Stop watching (one-time)
    }
  });
}, {
  root: null,           // Viewport (or specific scrollable element)
  rootMargin: "100px",  // Start loading 100px before visible
  threshold: [0, 0.5, 1] // Trigger at 0%, 50%, 100% visibility
});

// Observe elements
document.querySelectorAll("img[data-src]").forEach(img => {
  observer.observe(img);
});

// Infinite scroll
const sentinel = document.querySelector("#load-more-trigger");
const scrollObserver = new IntersectionObserver((entries) => {
  if (entries[0].isIntersecting) {
    loadMoreItems(); // Fetch next page
  }
});
scrollObserver.observe(sentinel);

// ─── MutationObserver ─────────────────────────────────────
// Watch for DOM changes (attributes, children, text)
const mutationObserver = new MutationObserver((mutations) => {
  mutations.forEach(mutation => {
    switch (mutation.type) {
      case "childList":
        console.log("Children added:", mutation.addedNodes);
        console.log("Children removed:", mutation.removedNodes);
        break;
      case "attributes":
        console.log(`${mutation.attributeName} changed`);
        break;
      case "characterData":
        console.log("Text changed:", mutation.target.data);
        break;
    }
  });
});

mutationObserver.observe(document.querySelector("#app"), {
  childList: true,      // Watch child additions/removals
  attributes: true,     // Watch attribute changes
  characterData: true,  // Watch text content changes
  subtree: true,        // Watch entire subtree (not just direct children)
  attributeFilter: ["class", "data-state"], // Only specific attributes
});

// Stop watching:
mutationObserver.disconnect();

// Angular uses MutationObserver internally for:
// - ContentChildren/ViewChildren change detection
// - Zone.js patches it to trigger change detection

10. Web Workers & Service Workers

📺 YouTube Tutorials
  • Web Workers — Fireship
  • Service Workers — Google Chrome Developers
  • Web Workers Crash Course — Traversy Media

Theory

// Web Workers — run JS in a background thread (heavy computation)
// worker.js
self.onmessage = (e) => {
  const result = heavyComputation(e.data);
  self.postMessage(result); // Send back to main thread
};

// main.js
const worker = new Worker("worker.js");
worker.postMessage({ data: largeArray }); // Send to worker
worker.onmessage = (e) => {
  console.log("Result:", e.data); // Receive from worker
};
worker.terminate(); // Kill the worker

// Worker limitations:
// - NO access to DOM (window, document)
// - NO access to parent's variables (separate context)
// - Communication only via postMessage (structured clone)
// - Can use fetch, IndexedDB, WebSockets

// Transferable objects (zero-copy transfer, not clone)
const buffer = new ArrayBuffer(1024 * 1024); // 1MB
worker.postMessage(buffer, [buffer]); // Transfer ownership
// buffer.byteLength === 0 now! (transferred, not copied)

// SharedArrayBuffer + Atomics (shared memory between threads)
const shared = new SharedArrayBuffer(1024);
const view = new Int32Array(shared);
Atomics.store(view, 0, 42);
Atomics.load(view, 0); // 42

// ─── Service Workers ──────────────────────────────────────
// Proxy between browser and network — enables offline, caching, push

// Register
if ("serviceWorker" in navigator) {
  navigator.serviceWorker.register("/sw.js")
    .then(reg => console.log("SW registered:", reg.scope));
}

// sw.js — Cache-first strategy
self.addEventListener("install", (e) => {
  e.waitUntil(
    caches.open("v1").then(cache => 
      cache.addAll(["/", "/index.html", "/style.css", "/app.js"])
    )
  );
});

self.addEventListener("fetch", (e) => {
  e.respondWith(
    caches.match(e.request).then(cached => {
      return cached || fetch(e.request); // Cache first, then network
    })
  );
});

// Angular: @angular/service-worker (ngsw) handles all this
// ng add @angular/pwa — sets up SW with configurable caching strategies

🎯 Interview Questions

Hard What's the difference between Web Workers and Service Workers?

Answer: Web Workers: background thread for CPU-heavy tasks, created per page, dies with the page, communicate via postMessage. Service Workers: network proxy, runs independent of page, survives page close, intercepts fetch requests, enables offline/caching/push notifications, lifecycle-based (install→activate→fetch). Angular's @angular/pwa uses Service Workers for PWA support.
