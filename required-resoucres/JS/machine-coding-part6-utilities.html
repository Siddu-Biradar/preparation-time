class DataTable {
  constructor(container, options = {}) {
    this.container = container;
    this.columns = options.columns || []; // [{ key, label, sortable?, filterable?, width? }]
    this.data = options.data || [];
    this.pageSize = options.pageSize || 10;
    this.onRowClick = options.onRowClick || null;
    this.selectable = options.selectable || false;

    this.currentPage = 1;
    this.sortKey = null;
    this.sortDir = "asc"; // asc | desc | null
    this.filters = {}; // { columnKey: filterValue }
    this.searchTerm = "";
    this.selectedRows = new Set();

    this.render();
  }

  getProcessedData() {
    let result = [...this.data];

    // Global search
    if (this.searchTerm) {
      const term = this.searchTerm.toLowerCase();
      result = result.filter((row) =>
        this.columns.some((col) =>
          String(row[col.key] || "").toLowerCase().includes(term)
        )
      );
    }

    // Column filters
    Object.entries(this.filters).forEach(([key, value]) => {
      if (!value) return;
      result = result.filter((row) =>
        String(row[key] || "").toLowerCase().includes(value.toLowerCase())
      );
    });

    // Sort
    if (this.sortKey) {
      result.sort((a, b) => {
        const aVal = a[this.sortKey] ?? "";
        const bVal = b[this.sortKey] ?? "";
        let cmp = 0;
        if (typeof aVal === "number" && typeof bVal === "number") {
          cmp = aVal - bVal;
        } else {
          cmp = String(aVal).localeCompare(String(bVal));
        }
        return this.sortDir === "desc" ? -cmp : cmp;
      });
    }

    return result;
  }

  getPageData() {
    const processed = this.getProcessedData();
    const start = (this.currentPage - 1) * this.pageSize;
    return {
      rows: processed.slice(start, start + this.pageSize),
      total: processed.length,
      totalPages: Math.ceil(processed.length / this.pageSize),
    };
  }

  render() {
    this.container.innerHTML = "";
    this.container.style.cssText = "font-family:sans-serif;font-size:13px";

    const { rows, total, totalPages } = this.getPageData();

    // Toolbar (search + page size)
    const toolbar = document.createElement("div");
    toolbar.style.cssText = "display:flex;justify-content:space-between;align-items:center;margin-bottom:12px;gap:12px;flex-wrap:wrap";

    const searchInput = document.createElement("input");
    searchInput.type = "text";
    searchInput.placeholder = "Search all columns...";
    searchInput.value = this.searchTerm;
    searchInput.style.cssText = "padding:8px 12px;border:1px solid #d1d5db;border-radius:6px;font-size:13px;width:250px;outline:none";
    searchInput.addEventListener("input", (e) => {
      this.searchTerm = e.target.value;
      this.currentPage = 1;
      this.render();
    });
    toolbar.appendChild(searchInput);

    const info = document.createElement("span");
    info.textContent = `${total} results | Page ${this.currentPage} of ${totalPages || 1}`;
    info.style.cssText = "color:#6b7280;font-size:12px";
    toolbar.appendChild(info);

    this.container.appendChild(toolbar);

    // Table
    const tableWrapper = document.createElement("div");
    tableWrapper.style.cssText = "overflow-x:auto;border:1px solid #e5e7eb;border-radius:8px";

    const table = document.createElement("table");
    table.style.cssText = "width:100%;border-collapse:collapse;font-size:13px";

    // Header
    const thead = document.createElement("thead");
    const headerRow = document.createElement("tr");

    if (this.selectable) {
      const th = document.createElement("th");
      th.style.cssText = "padding:10px;background:#f9fafb;border-bottom:2px solid #e5e7eb;width:40px";
      const selectAll = document.createElement("input");
      selectAll.type = "checkbox";
      selectAll.checked = rows.length > 0 && rows.every((r) => this.selectedRows.has(r.id));
      selectAll.addEventListener("change", () => {
        if (selectAll.checked) rows.forEach((r) => this.selectedRows.add(r.id));
        else rows.forEach((r) => this.selectedRows.delete(r.id));
        this.render();
      });
      th.appendChild(selectAll);
      headerRow.appendChild(th);
    }

    this.columns.forEach((col) => {
      const th = document.createElement("th");
      th.style.cssText = `
        padding:10px 12px;background:#f9fafb;border-bottom:2px solid #e5e7eb;
        text-align:left;font-weight:600;white-space:nowrap;
        ${col.width ? `width:${col.width}` : ""};
        ${col.sortable !== false ? "cursor:pointer;user-select:none" : ""};
      `;

      const labelWrap = document.createElement("div");
      labelWrap.style.cssText = "display:flex;align-items:center;gap:4px";
      const label = document.createElement("span");
      label.textContent = col.label;
      labelWrap.appendChild(label);

      // Sort indicator
      if (col.sortable !== false) {
        const arrow = document.createElement("span");
        arrow.style.cssText = "font-size:10px;color:#9ca3af";
        if (this.sortKey === col.key) {
          arrow.textContent = this.sortDir === "asc" ? " \u25b2" : " \u25bc";
          arrow.style.color = "#2563eb";
        } else {
          arrow.textContent = " \u2195";
        }
        labelWrap.appendChild(arrow);

        th.addEventListener("click", () => {
          if (this.sortKey === col.key) {
            this.sortDir = this.sortDir === "asc" ? "desc" : this.sortDir === "desc" ? null : "asc";
            if (!this.sortDir) this.sortKey = null;
          } else {
            this.sortKey = col.key;
            this.sortDir = "asc";
          }
          this.render();
        });
      }

      th.appendChild(labelWrap);

      // Column filter
      if (col.filterable !== false) {
        const filterInput = document.createElement("input");
        filterInput.type = "text";
        filterInput.placeholder = "Filter...";
        filterInput.value = this.filters[col.key] || "";
        filterInput.style.cssText = "width:100%;padding:4px 6px;border:1px solid #e5e7eb;border-radius:3px;font-size:11px;margin-top:4px;outline:none";
        filterInput.addEventListener("click", (e) => e.stopPropagation()); // Don't trigger sort
        filterInput.addEventListener("input", (e) => {
          this.filters[col.key] = e.target.value;
          this.currentPage = 1;
          this.render();
        });
        th.appendChild(filterInput);
      }

      headerRow.appendChild(th);
    });
    thead.appendChild(headerRow);
    table.appendChild(thead);

    // Body
    const tbody = document.createElement("tbody");
    if (rows.length === 0) {
      const tr = document.createElement("tr");
      const td = document.createElement("td");
      td.colSpan = this.columns.length + (this.selectable ? 1 : 0);
      td.textContent = "No data found";
      td.style.cssText = "padding:24px;text-align:center;color:#9ca3af";
      tr.appendChild(td);
      tbody.appendChild(tr);
    } else {
      rows.forEach((row) => {
        const tr = document.createElement("tr");
        tr.style.cssText = `
          border-bottom:1px solid #f3f4f6;transition:background 0.1s;
          ${this.selectedRows.has(row.id) ? "background:#eff6ff" : ""}
        `;
        tr.addEventListener("mouseenter", () => { if (!this.selectedRows.has(row.id)) tr.style.background = "#f9fafb"; });
        tr.addEventListener("mouseleave", () => { if (!this.selectedRows.has(row.id)) tr.style.background = ""; });

        if (this.onRowClick) {
          tr.style.cursor = "pointer";
          tr.addEventListener("click", () => this.onRowClick(row));
        }

        if (this.selectable) {
          const td = document.createElement("td");
          td.style.cssText = "padding:10px;text-align:center";
          const cb = document.createElement("input");
          cb.type = "checkbox";
          cb.checked = this.selectedRows.has(row.id);
          cb.addEventListener("click", (e) => e.stopPropagation());
          cb.addEventListener("change", () => {
            if (cb.checked) this.selectedRows.add(row.id);
            else this.selectedRows.delete(row.id);
            this.render();
          });
          td.appendChild(cb);
          tr.appendChild(td);
        }

        this.columns.forEach((col) => {
          const td = document.createElement("td");
          td.style.cssText = "padding:10px 12px";
          td.textContent = row[col.key] ?? "";
          tr.appendChild(td);
        });
        tbody.appendChild(tr);
      });
    }
    table.appendChild(tbody);
    tableWrapper.appendChild(table);
    this.container.appendChild(tableWrapper);

    // Pagination
    if (totalPages > 1) {
      const pager = document.createElement("div");
      pager.style.cssText = "display:flex;justify-content:center;gap:4px;margin-top:12px";

      const prevBtn = document.createElement("button");
      prevBtn.textContent = "\u2190";
      prevBtn.disabled = this.currentPage === 1;
      prevBtn.style.cssText = "padding:6px 10px;border:1px solid #d1d5db;border-radius:4px;background:white;cursor:pointer";
      prevBtn.addEventListener("click", () => { this.currentPage--; this.render(); });
      pager.appendChild(prevBtn);

      for (let i = 1; i <= Math.min(totalPages, 7); i++) {
        const btn = document.createElement("button");
        btn.textContent = i;
        btn.style.cssText = `
          padding:6px 10px;border:1px solid ${i === this.currentPage ? "#2563eb" : "#d1d5db"};
          border-radius:4px;cursor:pointer;font-weight:${i === this.currentPage ? "700" : "400"};
          background:${i === this.currentPage ? "#2563eb" : "white"};
          color:${i === this.currentPage ? "white" : "#333"};
        `;
        btn.addEventListener("click", () => { this.currentPage = i; this.render(); });
        pager.appendChild(btn);
      }

      const nextBtn = document.createElement("button");
      nextBtn.textContent = "\u2192";
      nextBtn.disabled = this.currentPage === totalPages;
      nextBtn.style.cssText = "padding:6px 10px;border:1px solid #d1d5db;border-radius:4px;background:white;cursor:pointer";
      nextBtn.addEventListener("click", () => { this.currentPage++; this.render(); });
      pager.appendChild(nextBtn);

      this.container.appendChild(pager);
    }

    // Selected count
    if (this.selectable && this.selectedRows.size > 0) {
      const selection = document.createElement("div");
      selection.textContent = `${this.selectedRows.size} row(s) selected`;
      selection.style.cssText = "margin-top:8px;font-size:12px;color:#2563eb;font-weight:500";
      this.container.appendChild(selection);
    }
  }

  getSelected() { return [...this.selectedRows]; }
  clearSelection() { this.selectedRows.clear(); this.render(); }
  setData(data) { this.data = data; this.currentPage = 1; this.render(); }
}

class TransferList {
  constructor(container, options = {}) {
    this.container = container;
    this.leftTitle = options.leftTitle || "Available";
    this.rightTitle = options.rightTitle || "Selected";
    this.leftItems = options.leftItems || [];
    this.rightItems = options.rightItems || [];
    this.onChange = options.onChange || (() => {});

    this.leftSelected = new Set();
    this.rightSelected = new Set();
    this.leftSearch = "";
    this.rightSearch = "";

    this.render();
  }

  render() {
    this.container.innerHTML = "";
    this.container.style.cssText = "display:flex;gap:12px;align-items:stretch;font-family:sans-serif";

    // Left panel
    const leftPanel = this.renderPanel("left", this.leftTitle, this.leftItems, this.leftSelected, this.leftSearch);
    this.container.appendChild(leftPanel);

    // Transfer buttons
    const btnCol = document.createElement("div");
    btnCol.style.cssText = "display:flex;flex-direction:column;justify-content:center;gap:8px";

    const moveRight = this.createBtn("\u25b6", "Move selected right", () => this.transferRight());
    const moveAllRight = this.createBtn("\u25b6\u25b6", "Move all right", () => this.transferAllRight());
    const moveLeft = this.createBtn("\u25c0", "Move selected left", () => this.transferLeft());
    const moveAllLeft = this.createBtn("\u25c0\u25c0", "Move all left", () => this.transferAllLeft());

    moveRight.disabled = this.leftSelected.size === 0;
    moveLeft.disabled = this.rightSelected.size === 0;

    btnCol.appendChild(moveAllRight);
    btnCol.appendChild(moveRight);
    btnCol.appendChild(moveLeft);
    btnCol.appendChild(moveAllLeft);
    this.container.appendChild(btnCol);

    // Right panel
    const rightPanel = this.renderPanel("right", this.rightTitle, this.rightItems, this.rightSelected, this.rightSearch);
    this.container.appendChild(rightPanel);
  }

  renderPanel(side, title, items, selected, searchTerm) {
    const panel = document.createElement("div");
    panel.style.cssText = "flex:1;border:1px solid #d1d5db;border-radius:8px;overflow:hidden;min-width:200px";

    // Header with count
    const header = document.createElement("div");
    header.style.cssText = "padding:10px 12px;background:#f9fafb;border-bottom:1px solid #e5e7eb;display:flex;justify-content:space-between;align-items:center";

    const titleEl = document.createElement("span");
    titleEl.textContent = title;
    titleEl.style.cssText = "font-weight:600;font-size:13px";
    header.appendChild(titleEl);

    const count = document.createElement("span");
    count.textContent = `${selected.size}/${items.length}`;
    count.style.cssText = "font-size:11px;color:#6b7280;background:#e5e7eb;padding:2px 6px;border-radius:10px";
    header.appendChild(count);
    panel.appendChild(header);

    // Search
    const searchWrap = document.createElement("div");
    searchWrap.style.cssText = "padding:8px";
    const searchInput = document.createElement("input");
    searchInput.type = "text";
    searchInput.placeholder = "Search...";
    searchInput.value = searchTerm;
    searchInput.style.cssText = "width:100%;padding:6px 10px;border:1px solid #e5e7eb;border-radius:4px;font-size:12px;outline:none";
    searchInput.addEventListener("input", (e) => {
      if (side === "left") this.leftSearch = e.target.value;
      else this.rightSearch = e.target.value;
      this.render();
    });
    searchWrap.appendChild(searchInput);
    panel.appendChild(searchWrap);

    // Select All
    const selectAllWrap = document.createElement("div");
    selectAllWrap.style.cssText = "padding:4px 12px;border-bottom:1px solid #f3f4f6";
    const selectAllLabel = document.createElement("label");
    selectAllLabel.style.cssText = "display:flex;align-items:center;gap:6px;font-size:12px;cursor:pointer;color:#6b7280";
    const selectAllCb = document.createElement("input");
    selectAllCb.type = "checkbox";
    const filtered = this.getFiltered(items, searchTerm);
    selectAllCb.checked = filtered.length > 0 && filtered.every((item) => selected.has(item.id));
    selectAllCb.indeterminate = filtered.some((item) => selected.has(item.id)) && !selectAllCb.checked;
    selectAllCb.addEventListener("change", () => {
      if (selectAllCb.checked) filtered.forEach((item) => selected.add(item.id));
      else filtered.forEach((item) => selected.delete(item.id));
      this.render();
    });
    selectAllLabel.appendChild(selectAllCb);
    selectAllLabel.appendChild(document.createTextNode("Select All"));
    selectAllWrap.appendChild(selectAllLabel);
    panel.appendChild(selectAllWrap);

    // Items list
    const list = document.createElement("div");
    list.style.cssText = "max-height:250px;overflow-y:auto";

    if (filtered.length === 0) {
      const empty = document.createElement("div");
      empty.textContent = searchTerm ? "No matches" : "Empty";
      empty.style.cssText = "padding:20px;text-align:center;color:#9ca3af;font-size:12px";
      list.appendChild(empty);
    } else {
      filtered.forEach((item) => {
        const row = document.createElement("label");
        row.style.cssText = `
          display:flex;align-items:center;gap:8px;padding:8px 12px;cursor:pointer;
          transition:background 0.1s;font-size:13px;
          background:${selected.has(item.id) ? "#eff6ff" : "transparent"};
        `;
        row.addEventListener("mouseenter", () => { if (!selected.has(item.id)) row.style.background = "#f9fafb"; });
        row.addEventListener("mouseleave", () => { if (!selected.has(item.id)) row.style.background = ""; });

        const cb = document.createElement("input");
        cb.type = "checkbox";
        cb.checked = selected.has(item.id);
        cb.addEventListener("change", () => {
          if (cb.checked) selected.add(item.id);
          else selected.delete(item.id);
          this.render();
        });
        row.appendChild(cb);

        const label = document.createElement("span");
        label.textContent = item.label;
        row.appendChild(label);

        list.appendChild(row);
      });
    }
    panel.appendChild(list);

    return panel;
  }

  createBtn(text, ariaLabel, onClick) {
    const btn = document.createElement("button");
    btn.textContent = text;
    btn.setAttribute("aria-label", ariaLabel);
    btn.style.cssText = "padding:8px 12px;border:1px solid #d1d5db;border-radius:4px;background:white;cursor:pointer;font-size:12px";
    btn.addEventListener("click", onClick);
    return btn;
  }

  getFiltered(items, search) {
    if (!search) return items;
    return items.filter((item) => item.label.toLowerCase().includes(search.toLowerCase()));
  }

  transferRight() {
    const toMove = this.leftItems.filter((item) => this.leftSelected.has(item.id));
    this.leftItems = this.leftItems.filter((item) => !this.leftSelected.has(item.id));
    this.rightItems.push(...toMove);
    this.leftSelected.clear();
    this.onChange({ left: this.leftItems, right: this.rightItems });
    this.render();
  }

  transferLeft() {
    const toMove = this.rightItems.filter((item) => this.rightSelected.has(item.id));
    this.rightItems = this.rightItems.filter((item) => !this.rightSelected.has(item.id));
    this.leftItems.push(...toMove);
    this.rightSelected.clear();
    this.onChange({ left: this.leftItems, right: this.rightItems });
    this.render();
  }

  transferAllRight() {
    this.rightItems.push(...this.leftItems);
    this.leftItems = [];
    this.leftSelected.clear();
    this.onChange({ left: this.leftItems, right: this.rightItems });
    this.render();
  }

  transferAllLeft() {
    this.leftItems.push(...this.rightItems);
    this.rightItems = [];
    this.rightSelected.clear();
    this.onChange({ left: this.leftItems, right: this.rightItems });
    this.render();
  }

  getResult() { return { left: this.leftItems, right: this.rightItems }; }
}

// ─── Basic Memoize ───
function memoize(fn, options = {}) {
  const {
    maxSize = Infinity,      // LRU cache max entries
    ttl = 0,                 // Time-to-live in ms (0 = no expiry)
    keyResolver = (...args) => JSON.stringify(args), // Custom key
  } = options;

  const cache = new Map(); // Map preserves insertion order (useful for LRU)

  function memoized(...args) {
    const key = keyResolver(...args);

    if (cache.has(key)) {
      const entry = cache.get(key);

      // Check TTL
      if (ttl > 0 && Date.now() - entry.timestamp > ttl) {
        cache.delete(key);
      } else {
        // Move to end (most recently used) for LRU
        cache.delete(key);
        cache.set(key, entry);
        return entry.value;
      }
    }

    // Compute
    const value = fn.apply(this, args);

    // Store
    cache.set(key, { value, timestamp: Date.now() });

    // Evict LRU if over maxSize
    if (cache.size > maxSize) {
      const firstKey = cache.keys().next().value;
      cache.delete(firstKey);
    }

    return value;
  }

  // API
  memoized.cache = cache;
  memoized.clear = () => cache.clear();
  memoized.delete = (...args) => cache.delete(keyResolver(...args));
  memoized.has = (...args) => {
    const key = keyResolver(...args);
    if (!cache.has(key)) return false;
    if (ttl > 0 && Date.now() - cache.get(key).timestamp > ttl) {
      cache.delete(key);
      return false;
    }
    return true;
  };

  return memoized;
}

// ─── Async Memoize (for API calls) ───
function memoizeAsync(fn, options = {}) {
  const { maxSize = 100, ttl = 60000, keyResolver = (...args) => JSON.stringify(args) } = options;
  const cache = new Map();
  const pending = new Map(); // Avoid duplicate in-flight requests

  async function memoized(...args) {
    const key = keyResolver(...args);

    // Return cached value
    if (cache.has(key)) {
      const entry = cache.get(key);
      if (ttl > 0 && Date.now() - entry.timestamp > ttl) {
        cache.delete(key);
      } else {
        cache.delete(key);
        cache.set(key, entry); // LRU refresh
        return entry.value;
      }
    }

    // Return pending promise (dedup)
    if (pending.has(key)) {
      return pending.get(key);
    }

    // Execute and cache
    const promise = fn.apply(this, args).then((value) => {
      cache.set(key, { value, timestamp: Date.now() });
      pending.delete(key);

      if (cache.size > maxSize) {
        const firstKey = cache.keys().next().value;
        cache.delete(firstKey);
      }

      return value;
    }).catch((err) => {
      pending.delete(key);
      throw err;
    });

    pending.set(key, promise);
    return promise;
  }

  memoized.cache = cache;
  memoized.clear = () => { cache.clear(); pending.clear(); };
  memoized.invalidate = (...args) => cache.delete(keyResolver(...args));

  return memoized;
}

// ─── WeakMap Memoize (for object arguments — avoids memory leaks) ───
function memoizeWeak(fn) {
  const cache = new WeakMap();

  return function (obj, ...rest) {
    if (!cache.has(obj)) {
      cache.set(obj, new Map());
    }
    const innerCache = cache.get(obj);
    const key = JSON.stringify(rest);

    if (innerCache.has(key)) return innerCache.get(key);

    const result = fn.call(this, obj, ...rest);
    innerCache.set(key, result);
    return result;
  };
}

function get(obj, path, defaultValue = undefined) {
  // Handle null/undefined object
  if (obj == null) return defaultValue;

  // Normalize path: "a.b[0].c" or ["a", "b", "0", "c"]
  const keys = Array.isArray(path) ? path : parsePath(path);

  let result = obj;
  for (const key of keys) {
    if (result == null) return defaultValue;
    result = result[key];
  }

  return result === undefined ? defaultValue : result;
}

// ─── lodash.set ───
function set(obj, path, value) {
  if (obj == null || typeof obj !== "object") return obj;

  const keys = Array.isArray(path) ? path : parsePath(path);

  let current = obj;
  for (let i = 0; i < keys.length - 1; i++) {
    const key = keys[i];
    const nextKey = keys[i + 1];

    // Create intermediate if doesn't exist or isn't an object
    if (current[key] == null || typeof current[key] !== "object") {
      // Determine if next level should be array or object
      current[key] = isArrayIndex(nextKey) ? [] : {};
    }

    current = current[key];
  }

  current[keys[keys.length - 1]] = value;
  return obj;
}

// ─── Path Parser ───
function parsePath(path) {
  if (typeof path !== "string") return [];

  const keys = [];
  // Match: word, [0], ["key"], ['key']
  const regex = /[^.[\]]+|\[(\d+|"[^"]*"|'[^']*')\]/g;
  let match;

  while ((match = regex.exec(path)) !== null) {
    if (match[1] !== undefined) {
      // Strip quotes from bracket notation
      const inner = match[1];
      if ((inner.startsWith('"') && inner.endsWith('"')) ||
          (inner.startsWith("'") && inner.endsWith("'"))) {
        keys.push(inner.slice(1, -1));
      } else {
        keys.push(inner); // numeric index as string
      }
    } else {
      keys.push(match[0]);
    }
  }

  return keys;
}

function isArrayIndex(key) {
  return /^\d+$/.test(key);
}

// ─── lodash.has ───
function has(obj, path) {
  if (obj == null) return false;
  const keys = Array.isArray(path) ? path : parsePath(path);

  let current = obj;
  for (const key of keys) {
    if (current == null || !Object.prototype.hasOwnProperty.call(current, key)) {
      return false;
    }
    current = current[key];
  }
  return true;
}

// ─── lodash.unset (delete deep property) ───
function unset(obj, path) {
  if (obj == null) return true;
  const keys = Array.isArray(path) ? path : parsePath(path);

  let current = obj;
  for (let i = 0; i < keys.length - 1; i++) {
    if (current[keys[i]] == null) return true; // Already doesn't exist
    current = current[keys[i]];
  }

  const lastKey = keys[keys.length - 1];
  if (Array.isArray(current)) {
    current.splice(Number(lastKey), 1);
  } else {
    delete current[lastKey];
  }
  return true;
}

class EventEmitter {
  constructor() {
    this.events = new Map();
    this.maxListeners = 10;
  }

  on(event, listener) {
    if (!this.events.has(event)) this.events.set(event, []);
    const listeners = this.events.get(event);
    
    if (listeners.length >= this.maxListeners) {
      console.warn(`MaxListenersExceededWarning: ${event} has ${listeners.length} listeners`);
    }
    listeners.push({ fn: listener, once: false });
    return this; // chainable
  }

  once(event, listener) {
    if (!this.events.has(event)) this.events.set(event, []);
    this.events.get(event).push({ fn: listener, once: true });
    return this;
  }

  emit(event, ...args) {
    if (!this.events.has(event)) return false;
    const listeners = this.events.get(event);
    
    // Filter out 'once' listeners after execution
    const remaining = [];
    for (const entry of listeners) {
      entry.fn(...args);
      if (!entry.once) remaining.push(entry);
    }
    this.events.set(event, remaining);
    return true;
  }

  off(event, listener) {
    if (!this.events.has(event)) return this;
    if (!listener) {
      this.events.delete(event); // Remove all
    } else {
      const filtered = this.events.get(event).filter(e => e.fn !== listener);
      filtered.length ? this.events.set(event, filtered) : this.events.delete(event);
    }
    return this;
  }

  listenerCount(event) {
    return this.events.get(event)?.length || 0;
  }

  eventNames() {
    return [...this.events.keys()];
  }

  removeAllListeners(event) {
    event ? this.events.delete(event) : this.events.clear();
    return this;
  }

  setMaxListeners(n) { this.maxListeners = n; return this; }

  // Async version — waits for all async listeners
  async emitAsync(event, ...args) {
    if (!this.events.has(event)) return false;
    const listeners = this.events.get(event);
    await Promise.all(listeners.map(entry => entry.fn(...args)));
    this.events.set(event, listeners.filter(e => !e.once));
    return true;
  }
}

// Usage:

  • Wildcard events: Support * pattern to listen to all events
  • Priority listeners: Allow specifying execution order
  • Memory leaks: Warn when listeners exceed threshold (maxListeners)
  • Error handling: If 'error' event has no listeners, throw

class PubSub {
  constructor() {
    this.topics = {};
    this.subId = 0;
  }

  subscribe(topic, callback) {
    if (!this.topics[topic]) this.topics[topic] = {};
    const id = `sub_${++this.subId}`;
    this.topics[topic][id] = callback;
    
    // Return unsubscribe function
    return {
      unsubscribe: () => { delete this.topics[topic][id]; },
      id,
    };
  }

  publish(topic, data) {
    if (!this.topics[topic]) return 0;
    const subscribers = Object.values(this.topics[topic]);
    subscribers.forEach(cb => {
      try { cb(data); } catch (err) { console.error(`PubSub error in ${topic}:`, err); }
    });
    return subscribers.length;
  }

  // Pattern-based subscription (e.g., "user.*" matches "user.login", "user.logout")
  subscribePattern(pattern, callback) {
    const regex = new RegExp('^' + pattern.replace(/\*/g, '[^.]+').replace(/\#/g, '.*') + '$');
    return this.subscribe('__pattern__', (msg) => {
      if (regex.test(msg.topic)) callback(msg.data, msg.topic);
    });
  }

  publishWithPattern(topic, data) {
    // Publish to exact subscribers
    this.publish(topic, data);
    // Publish to pattern subscribers
    if (this.topics['__pattern__']) {
      Object.values(this.topics['__pattern__']).forEach(cb => cb({ topic, data }));
    }
  }

  unsubscribeAll(topic) {
    if (topic) delete this.topics[topic];
    else this.topics = {};
  }

  getSubscribers(topic) {
    return Object.keys(this.topics[topic] || {}).length;
  }
}

7. Deep Clone (Handles Circular References)

function deepClone(obj, seen = new WeakMap()) {
  // Primitives and null
  if (obj === null || typeof obj !== 'object') return obj;
  
  // Circular reference detection
  if (seen.has(obj)) return seen.get(obj);
  
  // Date
  if (obj instanceof Date) return new Date(obj.getTime());
  
  // RegExp
  if (obj instanceof RegExp) return new RegExp(obj.source, obj.flags);
  
  // Map
  if (obj instanceof Map) {
    const clone = new Map();
    seen.set(obj, clone);
    obj.forEach((val, key) => clone.set(deepClone(key, seen), deepClone(val, seen)));
    return clone;
  }
  
  // Set
  if (obj instanceof Set) {
    const clone = new Set();
    seen.set(obj, clone);
    obj.forEach(val => clone.add(deepClone(val, seen)));
    return clone;
  }
  
  // ArrayBuffer
  if (obj instanceof ArrayBuffer) return obj.slice(0);
  
  // TypedArrays
  if (ArrayBuffer.isView(obj)) return new obj.constructor(obj.buffer.slice(0));
  
  // Array or Object
  const clone = Array.isArray(obj) ? [] : Object.create(Object.getPrototypeOf(obj));
  seen.set(obj, clone);
  
  // Copy all own properties (including symbols)
  const keys = [...Object.keys(obj), ...Object.getOwnPropertySymbols(obj)];
  for (const key of keys) {
    const descriptor = Object.getOwnPropertyDescriptor(obj, key);
    if (descriptor.value !== undefined) {
      descriptor.value = deepClone(descriptor.value, seen);
    }
    Object.defineProperty(clone, key, descriptor);
  }
  
  return clone;
}

      this.bindEvents();

    }
  }

    switch (type) {

      case 'boolean': return `<span class="json-bool">${value}</span>`;
      case 'number': return `<span class="json-num">${value}</span>`;

      case 'array':

    const close = isArray ? ']' : '}';
    const preview = count === 0 ? `${open}${close}` : `${open}...${close} // ${count} items`;

    let html = `<span class="json-toggle ${collapsed ? 'collapsed' : ''}">${collapsed ? '▸' : '▾'}</span>`;
    html += `<span class="json-bracket">${open}</span>`;
    html += `<span class="json-preview" ${collapsed ? '' : 'style="display:none"'}>${preview.slice(1)}</span>`;
    html += `<div class="json-content" ${collapsed ? 'style="display:none"' : ''}>`;

    if (isArray) {
      value.forEach((item, i) => {
        html += `<div class="json-line">${this.renderNode(item, depth + 1, i)}${i < count - 1 ? ',' : ''}</div>`;
      });
    } else {
      Object.entries(value).forEach(([k, v], i) => {
        html += `<div class="json-line"><span class="json-key">"${k}"</span>: ${this.renderNode(v, depth + 1, k)}${i < count - 1 ? ',' : ''}</div>`;
      });

class UndoRedoManager {
  constructor(options = {}) {
    this.undoStack = [];
    this.redoStack = [];
    this.maxHistory = options.maxHistory || 100;
    this.onChange = options.onChange || (() => {});
  }

  execute(command) {
    // command = { execute(), undo(), description }
    command.execute();
    this.undoStack.push(command);
    this.redoStack = []; // Clear redo on new action
    
    if (this.undoStack.length > this.maxHistory) this.undoStack.shift();
    this.onChange(this.getState());
  }

  undo() {
    if (!this.canUndo()) return null;
    const command = this.undoStack.pop();
    command.undo();
    this.redoStack.push(command);
    this.onChange(this.getState());
    return command;
  }

  redo() {
    if (!this.canRedo()) return null;
    const command = this.redoStack.pop();
    command.execute();
    this.undoStack.push(command);
    this.onChange(this.getState());
    return command;
  }

  canUndo() { return this.undoStack.length > 0; }
  canRedo() { return this.redoStack.length > 0; }

  getState() {
    return {
      canUndo: this.canUndo(),
      canRedo: this.canRedo(),
      undoCount: this.undoStack.length,
      redoCount: this.redoStack.length,
      lastAction: this.undoStack[this.undoStack.length - 1]?.description,
    };
  }

  clear() { this.undoStack = []; this.redoStack = []; }

  // Batch multiple commands into one undo step
  batch(commands) {
    const batchCmd = {
      description: `Batch (${commands.length} actions)`,
      execute: () => commands.forEach(c => c.execute()),
      undo: () => [...commands].reverse().forEach(c => c.undo()),
    };
    this.execute(batchCmd);
  }
}

class TextCommand {
  constructor(editor, index, inserted, removed) {

    this.index = index;
    this.inserted = inserted;

    if (this.inserted) this.editor.splice(this.index, 0, this.inserted);

  }
}

class Storage {
  constructor(namespace = 'app', options = {}) {
    this.namespace = namespace;
    this.defaultTTL = options.defaultTTL || null; // ms, null = no expiry
    this.storage = options.storage || localStorage;
    this.encryptionKey = options.encrypt ? options.encryptionKey : null;
  }

  _key(key) { return `${this.namespace}:${key}`; }

  set(key, value, ttl = this.defaultTTL) {
    const entry = {
      value,
      created: Date.now(),
      expires: ttl ? Date.now() + ttl : null,
    };
    try {
      this.storage.setItem(this._key(key), JSON.stringify(entry));
      return true;
    } catch (e) {
      if (e.name === 'QuotaExceededError') {
        this.evict(); // Remove expired items
        try {
          this.storage.setItem(this._key(key), JSON.stringify(entry));
          return true;
        } catch { return false; }
      }
      return false;
    }
  }

  get(key, defaultValue = null) {
    const raw = this.storage.getItem(this._key(key));
    if (!raw) return defaultValue;
    
    try {
      const entry = JSON.parse(raw);
      if (entry.expires && Date.now() > entry.expires) {
        this.remove(key); // Lazy expiration
        return defaultValue;
      }
      return entry.value;
    } catch {
      return defaultValue;
    }
  }

  remove(key) { this.storage.removeItem(this._key(key)); }

  has(key) { return this.get(key) !== null; }

  // Get all keys in this namespace
  keys() {
    const prefix = this.namespace + ':';
    const keys = [];
    for (let i = 0; i < this.storage.length; i++) {
      const k = this.storage.key(i);
      if (k.startsWith(prefix)) keys.push(k.slice(prefix.length));
    }
    return keys;
  }

  clear() { this.keys().forEach(k => this.remove(k)); }

  // Remove expired entries
  evict() {
    this.keys().forEach(key => this.get(key)); // Triggers lazy expiration
  }

  // Get remaining TTL in ms
  ttl(key) {
    const raw = this.storage.getItem(this._key(key));
    if (!raw) return -2; // Key doesn't exist
    const entry = JSON.parse(raw);
    if (!entry.expires) return -1; // No expiry
    const remaining = entry.expires - Date.now();
    return remaining > 0 ? remaining : 0;
  }

  // Get storage usage info
  info() {
    let totalSize = 0;
    this.keys().forEach(key => {
      totalSize += this.storage.getItem(this._key(key)).length * 2; // UTF-16
    });
    return { keys: this.keys().length, bytes: totalSize, namespace: this.namespace };
  }
}

// Transforms f(a, b, c) into f(a)(b)(c) or f(a, b)(c), etc.
function curry(fn) {
  const arity = fn.length;
  
  return function curried(...args) {
    if (args.length >= arity) {
      return fn.apply(this, args);
    }
    return function(...moreArgs) {
      return curried.apply(this, [...args, ...moreArgs]);
    };
  };
}

// With placeholder support (like lodash _.curry with _)
const _ = Symbol('placeholder');

function curryWithPlaceholder(fn) {
  const arity = fn.length;
  
  return function curried(...args) {
    // Replace placeholders with new args
    const complete = args.length >= arity && !args.slice(0, arity).includes(_);
    if (complete) return fn.apply(this, args);
    
    return function(...moreArgs) {
      const merged = args.map(a => a === _ && moreArgs.length ? moreArgs.shift() : a);
      return curried.apply(this, [...merged, ...moreArgs]);
    };
  };
}

// ──── COMPOSE (right-to-left) ────
function compose(...fns) {
  if (fns.length === 0) return (x) => x;
  if (fns.length === 1) return fns[0];
  
  return function composed(...args) {
    return fns.reduceRight((result, fn, i) => {
      return i === fns.length - 1 ? fn(...result) : fn(result);
    }, args);
  };
}

// ──── PIPE (left-to-right) — more readable ────
function pipe(...fns) {
  return function piped(...args) {
    return fns.reduce((result, fn, i) => {
      return i === 0 ? fn(...result) : fn(result);
    }, args);
  };
}

// ──── ASYNC COMPOSE ────
function composeAsync(...fns) {
  return function(input) {
    return fns.reduceRight((chain, fn) => chain.then(fn), Promise.resolve(input));
  };
}

// Usage:
const add = curry((a, b) => a + b);
const add5 = add(5);
add5(3); // 8
add(2, 3); // 5

const transform = pipe(
  (x) => x * 2,
  (x) => x + 1,
  (x) => `Result: ${x}`
);
transform(5); // "Result: 11"

// Async pipeline:
const process = composeAsync(
  async (data) => await fetch(data.url),
  async (res) => await res.json(),
  (json) => json.results,
);

class VirtualList {
  constructor(container, options) {
    this.container = container;
    this.items = options.items || []; // Array of data
    this.itemHeight = options.itemHeight || 40; // Fixed row height
    this.renderItem = options.renderItem; // (item, index) => HTML string
    this.bufferSize = options.bufferSize || 5; // Extra items above/below viewport
    this.overscan = options.overscan || 3;
    
    this.scrollTop = 0;
    this.containerHeight = container.clientHeight;
    
    this.setup();
    this.render();
  }

  setup() {
    this.container.style.overflow = 'auto';
    this.container.style.position = 'relative';
    
    // Spacer to set total scroll height
    this.spacer = document.createElement('div');
    this.spacer.style.height = `${this.items.length * this.itemHeight}px`;
    this.spacer.style.position = 'relative';
    
    // Viewport that holds visible items
    this.viewport = document.createElement('div');
    this.viewport.style.position = 'absolute';
    this.viewport.style.left = '0';
    this.viewport.style.right = '0';
    
    this.spacer.appendChild(this.viewport);
    this.container.appendChild(this.spacer);
    
    // Scroll listener with rAF throttling
    let ticking = false;
    this.container.addEventListener('scroll', () => {
      if (!ticking) {
        requestAnimationFrame(() => {
          this.scrollTop = this.container.scrollTop;
          this.render();
          ticking = false;
        });
        ticking = true;
      }
    });
  }

  render() {
    const startIndex = Math.max(0, Math.floor(this.scrollTop / this.itemHeight) - this.overscan);
    const visibleCount = Math.ceil(this.containerHeight / this.itemHeight) + this.overscan * 2;
    const endIndex = Math.min(this.items.length, startIndex + visibleCount);
    
    this.viewport.style.top = `${startIndex * this.itemHeight}px`;
    
    let html = '';
    for (let i = startIndex; i < endIndex; i++) {
      html += `<div class="virtual-item" style="height:${this.itemHeight}px" data-index="${i}">${this.renderItem(this.items[i], i)}</div>`;
    }
    this.viewport.innerHTML = html;
  }

  // Dynamic updates
  setItems(items) {
    this.items = items;
    this.spacer.style.height = `${items.length * this.itemHeight}px`;
    this.render();
  }

  scrollToIndex(index) {
    this.container.scrollTop = index * this.itemHeight;
  }

  // Variable height support (more complex)
  // Would need a height cache + binary search for start index
}

// Usage:
const list = new VirtualList(document.getElementById('list-container'), {
  items: Array.from({ length: 100000 }, (_, i) => ({ id: i, name: `Item ${i}` })),
  itemHeight: 50,
  renderItem: (item, i) => `<span>#${item.id}</span> ${item.name}`,
});

  • Infinite scroll: Trigger fetch when scrolled near bottom, append to items
  • Sticky headers: Group items by category, pin headers to top during scroll
  • Performance: Use will-change: transform on viewport, avoid layout thrashing
