1. Streaming, Backpressure & Data Pipelines

1.1 ReadableStream, WritableStream, TransformStream

The Streams API lets you process data chunk-by-chunk without buffering entire payloads in memory. Critical for: video players, large file uploads, server-sent data, AI token streaming.

1.2 Backpressure — The Most Important Concept

Problem: Data arrives at 10MB/s, but you can only process at 2MB/s. Without backpressure, memory grows unbounded → crash.

Solution: The consumer signals to the producer: "slow down, I'm not ready."

// ═══ Backpressure in action ═══
// ReadableStream with highWaterMark controls when pull() is called
const dataSource = new ReadableStream({
  start(controller) {
    this.position = 0;
  },
  async pull(controller) {
    // This is ONLY called when the internal queue has space!
    // If desiredSize <= 0, pull() won't be called (backpressure!)
    const chunk = await fetchNextChunk(this.position);
    this.position += chunk.length;
    
    controller.enqueue(chunk);
    
    // Check backpressure manually
    console.log('Desired size:', controller.desiredSize);
    // > 0 means "I want more data"
    // <= 0 means "I'm full, stop producing"
  }
}, new ByteLengthQueuingStrategy({ highWaterMark: 1024 * 64 })); // 64KB buffer

// ═══ Manual backpressure with async generator ═══
async function* generateWithBackpressure(source, batchSize = 100) {
  let buffer = [];
  for await (const item of source) {
    buffer.push(item);
    if (buffer.length >= batchSize) {
      yield buffer; // Consumer controls when to call next()
      buffer = [];
    }
  }
  if (buffer.length > 0) yield buffer;
}

// Consumer controls pace:
const gen = generateWithBackpressure(dataStream);
for await (const batch of gen) {
  await processBatch(batch); // Takes time → naturally creates backpressure
  await updateUI(); // Give browser time to paint
}

1.3 Streaming Fetch (NDJSON, Server-Sent Events, AI Tokens)

// ═══ Streaming NDJSON (Newline-Delimited JSON) ═══
// Used by: Netflix for real-time metrics, ChatGPT for token streaming
async function* streamNDJSON(url, signal) {
  const response = await fetch(url, { signal });
  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = '';

  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    
    buffer += decoder.decode(value, { stream: true });
    const lines = buffer.split('\n');
    buffer = lines.pop(); // Keep incomplete line in buffer
    
    for (const line of lines) {
      if (line.trim()) {
        yield JSON.parse(line);
      }
    }
  }
  // Process remaining buffer
  if (buffer.trim()) {
    yield JSON.parse(buffer);
  }
}

// Usage with AbortController

try {

    }
  }
} catch (e) {

// ═══ AI Token Streaming (ChatGPT/Claude style) ═══
async function streamAIResponse(prompt, onToken) {
  const response = await fetch('/api/chat', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ prompt })
  });

  const reader = response.body
    .pipeThrough(new TextDecoderStream())
    .getReader();
  
  let fullResponse = '';
  while (true) {
    const { done, value } = await reader.read();
    if (done) break;
    
    // SSE format: "data: {json}\n\n"
    const lines = value.split('\n');
    for (const line of lines) {
      if (line.startsWith('data: ')) {
        const data = JSON.parse(line.slice(6));
        if (data.token) {
          fullResponse += data.token;
          onToken(data.token, fullResponse);
        }
      }
    }
  }
  return fullResponse;
}

1.4 Large File Upload with Resumable Streaming

// ═══ Chunked Upload with Resume (Netflix uses for content ingestion) ═══
class ResumableUploader {
  constructor(file, { chunkSize = 5 * 1024 * 1024, endpoint }) {
    this.file = file;
    this.chunkSize = chunkSize;
    this.endpoint = endpoint;
    this.uploadedBytes = 0;
    this.uploadId = null;
    this.abortController = new AbortController();
  }

  async start() {
    // Initialize upload session
    const initRes = await fetch(`${this.endpoint}/init`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        filename: this.file.name,
        size: this.file.size,
        type: this.file.type
      })
    });
    const { uploadId, resumeFrom } = await initRes.json();
    this.uploadId = uploadId;
    this.uploadedBytes = resumeFrom || 0;

    // Stream chunks
    while (this.uploadedBytes < this.file.size) {
      const end = Math.min(this.uploadedBytes + this.chunkSize, this.file.size);
      const chunk = this.file.slice(this.uploadedBytes, end);
      
      await this.uploadChunk(chunk, this.uploadedBytes, end);
      this.uploadedBytes = end;
      
      this.onProgress?.(this.uploadedBytes / this.file.size);
    }
    
    // Finalize
    await fetch(`${this.endpoint}/complete`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ uploadId: this.uploadId })
    });
  }

  async uploadChunk(chunk, start, end) {
    const maxRetries = 3;
    for (let attempt = 0; attempt < maxRetries; attempt++) {
      try {
        const res = await fetch(`${this.endpoint}/chunk`, {
          method: 'PUT',
          headers: {
            'Content-Range': `bytes ${start}-${end - 1}/${this.file.size}`,
            'X-Upload-Id': this.uploadId
          },
          body: chunk,
          signal: this.abortController.signal
        });
        if (res.ok) return;
        throw new Error(`Upload failed: ${res.status}`);
      } catch (e) {
        if (e.name === 'AbortError') throw e;
        if (attempt === maxRetries - 1) throw e;
        await new Promise(r => setTimeout(r, 1000 * 2 ** attempt));
      }
    }
  }

  pause() { this.abortController.abort(); }
  resume() {
    this.abortController = new AbortController();
    return this.start(); // Resumes from uploadedBytes
  }
}
💡 Interview Answer: "Backpressure is the mechanism where a slow consumer signals to a fast producer to reduce its rate. In the Streams API, this is managed via highWaterMark and desiredSize. When the internal queue fills up, pull() isn't called until the consumer reads data. This prevents OOM crashes when processing large datasets."

2.1 Circuit Breaker Pattern (Complete Implementation)

// ═══ Circuit Breaker: Prevents cascading failures ═══
// States: CLOSED (normal) → OPEN (failing) → HALF_OPEN (testing recovery)
class CircuitBreaker {
  static CLOSED = 'CLOSED';
  static OPEN = 'OPEN';
  static HALF_OPEN = 'HALF_OPEN';

  constructor(options = {}) {
    this.failureThreshold = options.failureThreshold || 5;
    this.resetTimeout = options.resetTimeout || 30000; // 30s
    this.halfOpenMaxCalls = options.halfOpenMaxCalls || 3;
    this.monitorInterval = options.monitorInterval || 10000;
    
    this.state = CircuitBreaker.CLOSED;
    this.failureCount = 0;
    this.successCount = 0;
    this.halfOpenCalls = 0;
    this.lastFailureTime = null;
    this.listeners = new Map();
  }

  async execute(fn) {
    if (this.state === CircuitBreaker.OPEN) {
      if (Date.now() - this.lastFailureTime >= this.resetTimeout) {
        this.transitionTo(CircuitBreaker.HALF_OPEN);
      } else {
        throw new CircuitBreakerOpenError(
          `Circuit breaker is OPEN. Retry after ${this.resetTimeout - (Date.now() - this.lastFailureTime)}ms`
        );
      }
    }

    if (this.state === CircuitBreaker.HALF_OPEN) {
      if (this.halfOpenCalls >= this.halfOpenMaxCalls) {
        throw new CircuitBreakerOpenError('Half-open limit reached');
      }
      this.halfOpenCalls++;
    }

    try {
      const result = await fn();
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure(error);
      throw error;
    }
  }

  onSuccess() {
    if (this.state === CircuitBreaker.HALF_OPEN) {
      this.successCount++;
      if (this.successCount >= this.halfOpenMaxCalls) {
        this.transitionTo(CircuitBreaker.CLOSED);
      }
    }
    this.failureCount = 0;
  }

  onFailure(error) {
    this.failureCount++;
    this.lastFailureTime = Date.now();
    
    if (this.failureCount >= this.failureThreshold) {
      this.transitionTo(CircuitBreaker.OPEN);
    }
    this.emit('failure', { error, failureCount: this.failureCount, state: this.state });
  }

  transitionTo(newState) {
    const oldState = this.state;
    this.state = newState;
    
    if (newState === CircuitBreaker.CLOSED) {
      this.failureCount = 0;
      this.successCount = 0;
      this.halfOpenCalls = 0;
    }
    if (newState === CircuitBreaker.HALF_OPEN) {
      this.halfOpenCalls = 0;
      this.successCount = 0;
    }
    
    this.emit('stateChange', { from: oldState, to: newState });
  }

  on(event, handler) {
    if (!this.listeners.has(event)) this.listeners.set(event, []);
    this.listeners.get(event).push(handler);
  }
  emit(event, data) {
    (this.listeners.get(event) || []).forEach(h => h(data));
  }

  getState() {
    return { state: this.state, failureCount: this.failureCount, lastFailure: this.lastFailureTime };
  }
}

class CircuitBreakerOpenError extends Error {
  constructor(message) {
    super(message);
    this.name = 'CircuitBreakerOpenError';
  }
}

// ═══ Usage ═══

  failureThreshold: 3,
  resetTimeout: 10000
});

apiBreaker.on('stateChange', ({ from, to }) => {
  console.log(`Circuit: ${from} → ${to}`);

2.2 Bulkhead Pattern (Resource Isolation)

// ═══ Bulkhead: Isolate failures to prevent total system collapse ═══
// Like ship compartments — one flooding doesn't sink the whole ship
class Bulkhead {
  constructor(name, { maxConcurrent = 10, maxQueue = 50, timeout = 5000 }) {
    this.name = name;
    this.maxConcurrent = maxConcurrent;
    this.maxQueue = maxQueue;
    this.timeout = timeout;
    this.active = 0;
    this.queue = [];
    this.metrics = { total: 0, rejected: 0, timedOut: 0, succeeded: 0, failed: 0 };
  }

  async execute(fn) {
    if (this.active >= this.maxConcurrent) {
      if (this.queue.length >= this.maxQueue) {
        this.metrics.rejected++;
        throw new BulkheadRejectError(`${this.name}: Queue full (${this.maxQueue})`);
      }
      // Wait in queue
      await new Promise((resolve, reject) => {
        const timer = setTimeout(() => {
          const idx = this.queue.findIndex(q => q.resolve === resolve);
          if (idx !== -1) this.queue.splice(idx, 1);
          this.metrics.timedOut++;
          reject(new BulkheadTimeoutError(`${this.name}: Queue timeout`));
        }, this.timeout);
        this.queue.push({ resolve, reject, timer });
      });
    }

    this.active++;
    this.metrics.total++;
    try {
      const result = await Promise.race([
        fn(),
        new Promise((_, reject) => 
          setTimeout(() => reject(new BulkheadTimeoutError('Execution timeout')), this.timeout)
        )
      ]);
      this.metrics.succeeded++;
      return result;
    } catch (e) {
      this.metrics.failed++;
      throw e;
    } finally {
      this.active--;
      this.dequeue();
    }
  }

  dequeue() {
    if (this.queue.length > 0 && this.active < this.maxConcurrent) {
      const { resolve, timer } = this.queue.shift();
      clearTimeout(timer);
      resolve();
    }
  }

  getMetrics() { return { ...this.metrics, active: this.active, queued: this.queue.length }; }
}

// ═══ Isolate different API calls ═══
const criticalBulkhead = new Bulkhead('critical-api', { maxConcurrent: 20, maxQueue: 100 });
const analyticsBulkhead = new Bulkhead('analytics', { maxConcurrent: 5, maxQueue: 10 });

}

2.3 Fallback Strategies (Graceful Degradation)

// ═══ Multi-level Fallback Chain ═══
class FallbackChain {
  constructor(strategies) {
    this.strategies = strategies; // Array of { name, execute, condition? }
  }

  async execute(context) {
    const errors = [];
    
    for (const strategy of this.strategies) {
      if (strategy.condition && !strategy.condition(context)) continue;
      
      try {
        const result = await strategy.execute(context);
        if (errors.length > 0) {
          console.warn(`Degraded: using fallback "${strategy.name}" after ${errors.length} failures`);
        }
        return { result, strategy: strategy.name, degraded: errors.length > 0 };
      } catch (e) {
        errors.push({ strategy: strategy.name, error: e });
      }
    }
    
    throw new AggregateError(errors.map(e => e.error), 'All fallback strategies exhausted');
  }
}

// ═══ Stale-While-Revalidate (SWR) Pattern ═══
class SWRCache {
  constructor({ maxAge = 60000, staleWhileRevalidate = 300000 }) {
    this.maxAge = maxAge;
    this.staleWhileRevalidate = staleWhileRevalidate;
    this.cache = new Map();
    this.revalidating = new Set();
  }

  async get(key, fetcher) {
    const entry = this.cache.get(key);
    const now = Date.now();

    if (!entry) {
      // Cache miss — fetch fresh
      return this.fetchAndCache(key, fetcher);
    }

    const age = now - entry.timestamp;

    if (age < this.maxAge) {
      // Fresh — serve from cache
      return entry.data;
    }

    if (age < this.staleWhileRevalidate) {
      // Stale but acceptable — serve stale, revalidate in background
      if (!this.revalidating.has(key)) {
        this.revalidating.add(key);
        this.fetchAndCache(key, fetcher).finally(() => this.revalidating.delete(key));
      }
      return entry.data; // Return stale immediately
    }

    // Too stale — must refetch
    return this.fetchAndCache(key, fetcher);
  }

  async fetchAndCache(key, fetcher) {
    const data = await fetcher();
    this.cache.set(key, { data, timestamp: Date.now() });
    return data;
  }
}

// Usage:
const userCache = new SWRCache({ maxAge: 30000, staleWhileRevalidate: 120000 });

3.3 Communication Between Microfrontends

// ═══ Event Bus for Cross-MFE Communication ═══
// Shared library (loaded once via Module Federation shared)
class MicroFrontendBus {
  constructor() {
    // Use CustomEvent on window for cross-framework compatibility
    this.channel = new BroadcastChannel('mfe-bus');
    this.subscriptions = new Map();
  }

  // Publish event to all MFEs (same tab)
  emit(event, payload) {
    const detail = { event, payload, source: this.appName, timestamp: Date.now() };
    window.dispatchEvent(new CustomEvent('mfe:event', { detail }));
    // Also broadcast to other tabs
    this.channel.postMessage(detail);
  }

  // Subscribe to events from other MFEs
  on(event, handler) {
    const wrappedHandler = (e) => {
      if (e.detail.event === event) {
        handler(e.detail.payload, e.detail);
      }
    };
    window.addEventListener('mfe:event', wrappedHandler);

    if (!this.subscriptions.has(event)) this.subscriptions.set(event, []);
    this.subscriptions.get(event).push({ handler, wrappedHandler });
    
    // Return unsubscribe function
    return () => {
      window.removeEventListener('mfe:event', wrappedHandler);
      const subs = this.subscriptions.get(event);
      const idx = subs.findIndex(s => s.handler === handler);
      if (idx !== -1) subs.splice(idx, 1);
    };
  }

  // Shared state (minimal — just for cross-cutting concerns)
  setState(key, value) {
    const prev = this.getState(key);
    sessionStorage.setItem(`mfe:${key}`, JSON.stringify(value));
    if (prev !== value) {
      this.emit(`state:${key}`, { prev, next: value });
    }
  }

  getState(key) {
    const raw = sessionStorage.getItem(`mfe:${key}`);
    return raw ? JSON.parse(raw) : null;
  }
}

class SharedDependencyManager {
  static registry = new Map();
  
  static register(name, version, module) {
    if (!this.registry.has(name)) {
      this.registry.set(name, []);
    }
    this.registry.get(name).push({ version, module });
  }
  
  static resolve(name, requiredVersion) {
    const versions = this.registry.get(name) || [];
    // semver-compatible resolution
    const compatible = versions.find(v => satisfies(v.version, requiredVersion));
    if (!compatible) {
      console.warn(`No compatible ${name}@${requiredVersion}. Loading isolated copy.`);
      return null; // Will load its own copy
    }
    return compatible.module;
  }
}

// ═══ Production Error Reporter (Sentry-like) ═══
class ErrorReporter {
  constructor({ endpoint, appVersion, environment, sampleRate = 1.0 }) {
    this.endpoint = endpoint;
    this.appVersion = appVersion;
    this.environment = environment;
    this.sampleRate = sampleRate;
    this.breadcrumbs = [];
    this.maxBreadcrumbs = 50;
    this.context = {};
    
    this.install();
  }

  install() {
    // Capture unhandled errors
    window.addEventListener('error', (event) => {
      this.captureError(event.error || new Error(event.message), {
        type: 'unhandled',
        filename: event.filename,
        lineno: event.lineno,
        colno: event.colno
      });
    });

    // Capture unhandled promise rejections
    window.addEventListener('unhandledrejection', (event) => {
      this.captureError(
        event.reason instanceof Error ? event.reason : new Error(String(event.reason)),
        { type: 'unhandledrejection' }
      );
    });

    // Capture console.error
    const originalError = console.error;
    console.error = (...args) => {
      this.addBreadcrumb('console', 'error', args.map(String).join(' '));
      originalError.apply(console, args);
    };

    // Capture XHR/Fetch failures
    this.instrumentFetch();
    
    // Capture navigation
    this.instrumentNavigation();
  }

  instrumentFetch() {
    const originalFetch = window.fetch;
    window.fetch = async (url, options) => {
      const start = performance.now();
      try {
        const response = await originalFetch(url, options);
        const duration = performance.now() - start;
        this.addBreadcrumb('http', response.ok ? 'info' : 'error', 
          `${options?.method || 'GET'} ${url} → ${response.status} (${Math.round(duration)}ms)`
        );
        if (!response.ok && response.status >= 500) {
          this.captureError(new Error(`HTTP ${response.status}: ${url}`), {
            type: 'http', status: response.status, duration
          });
        }
        return response;
      } catch (e) {
        this.addBreadcrumb('http', 'error', `${options?.method || 'GET'} ${url} → NETWORK ERROR`);
        throw e;
      }
    };
  }

  instrumentNavigation() {
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (entry.entryType === 'navigation') {
          this.addBreadcrumb('navigation', 'info', 
            `${entry.type} → ${entry.name} (DOM: ${Math.round(entry.domContentLoadedEventEnd)}ms)`
          );
        }
      }
    });
    observer.observe({ type: 'navigation', buffered: true });
  }

  addBreadcrumb(category, level, message) {
    this.breadcrumbs.push({ category, level, message, timestamp: Date.now() });
    if (this.breadcrumbs.length > this.maxBreadcrumbs) {
      this.breadcrumbs.shift();
    }
  }

  setContext(key, value) { this.context[key] = value; }

  captureError(error, extra = {}) {
    if (Math.random() > this.sampleRate) return;

    const report = {
      error: {
        name: error.name,
        message: error.message,
        stack: this.parseStack(error.stack)
      },
      breadcrumbs: [...this.breadcrumbs],
      context: { ...this.context, ...extra },
      environment: this.environment,
      appVersion: this.appVersion,
      url: window.location.href,
      userAgent: navigator.userAgent,
      timestamp: new Date().toISOString(),
      // Performance context
      memory: performance?.memory ? {
        used: performance.memory.usedJSHeapSize,
        total: performance.memory.totalJSHeapSize,
        limit: performance.memory.jsHeapSizeLimit
      } : null,
      // Connection info
      connection: navigator?.connection ? {
        effectiveType: navigator.connection.effectiveType,
        downlink: navigator.connection.downlink,
        rtt: navigator.connection.rtt
      } : null
    };

    // Send with beacon (survives page unload)
    if (navigator.sendBeacon) {
      navigator.sendBeacon(this.endpoint, JSON.stringify(report));
    } else {
      fetch(this.endpoint, {
        method: 'POST',
        body: JSON.stringify(report),
        keepalive: true
      }).catch(() => {}); // Silent fail
    }
  }

  parseStack(stack) {
    if (!stack) return [];
    return stack.split('\n').slice(1).map(line => {
      const match = line.match(/at\s+(.+?)\s+\((.+?):(\d+):(\d+)\)/);
      if (match) {
        return { function: match[1], file: match[2], line: +match[3], column: +match[4] };
      }
      return { raw: line.trim() };
    });
  }
}

class WebVitalsCollector {

  observeLCP() {

      const entries = list.getEntries();
      const lastEntry = entries[entries.length - 1];
      this.metrics.lcp = {
        value: lastEntry.startTime,
        element: lastEntry.element?.tagName,
        url: lastEntry.url,
        rating: lastEntry.startTime <= 2500 ? 'good' : lastEntry.startTime <= 4000 ? 'needs-improvement' : 'poor'
      };
    });
    observer.observe({ type: 'largest-contentful-paint', buffered: true });
    
    // Report on page hide (user navigates away)
    document.addEventListener('visibilitychange', () => {
      if (document.visibilityState === 'hidden' && this.metrics.lcp) {
        this.report('LCP', this.metrics.lcp);
      }
    }, { once: true });
  }

  // Interaction to Next Paint (replaces FID)
  observeINP() {
    let maxDuration = 0;
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (entry.duration > maxDuration) {
          maxDuration = entry.duration;
          this.metrics.inp = {
            value: entry.duration,
            type: entry.name, // 'click', 'keydown', etc.
            target: entry.target?.tagName,
            rating: entry.duration <= 200 ? 'good' : entry.duration <= 500 ? 'needs-improvement' : 'poor'
          };
        }
      }
    });
    observer.observe({ type: 'event', buffered: true, durationThreshold: 16 });
  }

  // Cumulative Layout Shift
  observeCLS() {
    let clsValue = 0;
    let sessionValue = 0;
    let sessionEntries = [];
    
    const observer = new PerformanceObserver((list) => {
      for (const entry of list.getEntries()) {
        if (!entry.hadRecentInput) { // Ignore user-initiated shifts
          const firstEntry = sessionEntries[0];
          const lastEntry = sessionEntries[sessionEntries.length - 1];
          
          // Session window: gap < 1s, total < 5s
          if (sessionEntries.length && 
              entry.startTime - lastEntry.startTime < 1000 &&
              entry.startTime - firstEntry.startTime < 5000) {
            sessionValue += entry.value;
            sessionEntries.push(entry);
          } else {
            sessionValue = entry.value;
            sessionEntries = [entry];
          }
          
          if (sessionValue > clsValue) {
            clsValue = sessionValue;
            this.metrics.cls = {
              value: clsValue,
              rating: clsValue <= 0.1 ? 'good' : clsValue <= 0.25 ? 'needs-improvement' : 'poor',
              sources: sessionEntries.slice(-3).map(e => ({
                node: e.sources?.[0]?.node?.tagName,
                previousRect: e.sources?.[0]?.previousRect,
                currentRect: e.sources?.[0]?.currentRect
              }))
            };
          }
        }
      }
    });
    observer.observe({ type: 'layout-shift', buffered: true });
  }

  // First Input Delay
  observeFID() {
    const observer = new PerformanceObserver((list) => {
      const entry = list.getEntries()[0];
      this.metrics.fid = {
        value: entry.processingStart - entry.startTime,
        type: entry.name,
        rating: entry.processingStart - entry.startTime <= 100 ? 'good' : 'poor'
      };
      this.report('FID', this.metrics.fid);
    });
    observer.observe({ type: 'first-input', buffered: true });
  }

  // Time to First Byte
  observeTTFB() {
    const observer = new PerformanceObserver((list) => {
      const nav = list.getEntries()[0];
      this.metrics.ttfb = {
        value: nav.responseStart - nav.requestStart,
        dns: nav.domainLookupEnd - nav.domainLookupStart,
        tcp: nav.connectEnd - nav.connectStart,
        tls: nav.secureConnectionStart ? nav.connectEnd - nav.secureConnectionStart : 0,
        rating: nav.responseStart - nav.requestStart <= 800 ? 'good' : 'poor'
      };
      this.report('TTFB', this.metrics.ttfb);
    });
    observer.observe({ type: 'navigation', buffered: true });
  }

  getAll() { return { ...this.metrics }; }
}

// ═══ Trace Context Propagation (W3C Trace Context) ═══
class FrontendTracer {
  constructor({ serviceName, endpoint, sampleRate = 0.01 }) {
    this.serviceName = serviceName;
    this.endpoint = endpoint;
    this.sampleRate = sampleRate;
    this.spans = [];
  }

  // Generate W3C trace-id (32 hex chars) and span-id (16 hex chars)
  generateTraceId() {
    return Array.from(crypto.getRandomValues(new Uint8Array(16)))
      .map(b => b.toString(16).padStart(2, '0')).join('');
  }
  generateSpanId() {
    return Array.from(crypto.getRandomValues(new Uint8Array(8)))
      .map(b => b.toString(16).padStart(2, '0')).join('');
  }

  startSpan(name, parentSpanId = null) {
    const span = {
      traceId: this.currentTraceId || this.generateTraceId(),
      spanId: this.generateSpanId(),
      parentSpanId,
      name,
      serviceName: this.serviceName,
      startTime: performance.now(),
      endTime: null,
      attributes: {},
      events: []
    };
    if (!this.currentTraceId) this.currentTraceId = span.traceId;
    return span;
  }

  endSpan(span) {
    span.endTime = performance.now();
    span.duration = span.endTime - span.startTime;
    this.spans.push(span);
    
    // Batch send
    if (this.spans.length >= 10) this.flush();
  }

  // Inject trace context into outgoing fetch headers
  injectHeaders(span) {
    return {
      'traceparent': `00-${span.traceId}-${span.spanId}-01`,
      'tracestate': `${this.serviceName}=${span.spanId}`
    };
  }

  // Traced fetch — correlates frontend request with backend processing
  async tracedFetch(url, options = {}) {
    const span = this.startSpan(`HTTP ${options.method || 'GET'} ${new URL(url, location.origin).pathname}`);
    span.attributes = { 'http.url': url, 'http.method': options.method || 'GET' };

    const headers = { ...options.headers, ...this.injectHeaders(span) };
    
    try {
      const response = await fetch(url, { ...options, headers });
      span.attributes['http.status_code'] = response.status;
      span.attributes['http.response_size'] = response.headers.get('content-length');
      this.endSpan(span);
      return response;
    } catch (error) {
      span.attributes['error'] = true;
      span.attributes['error.message'] = error.message;
      span.events.push({ name: 'exception', timestamp: performance.now(), attributes: { message: error.message } });
      this.endSpan(span);
      throw error;
    }
  }

  flush() {
    if (this.spans.length === 0) return;
    const batch = this.spans.splice(0);
    navigator.sendBeacon(this.endpoint, JSON.stringify(batch));
  }
}

});

5.3 Builder Pattern with Type Safety

// ═══ Type-safe Query Builder (prevents invalid queries at compile time) ═══
type QueryConfig = {
  table?: string;
  conditions: string[];
  orderBy?: { column: string; direction: 'ASC' | 'DESC' };
  limit?: number;
  joins: string[];
};

class QueryBuilder<HasTable extends boolean = false, HasCondition extends boolean = false> {
  private config: QueryConfig = { conditions: [], joins: [] };

  from(table: string): QueryBuilder<true, HasCondition> {
    this.config.table = table;
    return this as any;
  }

  where(condition: string): QueryBuilder<HasTable, true> {
    this.config.conditions.push(condition);
    return this as any;
  }

  orderBy(column: string, direction: 'ASC' | 'DESC' = 'ASC'): this {
    this.config.orderBy = { column, direction };
    return this;
  }

  limit(n: number): this {
    this.config.limit = n;
    return this;
  }

  // Can only call build() if table AND condition are set
  build(this: QueryBuilder<true, true>): string {
    let sql = `SELECT * FROM ${this.config.table}`;
    if (this.config.conditions.length) {
      sql += ` WHERE ${this.config.conditions.join(' AND ')}`;
    }
    if (this.config.orderBy) {
      sql += ` ORDER BY ${this.config.orderBy.column} ${this.config.orderBy.direction}`;
    }
    if (this.config.limit) {
      sql += ` LIMIT ${this.config.limit}`;
    }
    return sql;
  }
}

class EntityAdapter {
  constructor(selectId = (entity) => entity.id) {
    this.selectId = selectId;
  }

  getInitialState(additionalState = {}) {
    return { ids: [], entities: {}, ...additionalState };
  }

  addOne(state, entity) {
    const id = this.selectId(entity);
    if (state.entities[id]) return state; // Already exists
    return {
      ...state,
      ids: [...state.ids, id],
      entities: { ...state.entities, [id]: entity }
    };
  }

  updateOne(state, { id, changes }) {
    if (!state.entities[id]) return state;
    return {
      ...state,
      entities: {
        ...state.entities,
        [id]: { ...state.entities[id], ...changes }
      }
    };
  }

  removeOne(state, id) {
    const { [id]: removed, ...entities } = state.entities;
    return {
      ...state,
      ids: state.ids.filter(i => i !== id),
      entities
    };
  }

  upsertMany(state, entities) {
    let newState = state;
    for (const entity of entities) {
      const id = this.selectId(entity);
      if (newState.entities[id]) {
        newState = this.updateOne(newState, { id, changes: entity });
      } else {
        newState = this.addOne(newState, entity);
      }
    }
    return newState;
  }

  // Selectors
  selectAll(state) { return state.ids.map(id => state.entities[id]); }
  selectById(state, id) { return state.entities[id]; }
  selectTotal(state) { return state.ids.length; }
}

6.2 Signals (The Future of Reactivity)

// ═══ Signals Implementation (Angular 16+ / SolidJS / Preact) ═══
// Fine-grained reactivity WITHOUT virtual DOM diffing

let currentEffect = null;

function createSignal(initialValue) {
  let value = initialValue;
  const subscribers = new Set();

  function read() {
    // Track which effect is reading this signal
    if (currentEffect) {
      subscribers.add(currentEffect);
    }
    return value;
  }

  function write(newValue) {
    if (Object.is(value, newValue)) return; // No change
    value = newValue;
    // Notify all subscribers (effects that depend on this signal)
    for (const effect of subscribers) {
      effect.execute();
    }
  }

  return [read, write];
}

function createEffect(fn) {
  const effect = {
    execute() {
      // Clean up old subscriptions
      currentEffect = effect;
      try {
        fn(); // During execution, signals track this effect
      } finally {
        currentEffect = null;
      }
    }
  };
  effect.execute(); // Run immediately to establish subscriptions
  return effect;
}

function createComputed(fn) {
  const [get, set] = createSignal(undefined);
  createEffect(() => set(fn()));
  return get;
}

// ═══ Usage (like Angular Signals) ═══
const [count, setCount] = createSignal(0);
const [multiplier, setMultiplier] = createSignal(2);

const doubled = createComputed(() => count() * multiplier());

createEffect(() => {
  console.log(`Count: ${count()}, Doubled: ${doubled()}`);
  // Only re-runs when count OR multiplier changes
  // NOT when unrelated signals change (unlike React re-renders!)
});

setCount(5);    // Logs: "Count: 5, Doubled: 10"
setMultiplier(3); // Logs: "Count: 5, Doubled: 15"

// React: Component re-renders entirely, THEN diffs virtual DOM
// Signals: Only the exact DOM nodes that depend on changed signals update
// Result: O(1) updates instead of O(n) tree diffing

      request.onupgradeneeded = (e) => {
        const db = e.target.result;
        if (!db.objectStoreNames.contains(this.storeName)) {
          db.createObjectStore(this.storeName, { keyPath: 'id' });
        }
        if (!db.objectStoreNames.contains(this.syncQueue)) {
          const store = db.createObjectStore(this.syncQueue, { keyPath: 'id', autoIncrement: true });
          store.createIndex('status', 'status');
        }
      };
      request.onsuccess = (e) => { this.db = e.target.result; resolve(this.db); };
      request.onerror = () => reject(request.error);
    });
  }

  // Write locally + queue for sync
  async put(item) {
    const tx = this.db.transaction([this.storeName, this.syncQueue], 'readwrite');
    
    // Save locally
    const localItem = { ...item, _localUpdatedAt: Date.now(), _synced: false };
    tx.objectStore(this.storeName).put(localItem);
    
    // Queue sync operation
    tx.objectStore(this.syncQueue).add({
      operation: 'PUT',
      storeName: this.storeName,
      data: item,
      timestamp: Date.now(),
      status: 'pending',
      retries: 0
    });

    await new Promise((resolve, reject) => {
      tx.oncomplete = resolve;
      tx.onerror = () => reject(tx.error);
    });

    // Try to sync immediately if online
    if (navigator.onLine) {
      this.processSyncQueue();
    }
  }

  async get(id) {
    return new Promise((resolve, reject) => {
      const tx = this.db.transaction(this.storeName, 'readonly');
      const request = tx.objectStore(this.storeName).get(id);
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }

  // Process queued operations when back online
  async processSyncQueue() {
    const tx = this.db.transaction(this.syncQueue, 'readonly');
    const index = tx.objectStore(this.syncQueue).index('status');
    const pending = await this.getAllFromIndex(index, 'pending');

    for (const op of pending) {
      try {
        await this.syncOperation(op);
        await this.updateSyncStatus(op.id, 'synced');
      } catch (e) {
        if (op.retries >= 3) {
          await this.updateSyncStatus(op.id, 'failed');
        } else {
          await this.incrementRetry(op.id);
        }
      }
    }
  }

  async syncOperation(op) {
    const response = await fetch(`/api/${op.storeName}`, {
      method: op.operation === 'PUT' ? 'PUT' : 'DELETE',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(op.data)
    });
    if (!response.ok) throw new Error(`Sync failed: ${response.status}`);
    return response.json();
  }

  getAllFromIndex(index, value) {
    return new Promise((resolve) => {
      const results = [];
      const request = index.openCursor(IDBKeyRange.only(value));
      request.onsuccess = (e) => {
        const cursor = e.target.result;
        if (cursor) { results.push(cursor.value); cursor.continue(); }
        else resolve(results);
      };
    });
  }

  async updateSyncStatus(id, status) {
    const tx = this.db.transaction(this.syncQueue, 'readwrite');
    const store = tx.objectStore(this.syncQueue);
    const item = await new Promise(r => { const req = store.get(id); req.onsuccess = () => r(req.result); });
    item.status = status;
    store.put(item);
  }

  async incrementRetry(id) {
    const tx = this.db.transaction(this.syncQueue, 'readwrite');
    const store = tx.objectStore(this.syncQueue);
    const item = await new Promise(r => { const req = store.get(id); req.onsuccess = () => r(req.result); });
    item.retries++;
    store.put(item);
  }
}

7.2 Conflict Resolution Strategies

// ═══ Last-Write-Wins (LWW) — Simplest strategy ═══
function resolveConflictLWW(local, remote) {
  return local._localUpdatedAt > remote.updatedAt ? local : remote;
}

// ═══ Three-Way Merge (like Git) ═══
function threeWayMerge(base, local, remote) {
  const result = { ...base };
  const conflicts = [];

  const allKeys = new Set([...Object.keys(local), ...Object.keys(remote)]);
  
  for (const key of allKeys) {
    const baseVal = base[key];
    const localVal = local[key];
    const remoteVal = remote[key];
    
    if (JSON.stringify(localVal) === JSON.stringify(remoteVal)) {
      // Both made same change — no conflict
      result[key] = localVal;
    } else if (JSON.stringify(localVal) === JSON.stringify(baseVal)) {
      // Only remote changed — accept remote
      result[key] = remoteVal;
    } else if (JSON.stringify(remoteVal) === JSON.stringify(baseVal)) {
      // Only local changed — accept local
      result[key] = localVal;
    } else {
      // CONFLICT: both changed differently
      conflicts.push({ key, base: baseVal, local: localVal, remote: remoteVal });
      result[key] = remoteVal; // Default to remote, flag for user resolution
    }
  }

  return { result, conflicts, hasConflicts: conflicts.length > 0 };
}

// ═══ CRDT: Counter (Conflict-free by design) ═══
// CRDTs guarantee convergence without coordination
class GCounter {
  constructor(nodeId) {
    this.nodeId = nodeId;
    this.counts = {}; // { nodeId: count }
  }

  increment(amount = 1) {
    this.counts[this.nodeId] = (this.counts[this.nodeId] || 0) + amount;
  }

  value() {
    return Object.values(this.counts).reduce((sum, c) => sum + c, 0);
  }

  merge(other) {
    // Take max of each node's counter — guarantees convergence
    for (const [nodeId, count] of Object.entries(other.counts)) {
      this.counts[nodeId] = Math.max(this.counts[nodeId] || 0, count);
    }
  }
}

// ═══ CRDT: LWW-Register (Last-Writer-Wins Register) ═══
class LWWRegister {
  constructor(nodeId) {
    this.nodeId = nodeId;
    this.value = null;
    this.timestamp = 0;
  }

  set(value) {
    this.value = value;
    this.timestamp = Date.now();
  }

  merge(other) {
    if (other.timestamp > this.timestamp) {
      this.value = other.value;
      this.timestamp = other.timestamp;
    }
    // If timestamps equal, use nodeId as tiebreaker (deterministic)
    if (other.timestamp === this.timestamp && other.nodeId > this.nodeId) {
      this.value = other.value;
    }
  }
}

class AccessibleCombobox {
  constructor(container) {
    this.container = container;
    this.input = container.querySelector('[role="combobox"]');
    this.listbox = container.querySelector('[role="listbox"]');
    this.activeIndex = -1;
    this.options = [];
    
    this.setup();
  }

  setup() {
    // Required ARIA attributes
    this.input.setAttribute('aria-expanded', 'false');
    this.input.setAttribute('aria-autocomplete', 'list');
    this.input.setAttribute('aria-controls', this.listbox.id);
    this.input.setAttribute('aria-activedescendant', '');
    
    this.input.addEventListener('keydown', (e) => this.handleKeydown(e));
    this.input.addEventListener('input', (e) => this.handleInput(e));
    this.input.addEventListener('focus', () => this.open());
    this.input.addEventListener('blur', () => setTimeout(() => this.close(), 150));
  }

  handleKeydown(e) {
    switch (e.key) {
      case 'ArrowDown':
        e.preventDefault();
        this.moveActive(1);
        break;
      case 'ArrowUp':
        e.preventDefault();
        this.moveActive(-1);
        break;
      case 'Enter':
        e.preventDefault();
        if (this.activeIndex >= 0) this.selectOption(this.activeIndex);
        break;
      case 'Escape':
        this.close();
        this.input.value = '';
        break;
      case 'Home':
        e.preventDefault();
        this.moveActive(-Infinity);
        break;
      case 'End':
        e.preventDefault();
        this.moveActive(Infinity);
        break;
    }
  }

  moveActive(delta) {
    const len = this.options.length;
    if (len === 0) return;
    
    if (delta === -Infinity) this.activeIndex = 0;
    else if (delta === Infinity) this.activeIndex = len - 1;
    else this.activeIndex = Math.max(0, Math.min(len - 1, this.activeIndex + delta));
    
    // Update ARIA
    this.options.forEach((opt, i) => {
      opt.setAttribute('aria-selected', i === this.activeIndex ? 'true' : 'false');
    });
    
    const activeOption = this.options[this.activeIndex];
    this.input.setAttribute('aria-activedescendant', activeOption.id);
    
    // Scroll into view
    activeOption.scrollIntoView({ block: 'nearest' });
    
    // Announce to screen reader
    this.announce(`${activeOption.textContent}, ${this.activeIndex + 1} of ${len}`);
  }

  announce(message) {
    // Live region for screen readers
    let live = document.getElementById('combobox-live');
    if (!live) {
      live = document.createElement('div');
      live.id = 'combobox-live';
      live.setAttribute('role', 'status');
      live.setAttribute('aria-live', 'polite');
      live.setAttribute('aria-atomic', 'true');
      live.className = 'sr-only'; // Visually hidden
      document.body.appendChild(live);
    }
    live.textContent = message;
  }

  open() {
    this.input.setAttribute('aria-expanded', 'true');
    this.listbox.hidden = false;
  }

  close() {
    this.input.setAttribute('aria-expanded', 'false');
    this.input.setAttribute('aria-activedescendant', '');
    this.listbox.hidden = true;
    this.activeIndex = -1;
  }

  selectOption(index) {
    const option = this.options[index];
    this.input.value = option.textContent;
    this.close();
    // Dispatch change event
    this.input.dispatchEvent(new Event('change', { bubbles: true }));
    this.announce(`${option.textContent} selected`);
  }
}

// ═══ Focus Trap (for Modals, Drawers) ═══
class FocusTrap {
  constructor(container) {
    this.container = container;
    this.previousFocus = null;
  }

  activate() {
    this.previousFocus = document.activeElement;
    
    // Find all focusable elements
    const focusable = this.getFocusableElements();
    if (focusable.length === 0) return;
    
    // Focus first element
    focusable[0].focus();
    
    // Trap focus within container
    this.handleKeydown = (e) => {
      if (e.key !== 'Tab') return;
      
      const focusable = this.getFocusableElements();
      const first = focusable[0];
      const last = focusable[focusable.length - 1];
      
      if (e.shiftKey && document.activeElement === first) {
        e.preventDefault();
        last.focus();
      } else if (!e.shiftKey && document.activeElement === last) {
        e.preventDefault();
        first.focus();
      }
    };
    
    document.addEventListener('keydown', this.handleKeydown);
  }

  deactivate() {
    document.removeEventListener('keydown', this.handleKeydown);
    // Return focus to trigger element
    this.previousFocus?.focus();
  }

  getFocusableElements() {
    const selector = [
      'a[href]', 'button:not([disabled])', 'input:not([disabled])',
      'select:not([disabled])', 'textarea:not([disabled])',
      '[tabindex]:not([tabindex="-1"])', '[contenteditable]'
    ].join(',');
    return [...this.container.querySelectorAll(selector)]
      .filter(el => !el.closest('[hidden]') && !el.closest('[aria-hidden="true"]'));
  }
}

// ═══ Roving Tabindex (for Toolbars, Tab Lists, Menus) ═══
class RovingTabindex {
  constructor(container, selector) {
    this.container = container;
    this.selector = selector;
    this.items = [...container.querySelectorAll(selector)];
    this.currentIndex = 0;
    
    this.init();
  }

  init() {
    // Only first item is in tab order
    this.items.forEach((item, i) => {
      item.setAttribute('tabindex', i === 0 ? '0' : '-1');
    });
    
    this.container.addEventListener('keydown', (e) => {
      let handled = true;
      switch (e.key) {
        case 'ArrowRight':
        case 'ArrowDown':
          this.move(1);
          break;
        case 'ArrowLeft':
        case 'ArrowUp':
          this.move(-1);
          break;
        case 'Home':
          this.moveTo(0);
          break;
        case 'End':
          this.moveTo(this.items.length - 1);
          break;
        default:
          handled = false;
      }
      if (handled) e.preventDefault();
    });
  }

  move(delta) {
    const newIndex = (this.currentIndex + delta + this.items.length) % this.items.length;
    this.moveTo(newIndex);
  }

  moveTo(index) {
    this.items[this.currentIndex].setAttribute('tabindex', '-1');
    this.currentIndex = index;
    this.items[this.currentIndex].setAttribute('tabindex', '0');
    this.items[this.currentIndex].focus();
  }
}

// ═══ Enterprise WebSocket with Reconnection, Heartbeat, Message Queue ═══
class WebSocketManager {
  constructor(url, options = {}) {
    this.url = url;
    this.options = {
      maxReconnectAttempts: 10,
      baseReconnectDelay: 1000,
      maxReconnectDelay: 30000,
      heartbeatInterval: 30000,
      heartbeatTimeout: 5000,
      messageQueueSize: 1000,
      protocols: [],
      ...options
    };
    
    this.ws = null;
    this.reconnectAttempts = 0;
    this.messageQueue = [];
    this.subscriptions = new Map();
    this.heartbeatTimer = null;
    this.heartbeatTimeoutTimer = null;
    this.state = 'disconnected'; // disconnected | connecting | connected | reconnecting
    this.listeners = { open: [], close: [], error: [], message: [], stateChange: [] };
  }

  connect() {
    if (this.state === 'connected' || this.state === 'connecting') return;
    this.setState('connecting');
    
    try {
      this.ws = new WebSocket(this.url, this.options.protocols);
      this.ws.binaryType = 'arraybuffer';
      
      this.ws.onopen = () => {
        this.setState('connected');
        this.reconnectAttempts = 0;
        this.startHeartbeat();
        this.flushQueue();
        this.emit('open');
      };
      
      this.ws.onmessage = (event) => {
        this.resetHeartbeatTimeout();
        const message = this.deserialize(event.data);
        
        if (message.type === 'pong') return; // Heartbeat response
        
        // Route to subscribers
        const handlers = this.subscriptions.get(message.type) || [];
        handlers.forEach(handler => handler(message.payload, message));
        
        this.emit('message', message);
      };
      
      this.ws.onclose = (event) => {
        this.stopHeartbeat();
        this.emit('close', { code: event.code, reason: event.reason });
        
        if (!event.wasClean && this.state !== 'disconnected') {
          this.reconnect();
        }
      };
      
      this.ws.onerror = (error) => {
        this.emit('error', error);
      };
    } catch (e) {
      this.reconnect();
    }
  }

  send(type, payload, options = {}) {
    const message = { type, payload, id: crypto.randomUUID(), timestamp: Date.now() };
    
    if (this.state === 'connected' && this.ws.readyState === WebSocket.OPEN) {
      this.ws.send(this.serialize(message));
    } else if (options.queue !== false) {
      // Queue for when connection is restored
      if (this.messageQueue.length >= this.options.messageQueueSize) {
        this.messageQueue.shift(); // Drop oldest
      }
      this.messageQueue.push(message);
    }
    
    return message.id;
  }

  subscribe(type, handler) {
    if (!this.subscriptions.has(type)) this.subscriptions.set(type, []);
    this.subscriptions.get(type).push(handler);
    return () => {
      const handlers = this.subscriptions.get(type);
      const idx = handlers.indexOf(handler);
      if (idx !== -1) handlers.splice(idx, 1);
    };
  }

  reconnect() {
    if (this.reconnectAttempts >= this.options.maxReconnectAttempts) {
      this.setState('disconnected');
      this.emit('error', new Error('Max reconnection attempts reached'));
      return;
    }
    
    this.setState('reconnecting');
    this.reconnectAttempts++;
    
    // Exponential backoff with jitter
    const delay = Math.min(
      this.options.baseReconnectDelay * 2 ** (this.reconnectAttempts - 1),
      this.options.maxReconnectDelay
    );
    const jitter = delay * 0.3 * Math.random();
    
    setTimeout(() => this.connect(), delay + jitter);
  }

  startHeartbeat() {
    this.heartbeatTimer = setInterval(() => {
      if (this.state === 'connected') {
        this.ws.send(this.serialize({ type: 'ping', timestamp: Date.now() }));
        this.heartbeatTimeoutTimer = setTimeout(() => {
          // No pong received — connection is dead
          this.ws.close(4000, 'Heartbeat timeout');
        }, this.options.heartbeatTimeout);
      }
    }, this.options.heartbeatInterval);
  }

  resetHeartbeatTimeout() {
    if (this.heartbeatTimeoutTimer) {
      clearTimeout(this.heartbeatTimeoutTimer);
      this.heartbeatTimeoutTimer = null;
    }
  }

  stopHeartbeat() {
    clearInterval(this.heartbeatTimer);
    this.resetHeartbeatTimeout();
  }

  flushQueue() {
    while (this.messageQueue.length > 0 && this.state === 'connected') {
      const message = this.messageQueue.shift();
      this.ws.send(this.serialize(message));
    }
  }

  disconnect() {
    this.setState('disconnected');
    this.stopHeartbeat();
    if (this.ws) {
      this.ws.close(1000, 'Client disconnect');
      this.ws = null;
    }
  }

  setState(state) {
    const prev = this.state;
    this.state = state;
    this.emit('stateChange', { from: prev, to: state });
  }

  serialize(data) { return JSON.stringify(data); }
  deserialize(data) { return JSON.parse(data); }
  
  on(event, handler) { this.listeners[event]?.push(handler); }
  emit(event, data) { (this.listeners[event] || []).forEach(h => h(data)); }
}

class EventSourceManager {

    this.url = url;
    this.lastEventId = options.lastEventId || null;
    this.handlers = new Map();
    this.connect();
  }

    if (this.lastEventId) url.searchParams.set('lastEventId', this.lastEventId);

      this.lastEventId = e.lastEventId;

    };
    
    // Named events
    for (const [event, handlers] of this.handlers) {
      this.source.addEventListener(event, (e) => {
        this.lastEventId = e.lastEventId;
        handlers.forEach(h => h(JSON.parse(e.data)));
      });
    }
    
    this.source.onerror = () => {
      // EventSource auto-reconnects, but we track state
      console.log('SSE reconnecting...');
    };
  }

  on(event, handler) {
    if (!this.handlers.has(event)) this.handlers.set(event, []);
    this.handlers.get(event).push(handler);
    
    // Add to existing source
    this.source.addEventListener(event, (e) => {
      this.lastEventId = e.lastEventId;
      handler(JSON.parse(e.data));
    });
  }

    (this.handlers.get(event) || []).forEach(h => h(data));
  }

async function compareBundles(baseStats, prStats) {
  const changes = [];
  
  for (const [chunk, size] of Object.entries(prStats.chunks)) {
    const baseSize = baseStats.chunks[chunk] || 0;
    const diff = size - baseSize;
    const percentChange = baseSize ? ((diff / baseSize) * 100).toFixed(1) : 'NEW';
    
    if (Math.abs(diff) > 1024) { // Only report > 1KB changes
      changes.push({
        chunk,
        baseSize: formatBytes(baseSize),
        prSize: formatBytes(size),
        diff: formatBytes(diff),
        percentChange,
        status: diff > 0 ? '🔴' : '🟢'
      });
    }
  }
  
  // Generate markdown table for PR comment
  let report = '## Bundle Size Report\n\n';
  report += '| Chunk | Base | PR | Diff | % |\n';
  report += '|-------|------|----|------|---|\n';
  changes.forEach(c => {
    report += `| ${c.status} ${c.chunk} | ${c.baseSize} | ${c.prSize} | ${c.diff} | ${c.percentChange}% |\n`;
  });
  
  const totalBase = Object.values(baseStats.chunks).reduce((s, v) => s + v, 0);
  const totalPR = Object.values(prStats.chunks).reduce((s, v) => s + v, 0);
  report += `\n**Total:** ${formatBytes(totalBase)} → ${formatBytes(totalPR)} (${formatBytes(totalPR - totalBase)})\n`;
  
  return { report, hasRegressions: changes.some(c => c.status === '🔴') };
}

function formatBytes(bytes) {
  if (bytes === 0) return '0 B';
  const k = 1024;
  const sizes = ['B', 'KB', 'MB'];
  const i = Math.floor(Math.log(Math.abs(bytes)) / Math.log(k));
  return (bytes / Math.pow(k, i)).toFixed(1) + ' ' + sizes[i];
}

// ═══ Production Feature Flag Architecture ═══
class FeatureFlagClient {
  constructor({ endpoint, refreshInterval = 60000, userId }) {
    this.endpoint = endpoint;
    this.userId = userId;
    this.flags = new Map();
    this.overrides = new Map(); // Local dev overrides
    this.listeners = new Map();
    this.ready = this.initialize();
    
    // Periodic refresh
    setInterval(() => this.refresh(), refreshInterval);
    
    // Real-time updates via SSE
    this.connectSSE();
  }

  async initialize() {
    try {
      const response = await fetch(`${this.endpoint}/flags?userId=${this.userId}`);
      const data = await response.json();
      this.updateFlags(data.flags);
    } catch (e) {
      // Fall back to localStorage cache
      const cached = localStorage.getItem('feature-flags');
      if (cached) this.updateFlags(JSON.parse(cached));
    }
  }

  isEnabled(flagName, defaultValue = false) {
    // Dev override takes priority
    if (this.overrides.has(flagName)) return this.overrides.get(flagName);
    
    const flag = this.flags.get(flagName);
    if (!flag) return defaultValue;
    
    // Check targeting rules
    return this.evaluateFlag(flag);
  }

  evaluateFlag(flag) {
    if (!flag.enabled) return false;
    
    // Percentage rollout
    if (flag.percentage !== undefined) {
      const hash = this.hashUser(this.userId + flag.name);
      return hash < flag.percentage;
    }
    
    // User targeting
    if (flag.targetUsers?.includes(this.userId)) return true;
    
    // Segment targeting
    if (flag.segments) {
      return flag.segments.some(segment => this.userMatchesSegment(segment));
    }
    
    return flag.enabled;
  }

  // Deterministic hash (same user always gets same result)
  hashUser(input) {
    let hash = 0;
    for (let i = 0; i < input.length; i++) {
      const char = input.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash; // Convert to 32-bit integer
    }
    return Math.abs(hash) % 100;
  }

  // Subscribe to flag changes (for reactive UI)
  onChange(flagName, handler) {
    if (!this.listeners.has(flagName)) this.listeners.set(flagName, []);
    this.listeners.get(flagName).push(handler);
    return () => {
      const handlers = this.listeners.get(flagName);
      const idx = handlers.indexOf(handler);
      if (idx !== -1) handlers.splice(idx, 1);
    };
  }

  updateFlags(newFlags) {
    for (const [name, config] of Object.entries(newFlags)) {
      const oldEnabled = this.isEnabled(name);
      this.flags.set(name, config);
      const newEnabled = this.isEnabled(name);
      
      if (oldEnabled !== newEnabled) {
        (this.listeners.get(name) || []).forEach(h => h(newEnabled));
      }
    }
    // Cache in localStorage
    localStorage.setItem('feature-flags', JSON.stringify(Object.fromEntries(this.flags)));
  }

  connectSSE() {
    const source = new EventSource(`${this.endpoint}/stream?userId=${this.userId}`);
    source.addEventListener('flag-update', (e) => {
      const update = JSON.parse(e.data);
      this.updateFlags({ [update.name]: update.config });
    });
  }

  // Development helpers
  override(flagName, value) { this.overrides.set(flagName, value); }
  clearOverrides() { this.overrides.clear(); }
}
