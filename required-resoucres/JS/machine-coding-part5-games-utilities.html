class CommentSystem {
  constructor(container, options = {}) {
    this.container = container;
    this.comments = options.comments || [];
    this.currentUser = options.currentUser || { id: "user1", name: "You" };
    this.maxDepth = options.maxDepth || 5;
    this.onAdd = options.onAdd || (() => {});
    this.onDelete = options.onDelete || (() => {});
    this.onEdit = options.onEdit || (() => {});

    this.replyingTo = null;
    this.editingId = null;
    this.collapsedIds = new Set();
    this.nextId = this.getMaxId(this.comments) + 1;

    this.render();
  }

  getMaxId(comments) {
    let max = 0;
    comments.forEach((c) => {
      max = Math.max(max, c.id);
      if (c.replies) max = Math.max(max, this.getMaxId(c.replies));
    });
    return max;
  }

  render() {
    this.container.innerHTML = "";
    this.container.style.cssText = "font-family:sans-serif;max-width:700px";

    // Top-level comment input
    const topInput = this.createCommentInput(null);
    this.container.appendChild(topInput);

    // Render comment tree
    const tree = this.renderComments(this.comments, 0);
    this.container.appendChild(tree);
  }

  renderComments(comments, depth) {
    const wrapper = document.createElement("div");

    comments.forEach((comment) => {
      const commentEl = document.createElement("div");
      commentEl.className = "comment";
      commentEl.style.cssText = `
        margin-left:${depth * 24}px;margin-top:12px;
        border-left:${depth > 0 ? "2px solid #e5e7eb" : "none"};
        padding-left:${depth > 0 ? "12px" : "0"};
      `;

      const isCollapsed = this.collapsedIds.has(comment.id);

      // Header
      const header = document.createElement("div");
      header.style.cssText = "display:flex;align-items:center;gap:8px;margin-bottom:4px";

      // Collapse toggle
      if (comment.replies && comment.replies.length > 0) {
        const toggle = document.createElement("button");
        toggle.textContent = isCollapsed ? "[+]" : "[\u2013]";
        toggle.style.cssText = "border:none;background:none;cursor:pointer;color:#6b7280;font-size:12px;font-family:monospace;padding:0";
        toggle.addEventListener("click", () => {
          if (isCollapsed) this.collapsedIds.delete(comment.id);
          else this.collapsedIds.add(comment.id);
          this.render();
        });
        header.appendChild(toggle);
      }

      const author = document.createElement("span");
      author.textContent = comment.author.name;
      author.style.cssText = "font-weight:600;font-size:13px;color:#1f2937";
      header.appendChild(author);

      const time = document.createElement("span");
      time.textContent = this.timeAgo(comment.timestamp);
      time.style.cssText = "font-size:11px;color:#9ca3af";
      header.appendChild(time);

      if (comment.edited) {
        const edited = document.createElement("span");
        edited.textContent = "(edited)";
        edited.style.cssText = "font-size:11px;color:#9ca3af;font-style:italic";
        header.appendChild(edited);
      }

      commentEl.appendChild(header);

      if (!isCollapsed) {
        // Body
        if (this.editingId === comment.id) {
          const editInput = this.createEditInput(comment);
          commentEl.appendChild(editInput);
        } else {
          const body = document.createElement("p");
          body.textContent = comment.text;
          body.style.cssText = "margin:4px 0 8px;font-size:14px;line-height:1.5;color:#374151";
          commentEl.appendChild(body);
        }

        // Actions
        const actions = document.createElement("div");
        actions.style.cssText = "display:flex;gap:12px;font-size:12px";

        // Reply button (if not at max depth)
        if (depth < this.maxDepth) {
          const replyBtn = document.createElement("button");
          replyBtn.textContent = "Reply";
          replyBtn.style.cssText = "border:none;background:none;cursor:pointer;color:#6b7280;font-size:12px;font-weight:500";
          replyBtn.addEventListener("click", () => {
            this.replyingTo = this.replyingTo === comment.id ? null : comment.id;
            this.render();
          });
          actions.appendChild(replyBtn);
        }

        // Edit/Delete (only own comments)
        if (comment.author.id === this.currentUser.id) {
          const editBtn = document.createElement("button");
          editBtn.textContent = "Edit";
          editBtn.style.cssText = "border:none;background:none;cursor:pointer;color:#6b7280;font-size:12px";
          editBtn.addEventListener("click", () => {
            this.editingId = comment.id;
            this.render();
          });
          actions.appendChild(editBtn);

          const deleteBtn = document.createElement("button");
          deleteBtn.textContent = "Delete";
          deleteBtn.style.cssText = "border:none;background:none;cursor:pointer;color:#dc3545;font-size:12px";
          deleteBtn.addEventListener("click", () => {
            if (confirm("Delete this comment?")) {
              this.deleteComment(this.comments, comment.id);
              this.onDelete(comment.id);
              this.render();
            }
          });
          actions.appendChild(deleteBtn);
        }

        commentEl.appendChild(actions);

        // Reply input
        if (this.replyingTo === comment.id) {
          const replyInput = this.createCommentInput(comment.id);
          commentEl.appendChild(replyInput);
        }

        // Nested replies
        if (comment.replies && comment.replies.length > 0) {
          const replies = this.renderComments(comment.replies, depth + 1);
          commentEl.appendChild(replies);
        }
      } else {
        // Collapsed summary
        const summary = document.createElement("span");
        const replyCount = this.countReplies(comment);
        summary.textContent = `${replyCount} ${replyCount === 1 ? "reply" : "replies"} hidden`;
        summary.style.cssText = "font-size:11px;color:#9ca3af;font-style:italic";
        commentEl.appendChild(summary);
      }

      wrapper.appendChild(commentEl);
    });

    return wrapper;
  }

  createCommentInput(parentId) {
    const wrapper = document.createElement("div");
    wrapper.style.cssText = "margin:12px 0;display:flex;gap:8px";

    const textarea = document.createElement("textarea");
    textarea.placeholder = parentId ? "Write a reply..." : "Add a comment...";
    textarea.rows = parentId ? 2 : 3;
    textarea.style.cssText = `
      flex:1;padding:10px 12px;border:1px solid #d1d5db;border-radius:6px;
      font-size:13px;font-family:sans-serif;resize:vertical;outline:none;
      transition:border-color 0.2s;
    `;
    textarea.addEventListener("focus", () => { textarea.style.borderColor = "#2563eb"; });
    textarea.addEventListener("blur", () => { textarea.style.borderColor = "#d1d5db"; });

    const submitBtn = document.createElement("button");
    submitBtn.textContent = parentId ? "Reply" : "Comment";
    submitBtn.style.cssText = `
      padding:8px 16px;background:#2563eb;color:white;border:none;
      border-radius:6px;cursor:pointer;font-size:13px;font-weight:500;
      align-self:flex-end;
    `;
    submitBtn.addEventListener("click", () => {
      const text = textarea.value.trim();
      if (!text) return;

      const newComment = {
        id: this.nextId++,
        text,
        author: this.currentUser,
        timestamp: Date.now(),
        replies: [],
      };

      if (parentId) {
        this.addReply(this.comments, parentId, newComment);
      } else {
        this.comments.unshift(newComment);
      }

      this.replyingTo = null;
      this.onAdd(newComment, parentId);
      this.render();
    });

    wrapper.appendChild(textarea);
    wrapper.appendChild(submitBtn);
    return wrapper;
  }

  createEditInput(comment) {
    const wrapper = document.createElement("div");
    wrapper.style.cssText = "margin:4px 0 8px";

    const textarea = document.createElement("textarea");
    textarea.value = comment.text;
    textarea.rows = 2;
    textarea.style.cssText = `
      width:100%;padding:8px 10px;border:1px solid #2563eb;border-radius:4px;
      font-size:13px;font-family:sans-serif;resize:vertical;outline:none;
    `;

    const btnRow = document.createElement("div");
    btnRow.style.cssText = "display:flex;gap:8px;margin-top:6px";

    const saveBtn = document.createElement("button");
    saveBtn.textContent = "Save";
    saveBtn.style.cssText = "padding:4px 12px;background:#2563eb;color:white;border:none;border-radius:4px;font-size:12px;cursor:pointer";
    saveBtn.addEventListener("click", () => {
      const newText = textarea.value.trim();
      if (newText && newText !== comment.text) {
        comment.text = newText;
        comment.edited = true;
        this.onEdit(comment.id, newText);
      }
      this.editingId = null;
      this.render();
    });

    const cancelBtn = document.createElement("button");
    cancelBtn.textContent = "Cancel";
    cancelBtn.style.cssText = "padding:4px 12px;background:#f3f4f6;border:1px solid #d1d5db;border-radius:4px;font-size:12px;cursor:pointer";
    cancelBtn.addEventListener("click", () => {
      this.editingId = null;
      this.render();
    });

    btnRow.appendChild(saveBtn);
    btnRow.appendChild(cancelBtn);
    wrapper.appendChild(textarea);
    wrapper.appendChild(btnRow);
    return wrapper;
  }

  addReply(comments, parentId, newComment) {
    for (const comment of comments) {
      if (comment.id === parentId) {
        if (!comment.replies) comment.replies = [];
        comment.replies.push(newComment);
        return true;
      }
      if (comment.replies && this.addReply(comment.replies, parentId, newComment)) {
        return true;
      }
    }
    return false;
  }

  deleteComment(comments, id) {
    const index = comments.findIndex((c) => c.id === id);
    if (index !== -1) {
      comments.splice(index, 1);
      return true;
    }
    for (const comment of comments) {
      if (comment.replies && this.deleteComment(comment.replies, id)) {
        return true;
      }
    }
    return false;
  }

  countReplies(comment) {
    if (!comment.replies) return 0;
    return comment.replies.reduce((sum, r) => sum + 1 + this.countReplies(r), 0);
  }

  timeAgo(timestamp) {
    const seconds = Math.floor((Date.now() - timestamp) / 1000);
    if (seconds < 60) return "just now";
    if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`;
    if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`;
    return `${Math.floor(seconds / 86400)}d ago`;
  }
}

Interview Follow-ups

  • Recursive vs iterative rendering: Recursive is cleaner, but stack overflow risk at 100+ depth — use iterative with explicit stack
  • Optimistic updates: Show comment immediately, revert on API failure
  • Pagination: "Load more replies" button at depth limits or for 10+ replies
  • Real-time: WebSocket for live comment updates from other users

class MultiStepForm {
  constructor(container, options = {}) {
    this.container = container;
    this.steps = options.steps || [];
    this.onSubmit = options.onSubmit || (() => {});
    this.onStepChange = options.onStepChange || (() => {});
    this.allowSkip = options.allowSkip || false;

    this.currentStep = 0;
    this.formData = {};
    this.errors = {};
    this.visited = new Set([0]);

    this.render();
  }

  render() {
    this.container.innerHTML = "";
    this.container.style.cssText = "font-family:sans-serif;max-width:600px;margin:0 auto";

    // Progress indicator
    const progress = this.renderProgress();
    this.container.appendChild(progress);

    // Step content
    const stepWrapper = document.createElement("div");
    stepWrapper.style.cssText = "padding:24px;border:1px solid #e5e7eb;border-radius:8px;margin-top:20px;background:white";

    const step = this.steps[this.currentStep];

    // Step title
    const title = document.createElement("h3");
    title.textContent = step.title;
    title.style.cssText = "margin:0 0 4px;font-size:18px;color:#1f2937";
    stepWrapper.appendChild(title);

    if (step.description) {
      const desc = document.createElement("p");
      desc.textContent = step.description;
      desc.style.cssText = "margin:0 0 20px;font-size:13px;color:#6b7280";
      stepWrapper.appendChild(desc);
    }

    // Render fields
    step.fields.forEach((field) => {
      const fieldEl = this.renderField(field);
      stepWrapper.appendChild(fieldEl);
    });

    // Navigation buttons
    const nav = document.createElement("div");
    nav.style.cssText = "display:flex;justify-content:space-between;margin-top:24px;padding-top:16px;border-top:1px solid #f3f4f6";

    if (this.currentStep > 0) {
      const prevBtn = document.createElement("button");
      prevBtn.textContent = "\u2190 Back";
      prevBtn.style.cssText = "padding:10px 20px;border:1px solid #d1d5db;border-radius:6px;background:white;cursor:pointer;font-size:14px";
      prevBtn.addEventListener("click", () => this.prev());
      nav.appendChild(prevBtn);
    } else {
      nav.appendChild(document.createElement("span"));
    }

    const isLast = this.currentStep === this.steps.length - 1;
    const nextBtn = document.createElement("button");
    nextBtn.textContent = isLast ? "Submit \u2713" : "Next \u2192";
    nextBtn.style.cssText = `
      padding:10px 24px;border:none;border-radius:6px;cursor:pointer;font-size:14px;
      font-weight:600;color:white;
      background:${isLast ? "#16a34a" : "#2563eb"};
    `;
    nextBtn.addEventListener("click", () => {
      if (isLast) this.submit();
      else this.next();
    });
    nav.appendChild(nextBtn);

    stepWrapper.appendChild(nav);
    this.container.appendChild(stepWrapper);

    // Summary (on last step, show all collected data)
    if (isLast && Object.keys(this.formData).length > 0) {
      const summary = this.renderSummary();
      this.container.appendChild(summary);
    }
  }

  renderProgress() {
    const wrapper = document.createElement("div");
    wrapper.style.cssText = "display:flex;align-items:center;justify-content:center;gap:0";

    this.steps.forEach((step, i) => {
      const isActive = i === this.currentStep;
      const isCompleted = i < this.currentStep;
      const isVisited = this.visited.has(i);

      const stepEl = document.createElement("div");
      stepEl.style.cssText = "display:flex;flex-direction:column;align-items:center;position:relative;min-width:80px";

      const circle = document.createElement("div");
      circle.textContent = isCompleted ? "\u2713" : i + 1;
      circle.style.cssText = `
        width:32px;height:32px;border-radius:50%;display:flex;align-items:center;
        justify-content:center;font-size:13px;font-weight:600;
        border:2px solid ${isActive ? "#2563eb" : isCompleted ? "#16a34a" : "#d1d5db"};
        background:${isActive ? "#2563eb" : isCompleted ? "#16a34a" : "white"};
        color:${isActive || isCompleted ? "white" : "#6b7280"};
        transition:all 0.3s ease;
      `;

      const label = document.createElement("span");
      label.textContent = step.title;
      label.style.cssText = `
        font-size:11px;margin-top:4px;color:${isActive ? "#2563eb" : "#6b7280"};
        font-weight:${isActive ? "600" : "400"};text-align:center;
      `;

      stepEl.appendChild(circle);
      stepEl.appendChild(label);
      wrapper.appendChild(stepEl);

      // Connector line
      if (i < this.steps.length - 1) {
        const line = document.createElement("div");
        line.style.cssText = `
          height:2px;width:40px;margin:0 4px;margin-bottom:18px;
          background:${i < this.currentStep ? "#16a34a" : "#e5e7eb"};
          transition:background 0.3s;
        `;
        wrapper.appendChild(line);
      }
    });

    return wrapper;
  }

  renderField(field) {
    const wrapper = document.createElement("div");
    wrapper.style.cssText = "margin-bottom:16px";

    const label = document.createElement("label");
    label.textContent = field.label + (field.required ? " *" : "");
    label.style.cssText = "display:block;font-size:13px;font-weight:500;color:#374151;margin-bottom:4px";
    wrapper.appendChild(label);

    let input;
    const value = this.formData[field.name] || field.defaultValue || "";

    switch (field.type) {
      case "select":
        input = document.createElement("select");
        input.style.cssText = "width:100%;padding:10px;border:1px solid #d1d5db;border-radius:6px;font-size:14px;outline:none";
        const defaultOpt = document.createElement("option");
        defaultOpt.value = "";
        defaultOpt.textContent = field.placeholder || "Select...";
        input.appendChild(defaultOpt);
        field.options.forEach((opt) => {
          const option = document.createElement("option");
          option.value = opt.value || opt;
          option.textContent = opt.label || opt;
          if (value === (opt.value || opt)) option.selected = true;
          input.appendChild(option);
        });
        break;

      case "textarea":
        input = document.createElement("textarea");
        input.rows = 3;
        input.value = value;
        input.placeholder = field.placeholder || "";
        input.style.cssText = "width:100%;padding:10px;border:1px solid #d1d5db;border-radius:6px;font-size:14px;outline:none;resize:vertical;font-family:inherit";
        break;

      case "radio":
        input = document.createElement("div");
        input.style.cssText = "display:flex;flex-wrap:wrap;gap:12px;margin-top:4px";
        field.options.forEach((opt) => {
          const radioLabel = document.createElement("label");
          radioLabel.style.cssText = "display:flex;align-items:center;gap:6px;cursor:pointer;font-size:14px";
          const radio = document.createElement("input");
          radio.type = "radio";
          radio.name = field.name;
          radio.value = opt.value || opt;
          if (value === (opt.value || opt)) radio.checked = true;
          radio.addEventListener("change", () => {
            this.formData[field.name] = radio.value;
          });
          radioLabel.appendChild(radio);
          radioLabel.appendChild(document.createTextNode(opt.label || opt));
          input.appendChild(radioLabel);
        });
        wrapper.appendChild(input);
        // Show error if exists
        if (this.errors[field.name]) {
          const err = document.createElement("span");
          err.textContent = this.errors[field.name];
          err.style.cssText = "color:#dc3545;font-size:12px;margin-top:4px;display:block";
          wrapper.appendChild(err);
        }
        return wrapper;

      default: // text, email, number, tel, date
        input = document.createElement("input");
        input.type = field.type || "text";
        input.value = value;
        input.placeholder = field.placeholder || "";
        input.style.cssText = "width:100%;padding:10px;border:1px solid #d1d5db;border-radius:6px;font-size:14px;outline:none";
    }

    if (input.tagName !== "DIV") {
      input.addEventListener("input", (e) => {
        this.formData[field.name] = e.target.value;
        // Clear error on input
        if (this.errors[field.name]) {
          delete this.errors[field.name];
          const errEl = wrapper.querySelector(".field-error");
          if (errEl) errEl.remove();
        }
      });
      input.addEventListener("change", (e) => {
        this.formData[field.name] = e.target.value;
      });
      wrapper.appendChild(input);
    }

    // Error message
    if (this.errors[field.name]) {
      const err = document.createElement("span");
      err.className = "field-error";
      err.textContent = this.errors[field.name];
      err.style.cssText = "color:#dc3545;font-size:12px;margin-top:4px;display:block";
      wrapper.appendChild(err);
      if (input.style) input.style.borderColor = "#dc3545";
    }

    return wrapper;
  }

  renderSummary() {
    const wrapper = document.createElement("div");
    wrapper.style.cssText = "margin-top:20px;padding:16px;background:#f9fafb;border:1px solid #e5e7eb;border-radius:8px";

    const title = document.createElement("h4");
    title.textContent = "Summary";
    title.style.cssText = "margin:0 0 12px;font-size:14px;color:#374151";
    wrapper.appendChild(title);

    Object.entries(this.formData).forEach(([key, value]) => {
      if (!value) return;
      const row = document.createElement("div");
      row.style.cssText = "display:flex;justify-content:space-between;padding:4px 0;font-size:13px;border-bottom:1px solid #f3f4f6";
      const keyEl = document.createElement("span");
      keyEl.textContent = key.replace(/([A-Z])/g, " $1").replace(/^./, (s) => s.toUpperCase());
      keyEl.style.color = "#6b7280";
      const valEl = document.createElement("span");
      valEl.textContent = value;
      valEl.style.fontWeight = "500";
      row.appendChild(keyEl);
      row.appendChild(valEl);
      wrapper.appendChild(row);
    });

    return wrapper;
  }

  validateStep() {
    const step = this.steps[this.currentStep];
    this.errors = {};
    let isValid = true;

    step.fields.forEach((field) => {
      const value = this.formData[field.name] || "";

      if (field.required && !value.trim()) {
        this.errors[field.name] = `${field.label} is required`;
        isValid = false;
      } else if (field.validate && value) {
        const error = field.validate(value, this.formData);
        if (error) {
          this.errors[field.name] = error;
          isValid = false;
        }
      }
    });

    return isValid;
  }

  next() {
    if (!this.validateStep()) {
      this.render();
      return;
    }
    this.currentStep = Math.min(this.currentStep + 1, this.steps.length - 1);
    this.visited.add(this.currentStep);
    this.onStepChange(this.currentStep, this.formData);
    this.render();
  }

  prev() {
    this.currentStep = Math.max(this.currentStep - 1, 0);
    this.render();
  }

  goTo(step) {
    if (this.allowSkip || this.visited.has(step)) {
      this.currentStep = step;
      this.render();
    }
  }

  submit() {
    if (!this.validateStep()) {
      this.render();
      return;
    }
    this.onSubmit(this.formData);
  }

  getData() { return { ...this.formData }; }
  reset() {
    this.formData = {};
    this.errors = {};
    this.currentStep = 0;
    this.visited = new Set([0]);
    this.render();
  }
}

  init() {
    const totalCards = this.gridSize * this.gridSize;
    const pairsNeeded = totalCards / 2;

  flipCard(index) {
    // Guards
    if (this.isLocked) return;
    if (this.flippedCards.includes(index)) return;
    if (this.matchedPairs.has(this.cards[index].symbol)) return;
    if (this.flippedCards.length >= 2) return;

    // Start timer on first flip
    if (!this.startTime) {
      this.startTime = Date.now();
      this.timerInterval = setInterval(() => this.updateTimer(), 1000);
    }

    this.flippedCards.push(index);
    this.render();

    // Check for match when 2 cards are flipped
    if (this.flippedCards.length === 2) {
      this.moves++;
      this.movesEl.textContent = `Moves: ${this.moves}`;
      this.checkMatch();
    }
  }

  checkMatch() {
    const [first, second] = this.flippedCards;
    const card1 = this.cards[first];
    const card2 = this.cards[second];

    if (card1.symbol === card2.symbol) {
      // Match found!
      this.matchedPairs.add(card1.symbol);
      this.flippedCards = [];

      // Check win condition
      if (this.matchedPairs.size === this.cards.length / 2) {
        clearInterval(this.timerInterval);
        this.onWin({ moves: this.moves, time: this.getElapsedTime() });
      }
      this.render();
    } else {
      // No match — flip back after delay
      this.isLocked = true;
      setTimeout(() => {
        this.flippedCards = [];
        this.isLocked = false;
        this.render();
      }, 800);
    }
  }

  updateTimer() {
    if (this.timerEl) {
      this.timerEl.textContent = `Time: ${this.getElapsedTime()}`;
    }
  }

  getElapsedTime() {
    if (!this.startTime) return "0:00";
    const elapsed = Math.floor((Date.now() - this.startTime) / 1000);
    const min = Math.floor(elapsed / 60);
    const sec = (elapsed % 60).toString().padStart(2, "0");
    return `${min}:${sec}`;
  }

  showWinMessage() {
    const overlay = document.createElement("div");
    overlay.style.cssText = `
      margin-top:16px;padding:20px;background:#d4edda;border-radius:8px;
      text-align:center;border:1px solid #c3e6cb;
    `;
    overlay.innerHTML = `
      <h3 style="margin:0 0 8px;color:#155724">\ud83c\udf89 Congratulations!</h3>
      <p style="margin:0;color:#155724;font-size:14px">
        Completed in ${this.moves} moves and ${this.getElapsedTime()}
      </p>
    `.replace(/</g, '<').replace(/>/g, '>');
    // Note: Using DOM methods in real implementation
    const h3 = document.createElement("h3");
    h3.textContent = "\ud83c\udf89 Congratulations!";
    h3.style.cssText = "margin:0 0 8px;color:#155724";
    const p = document.createElement("p");
    p.textContent = `Completed in ${this.moves} moves and ${this.getElapsedTime()}`;
    p.style.cssText = "margin:0;color:#155724;font-size:14px";
    overlay.innerHTML = "";
    overlay.appendChild(h3);
    overlay.appendChild(p);
    this.container.appendChild(overlay);
  }

  reset() {
    clearInterval(this.timerInterval);
    this.flippedCards = [];
    this.matchedPairs = new Set();
    this.moves = 0;
    this.startTime = null;
    this.isLocked = false;
    // Reshuffle
    this.cards.sort(() => Math.random() - 0.5);
    this.render();
  }
}

    clearInterval(this.intervalId);
    this.render();
  }

    this.isPaused = false;
    this.render();

    // Resume with remaining time
    this.intervalId = setInterval(() => {
      this.remainingTime -= 100;
      if (this.countdownEl) {
        this.countdownEl.textContent = `${Math.max(0, Math.ceil(this.remainingTime / 1000))}s`;
      }
    }, 100);

    this.timeoutId = setTimeout(() => {
      clearInterval(this.intervalId);
      this.currentIndex = (this.currentIndex + 1) % this.config.length;
      this.cycle();
    }, this.remainingTime);
  }

  triggerPedestrian() {
    if (this.isPedestrianActive) return;
    this.isPedestrianActive = true;

    // Interrupt current cycle — force RED for vehicles
    clearTimeout(this.timeoutId);
    clearInterval(this.intervalId);

    // Set to red (pedestrian walk)
    this.currentIndex = 0; // Red
    this.remainingTime = 6000; // 6 seconds for pedestrian
    this.render();

    this.intervalId = setInterval(() => {
      this.remainingTime -= 100;
      if (this.countdownEl) {
        this.countdownEl.textContent = `${Math.max(0, Math.ceil(this.remainingTime / 1000))}s`;
      }
    }, 100);

    this.timeoutId = setTimeout(() => {
      clearInterval(this.intervalId);
      this.isPedestrianActive = false;
      // Resume from yellow (transitioning back to green)
      this.currentIndex = 1;
      this.cycle();
    }, 6000);
  }

  destroy() {
    clearTimeout(this.timeoutId);
    clearInterval(this.intervalId);
  }
}

class SnakeGame {
  constructor(canvas, options = {}) {
    this.canvas = canvas;
    this.ctx = canvas.getContext('2d');
    this.gridSize = options.gridSize || 20;
    this.speed = options.speed || 150; // ms per frame
    this.cols = Math.floor(canvas.width / this.gridSize);
    this.rows = Math.floor(canvas.height / this.gridSize);
    
    this.reset();
    this.bindControls();
  }

  reset() {
    this.snake = [{ x: 5, y: 5 }, { x: 4, y: 5 }, { x: 3, y: 5 }];
    this.direction = { x: 1, y: 0 };
    this.nextDirection = { x: 1, y: 0 };
    this.food = this.randomFood();
    this.score = 0;
    this.gameOver = false;
    this.intervalId = null;
  }

  start() {
    this.reset();
    this.intervalId = setInterval(() => this.tick(), this.speed);
  }

  tick() {
    // Update direction (prevents 180-degree turns)
    this.direction = this.nextDirection;
    
    // Calculate new head
    const head = {
      x: this.snake[0].x + this.direction.x,
      y: this.snake[0].y + this.direction.y,
    };

    // Wall collision (wrap-around or game over)
    if (head.x < 0 || head.x >= this.cols || head.y < 0 || head.y >= this.rows) {
      return this.endGame();
    }

    // Self collision
    if (this.snake.some(seg => seg.x === head.x && seg.y === head.y)) {
      return this.endGame();
    }

    this.snake.unshift(head);

    // Eat food
    if (head.x === this.food.x && head.y === this.food.y) {
      this.score += 10;
      this.food = this.randomFood();
    } else {
      this.snake.pop(); // Remove tail if no food eaten
    }

    this.draw();
  }

  draw() {
    const { ctx, gridSize } = this;
    ctx.fillStyle = '#1a1a2e';
    ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);

    // Snake
    this.snake.forEach((seg, i) => {
      ctx.fillStyle = i === 0 ? '#27ae60' : '#2ecc71';
      ctx.fillRect(seg.x * gridSize + 1, seg.y * gridSize + 1, gridSize - 2, gridSize - 2);
    });

    // Food
    ctx.fillStyle = '#e74c3c';
    ctx.beginPath();
    ctx.arc(this.food.x * gridSize + gridSize/2, this.food.y * gridSize + gridSize/2, gridSize/2 - 2, 0, Math.PI * 2);
    ctx.fill();

    // Score
    ctx.fillStyle = 'white';
    ctx.font = '14px monospace';
    ctx.fillText(`Score: ${this.score}`, 10, 20);
  }

  bindControls() {
    const dirMap = { ArrowUp: {x:0,y:-1}, ArrowDown: {x:0,y:1}, ArrowLeft: {x:-1,y:0}, ArrowRight: {x:1,y:0} };
    document.addEventListener('keydown', (e) => {
      const dir = dirMap[e.key];
      if (!dir) return;
      // Prevent reversing
      if (dir.x !== -this.direction.x || dir.y !== -this.direction.y) {
        this.nextDirection = dir;
      }
    });
  }

  randomFood() {
    let pos;
    do {
      pos = { x: Math.floor(Math.random() * this.cols), y: Math.floor(Math.random() * this.rows) };
    } while (this.snake.some(s => s.x === pos.x && s.y === pos.y));
    return pos;
  }

  endGame() {
    this.gameOver = true;
    clearInterval(this.intervalId);
    this.ctx.fillStyle = 'rgba(0,0,0,0.7)';
    this.ctx.fillRect(0, 0, this.canvas.width, this.canvas.height);
    this.ctx.fillStyle = 'white';
    this.ctx.font = '24px sans-serif';
    this.ctx.textAlign = 'center';
    this.ctx.fillText(`Game Over! Score: ${this.score}`, this.canvas.width/2, this.canvas.height/2);
  }
}

      this.container.querySelector('#time').textContent = this.timeLeft + 's';
      if (this.timeLeft <= 0) this.finish();
    }, 1000);
  }

    this.isRunning = false;
    clearInterval(this.timerId);
    this.updateStats();
    this.container.querySelector('#type-input').disabled = true;

    if (this.allowClickNavigation) {
      this.container.querySelectorAll('.step').forEach(el => {
        el.addEventListener('click', () => {
          const idx = parseInt(el.dataset.step);
          if (idx <= this.currentStep || this.completedSteps.has(idx - 1)) this.goTo(idx);
        });
      });
    }
  }

  next() {
    this.completedSteps.add(this.currentStep);
    if (this.currentStep < this.steps.length - 1) {
      this.currentStep++;
      this.onStepChange(this.currentStep);
      this.render();
    }
  }

  prev() {
    if (this.currentStep > 0) {
      this.currentStep--;
      this.onStepChange(this.currentStep);

    }

class Breadcrumb {
  constructor(container, options = {}) {
    this.container = container;
    this.items = options.items || []; // [{label, href, icon?}]
    this.separator = options.separator || '›';
    this.maxItems = options.maxItems || 5; // Collapse if more
    this.onNavigate = options.onNavigate || (() => {});

    this.render();
  }

  render() {
    let displayItems = [...this.items];
    let collapsed = false;

    // Collapse middle items if too many
    if (displayItems.length > this.maxItems) {
      const first = displayItems.slice(0, 1);
      const last = displayItems.slice(-2);
      displayItems = [...first, { label: '...', isCollapsed: true }, ...last];
      collapsed = true;
    }

    this.container.innerHTML = `
      <nav class="breadcrumb" aria-label="Breadcrumb">
        <ol>
          ${displayItems.map((item, i) => {
            const isLast = i === displayItems.length - 1;
            if (item.isCollapsed) {
              return `<li class="breadcrumb-collapsed">
                <button class="collapse-btn" aria-label="Show hidden items">•••</button>
                <div class="collapse-menu">
                  ${this.items.slice(1, -2).map(it => `<a href="${it.href || '#'}">${it.label}</a>`).join('')}
                </div>
              </li>`;
            }
            return `<li>
              ${isLast 
                ? `<span class="current" aria-current="page">${item.icon || ''}${item.label}</span>`
                : `<a href="${item.href || '#'}" data-index="${i}">${item.icon || ''}${item.label}</a>`}
              ${!isLast ? `<span class="separator">${this.separator}</span>` : ''}
            </li>`;
          }).join('')}
        </ol>
      </nav>
    `;
    this.bindEvents();
  }

  bindEvents() {
    this.container.querySelectorAll('a[data-index]').forEach(link => {
      link.addEventListener('click', (e) => {
        e.preventDefault();
        const idx = parseInt(link.dataset.index);
        this.onNavigate(this.items[idx], idx);
      });
    });

    const collapseBtn = this.container.querySelector('.collapse-btn');
    if (collapseBtn) {
      collapseBtn.addEventListener('click', () => {
        collapseBtn.nextElementSibling.classList.toggle('visible');
      });
    }
  }

  push(item) { this.items.push(item); this.render(); }
  pop() { this.items.pop(); this.render(); }
  setItems(items) { this.items = items; this.render(); }
}
