Array.prototype.map

// Polyfill for Array.prototype.map
Array.prototype.myMap = function(callback, thisArg) {
  if (this == null) throw new TypeError("Cannot read properties of null/undefined");
  if (typeof callback !== "function") throw new TypeError(callback + " is not a function");
  
  const result = new Array(this.length);
  for (let i = 0; i < this.length; i++) {
    // Skip holes in sparse arrays (like native map does)
    if (i in this) {
      result[i] = callback.call(thisArg, this[i], i, this);
    }
  }
  return result;
};

// Test:
[1, 2, 3].myMap(x => x * 2); // [2, 4, 6]
[1, , 3].myMap(x => x * 2);  // [2, empty, 6] — handles sparse arrays!

Array.prototype.filter

Array.prototype.myFilter = function(callback, thisArg) {
  if (this == null) throw new TypeError("Cannot read properties of null/undefined");
  if (typeof callback !== "function") throw new TypeError(callback + " is not a function");
  
  const result = [];
  for (let i = 0; i < this.length; i++) {
    if (i in this) {
      if (callback.call(thisArg, this[i], i, this)) {
        result.push(this[i]);
      }
    }
  }
  return result;
};

// Test:
[1, 2, 3, 4, 5].myFilter(x => x % 2 === 0); // [2, 4]

Array.prototype.reduce

Array.prototype.myReduce = function(callback, initialValue) {
  if (this == null) throw new TypeError("Cannot read properties of null/undefined");
  if (typeof callback !== "function") throw new TypeError(callback + " is not a function");
  
  const arr = this;
  const len = arr.length;
  let accumulator;
  let startIndex;
  
  if (arguments.length >= 2) {
    // initialValue provided
    accumulator = initialValue;
    startIndex = 0;
  } else {
    // No initialValue — use first element
    if (len === 0) throw new TypeError("Reduce of empty array with no initial value");
    
    // Find first non-hole index
    let found = false;
    for (let i = 0; i < len; i++) {
      if (i in arr) {
        accumulator = arr[i];
        startIndex = i + 1;
        found = true;
        break;
      }
    }
    if (!found) throw new TypeError("Reduce of empty array with no initial value");
  }
  
  for (let i = startIndex; i < len; i++) {
    if (i in arr) {
      accumulator = callback(accumulator, arr[i], i, arr);
    }
  }
  
  return accumulator;
};

// Tests:
[1, 2, 3].myReduce((acc, val) => acc + val, 0); // 6
[1, 2, 3].myReduce((acc, val) => acc + val);    // 6 (no initial value)
// [].myReduce((a, b) => a + b); // TypeError!

Array.prototype.find & findIndex

Array.prototype.myFind = function(callback, thisArg) {
  if (typeof callback !== "function") throw new TypeError(callback + " is not a function");
  
  for (let i = 0; i < this.length; i++) {
    if (callback.call(thisArg, this[i], i, this)) {
      return this[i];
    }
  }
  return undefined;
};

Array.prototype.myFindIndex = function(callback, thisArg) {
  if (typeof callback !== "function") throw new TypeError(callback + " is not a function");
  
  for (let i = 0; i < this.length; i++) {
    if (callback.call(thisArg, this[i], i, this)) {
      return i;
    }
  }
  return -1;
};

Array.prototype.flat

Array.prototype.myFlat = function(depth = 1) {
  if (depth < 0) return this.slice();
  
  return this.reduce((acc, item) => {
    if (Array.isArray(item) && depth > 0) {
      acc.push(...item.myFlat(depth - 1));
    } else {
      acc.push(item);
    }
    return acc;
  }, []);
};

// Iterative approach (no recursion — better for deep nesting):
Array.prototype.myFlatIterative = function(depth = 1) {
  let result = [...this];
  
  for (let d = 0; d < depth; d++) {
    const temp = [];
    let hasNestedArray = false;
    
    for (const item of result) {
      if (Array.isArray(item)) {
        temp.push(...item);
        hasNestedArray = true;
      } else {
        temp.push(item);
      }
    }
    
    result = temp;
    if (!hasNestedArray) break; // No more arrays to flatten
  }
  
  return result;
};

// Tests:
[1, [2, [3, [4]]]].myFlat();    // [1, 2, [3, [4]]]
[1, [2, [3, [4]]]].myFlat(2);   // [1, 2, 3, [4]]
[1, [2, [3, [4]]]].myFlat(Infinity); // [1, 2, 3, 4]

Array.prototype.every & some

Array.prototype.myEvery = function(callback, thisArg) {
  for (let i = 0; i < this.length; i++) {
    if (i in this && !callback.call(thisArg, this[i], i, this)) {
      return false; // Short-circuits on first false
    }
  }
  return true;
};

Array.prototype.mySome = function(callback, thisArg) {
  for (let i = 0; i < this.length; i++) {
    if (i in this && callback.call(thisArg, this[i], i, this)) {
      return true; // Short-circuits on first true
    }
  }
  return false;
};

Array.from

Array.myFrom = function(arrayLike, mapFn, thisArg) {
  if (arrayLike == null) throw new TypeError("Cannot convert undefined or null to object");
  
  const items = Object(arrayLike);
  const len = Math.floor(Number(items.length)) || 0;
  const result = new Array(len);
  
  for (let i = 0; i < len; i++) {
    const value = items[i];
    result[i] = mapFn ? mapFn.call(thisArg, value, i) : value;
  }
  
  return result;
};

Array.myFrom([1,2,3], x => x*2); // [2, 4, 6]

Function.prototype.call

Function.prototype.myCall = function(context, ...args) {
  // Handle null/undefined → globalThis (window in browser)
  context = context != null ? Object(context) : globalThis;
  
  // Use Symbol to avoid property collision
  const fnKey = Symbol("fn");
  context[fnKey] = this; // 'this' is the function being called
  
  const result = context[fnKey](...args);
  delete context[fnKey];
  
  return result;
};

// Test:
function greet(greeting, punctuation) {
  return `${greeting}, ${this.name}${punctuation}`;
}
greet.myCall({ name: "Sid" }, "Hello", "!"); // "Hello, Sid!"

Function.prototype.apply

Function.prototype.myApply = function(context, argsArray) {
  context = context != null ? Object(context) : globalThis;
  
  // Validate argsArray
  if (argsArray != null && !Array.isArray(argsArray) && typeof argsArray[Symbol.iterator] !== "function") {
    throw new TypeError("CreateListFromArrayLike called on non-object");
  }
  
  const fnKey = Symbol("fn");
  context[fnKey] = this;
  
  const result = argsArray ? context[fnKey](...argsArray) : context[fnKey]();
  delete context[fnKey];
  
  return result;
};

// Test:
Math.max.myApply(null, [1, 5, 3, 9, 2]); // 9

Function.prototype.bind

Function.prototype.myBind = function(context, ...boundArgs) {
  if (typeof this !== "function") {
    throw new TypeError("Bind must be called on a function");
  }
  
  const originalFn = this;
  context = context != null ? Object(context) : globalThis;
  
  // Bound function
  function boundFunction(...callArgs) {
    // Handle 'new' keyword — if called with new, ignore bound context
    const isNewCall = this instanceof boundFunction;
    
    return originalFn.apply(
      isNewCall ? this : context,
      [...boundArgs, ...callArgs]
    );
  }
  
  // Preserve prototype chain (for new keyword support)
  if (originalFn.prototype) {
    boundFunction.prototype = Object.create(originalFn.prototype);
  }
  
  return boundFunction;
};

// Tests:
const obj = { name: "Sid" };
function sayName(greeting) { return `${greeting} ${this.name}`; }

const bound = sayName.myBind(obj, "Hi");
bound(); // "Hi Sid"

// Works with new:
function Person(name, age) { this.name = name; this.age = age; }
const BoundPerson = Person.myBind(null, "Sid");
const p = new BoundPerson(25);
p.name; // "Sid"
p.age;  // 25

Object.create

Object.myCreate = function(proto, propertiesObject) {
  if (proto !== null && typeof proto !== "object" && typeof proto !== "function") {
    throw new TypeError("Object prototype may only be an Object or null");
  }
  
  // Create a constructor function
  function F() {}
  F.prototype = proto;
  
  const obj = new F();
  
  // Handle null prototype
  if (proto === null) {
    Object.setPrototypeOf(obj, null);
  }
  
  // Define additional properties if provided
  if (propertiesObject !== undefined) {
    Object.defineProperties(obj, propertiesObject);
  }
  
  return obj;
};

// Test:

child.greet(); // "Hello, Sid"

Object.assign

Object.myAssign = function(target, ...sources) {
  if (target == null) throw new TypeError("Cannot convert undefined or null to object");
  
  const result = Object(target);
  
  for (const source of sources) {
    if (source == null) continue; // Skip null/undefined sources
    
    // Copy own enumerable string-keyed properties
    for (const key of Object.keys(source)) {
      result[key] = source[key];
    }
    
    // Also copy own enumerable symbol-keyed properties
    for (const sym of Object.getOwnPropertySymbols(source)) {
      if (Object.getOwnPropertyDescriptor(source, sym).enumerable) {
        result[sym] = source[sym];
      }
    }
  }
  
  return result;
};

// Test:
Object.myAssign({}, { a: 1 }, { b: 2 }, { a: 3 }); // { a: 3, b: 2 }

Object.keys / values / entries

Object.myKeys = function(obj) {
  if (obj == null) throw new TypeError("Cannot convert undefined or null to object");
  obj = Object(obj);
  const keys = [];
  for (const key in obj) {
    if (Object.prototype.hasOwnProperty.call(obj, key)) {
      keys.push(key);
    }
  }
  return keys;
};

Object.myValues = function(obj) {
  return Object.myKeys(obj).map(key => obj[key]);
};

Object.myEntries = function(obj) {
  return Object.myKeys(obj).map(key => [key, obj[key]]);
};

Object.myFromEntries = function(iterable) {
  const obj = {};
  for (const [key, value] of iterable) {
    obj[key] = value;
  }
  return obj;
};

Object.freeze (Deep)

// Shallow freeze (native behavior):
Object.myFreeze = function(obj) {
  if (typeof obj !== "object" || obj === null) return obj;
  
  Object.getOwnPropertyNames(obj).forEach(name => {
    Object.defineProperty(obj, name, { writable: false, configurable: false });
  });
  Object.preventExtensions(obj);
  return obj;
};

// Deep freeze (interview favorite!):
function deepFreeze(obj) {
  if (typeof obj !== "object" || obj === null || Object.isFrozen(obj)) return obj;
  
  Object.freeze(obj);
  
  Object.getOwnPropertyNames(obj).forEach(name => {
    const value = obj[name];
    if (typeof value === "object" && value !== null) {
      deepFreeze(value);
    }
  });
  
  return obj;
}

const frozen = deepFreeze({ a: { b: { c: 1 } } });
frozen.a.b.c = 99; // Silently fails (or throws in strict mode)
console.log(frozen.a.b.c); // 1

4. Promise from Scratch (Full Implementation)

📺 YouTube Tutorials

⚠️ Most Asked at FAANG: "Implement Promise from scratch" is the #1 most-asked machine coding question at Google, Amazon, and Microsoft for senior frontend roles. You MUST be able to write this in 30 minutes. 
class MyPromise {
  #state = "pending";    // "pending" | "fulfilled" | "rejected"
  #value = undefined;    // Resolved value or rejection reason
  #handlers = [];        // Array of { onFulfilled, onRejected, resolve, reject }
  
  constructor(executor) {
    if (typeof executor !== "function") {
      throw new TypeError("Promise resolver is not a function");
    }
    
    try {
      executor(this.#resolve.bind(this), this.#reject.bind(this));
    } catch (error) {
      this.#reject(error);
    }
  }
  
  #resolve(value) {
    // Resolution can only happen once
    if (this.#state !== "pending") return;
    
    // If resolved with a thenable/promise, unwrap it
    if (value !== null && (typeof value === "object" || typeof value === "function")) {
      const then = value?.then;
      if (typeof then === "function") {
        try {
          then.call(value, this.#resolve.bind(this), this.#reject.bind(this));
        } catch (e) {
          this.#reject(e);
        }
        return;
      }
    }
    
    this.#state = "fulfilled";
    this.#value = value;
    this.#executeHandlers();
  }
  
  #reject(reason) {
    if (this.#state !== "pending") return;
    
    this.#state = "rejected";
    this.#value = reason;
    this.#executeHandlers();
  }
  
  #executeHandlers() {
    // Execute asynchronously (microtask)
    queueMicrotask(() => {
      while (this.#handlers.length) {
        const { onFulfilled, onRejected, resolve, reject } = this.#handlers.shift();
        
        try {
          if (this.#state === "fulfilled") {
            if (typeof onFulfilled === "function") {
              resolve(onFulfilled(this.#value));
            } else {
              resolve(this.#value); // Pass-through
            }
          } else if (this.#state === "rejected") {
            if (typeof onRejected === "function") {
              resolve(onRejected(this.#value)); // catch returns resolved promise
            } else {
              reject(this.#value); // Pass-through rejection
            }
          }
        } catch (error) {
          reject(error);
        }
      }
    });
  }
  
  then(onFulfilled, onRejected) {
    // .then() ALWAYS returns a new Promise (chaining!)
    return new MyPromise((resolve, reject) => {
      this.#handlers.push({ onFulfilled, onRejected, resolve, reject });
      
      // If already settled, execute immediately
      if (this.#state !== "pending") {
        this.#executeHandlers();
      }
    });
  }
  
  catch(onRejected) {
    return this.then(null, onRejected);
  }
  
  finally(onFinally) {
    return this.then(
      value => MyPromise.resolve(onFinally()).then(() => value),
      reason => MyPromise.resolve(onFinally()).then(() => { throw reason; })
    );
  }
  
  // Static methods
  static resolve(value) {
    if (value instanceof MyPromise) return value;
    return new MyPromise(resolve => resolve(value));
  }
  
  static reject(reason) {
    return new MyPromise((_, reject) => reject(reason));
  }
}

// ═══════ VERIFICATION ═══════
const p = new MyPromise((resolve) => {
  setTimeout(() => resolve("hello"), 100);
});

p.then(val => val + " world")
 .then(val => console.log(val)) // "hello world"
 .catch(err => console.error(err));

// Chaining:
MyPromise.resolve(1)
  .then(x => x + 1)
  .then(x => { throw new Error("oops"); })
  .catch(e => console.log(e.message)) // "oops"
  .finally(() => console.log("done")); // "done"
💡 Key Points Interviewers Look For:
  • State can only transition ONCE (pending → fulfilled OR pending → rejected)
  • .then() returns a NEW promise (enables chaining)
  • Handlers execute as microtasks (asynchronously)
  • If resolved with a thenable, it must be unwrapped recursively
  • .catch() is just .then(null, onRejected)
  • .finally() preserves the value/reason — it's transparent

Promise.all

MyPromise.all = function(promises) {
  return new MyPromise((resolve, reject) => {
    const results = [];
    let pending = 0;
    
    const iterable = [...promises]; // Handle any iterable
    
    if (iterable.length === 0) {
      resolve(results);
      return;
    }
    
    iterable.forEach((promise, index) => {
      pending++;
      MyPromise.resolve(promise).then(
        value => {
          results[index] = value; // Maintain order!
          pending--;
          if (pending === 0) resolve(results);
        },
        reason => reject(reason) // Reject on FIRST failure
      );
    });
  });
};

  MyPromise.resolve(2),

Promise.allSettled

MyPromise.allSettled = function(promises) {
  return new MyPromise((resolve) => {
    const results = [];
    let remaining = 0;
    
    const iterable = [...promises];
    
    if (iterable.length === 0) {
      resolve(results);
      return;
    }
    
    iterable.forEach((promise, index) => {
      remaining++;
      MyPromise.resolve(promise).then(
        value => {
          results[index] = { status: "fulfilled", value };
          remaining--;
          if (remaining === 0) resolve(results);
        },
        reason => {
          results[index] = { status: "rejected", reason };
          remaining--;
          if (remaining === 0) resolve(results);
        }
      );
    });
  });
};

// NEVER rejects! Always resolves with all statuses.

Promise.race

MyPromise.race = function(promises) {
  return new MyPromise((resolve, reject) => {
    const iterable = [...promises];
    
    // Note: if empty iterable, promise stays pending forever (per spec)
    iterable.forEach(promise => {
      MyPromise.resolve(promise).then(resolve, reject);
      // First one to settle wins — others are ignored
    });
  });
};

Promise.any

MyPromise.any = function(promises) {
  return new MyPromise((resolve, reject) => {
    const errors = [];
    let remaining = 0;
    
    const iterable = [...promises];
    
    if (iterable.length === 0) {
      reject(new AggregateError([], "All promises were rejected"));
      return;
    }
    
    iterable.forEach((promise, index) => {
      remaining++;
      MyPromise.resolve(promise).then(
        resolve, // First success wins!
        error => {
          errors[index] = error;
          remaining--;
          if (remaining === 0) {
            reject(new AggregateError(errors, "All promises were rejected"));
          }
        }
      );
    });
  });
};

// Opposite of Promise.all:
// .all → rejects on first failure
// .any → resolves on first success

function myNew(Constructor, ...args) {
  // 1. Create a new object with Constructor's prototype
  const obj = Object.create(Constructor.prototype);
  
  // 2. Call the constructor with the new object as 'this'
  const result = Constructor.apply(obj, args);
  
  // 3. If constructor returns an object, use that; otherwise use obj
  return (result !== null && typeof result === "object") ? result : obj;
}

Implement 'instanceof'

function myInstanceOf(obj, Constructor) {
  if (obj === null || (typeof obj !== "object" && typeof obj !== "function")) {
    return false;
  }
  
  let proto = Object.getPrototypeOf(obj);
  const target = Constructor.prototype;
  
  while (proto !== null) {
    if (proto === target) return true;
    proto = Object.getPrototypeOf(proto);
  }
  
  return false;
}

Deep Equality Check

function deepEqual(a, b, seen = new WeakMap()) {
  // Primitive comparison
  if (a === b) return true;
  
  // NaN check (NaN !== NaN, but should be equal)
  if (Number.isNaN(a) && Number.isNaN(b)) return true;
  
  // Type check
  if (typeof a !== "object" || typeof b !== "object") return false;
  if (a === null || b === null) return false;
  
  // Circular reference check
  if (seen.has(a)) return seen.get(a) === b;
  seen.set(a, b);
  
  // Date comparison
  if (a instanceof Date && b instanceof Date) {
    return a.getTime() === b.getTime();
  }
  
  // RegExp comparison
  if (a instanceof RegExp && b instanceof RegExp) {
    return a.source === b.source && a.flags === b.flags;
  }
  
  // Array/Object comparison
  const keysA = Object.keys(a);
  const keysB = Object.keys(b);
  
  if (keysA.length !== keysB.length) return false;
  
  return keysA.every(key => 
    Object.prototype.hasOwnProperty.call(b, key) && deepEqual(a[key], b[key], seen)
  );
}

// Tests:

deepEqual(NaN, NaN); // true

Implement typeof (accurate version)

// The problem: typeof null === "object" (historic bug)
// Accurate type checking:
function typeOf(value) {
  if (value === null) return "null";
  if (value === undefined) return "undefined";
  if (typeof value === "function") return "function";
  
  // Use Object.prototype.toString for accurate type
  const type = Object.prototype.toString.call(value);
  // "[object Array]" → "array"
  return type.slice(8, -1).toLowerCase();

// Basic memoize:
function memoize(fn) {
  const cache = new Map();
  
  return function(...args) {
    const key = JSON.stringify(args);
    
    if (cache.has(key)) {
      return cache.get(key);
    }
    
    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}

// Advanced memoize with custom key resolver & cache limit (LRU):
function memoizeAdvanced(fn, { maxSize = 100, keyResolver = JSON.stringify } = {}) {
  const cache = new Map();
  
  function memoized(...args) {
    const key = keyResolver(args);
    
    if (cache.has(key)) {
      // Move to end (most recently used)
      const value = cache.get(key);
      cache.delete(key);
      cache.set(key, value);
      return value;
    }
    
    const result = fn.apply(this, args);
    cache.set(key, result);
    
    // Evict oldest if over limit
    if (cache.size > maxSize) {
      const firstKey = cache.keys().next().value;
      cache.delete(firstKey);
    }
    
    return result;
  }
  
  memoized.cache = cache;
  memoized.clear = () => cache.clear();
  return memoized;
}

Debounce & Throttle (Production-Grade)

// Full debounce (Lodash-compatible):
function debounce(fn, wait, { leading = false, trailing = true, maxWait } = {}) {
  let timeoutId, lastCallTime, lastInvokeTime = 0, result;
  
  function invoke(time) {
    lastInvokeTime = time;
    result = fn.apply(this, arguments);
    return result;
  }
  
  function shouldInvoke(time) {
    const timeSinceLastCall = time - (lastCallTime || 0);
    const timeSinceLastInvoke = time - lastInvokeTime;
    
    return (lastCallTime === undefined) ||
           (timeSinceLastCall >= wait) ||
           (maxWait !== undefined && timeSinceLastInvoke >= maxWait);
  }
  
  function debounced(...args) {
    const time = Date.now();
    const isInvoking = shouldInvoke(time);
    
    lastCallTime = time;
    
    if (isInvoking && leading && !timeoutId) {
      result = fn.apply(this, args);
      lastInvokeTime = time;
    }
    
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => {
      timeoutId = null;
      if (trailing) {
        result = fn.apply(this, args);
        lastInvokeTime = Date.now();
      }
    }, wait);
    
    return result;
  }
  
  debounced.cancel = () => { clearTimeout(timeoutId); timeoutId = null; };
  debounced.flush = () => { if (timeoutId) { clearTimeout(timeoutId); timeoutId = null; } };
  
  return debounced;
}

// Full throttle:
function throttle(fn, limit, { leading = true, trailing = true } = {}) {
  let lastTime = 0, timeoutId, lastArgs, lastContext;
  
  function invoke() {
    lastTime = leading ? Date.now() : 0;
    timeoutId = null;
    fn.apply(lastContext, lastArgs);
    lastArgs = lastContext = null;
  }
  
  return function(...args) {
    const now = Date.now();
    if (!lastTime && !leading) lastTime = now;
    
    const remaining = limit - (now - lastTime);
    lastArgs = args;
    lastContext = this;
    
    if (remaining <= 0 || remaining > limit) {
      if (timeoutId) { clearTimeout(timeoutId); timeoutId = null; }
      lastTime = now;
      fn.apply(this, args);
    } else if (!timeoutId && trailing) {
      timeoutId = setTimeout(invoke, remaining);
    }
  };
}

function mySetInterval(callback, delay, ...args) {
  let timerId;
  let cancelled = false;
  
  function repeat() {
    if (cancelled) return;
    callback(...args);
    timerId = setTimeout(repeat, delay);
  }
  
  timerId = setTimeout(repeat, delay);
  
  // Return a way to clear it
  return {
    clear() {
      cancelled = true;
      clearTimeout(timerId);
    }
  };
}

function querySelectorAll(root, selector) {

    if (node.nodeType !== 1) return; // Element nodes only
    
    if (node.matches(selector)) {
      results.push(node);
    }
    
    for (const child of node.children) {
      walk(child);

Event Target (addEventListener/removeEventListener)

class MyEventTarget {
  #listeners = new Map();
  
  addEventListener(type, callback, options = {}) {
    if (!this.#listeners.has(type)) {
      this.#listeners.set(type, []);
    }
    
    const listener = { callback, once: options.once || false };
    this.#listeners.get(type).push(listener);
  }
  
  removeEventListener(type, callback) {
    const listeners = this.#listeners.get(type);
    if (!listeners) return;
    
    this.#listeners.set(
      type,
      listeners.filter(l => l.callback !== callback)
    );
  }
  
  dispatchEvent(event) {
    const type = event.type || event;
    const listeners = this.#listeners.get(type);
    if (!listeners) return;
    
    const toRemove = [];
    
    for (const listener of [...listeners]) {
      listener.callback(event);
      if (listener.once) toRemove.push(listener);
    }
    
    // Remove once listeners
    toRemove.forEach(l => {
      const idx = listeners.indexOf(l);
      if (idx > -1) listeners.splice(idx, 1);
    });
  }
}

function myJSONStringify(value, replacer, space) {
  const indent = typeof space === "number" ? " ".repeat(space) : (space || "");
  
  function stringify(val, depth, key) {
    // Apply replacer if provided
    if (typeof replacer === "function") {
      val = replacer(key, val);
    }
    
    // Handle primitives
    if (val === null) return "null";
    if (val === undefined || typeof val === "function" || typeof val === "symbol") {
      return undefined; // Omitted from objects, becomes null in arrays
    }
    if (typeof val === "boolean") return String(val);
    if (typeof val === "number") {
      if (!isFinite(val)) return "null"; // Infinity, -Infinity, NaN → null
      return String(val);
    }
    if (typeof val === "string") return `"${escapeString(val)}"`;
    if (typeof val === "bigint") throw new TypeError("BigInt value can't be serialized in JSON");
    
    // Handle toJSON method
    if (val !== null && typeof val === "object" && typeof val.toJSON === "function") {
      return stringify(val.toJSON(key), depth, key);
    }
    
    // Handle Date
    if (val instanceof Date) {
      return `"${val.toISOString()}"`;
    }
    
    // Handle Array
    if (Array.isArray(val)) {
      if (val.length === 0) return "[]";
      
      const items = val.map((item, i) => {
        const str = stringify(item, depth + 1, String(i));
        return str === undefined ? "null" : str; // undefined → null in arrays
      });
      
      if (!indent) return `[${items.join(",")}]`;
      
      const currentIndent = indent.repeat(depth + 1);
      const closingIndent = indent.repeat(depth);
      return `[\n${items.map(i => currentIndent + i).join(",\n")}\n${closingIndent}]`;
    }
    
    // Handle Object
    if (typeof val === "object") {
      let keys = Object.keys(val);
      
      // Array replacer acts as allowlist
      if (Array.isArray(replacer)) {
        keys = keys.filter(k => replacer.includes(k));
      }
      
      const pairs = [];
      for (const k of keys) {
        const str = stringify(val[k], depth + 1, k);
        if (str !== undefined) { // Skip undefined values
          pairs.push(`"${escapeString(k)}":${indent ? " " : ""}${str}`);
        }
      }
      
      if (pairs.length === 0) return "{}";
      
      if (!indent) return `{${pairs.join(",")}}`;
      
      const currentIndent = indent.repeat(depth + 1);
      const closingIndent = indent.repeat(depth);
      return `{\n${pairs.map(p => currentIndent + p).join(",\n")}\n${closingIndent}}`;
    }
    
    return undefined;
  }
  
  function escapeString(str) {
    return str
      .replace(/\\/g, "\\\\")
      .replace(/"/g, '\\"')
      .replace(/\n/g, "\\n")
      .replace(/\r/g, "\\r")
      .replace(/\t/g, "\\t")
      .replace(/[\b]/g, "\\b")
      .replace(/\f/g, "\\f");
  }
  
  const result = stringify(value, 0, "");
  return result === undefined ? undefined : result;
}

// Async Task Scheduler with concurrency limit
// Asked at: Amazon, Google, Uber, Atlassian
class TaskScheduler {
  #concurrency;
  #running = 0;
  #queue = [];
  #results = new Map();
  #taskId = 0;
  
  constructor(concurrency = 3) {
    this.#concurrency = concurrency;
  }
  
  // Add a task (returns a promise that resolves with the task result)
  add(task, priority = 0) {
    const id = ++this.#taskId;
    
    return new Promise((resolve, reject) => {
      const wrappedTask = { id, task, priority, resolve, reject };
      
      // Insert in priority order (higher priority first)
      const insertIndex = this.#queue.findIndex(t => t.priority < priority);
      if (insertIndex === -1) {
        this.#queue.push(wrappedTask);
      } else {
        this.#queue.splice(insertIndex, 0, wrappedTask);
      }
      
      this.#run();
    });
  }
  
  #run() {
    while (this.#running < this.#concurrency && this.#queue.length > 0) {
      const { task, resolve, reject, id } = this.#queue.shift();
      this.#running++;
      
      Promise.resolve(task())
        .then(result => {
          this.#results.set(id, { status: "fulfilled", value: result });
          resolve(result);
        })
        .catch(error => {
          this.#results.set(id, { status: "rejected", reason: error });
          reject(error);
        })
        .finally(() => {
          this.#running--;
          this.#run(); // Pick next task
        });
    }
  }
  
  get pending() { return this.#queue.length; }
  get active() { return this.#running; }
  
  // Wait for all current tasks to complete
  async drain() {
    while (this.#running > 0 || this.#queue.length > 0) {
      await new Promise(resolve => setTimeout(resolve, 10));
    }
  }
}

async function parallelLimit(tasks, limit) {

  const executing = new Set();
  
  for (const [index, task] of tasks.entries()) {
    const promise = Promise.resolve(task()).then(result => {
      executing.delete(promise);
      results[index] = result;
    });
    executing.add(promise);

Retry with Exponential Backoff

async function retryWithBackoff(fn, options = {}) {
  const {
    maxRetries = 3,
    baseDelay = 1000,
    maxDelay = 30000,
    backoffFactor = 2,
    jitter = true,         // Add randomness to prevent thundering herd
    retryOn = () => true,  // Predicate: should we retry this error?
    onRetry = () => {},    // Callback before each retry
  } = options;
  
  let lastError;
  
  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    try {
      return await fn(attempt);
    } catch (error) {
      lastError = error;
      
      if (attempt === maxRetries || !retryOn(error, attempt)) {
        throw error;
      }
      
      // Calculate delay with exponential backoff
      let delay = Math.min(baseDelay * backoffFactor ** attempt, maxDelay);
      
      // Add jitter (±25%) to prevent all clients retrying simultaneously
      if (jitter) {
        delay = delay * (0.75 + Math.random() * 0.5);
      }
      
      onRetry({ attempt, delay, error });
      
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }
  
  throw lastError;
}

Circuit Breaker Pattern

// Prevents cascading failures by stopping requests to a failing service
class CircuitBreaker {
  #state = "CLOSED";      // CLOSED → requests pass through
  #failureCount = 0;      // OPEN → requests immediately rejected
  #successCount = 0;      // HALF_OPEN → limited requests pass through
  #lastFailureTime = null;
  #options;
  
  constructor(fn, options = {}) {
    this.fn = fn;
    this.#options = {
      failureThreshold: options.failureThreshold || 5,
      successThreshold: options.successThreshold || 3,
      timeout: options.timeout || 30000, // Time before trying again
      ...options
    };
  }
  
  get state() { return this.#state; }
  
  async call(...args) {
    if (this.#state === "OPEN") {
      // Check if timeout has elapsed → transition to HALF_OPEN
      if (Date.now() - this.#lastFailureTime >= this.#options.timeout) {
        this.#state = "HALF_OPEN";
        this.#successCount = 0;
      } else {
        throw new Error(`Circuit breaker is OPEN. Service unavailable.`);
      }
    }
    
    try {
      const result = await this.fn(...args);
      this.#onSuccess();
      return result;
    } catch (error) {
      this.#onFailure();
      throw error;
    }
  }
  
  #onSuccess() {
    if (this.#state === "HALF_OPEN") {
      this.#successCount++;
      if (this.#successCount >= this.#options.successThreshold) {
        this.#state = "CLOSED"; // Recovery!
        this.#failureCount = 0;
      }
    } else {
      this.#failureCount = 0;
    }
  }
  
  #onFailure() {
    this.#failureCount++;
    this.#lastFailureTime = Date.now();
    
    if (this.#failureCount >= this.#options.failureThreshold) {
      this.#state = "OPEN"; // Trip the circuit!
    }
  }
  
  reset() {
    this.#state = "CLOSED";
    this.#failureCount = 0;
    this.#successCount = 0;
  }
}

  (url) => fetch(url).then(r => { if (!r.ok) throw new Error(`${r.status}`); return r.json(); }),
  { failureThreshold: 3, timeout: 10000 }
);

  }

// LRU Cache with TTL, max size, and event hooks
class LRUCache {
  #capacity;
  #cache = new Map(); // Map maintains insertion order
  #ttl;
  
  constructor(capacity = 100, ttl = Infinity) {
    this.#capacity = capacity;
    this.#ttl = ttl;
  }
  
  get(key) {
    if (!this.#cache.has(key)) return undefined;
    
    const entry = this.#cache.get(key);
    
    // Check TTL
    if (this.#isExpired(entry)) {
      this.#cache.delete(key);
      return undefined;
    }
    
    // Move to end (most recently used)
    this.#cache.delete(key);
    this.#cache.set(key, entry);
    
    return entry.value;
  }
  
  set(key, value, ttl) {
    // Delete existing to update position
    if (this.#cache.has(key)) {
      this.#cache.delete(key);
    }
    
    // Evict least recently used if at capacity
    if (this.#cache.size >= this.#capacity) {
      const lruKey = this.#cache.keys().next().value; // First = oldest
      this.#cache.delete(lruKey);
    }
    
    this.#cache.set(key, {
      value,
      createdAt: Date.now(),
      ttl: ttl || this.#ttl
    });
    
    return this;
  }
  
  has(key) {
    if (!this.#cache.has(key)) return false;
    if (this.#isExpired(this.#cache.get(key))) {
      this.#cache.delete(key);
      return false;
    }
    return true;
  }
  
  delete(key) {
    return this.#cache.delete(key);
  }
  
  clear() {
    this.#cache.clear();
  }
  
  get size() {
    // Purge expired entries and return actual size
    this.#purgeExpired();
    return this.#cache.size;
  }
  
  #isExpired(entry) {
    if (entry.ttl === Infinity) return false;
    return Date.now() - entry.createdAt > entry.ttl;
  }
  
  #purgeExpired() {
    for (const [key, entry] of this.#cache) {
      if (this.#isExpired(entry)) this.#cache.delete(key);
    }
  }
  
  // Get all valid entries (useful for persistence)
  entries() {
    this.#purgeExpired();
    return [...this.#cache.entries()].map(([key, entry]) => [key, entry.value]);
  }
}
