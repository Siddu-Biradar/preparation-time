// ─── Debounce ─────────────────────────────────────────────
// Execute AFTER user stops triggering (search input, resize)
function debounce(fn, delay) {
  let timeoutId;
  return function(...args) {
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => fn.apply(this, args), delay);
  };
}
const debouncedSearch = debounce((query) => fetch(`/search?q=${query}`), 300);

// ─── Throttle ─────────────────────────────────────────────
// Execute at most once per interval (scroll, mousemove)
function throttle(fn, limit) {
  let inThrottle = false;
  return function(...args) {
    if (!inThrottle) {
      fn.apply(this, args);
      inThrottle = true;
      setTimeout(() => inThrottle = false, limit);
    }
  };
}
window.addEventListener("scroll", throttle(updatePosition, 100));

// ─── CSRF (Cross-Site Request Forgery) ────────────────────
// Attacker tricks user's browser into making unwanted requests
// Prevention:
// 1. CSRF tokens (Angular HttpClient sends X-XSRF-TOKEN automatically)
// 2. SameSite cookies (SameSite=Strict or Lax)
// 3. Check Origin/Referer headers

// ─── CSP (Content Security Policy) ───────────────────────
// HTTP header that restricts what resources can load
// Content-Security-Policy: default-src 'self'; script-src 'self' https://cdn.example.com

// ─── Other security practices ─────────────────────────────
// - Never store sensitive data in localStorage (XSS can read it)
// - Use HttpOnly cookies for auth tokens
// - Validate on server (client validation is UX only)
// - Use Subresource Integrity (SRI) for CDN scripts
// - Avoid eval(), new Function(), setTimeout(string)
// - Use Object.freeze() for config objects
// - Input sanitization at BOTH client and server

// ─── Prototype Pollution (Senior Interview Topic!) ────────
// Attack: Modifying Object.prototype to inject properties into ALL objects
// Example vulnerability:
function merge(target, source) {
  for (const key in source) {
    if (typeof source[key] === "object") {
      target[key] = target[key] || {};
      merge(target[key], source[key]); // Recursively merge
    } else {
      target[key] = source[key];
    }
  }
}

// Attack payload:
// merge({}, JSON.parse('{"__proto__":{"isAdmin":true}}'));
// Now EVERY object has .isAdmin === true!
// const user = {}; user.isAdmin → true! (polluted prototype)

// ✅ FIX: Block dangerous keys
function safeMerge(target, source) {
  for (const key of Object.keys(source)) { // Use Object.keys (not for...in)
    if (key === "__proto__" || key === "constructor" || key === "prototype") {
      continue; // Block prototype pollution vectors!
    }
    if (typeof source[key] === "object" && source[key] !== null) {
      target[key] = target[key] || {};
      safeMerge(target[key], source[key]);
    } else {
      target[key] = source[key];
    }
  }
  return target;
}

// ✅ FIX 2: Use Object.create(null) for dictionaries (no prototype!)
const safeDict = Object.create(null); // No __proto__!
safeDict.key = "value"; // Safe — can't pollute prototype

// ─── Clickjacking Prevention ─────────────────────────────
// X-Frame-Options: DENY (or SAMEORIGIN)
// CSP: frame-ancestors 'self'

// ─── Secure Cookie Best Practices ─────────────────────────
// Set-Cookie: token=abc; HttpOnly; Secure; SameSite=Strict; Path=/; Max-Age=3600

// ─── Interview: "How would you secure a frontend app?" ────
// 1. Sanitize all user input (DOMPurify for HTML, escape for text)
// 2. CSP headers (script-src 'self'; no inline scripts)
// 3. HttpOnly + Secure cookies for auth
// 4. CSRF tokens on state-changing requests
// 5. SRI (Subresource Integrity) for CDN scripts
// 6. No secrets in frontend code/localStorage
// 7. Rate limiting on APIs
// 8. Prototype pollution protection in object merges
// 9. Input validation on server (client is UX only)

6. Error Handling Architecture

📺 YouTube Tutorials
  • Error Handling Patterns — Academind

Theory

// ─── Custom Error Classes ─────────────────────────────────
class AppError extends Error {
  constructor(message, code, statusCode) {
    super(message);
    this.name = this.constructor.name;
    this.code = code;
    this.statusCode = statusCode;
    Error.captureStackTrace?.(this, this.constructor);
  }
}

class ValidationError extends AppError {
  constructor(field, message) {
    super(`Validation failed: ${field} — ${message}`, "VALIDATION", 400);
    this.field = field;
  }
}

class NotFoundError extends AppError {
  constructor(resource, id) {
    super(`${resource} ${id} not found`, "NOT_FOUND", 404);
  }
}

// ─── Error Boundary Pattern ───────────────────────────────
class ErrorBoundary {
  #handlers = new Map();
  
  register(ErrorType, handler) {
    this.#handlers.set(ErrorType, handler);
  }
  
  async execute(fn) {
    try {
      return await fn();
    } catch (error) {
      for (const [Type, handler] of this.#handlers) {
        if (error instanceof Type) return handler(error);
      }
      throw error; // Unhandled — re-throw
    }
  }
}

const boundary = new ErrorBoundary();
boundary.register(ValidationError, (e) => ({ error: e.message, field: e.field }));
boundary.register(NotFoundError, (e) => ({ error: e.message, status: 404 }));

// ─── Result type (no-throw pattern) ──────────────────────
class Result {
  #value; #error; #isOk;
  
  static ok(value) { const r = new Result(); r.#value = value; r.#isOk = true; return r; }
  static err(error) { const r = new Result(); r.#error = error; r.#isOk = false; return r; }
  
  isOk() { return this.#isOk; }
  unwrap() { if (!this.#isOk) throw this.#error; return this.#value; }
  unwrapOr(fallback) { return this.#isOk ? this.#value : fallback; }
  
  map(fn) { return this.#isOk ? Result.ok(fn(this.#value)) : this; }
}

// Usage:
function divide(a, b) {
  if (b === 0) return Result.err(new Error("Division by zero"));
  return Result.ok(a / b);
}
divide(10, 2).map(x => x * 2).unwrap(); // 10

Theory

// Metaprogramming: code that manipulates code/objects at runtime

// ─── Proxy (Intercept ALL object operations) ──────────────
// new Proxy(target, handler) — handler has "traps"

// Validation Proxy:
const schema = {
  name: (v) => typeof v === "string" && v.length > 0,
  age: (v) => Number.isInteger(v) && v >= 0 && v <= 150,
  email: (v) => typeof v === "string" && v.includes("@")
};

function createValidated(schema) {
  return new Proxy({}, {
    set(target, prop, value) {
      if (schema[prop] && !schema[prop](value)) {
        throw new TypeError(`Invalid value for ${prop}: ${value}`);
      }
      return Reflect.set(target, prop, value);
    }
  });
}

const user = createValidated(schema);
user.name = "Sid";  // ✅
user.age = 25;      // ✅
// user.age = -1;   // ❌ TypeError: Invalid value for age: -1

// Logging/Debugging Proxy (wrap any object to trace access):
function traceable(obj, label = "Object") {
  return new Proxy(obj, {
    get(target, prop) {
      console.log(`[${label}] GET ${String(prop)}`);
      return Reflect.get(target, prop);
    },
    set(target, prop, value) {
      console.log(`[${label}] SET ${String(prop)} = ${value}`);
      return Reflect.set(target, prop, value);
    }
  });
}

// ─── Reflect API ──────────────────────────────────────────
// Mirror of Object methods + Proxy traps — more consistent, returns booleans

Reflect.get(obj, "prop");              // obj.prop
Reflect.set(obj, "prop", value);       // obj.prop = value (returns boolean)
Reflect.has(obj, "prop");              // "prop" in obj
Reflect.deleteProperty(obj, "prop");   // delete obj.prop
Reflect.ownKeys(obj);                  // All own keys (strings + symbols)
Reflect.construct(Class, args);        // new Class(...args)
Reflect.apply(fn, thisArg, args);      // fn.apply(thisArg, args)
Reflect.defineProperty(obj, key, desc); // Like Object.defineProperty but returns boolean

// Why Reflect over Object methods?
// 1. Returns boolean (success/failure) instead of throwing
// 2. Consistent with Proxy trap names
// 3. First-class functions (no need for Object.prototype.hasOwnProperty.call)

// ─── Property Descriptors ─────────────────────────────────
Object.defineProperty(obj, "immutable", {
  value: 42,
  writable: false,       // Cannot reassign
  enumerable: false,     // Hidden from for...in, Object.keys
  configurable: false    // Cannot delete or redefine
});

// Accessor descriptors:
Object.defineProperty(obj, "fullName", {
  get() { return `${this.first} ${this.last}`; },
  set(value) { [this.first, this.last] = value.split(" "); },
  enumerable: true,
  configurable: true
});

// ─── Object immutability levels ───────────────────────────
Object.preventExtensions(obj); // No new properties
Object.seal(obj);              // No new/delete, existing non-configurable
Object.freeze(obj);            // seal + no value changes (shallow!)
// Deep freeze:
function deepFreeze(obj) {
  Object.freeze(obj);
  Object.getOwnPropertyNames(obj).forEach(prop => {
    if (typeof obj[prop] === "object" && obj[prop] !== null) {
      deepFreeze(obj[prop]);
    }
  });
  return obj;
}

8. V8 Engine Internals & Optimization

📺 YouTube Tutorials
  • How V8 Works — Franziska Hinkelmann
  • JS Engine Internals — Lydia Hallie

Theory

/*
V8 Pipeline:
Source Code → Parser → AST → Ignition (Bytecode Interpreter)
                                    ↓ (hot functions)
                              TurboFan (Optimizing JIT Compiler)
                                    ↓
                              Optimized Machine Code
                                    ↓ (if assumptions invalidated)
                              Deoptimization → back to Ignition
*/

// Hidden Classes (Shapes/Maps):
// V8 creates hidden classes for objects with same structure
// ✅ Always initialize properties in same order
function Point(x, y) { this.x = x; this.y = y; }
// All Point instances share the same hidden class → fast access

// ❌ Bad: Adding properties in different orders
const a = {}; a.x = 1; a.y = 2; // HiddenClass1 → HC2 → HC3
const b = {}; b.y = 2; b.x = 1; // HC1 → HC4 → HC5 (different!)
// 'a' and 'b' have DIFFERENT hidden classes → slower

// Inline Caches (ICs):
// V8 caches property lookup results at call sites
// Monomorphic (1 shape) → fast
// Polymorphic (2-4 shapes) → slower
// Megamorphic (5+ shapes) → very slow (dictionary lookup)

// ✅ Monomorphic: same shape objects
function getX(obj) { return obj.x; } // Always same shape → fast
getX({ x: 1, y: 2 });
getX({ x: 3, y: 4 });

// ❌ Megamorphic: different shapes
getX({ x: 1 });
getX({ x: 1, y: 2 });
getX({ x: 1, y: 2, z: 3 });

// Optimization tips:
// 1. Use consistent object shapes (constructors/classes help)
// 2. Avoid delete (changes hidden class) — set to undefined instead
// 3. Use TypedArrays for numeric data (Float64Array, Int32Array)
// 4. Avoid try/catch in hot paths (prevented optimization in old V8)
// 5. Don't change variable types: let x = 1; x = "string"; // Bad!
// 6. Small functions get inlined by TurboFan

🏋️ Challenge 2: Implement Promise.all

function promiseAll(promises) {
  return new Promise((resolve, reject) => {
    if (!promises.length) return resolve([]);
    const results = new Array(promises.length);
    let pending = promises.length;
    
    promises.forEach((p, i) => {
      Promise.resolve(p).then(
        value => { results[i] = value; if (--pending === 0) resolve(results); },
        reject
      );
    });
  });
}

🏋️ Challenge 3: Implement debounce with leading/trailing options

function debounce(fn, delay, { leading = false, trailing = true } = {}) {
  let timeoutId, lastArgs;
  
  return function(...args) {
    const callNow = leading && !timeoutId;
    lastArgs = args;
    
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => {
      timeoutId = null;
      if (trailing && lastArgs) fn.apply(this, lastArgs);
      lastArgs = null;
    }, delay);
    
    if (callNow) fn.apply(this, args);
  };
}

🏋️ Challenge 4: Deep clone (handling circular references)

function deepClone(obj, seen = new WeakMap()) {
  if (obj === null || typeof obj !== "object") return obj;
  if (obj instanceof Date) return new Date(obj);
  if (obj instanceof RegExp) return new RegExp(obj);
  if (seen.has(obj)) return seen.get(obj); // Handle circular refs!
  
  const clone = Array.isArray(obj) ? [] : Object.create(Object.getPrototypeOf(obj));
  seen.set(obj, clone);
  
  for (const key of Reflect.ownKeys(obj)) {
    clone[key] = deepClone(obj[key], seen);
  }
  return clone;
}
// Or just: structuredClone(obj) — native ES2022!

Array.prototype.myFlat = function(depth = 1) {
  if (depth <= 0) return this.slice();
  return this.reduce((acc, item) => {
    if (Array.isArray(item)) {
      acc.push(...item.myFlat(depth - 1));

🏋️ Challenge 6: Implement EventEmitter

class EventEmitter {
  #events = new Map();
  
  on(event, fn) {
    if (!this.#events.has(event)) this.#events.set(event, []);
    this.#events.get(event).push(fn);
    return this;
  }
  
  off(event, fn) {
    const fns = this.#events.get(event);
    if (fns) this.#events.set(event, fns.filter(f => f !== fn && f._original !== fn));
    return this;
  }
  
  emit(event, ...args) {
    this.#events.get(event)?.forEach(fn => fn(...args));
    return this;
  }
  
  once(event, fn) {
    const wrapper = (...args) => { this.off(event, wrapper); fn(...args); };
    wrapper._original = fn;
    return this.on(event, wrapper);
  }
}

🏋️ Challenge 7: Implement throttle with leading/trailing

function throttle(fn, limit) {
  let lastTime = 0;
  let timeoutId;
  
  return function(...args) {
    const now = Date.now();
    const remaining = limit - (now - lastTime);
    
    clearTimeout(timeoutId);
    
    if (remaining <= 0) {
      lastTime = now;
      fn.apply(this, args);
    } else {
      timeoutId = setTimeout(() => {
        lastTime = Date.now();
        fn.apply(this, args);
      }, remaining);
    }
  };
}

🏋️ Challenge 8: Implement currying for unlimited arguments

// sum(1)(2)(3)() === 6
function sum(a) {
  let total = a;
  function inner(b) {
    if (b === undefined) return total;
    total += b;
    return inner;
  }
  inner.valueOf = () => total; // For implicit conversion
  return inner;
}
sum(1)(2)(3)(); // 6

// Generic curry:
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) return fn(...args);
    return (...more) => curried(...args, ...more);
  };
}

10. Top 50 Interview Questions & Answers

🎯 Rapid Fire — Commonly Asked

Easy 1. What's the difference between == and ===?
== does type coercion before comparison. === checks value AND type without coercion. Always use === except for null checks (val == null catches both null and undefined).

Easy 2. What are falsy values in JS?
false, 0, -0, 0n, "", null, undefined, NaN — everything else is truthy (including [], {}, "0").

Medium 3. Explain closure with an example.
A closure is a function that remembers variables from its outer scope even after the outer function has returned. function counter() { let count = 0; return () => ++count; } — the returned function "closes over" count.

Medium 4. What's the difference between var, let, and const?
var: function-scoped, hoisted (initialized to undefined), can redeclare. let: block-scoped, hoisted (TDZ), can reassign. const: block-scoped, hoisted (TDZ), cannot reassign (but objects/arrays can be mutated).

Medium 5. What is hoisting?
Declarations are moved to top of scope during compilation. var declarations initialized to undefined. function declarations fully hoisted (can call before declaration). let/const hoisted but in TDZ until declaration line.

Medium 6. Explain 'this' keyword.
'this' depends on HOW a function is called: global call → window/undefined; method → object; new → new instance; call/apply/bind → explicit; arrow → lexical (inherits from enclosing).

Medium 7. What is event delegation?
Attach one listener to parent instead of many to children. Uses event bubbling — check e.target to identify which child was clicked. Works for dynamically added elements, uses less memory.

Medium 8. Difference between null and undefined?
undefined: variable declared but not assigned, missing function params, missing object properties. null: intentional absence of value, explicitly set. typeof null === "object" (historic bug). typeof undefined === "undefined".

Medium 9. What is the event loop?
Mechanism that handles async operations in single-threaded JS. Continuously checks: call stack empty? → drain microtask queue → pick one macrotask → repeat. Enables non-blocking I/O despite single thread.

Medium 10. What's the difference between map, filter, and reduce?
map: transform each element, returns same-length array. filter: select elements by predicate, returns subset. reduce: accumulate into single value (sum, grouping, flattening).

Hard 11. Explain prototypal inheritance.
Objects inherit from other objects via [[Prototype]] chain. When accessing a property, JS walks up the chain until found or null. Constructor.prototype becomes __proto__ of instances created with new.

Hard 12. What's the difference between call, apply, and bind?
call: invoke immediately with args list. apply: invoke immediately with args array. bind: returns NEW function with 'this' permanently set — doesn't invoke. All set explicit 'this' context.

Hard 13. Explain Promises. How do they differ from callbacks?
Promise: object representing future value. States: pending→fulfilled/rejected (immutable). Advantages over callbacks: chainable (.then), centralized errors (.catch), composable (Promise.all), no inversion of control, guaranteed single resolution.

Hard 14. What is a memory leak? How do you find and fix them?
Memory that should be freed but isn't (still referenced). Common: forgotten timers, detached DOM refs, closures holding large scopes, event listeners not removed. Debug with Chrome DevTools Memory tab (heap snapshots, allocation timeline).

Hard 15. Explain async/await. What happens under the hood?
Syntactic sugar over Promises + generators. async function always returns Promise. await pauses execution (remainder becomes microtask). Under the hood: similar to yield in a generator consumed by a Promise runner.

Hard 16. What is debouncing and throttling?
Debounce: execute AFTER user stops triggering for N ms (search input). Throttle: execute at most once per N ms (scroll handler). Both prevent excessive function calls for performance.

Hard 17. Explain the Module pattern.
IIFE that returns an object exposing public API while keeping variables private via closure. Predecessor to ES modules. Still relevant for understanding encapsulation and Angular service patterns.

Hard 18. What are generators? Use cases?
Functions that can pause (yield) and resume. Use cases: lazy evaluation, infinite sequences, async flow control (redux-saga), custom iterables, state machines, data streaming.

Hard 19. Explain WeakMap vs Map.
Map: any key type, strong references, iterable, has size. WeakMap: object keys only, weak references (keys can be GC'd), NOT iterable, no size. Use WeakMap for metadata that shouldn't prevent GC (caching, private data).

Hard 20. How does JavaScript handle concurrency despite being single-threaded?
Event loop + Web APIs. Async operations (setTimeout, fetch, I/O) are offloaded to browser/OS threads. When complete, callbacks are queued. Event loop picks them up when call stack is empty. Result: non-blocking despite single thread.

Expert 21. Implement a deep equality check.
function deepEqual(a, b) { if (a === b) return true; if (typeof a !== "object" || typeof b !== "object" || !a || !b) return false; const keysA = Object.keys(a), keysB = Object.keys(b); if (keysA.length !== keysB.length) return false; return keysA.every(key => deepEqual(a[key], b[key])); }

Expert 22. What are Proxy and Reflect? Real-world uses?
Proxy intercepts operations on objects (get, set, delete, has, etc.). Reflect provides matching methods for forwarding. Used in: Vue 3 reactivity, validation layers, logging, observable objects, auto-vivification, negative array indexing.

Expert 23. Explain V8's hidden classes and inline caches.
V8 creates hidden classes (shapes) for objects with same property structure. Inline caches memoize property lookup results. Monomorphic (1 shape) is fast. Mixing shapes causes megamorphic (slow dictionary lookup). Keep object shapes consistent for performance.

Expert 24. How would you implement a simple reactive system (like Vue/Angular Signals)?
Track dependencies during read (Proxy get trap adds current effect to dependency set). Notify subscribers during write (Proxy set trap triggers all effects in dependency set). Key: automatic dependency tracking without manual subscription.

Expert 25. What is tail call optimization?
When the last action of a function is calling another function, the engine can reuse the current stack frame instead of adding a new one. Prevents stack overflow in recursion. Only Safari implements TCO; other engines don't. Use iteration or trampolining as alternatives.

🎯 Scenario-Based Questions

Hard Q36: How would you implement infinite scroll?
IntersectionObserver on sentinel element at bottom. When intersects: fetch next page, append items, move sentinel. Include loading state, error handling, and "no more items" detection. Debounce rapid scrolling.

Hard Q37: How would you implement a rate limiter in JS?
Token bucket: class RateLimiter { #tokens; #maxTokens; #refillRate; constructor(max, refillPerSec) { this.#tokens = max; this.#maxTokens = max; setInterval(() => { this.#tokens = Math.min(this.#maxTokens, this.#tokens + 1); }, 1000/refillPerSec); } tryAcquire() { if (this.#tokens > 0) { this.#tokens--; return true; } return false; } }

Hard Q38: How do you handle race conditions in frontend?
1. AbortController (cancel stale requests). 2. Request ID matching (only apply latest). 3. Mutex/lock for critical sections. 4. Debounce rapid inputs. 5. switchMap in RxJS (Angular: cancels previous observable). 6. Optimistic UI with rollback.

Expert Q39: Design a client-side caching system with TTL, LRU eviction, and cache invalidation.
Use Map (insertion-ordered). On get: move to end (delete + set). On set: if at capacity, delete first entry (LRU). Store {value, expiresAt}. On get: check expiry, delete if stale. Invalidation: pattern-based keys or tag-based grouping.

Expert Q40: How would you implement undo/redo in a text editor?
Command pattern: each action is an object with execute() and undo(). Two stacks: undoStack and redoStack. Execute → push to undo, clear redo. Undo → pop from undo, push to redo, call undo(). Redo → pop from redo, push to undo, call execute().

🎯 Angular-Specific JS Questions

Hard Q41: How does Zone.js work?
Zone.js monkey-patches all async APIs (setTimeout, Promise, addEventListener, fetch, etc.) to notify Angular when async operations complete. This triggers change detection. Angular 17+ signals reduce reliance on Zone.js.

Hard Q42: How does Angular change detection relate to the event loop?
Zone.js hooks into the end of microtask queue processing. After each async task completes and microtasks drain, Zone triggers ApplicationRef.tick() → runs change detection from root. OnPush strategy skips components unless inputs change or events fire within them.

Hard Q43: Why are RxJS Observables preferred over Promises in Angular?
Observables: lazy (don't execute until subscribed), cancellable (unsubscribe), can emit multiple values over time, composable (operators: map, filter, switchMap, debounceTime), retry/error handling built-in. Promises: eager, non-cancellable, single value only.

Expert Q44: How would you detect and fix a memory leak in an Angular app?
Symptoms: growing memory in DevTools. Common causes: unsubscribed Observables (fix: takeUntil + destroy$, or async pipe), event listeners in directives not cleaned in ngOnDestroy, large data cached in services. Debug: Chrome Memory → Heap snapshots → compare → find retained objects.

Expert Q45: Explain how Angular's DI system uses JavaScript concepts.
Uses class constructors + metadata (decorators/TypeScript reflect-metadata) to determine dependencies. Hierarchical injector tree mirrors component tree. Providers map tokens to values/factories. At runtime: look up token in injector hierarchy → instantiate with dependencies → cache as singleton (or create per component with component-level providers).
