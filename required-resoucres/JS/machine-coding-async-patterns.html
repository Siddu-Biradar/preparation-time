1. Event Loop — Theory & Output Prediction

HARD

Asked at: Google, Amazon, Microsoft, Flipkart, Uber, Razorpay — literally everywhere

Theory — The Mental Model

JavaScript is single-threaded. It has ONE call stack. But it handles async operations through the Event Loop:

/*
┌──────────────────────────────────────────────────────────────────┐
│                   HOW THE EVENT LOOP WORKS                        │
├──────────────────────────────────────────────────────────────────┤
│                                                                  │
│  1. Execute ALL synchronous code (call stack)                    │
│  2. Call stack empty? → Drain ALL microtasks                     │
│  3. Microtask queue empty? → Pick ONE macrotask                  │
│  4. After that macrotask → Drain ALL microtasks again            │
│  5. Repeat forever                                               │
│                                                                  │
├──────────────────────────────────────────────────────────────────┤
│  MICROTASKS (higher priority):                                   │
│  • Promise.then / catch / finally                                │
│  • queueMicrotask()                                              │
│  • MutationObserver                                              │
│  • process.nextTick() [Node.js only, even higher priority]       │
│                                                                  │
├──────────────────────────────────────────────────────────────────┤
│  MACROTASKS (lower priority):                                    │
│  • setTimeout / setInterval                                      │
│  • setImmediate() [Node.js only]                                 │
│  • I/O callbacks                                                 │
│  • UI rendering (between macrotasks in browsers)                 │
│  • requestAnimationFrame (before repaint)                        │
│                                                                  │
├──────────────────────────────────────────────────────────────────┤
│  KEY RULE: Microtasks can STARVE macrotasks!                     │
│  If a microtask enqueues another microtask, it runs immediately  │
│  before any macrotask gets a chance.                             │
└──────────────────────────────────────────────────────────────────┘
*/

The Golden Rules

  1. Promise executor is SYNC — new Promise((resolve, reject) => { /* THIS IS SYNC */ })
  2. await pauses the function, not the thread — code AFTER await becomes a microtask
  3. Microtasks always run before macrotasks — setTimeout(0) loses to Promise.then()
  4. Nested microtasks run immediately — they don't wait for the next "tick"

2. Implement Promise from Scratch

HARD

Asked at: Google, Microsoft, Flipkart, Atlassian, Razorpay, PhonePe

Theory — How Promises Work Internally

A Promise is a state machine with 3 states: pending → fulfilled OR pending → rejected. Once settled, it NEVER changes state again.

/*
  Promise Internals:
  ┌─────────┐     resolve(value)     ┌───────────┐
  │ PENDING │ ─────────────────────→ │ FULFILLED │
  │         │                        └───────────┘
  │         │     reject(reason)     ┌──────────┐
  │         │ ─────────────────────→ │ REJECTED │
  └─────────┘                        └──────────┘

  Key implementation details:
  1. Store callbacks array (for .then called before resolution)
  2. Handle "thenable" values (if resolve(anotherPromise))
  3. .then() ALWAYS returns a new Promise (chaining!)
  4. Callbacks must be async (queueMicrotask)
*/

Complete Implementation

class MyPromise {
  constructor(executor) {
    this.state = 'pending';   // pending | fulfilled | rejected
    this.value = undefined;   // resolved value or rejection reason
    this.callbacks = [];      // [{onFulfilled, onRejected}] — queued .then handlers

    const resolve = (value) => {
      if (this.state !== 'pending') return; // Ignore if already settled
      
      // If value is a thenable, unwrap it
      if (value && (typeof value === 'object' || typeof value === 'function') && typeof value.then === 'function') {
        value.then(resolve, reject);
        return;
      }
      
      this.state = 'fulfilled';
      this.value = value;
      this.callbacks.forEach(cb => cb.onFulfilled(value));
    };

    const reject = (reason) => {
      if (this.state !== 'pending') return;
      this.state = 'rejected';
      this.value = reason;
      this.callbacks.forEach(cb => cb.onRejected(reason));
    };

    try {
      executor(resolve, reject);
    } catch (err) {
      reject(err); // Executor throws → auto-reject
    }
  }

  then(onFulfilled, onRejected) {
    // .then ALWAYS returns a new Promise for chaining
    return new MyPromise((resolve, reject) => {
      const handleFulfilled = (value) => {
        queueMicrotask(() => {
          try {
            if (typeof onFulfilled === 'function') {
              resolve(onFulfilled(value));
            } else {
              resolve(value); // Pass through if no handler
            }
          } catch (err) {
            reject(err);
          }
        });
      };

      const handleRejected = (reason) => {
        queueMicrotask(() => {
          try {
            if (typeof onRejected === 'function') {
              resolve(onRejected(reason)); // catch returns resolved promise!
            } else {
              reject(reason); // Propagate rejection
            }
          } catch (err) {
            reject(err);
          }
        });
      };

      // If already settled, run immediately (async)
      if (this.state === 'fulfilled') handleFulfilled(this.value);
      else if (this.state === 'rejected') handleRejected(this.value);
      else this.callbacks.push({ onFulfilled: handleFulfilled, onRejected: handleRejected });
    });
  }

  catch(onRejected) {
    return this.then(null, onRejected);
  }

  finally(fn) {
    return this.then(
      value => { fn(); return value; },
      reason => { fn(); throw reason; }
    );
  }
}

// Usage:
const p = new MyPromise((resolve) => {
  setTimeout(() => resolve(42), 1000);
});

p.then(val => val * 2)
 .then(val => console.log(val)) // 84
 .catch(err => console.error(err));
🔑 Interview Key Points:
  • Executor runs synchronously in constructor
  • .then() returns a NEW promise (not the same one)
  • Resolution with a thenable recursively unwraps
  • Must use queueMicrotask (or setTimeout) for async execution
  • Catch is just .then(null, onRejected)

3. Promise.all / race / allSettled / any (Polyfills)

MEDIUM

Asked at: Amazon, Google, Flipkart, Razorpay, PhonePe, Paytm, Swiggy

Theory — When to Use Which

MethodResolves WhenRejects WhenUse Case
Promise.allALL resolveFIRST rejects (fail-fast)Load all resources, parallel API calls that ALL must succeed
Promise.raceFIRST settles (resolve OR reject)FIRST settlesTimeout pattern, fastest server wins
Promise.allSettledALL settle (never rejects)NeverFire-and-forget, batch where some can fail
Promise.anyFIRST resolvesALL reject (AggregateError)Fastest success, CDN fallback, first available

Complete Polyfill Implementations

// ━━━━━ Promise.all ━━━━━
// Resolves with array of all values. Rejects immediately on first rejection.
Promise.myAll = function(promises) {
  return new Promise((resolve, reject) => {
    if (!promises.length) return resolve([]);
    
    const results = [];
    let completed = 0;
    
    promises.forEach((p, i) => {
      Promise.resolve(p).then(
        value => {
          results[i] = value; // Maintain order!
          if (++completed === promises.length) resolve(results);
        },
        reject // First rejection rejects entire Promise.all
      );
    });
  });
};

// ━━━━━ Promise.race ━━━━━
// First promise to settle (resolve OR reject) wins.
Promise.myRace = function(promises) {
  return new Promise((resolve, reject) => {
    promises.forEach(p => Promise.resolve(p).then(resolve, reject));
  });
};

// ━━━━━ Promise.allSettled ━━━━━
// NEVER rejects. Returns [{status, value/reason}] for each promise.
Promise.myAllSettled = function(promises) {
  return new Promise((resolve) => {
    if (!promises.length) return resolve([]);
    
    const results = [];
    let completed = 0;
    
    promises.forEach((p, i) => {
      Promise.resolve(p).then(
        value => {
          results[i] = { status: 'fulfilled', value };
          if (++completed === promises.length) resolve(results);
        },
        reason => {
          results[i] = { status: 'rejected', reason };
          if (++completed === promises.length) resolve(results);
        }
      );
    });
  });
};

// ━━━━━ Promise.any ━━━━━
// First to RESOLVE wins. Only rejects if ALL reject (AggregateError).
Promise.myAny = function(promises) {
  return new Promise((resolve, reject) => {
    if (!promises.length) return reject(new AggregateError([], 'All promises rejected'));
    
    const errors = [];
    let rejectedCount = 0;
    
    promises.forEach((p, i) => {
      Promise.resolve(p).then(
        resolve, // First resolve wins!
        reason => {
          errors[i] = reason;
          if (++rejectedCount === promises.length) {
            reject(new AggregateError(errors, 'All promises were rejected'));
          }
        }
      );
    });
  });
};

// CUSTOM PROMISE IMPLEMENTATION (FROM SCRATCH)
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
class MyPromise {
  constructor(executor) {
    this.state = 'pending';
    this.value = undefined;
    this.callbacks = [];

    const resolve = (value) => {
      if (this.state !== 'pending') return;
      // Handle thenable
      if (value && typeof value.then === 'function') {
        value.then(resolve, reject);
        return;
      }
      this.state = 'fulfilled';
      this.value = value;
      this.callbacks.forEach(cb => cb.onFulfilled(value));
    };

    const reject = (reason) => {
      if (this.state !== 'pending') return;
      this.state = 'rejected';
      this.value = reason;
      this.callbacks.forEach(cb => cb.onRejected(reason));
    };

    try {
      executor(resolve, reject);
    } catch (err) {
      reject(err);
    }
  }

  then(onFulfilled, onRejected) {
    return new MyPromise((resolve, reject) => {
      const handle = (fn, fallback) => (value) => {
        try {
          if (typeof fn === 'function') {
            resolve(fn(value));
          } else {
            fallback(value);
          }
        } catch (err) {
          reject(err);
        }
      };

      if (this.state === 'fulfilled') {
        queueMicrotask(() => handle(onFulfilled, resolve)(this.value));
      } else if (this.state === 'rejected') {
        queueMicrotask(() => handle(onRejected, reject)(this.value));
      } else {
        this.callbacks.push({
          onFulfilled: handle(onFulfilled, resolve),
          onRejected: handle(onRejected, reject),
        });
      }
    });
  }

  catch(onRejected) { return this.then(null, onRejected); }
  finally(fn) { return this.then(v => { fn(); return v; }, r => { fn(); throw r; }); }

  // ━━━━━━━ STATIC METHODS ━━━━━━━
  static resolve(val) { return new MyPromise(res => res(val)); }
  static reject(val) { return new MyPromise((_, rej) => rej(val)); }

  static all(promises) {
    return new MyPromise((resolve, reject) => {
      const results = [];
      let completed = 0;
      if (promises.length === 0) return resolve([]);

      promises.forEach((p, i) => {
        MyPromise.resolve(p).then(val => {
          results[i] = val;
          if (++completed === promises.length) resolve(results);
        }, reject); // First rejection rejects all
      });
    });
  }

  static race(promises) {
    return new MyPromise((resolve, reject) => {
      promises.forEach(p => MyPromise.resolve(p).then(resolve, reject));
    });
  }

  static allSettled(promises) {
    return new MyPromise((resolve) => {
      const results = [];
      let completed = 0;
      if (promises.length === 0) return resolve([]);

      promises.forEach((p, i) => {
        MyPromise.resolve(p).then(
          val => { results[i] = { status: 'fulfilled', value: val }; if (++completed === promises.length) resolve(results); },
          reason => { results[i] = { status: 'rejected', reason }; if (++completed === promises.length) resolve(results); }
        );
      });
    });
  }

  static any(promises) {
    return new MyPromise((resolve, reject) => {
      const errors = [];
      let rejectedCount = 0;
      if (promises.length === 0) return reject(new AggregateError([], 'All promises were rejected'));

      promises.forEach((p, i) => {
        MyPromise.resolve(p).then(resolve, err => {
          errors[i] = err;
          if (++rejectedCount === promises.length) {
            reject(new AggregateError(errors, 'All promises were rejected'));
          }
        });
      });
    });
  }
}

4. Retry with Exponential Backoff

MEDIUM

Asked at: Google, Amazon, Uber, Flipkart, Razorpay, Swiggy

Theory

Exponential Backoff: Wait time doubles after each failure: 200ms → 400ms → 800ms → 1600ms. This prevents thundering herd (all clients retrying simultaneously after a service recovers).

Jitter: Add randomness to prevent synchronized retries across clients. Without jitter, 10,000 clients all retry at exactly the same exponential intervals.

/*
  Retry Strategy:
  ┌──────────────┐     ┌──────────────┐     ┌──────────────┐
  │  Attempt 1   │────→│  Wait 200ms  │────→│  Attempt 2   │───→ ...
  │  (FAIL)      │     │  (backoff)   │     │  (FAIL)      │
  └──────────────┘     └──────────────┘     └──────────────┘
  
  Backoff formula: delay = baseDelay * (factor ^ attemptNumber)
  With jitter:     delay = baseDelay * (factor ^ attempt) * random(0.5, 1.5)
*/

Complete Implementation

async function retry(fn, options = {}) {
  const {
    retries = 3,          // Max retry attempts
    baseDelay = 200,      // Initial delay in ms
    factor = 2,           // Exponential factor
    maxDelay = 10000,     // Cap maximum delay
    jitter = true,        // Add randomness
    retryOn = null,       // Function to decide if error is retryable
    onRetry = null,       // Callback on each retry
  } = options;

  let lastError;

  for (let attempt = 0; attempt <= retries; attempt++) {
    try {
      return await fn(attempt);
    } catch (err) {
      lastError = err;

      // Check if we should retry this specific error
      if (retryOn && !retryOn(err)) throw err;
      
      // No more retries left
      if (attempt === retries) break;

      // Calculate delay with exponential backoff
      let delay = Math.min(baseDelay * Math.pow(factor, attempt), maxDelay);
      
      // Add jitter (±50%)
      if (jitter) {
        delay *= 0.5 + Math.random();
      }

      if (onRetry) onRetry(attempt + 1, delay, err);
      
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }

  throw lastError;
}

// ━━━━━ Usage Examples ━━━━━

// Basic retry
const data = await retry(() => fetch('/api/data').then(r => {
  if (!r.ok) throw new Error(`HTTP ${r.status}`);
  return r.json();
}), { retries: 3, baseDelay: 1000 });

// Retry only on network errors (not 4xx)
await retry(fetchUserData, {
  retries: 5,
  retryOn: (err) => err.message.includes('503') || err.message.includes('timeout'),
  onRetry: (attempt, delay) => console.log(`Retry ${attempt}, waiting ${delay}ms`),
});

  }

5. Promise Pool / Concurrency Limiter

HARD

Asked at: Google, Amazon, Uber, Atlassian, Flipkart, Razorpay

Theory

Problem: You have 100 URLs to fetch but don't want 100 simultaneous connections (server overload, browser limit is ~6 per domain). Solution: Run at most N tasks concurrently.

/*
  Concurrency Pool (limit = 3):
  
  Time ──────────────────────────────────────────→
  
  Slot 1: [Task 1████] [Task 4████] [Task 7██]
  Slot 2: [Task 2██████] [Task 5██] [Task 8████]
  Slot 3: [Task 3████] [Task 6████████]
  
  When a slot finishes, next queued task immediately starts.
*/

Complete Implementation

// ━━━━━ Approach 1: Worker-based (clean & readable) ━━━━━
async function promisePool(tasks, concurrency = 3) {
  const results = [];
  let index = 0;

  async function worker() {
    while (index < tasks.length) {
      const currentIndex = index++;
      results[currentIndex] = await tasks[currentIndex]();
    }
  }

  // Create N workers that pull from shared task list
  await Promise.all(
    Array.from({ length: Math.min(concurrency, tasks.length) }, () => worker())
  );

  return results;
}

// ━━━━━ Approach 2: Queue-based (more control) ━━━━━
class PromisePool {
  constructor(concurrency) {
    this.concurrency = concurrency;
    this.running = 0;
    this.queue = [];
  }

  async add(taskFn) {
    if (this.running >= this.concurrency) {
      // Wait until a slot opens up
      await new Promise(resolve => this.queue.push(resolve));
    }
    
    this.running++;
    try {
      return await taskFn();
    } finally {
      this.running--;
      if (this.queue.length > 0) {
        this.queue.shift()(); // Release next waiting task
      }
    }
  }
}

// ━━━━━ Usage ━━━━━
const urls = Array.from({ length: 20 }, (_, i) => `/api/item/${i}`);

// Convert URLs to task factories (functions that return promises)
const tasks = urls.map(url => () => fetch(url).then(r => r.json()));

const results = await promisePool(tasks, 5); // Only 5 concurrent requests

// Class-based usage:

const promises = urls.map(url => pool.add(() => fetch(url)));
const results2 = await Promise.all(promises);

6. Async Task Queue (FIFO)

MEDIUM

Asked at: Razorpay, Flipkart, Atlassian, Microsoft

Theory

An async queue processes tasks in FIFO order, respecting a concurrency limit. Unlike Promise Pool (which gets all tasks upfront), a queue allows dynamic enqueuing at any time.

Complete Implementation

class AsyncQueue {
  constructor(concurrency = 1) {
    this.concurrency = concurrency;
    this.queue = [];
    this.running = 0;
    this.paused = false;
  }

  enqueue(task, priority = 0) {
    return new Promise((resolve, reject) => {
      const item = { task, resolve, reject, priority };
      
      // Insert by priority (higher priority = earlier execution)
      const insertIdx = this.queue.findIndex(q => q.priority < priority);
      if (insertIdx === -1) this.queue.push(item);
      else this.queue.splice(insertIdx, 0, item);
      
      this.process();
    });
  }

  async process() {
    if (this.paused) return;
    
    while (this.running < this.concurrency && this.queue.length > 0) {
      const { task, resolve, reject } = this.queue.shift();
      this.running++;
      
      // Don't await here — let multiple tasks run concurrently
      task()
        .then(resolve)
        .catch(reject)
        .finally(() => {
          this.running--;
          this.process(); // Check if more tasks can run
        });
    }
  }

  pause() { this.paused = true; }
  resume() { this.paused = false; this.process(); }
  clear() { this.queue = []; }
  get size() { return this.queue.length; }
  get pending() { return this.running; }
}

// ━━━━━ Usage ━━━━━
const queue = new AsyncQueue(2); // Max 2 concurrent

// Enqueue tasks dynamically
queue.enqueue(() => fetch('/api/1').then(r => r.json()));
queue.enqueue(() => fetch('/api/2').then(r => r.json()));
queue.enqueue(() => fetch('/api/critical').then(r => r.json()), 10); // High priority!
queue.enqueue(() => fetch('/api/3').then(r => r.json()));

7. Sequential vs Parallel Execution

EASY

Asked at: Flipkart, Amazon, Razorpay, Paytm, PhonePe

Theory

/*
  Sequential: Total time = SUM of all task times
  ────[T1 300ms]────[T2 200ms]────[T3 400ms]────  = 900ms

  Parallel:   Total time = MAX of all task times  
  ────[T1 300ms]────
  ────[T2 200ms]──
  ────[T3 400ms]──────   = 400ms

  When to use which:
  • Sequential: Each task depends on previous result (waterfall)
  • Parallel: Tasks are independent (fetch multiple APIs)
  • Pool: Parallel but with a concurrency limit
*/

Complete Implementation

// ━━━━━ Sequential (for...of + await) ━━━━━
async function sequential(tasks) {
  const results = [];
  for (const task of tasks) {
    results.push(await task()); // Wait for each before starting next
  }
  return results;
}

// ━━━━━ Parallel (Promise.all) ━━━━━
async function parallel(tasks) {
  return Promise.all(tasks.map(task => task()));
}

// ━━━━━ Waterfall (each gets previous result) ━━━━━
async function waterfall(tasks, initialValue) {
  let result = initialValue;
  for (const task of tasks) {
    result = await task(result); // Pass previous result to next
  }
  return result;
}

// ━━━━━ reduce pattern (sequential with accumulator) ━━━━━
async function sequentialReduce(urls) {
  return urls.reduce(async (prevPromise, url) => {
    const results = await prevPromise;
    const data = await fetch(url).then(r => r.json());
    return [...results, data];
  }, Promise.resolve([]));
}

// ━━━━━ Practical Example: Dependent API calls ━━━━━
async function getUserPosts(userId) {
  // Sequential — each depends on previous
  const user = await fetch(`/api/user/${userId}`).then(r => r.json());
  const posts = await fetch(`/api/posts?author=${user.id}`).then(r => r.json());
  const comments = await Promise.all(
    posts.map(post => fetch(`/api/comments?post=${post.id}`).then(r => r.json()))
  );
  return { user, posts, comments };
}

// ━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 4. ASYNC TASK QUEUE (FIFO)
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━
class AsyncQueue {
  constructor(concurrency = 2) {
    this.concurrency = concurrency;
    this.queue = [];
    this.running = 0;
  }

  enqueue(task) {
    return new Promise((resolve, reject) => {
      this.queue.push({ task, resolve, reject });
      this.process();
    });
  }

  async process() {
    while (this.running < this.concurrency && this.queue.length > 0) {
      const { task, resolve, reject } = this.queue.shift();
      this.running++;
      try {
        const result = await task();
        resolve(result);
      } catch (err) {
        reject(err);
      } finally {
        this.running--;
        this.process();
      }
    }
  }
}

async function runQueue() {
  const queue = new AsyncQueue(2);
  log('log4', '▶ Enqueuing 6 tasks (concurrency=2)...', 'warn');

  for (let i = 1; i <= 6; i++) {
    const ms = 400 + Math.floor(Math.random() * 600);
    queue.enqueue(() => new Promise(resolve => {
      log('log4', `  🟡 Task ${i} executing (${ms}ms)...`, 'warn');
      setTimeout(() => { log('log4', `  ✅ Task ${i} complete`, 'ok'); resolve(); }, ms);
    }));
  }
}

8. Cancellable Fetch (AbortController)

MEDIUM

Asked at: Google, Uber, Flipkart, Atlassian, Microsoft

Theory

AbortController is the standard way to cancel async operations. It creates a signal that you pass to fetch (or any async operation). Calling abort() triggers an AbortError.

/*
  AbortController flow:
  
  controller = new AbortController()
       │
       ├── signal = controller.signal  ──→  Pass to fetch/listener
       │
       └── controller.abort()  ──→  signal fires 'abort' event
                                   ──→  fetch rejects with AbortError
                                   ──→  signal.aborted = true
*/

Complete Implementation

// ━━━━━ Basic cancellable fetch ━━━━━
function createCancellableFetch() {
  let controller = null;

  return {
    fetch(url, options = {}) {
      // Cancel previous request if still in-flight
      if (controller) controller.abort();
      
      controller = new AbortController();
      
      return fetch(url, {
        ...options,
        signal: controller.signal,
      });
    },
    cancel() {
      if (controller) controller.abort();
    }
  };
}

// ━━━━━ Fetch with timeout (auto-cancel) ━━━━━
async function fetchWithTimeout(url, { timeout = 5000, ...options } = {}) {
  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), timeout);
  
  try {
    const response = await fetch(url, { ...options, signal: controller.signal });
    if (!response.ok) throw new Error(`HTTP ${response.status}`);
    return await response.json();
  } catch (err) {
    if (err.name === 'AbortError') {
      throw new Error(`Request timed out after ${timeout}ms`);
    }
    throw err;
  } finally {
    clearTimeout(timeoutId);
  }
}

// ━━━━━ Cancel-on-unmount pattern (React-style) ━━━━━
function useEffect_fetchPattern() {
  const controller = new AbortController();

  fetch('/api/data', { signal: controller.signal })
    .then(r => r.json())
    .then(data => { /* update state only if not aborted */ })
    .catch(err => {
      if (err.name !== 'AbortError') console.error(err);
    });

  // Cleanup (runs on component unmount)
  return () => controller.abort();
}

// ━━━━━ Cancellable Promise wrapper ━━━━━
function makeCancellable(promise) {
  let cancelled = false;
  
  const wrappedPromise = new Promise((resolve, reject) => {
    promise.then(
      val => cancelled ? reject({ isCancelled: true }) : resolve(val),
      err => cancelled ? reject({ isCancelled: true }) : reject(err)
    );
  });

  return { promise: wrappedPromise, cancel: () => { cancelled = true; } };
}

9. Rate Limiter

HARD

Asked at: Google, Uber, Razorpay, Flipkart, Stripe

Theory

Rate limiting ensures a function executes at most N times within a time window. Common algorithms:

  • Token Bucket: Tokens refill at a constant rate. Each call uses a token. No tokens? Wait.
  • Sliding Window: Track timestamps of recent calls, reject if count exceeds limit in window.
  • Leaky Bucket: Queue requests, process at a fixed rate (like a leaking bucket).

Complete Implementation

// ━━━━━ Token Bucket Rate Limiter ━━━━━
class RateLimiter {
  constructor({ maxTokens = 10, refillRate = 1, refillInterval = 1000 } = {}) {
    this.maxTokens = maxTokens;
    this.tokens = maxTokens;
    this.refillRate = refillRate; // Tokens added per interval
    this.refillInterval = refillInterval;
    this.queue = [];
    
    // Refill tokens periodically
    this.intervalId = setInterval(() => {
      this.tokens = Math.min(this.maxTokens, this.tokens + this.refillRate);
      this.processQueue();
    }, this.refillInterval);
  }

  async execute(fn) {
    if (this.tokens > 0) {
      this.tokens--;
      return fn();
    }
    
    // Queue the request — wait for a token
    return new Promise((resolve, reject) => {
      this.queue.push({ fn, resolve, reject });
    });
  }

  processQueue() {
    while (this.tokens > 0 && this.queue.length > 0) {
      this.tokens--;
      const { fn, resolve, reject } = this.queue.shift();
      fn().then(resolve).catch(reject);
    }
  }

  destroy() { clearInterval(this.intervalId); }
}

// ━━━━━ Sliding Window Rate Limiter ━━━━━
class SlidingWindowLimiter {
  constructor(maxRequests, windowMs) {
    this.maxRequests = maxRequests;
    this.windowMs = windowMs;
    this.timestamps = [];
  }

  async execute(fn) {
    const now = Date.now();
    
    // Remove timestamps outside the window
    this.timestamps = this.timestamps.filter(t => now - t < this.windowMs);
    
    if (this.timestamps.length >= this.maxRequests) {
      // Calculate wait time
      const oldestInWindow = this.timestamps[0];
      const waitTime = this.windowMs - (now - oldestInWindow);
      await new Promise(r => setTimeout(r, waitTime));
      return this.execute(fn); // Retry after waiting
    }
    
    this.timestamps.push(now);
    return fn();
  }
}

// ━━━━━ Usage ━━━━━
const limiter = new RateLimiter({ maxTokens: 5, refillRate: 2, refillInterval: 1000 });

// Only 5 calls allowed initially, then 2 per second
async function makeAPICall(id) {
  return limiter.execute(async () => {
    const res = await fetch(`/api/item/${id}`);
    return res.json();
  });
}

// Fire 20 requests — they'll be rate-limited automatically
const results = await Promise.all(
  Array.from({ length: 20 }, (_, i) => makeAPICall(i))
);

10. Auto-Retry Fetch Wrapper

MEDIUM

Asked at: Flipkart, Swiggy, Razorpay, Amazon

Complete Implementation

// Production-ready fetch wrapper with retry, timeout, and interceptors
function createFetchClient(defaultOptions = {}) {
  const {
    baseURL = '',
    timeout = 10000,
    retries = 3,
    retryDelay = 1000,
    retryOn = [408, 429, 500, 502, 503, 504], // Status codes to retry
    onRequest = null,  // Request interceptor
    onResponse = null, // Response interceptor
    onError = null,    // Error interceptor
  } = defaultOptions;

  async function request(url, options = {}) {
    const fullURL = baseURL + url;
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), options.timeout || timeout);
    
    let lastError;

    for (let attempt = 0; attempt <= retries; attempt++) {
      try {
        // Request interceptor
        let requestOptions = { ...options, signal: controller.signal };
        if (onRequest) requestOptions = await onRequest(requestOptions);

        const response = await fetch(fullURL, requestOptions);
        clearTimeout(timeoutId);

        // Check if we should retry based on status code
        if (!response.ok && retryOn.includes(response.status) && attempt < retries) {
          const delay = retryDelay * Math.pow(2, attempt);
          await new Promise(r => setTimeout(r, delay));
          continue;
        }

        if (!response.ok) {
          throw new Error(`HTTP ${response.status}: ${response.statusText}`);
        }

        const data = await response.json();
        
        // Response interceptor
        return onResponse ? await onResponse(data, response) : data;
      } catch (err) {
        lastError = err;
        clearTimeout(timeoutId);

        if (err.name === 'AbortError') {
          throw new Error(`Request timeout after ${timeout}ms`);
        }

        if (attempt < retries) {
          const delay = retryDelay * Math.pow(2, attempt);
          await new Promise(r => setTimeout(r, delay));
          continue;
        }
      }
    }

    if (onError) return onError(lastError);
    throw lastError;
  }

  return {
    get: (url, opts) => request(url, { ...opts, method: 'GET' }),
    post: (url, body, opts) => request(url, { ...opts, method: 'POST', body: JSON.stringify(body), headers: { 'Content-Type': 'application/json', ...opts?.headers } }),
    put: (url, body, opts) => request(url, { ...opts, method: 'PUT', body: JSON.stringify(body), headers: { 'Content-Type': 'application/json', ...opts?.headers } }),
    delete: (url, opts) => request(url, { ...opts, method: 'DELETE' }),
  };
}

// Usage:
const api = createFetchClient({
  baseURL: 'https://api.example.com',
  timeout: 8000,
  retries: 3,
  onRequest: (opts) => {
    opts.headers = { ...opts.headers, Authorization: `Bearer ${getToken()}` };
    return opts;
  },
});

const user = await api.get('/users/123');
await api.post('/users', { name: 'John', email: 'john@example.com' });

// ━━━━━ Promise.race approach (standard) ━━━━━
function withTimeout(promise, ms, message = 'Operation timed out') {
  const timeout = new Promise((_, reject) => {
    setTimeout(() => reject(new Error(message)), ms);
  });
  return Promise.race([promise, timeout]);
}

// ━━━━━ With AbortController (cancels the actual operation) ━━━━━
function fetchWithTimeout(url, ms = 5000) {
  const controller = new AbortController();
  const timeoutId = setTimeout(() => controller.abort(), ms);
  
  return fetch(url, { signal: controller.signal })
    .then(response => { clearTimeout(timeoutId); return response; })
    .catch(err => {
      clearTimeout(timeoutId);
      if (err.name === 'AbortError') throw new Error(`Fetch timed out after ${ms}ms`);
      throw err;
    });
}

// ━━━━━ Configurable timeout with cleanup ━━━━━
function timeoutPromise(fn, ms) {
  let timeoutId;
  let aborted = false;
  
  const promise = new Promise(async (resolve, reject) => {
    timeoutId = setTimeout(() => {
      aborted = true;
      reject(new Error(`Timed out after ${ms}ms`));
    }, ms);
    
    try {
      const result = await fn(() => aborted); // Pass "isAborted" checker
      if (!aborted) resolve(result);
    } catch (err) {
      if (!aborted) reject(err);
    } finally {
      clearTimeout(timeoutId);
    }
  });

  return promise;
}

  fetch('/api/slow-endpoint').then(r => r.json()),

12. Lazy / Deferred Promise

EASY

Asked at: Atlassian, Razorpay, Freshworks

Theory

A deferred promise doesn't start executing until you explicitly call it. Useful for lazy initialization, on-demand API calls, and conditional execution.

Complete Implementation

// ━━━━━ Deferred (resolve/reject from outside) ━━━━━
function createDeferred() {
  let resolve, reject;
  const promise = new Promise((res, rej) => {
    resolve = res;
    reject = rej;
  });
  return { promise, resolve, reject };
}

// ━━━━━ Lazy Promise (executes only when awaited) ━━━━━
class LazyPromise {
  constructor(executor) {
    this.executor = executor;
    this.promise = null;
  }

  then(onFulfilled, onRejected) {
    if (!this.promise) {
      this.promise = new Promise(this.executor);
    }
    return this.promise.then(onFulfilled, onRejected);
  }

  catch(onRejected) { return this.then(null, onRejected); }
  finally(fn) { return this.then(v => { fn(); return v; }, r => { fn(); throw r; }); }
}

// ━━━━━ Memoized async (execute once, cache forever) ━━━━━
function memoizeAsync(fn) {
  let cache = null;
  let pending = null;

  return async function(...args) {
    if (cache) return cache;
    if (pending) return pending; // Deduplicate concurrent calls
    
    pending = fn(...args).then(result => {
      cache = result;
      pending = null;
      return result;
    }).catch(err => {
      pending = null; // Allow retry on error
      throw err;
    });

    return pending;
  };
}

// ━━━━━ Usage ━━━━━
// Deferred — control from outside
const { promise, resolve } = createDeferred();
setTimeout(() => resolve('done!'), 1000);
const result = await promise; // Waits until resolve is called

// Lazy — only fetches when you actually need it
const lazyUser = new LazyPromise((resolve) => {
  console.log('Fetching user...'); // Only runs when awaited!
  fetch('/api/user').then(r => r.json()).then(resolve);
});
// ... later, when actually needed:
const user = await lazyUser;

// Memoized — fetch config once, cache for lifetime
const getConfig = memoizeAsync(() => fetch('/api/config').then(r => r.json()));
const config1 = await getConfig(); // Fetches
const config2 = await getConfig(); // Returns cached
