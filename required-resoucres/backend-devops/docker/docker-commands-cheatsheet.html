1. Image Management

# Build an image from the Dockerfile in the current directory
docker build -t di-jfrog:local .                         # DAILY

# Build with a specific Dockerfile path and build-time args
docker build -f docker/Dockerfile --build-arg API_URL=https://x -t app:dev .   # MEDIUM

# List local images
docker images                                             # DAILY

# Remove an image (add -f to force if containers reference it)
docker rmi di-jfrog:local                                 # MEDIUM

# Tag an image (e.g. before pushing to a registry)
docker tag di-jfrog:local kyndryl.jfrog.io/ibmcb-docker-local/dash/portal_jfrog_app:1.2.0   # DAILY

# Pull / push
docker pull mongo:7                                       # DAILY
docker push kyndryl.jfrog.io/ibmcb-docker-local/dash/portal_jfrog_app:1.2.0   # DAILY

# Show the layer history of an image (great for debugging image size)
docker history di-jfrog:local                             # MEDIUM

# Inspect full image metadata (env, entrypoint, exposed ports, layers)
docker inspect di-jfrog:local                             # MEDIUM

# Save/load an image as a tar file (air-gapped transfer, no registry access)
docker save -o app.tar di-jfrog:local                     # MEDIUM
docker load -i app.tar                                    # MEDIUM

2. Container Lifecycle

# Run in background, map a port, name it
docker run -d -p 8080:8080 --name jfrog-svc di-jfrog:local     # DAILY

# Run interactively and auto-remove when it exits (great for one-off debugging)
docker run -it --rm alpine sh                              # DAILY

# Run with env vars and a volume mounted
docker run -d -p 8080:8080 \
  -e DATABASE_URL=mongodb://mongo:27017/jfrog \
  -v $(pwd)/config.json:/app/config.json:ro \
  --name jfrog-svc di-jfrog:local                          # DAILY

# List running containers / all containers (including stopped)
docker ps                                                  # DAILY
docker ps -a                                                # DAILY

# Start / stop / restart an existing (stopped) container
docker start jfrog-svc                                     # DAILY
docker stop jfrog-svc                                       # DAILY
docker restart jfrog-svc                                    # DAILY

# Remove a container (must be stopped, or add -f)
docker rm jfrog-svc                                         # DAILY
docker rm -f jfrog-svc                                       # MEDIUM

# Rename a running/stopped container
docker rename jfrog-svc jfrog-svc-old                       # MEDIUM

# Pause/unpause all processes in a container (freezes without stopping)
docker pause jfrog-svc && docker unpause jfrog-svc          # MEDIUM

3. Inspecting, Logs & Debugging

# Tail logs (follow), with timestamps, last N lines
docker logs -f jfrog-svc                                    # DAILY
docker logs --tail 100 -t jfrog-svc                         # DAILY

# Shell into a running container
docker exec -it jfrog-svc sh                                 # DAILY
docker exec -it jfrog-svc bash                               # DAILY (if bash is available)

# Run a one-off command inside a running container without a full shell
docker exec jfrog-svc env                                     # MEDIUM
docker exec jfrog-svc cat /app/config.json                    # MEDIUM

# Full container metadata: env vars, mounts, network, restart count, exit code
docker inspect jfrog-svc                                      # DAILY
docker inspect --format '{{.State.ExitCode}}' jfrog-svc        # MEDIUM
docker inspect --format '{{.State.OOMKilled}}' jfrog-svc       # MEDIUM

# Live resource usage (CPU/mem) — like `top` for containers
docker stats                                                   # MEDIUM
docker stats jfrog-svc --no-stream                              # MEDIUM

# Processes running inside a container
docker top jfrog-svc                                            # MEDIUM

# Diff of filesystem changes vs the image (added/changed/deleted files)
docker diff jfrog-svc                                            # MEDIUM

# Bypass a crashing ENTRYPOINT to get a shell for debugging
docker run -it --entrypoint sh di-jfrog:local                    # MEDIUM

# Which ports are published, from where
docker port jfrog-svc                                             # MEDIUM

4. Volumes & Copying Data

# Create/list/inspect/remove a named volume
docker volume create mongo-data                             # MEDIUM
docker volume ls                                              # DAILY
docker volume inspect mongo-data                               # MEDIUM
docker volume rm mongo-data                                     # MEDIUM

# Run with a named volume (persists Mongo data across container recreation)
docker run -d -v mongo-data:/data/db --name mongo mongo:7      # DAILY

# Copy a file OUT of a container (e.g. grab a generated log/config)
docker cp jfrog-svc:/app/config.json ./local-config.json        # MEDIUM

# Copy a file INTO a running container
docker cp ./patched-config.json jfrog-svc:/app/config.json        # MEDIUM

5. Networking

# Create a user-defined network (enables name-based DNS between containers)
docker network create di-net                                   # MEDIUM

# Run containers attached to it
docker run -d --network di-net --name mongo mongo:7              # DAILY
docker run -d --network di-net --name jfrog-svc di-jfrog:local    # DAILY

# List / inspect networks (see which containers are attached, subnet, etc.)
docker network ls                                                 # MEDIUM
docker network inspect di-net                                       # MEDIUM

# Connect/disconnect a running container to/from a network
docker network connect di-net some-other-container                    # MEDIUM
docker network disconnect di-net some-other-container                  # MEDIUM

6. Docker Compose

# Build (if needed) and start everything in the background
docker compose up -d                                            # DAILY

# Rebuild images before starting (after a Dockerfile/dependency change)
docker compose up -d --build                                     # DAILY

# Tail logs of one service, or all services
docker compose logs -f jfrog-service                               # DAILY
docker compose logs -f                                              # DAILY

# List running services in this Compose project
docker compose ps                                                    # DAILY

# Shell into a service container (by service name, not container name)
docker compose exec jfrog-service sh                                   # DAILY

# Stop and remove containers/networks (add -v to also remove volumes = wipes data)
docker compose down                                                     # DAILY
docker compose down -v                                                    # MEDIUM

# Restart just one service
docker compose restart jfrog-service                                        # MEDIUM

# Validate/print the fully-resolved compose config (great for debugging env var interpolation)
docker compose config                                                        # MEDIUM

7. Cleanup & Maintenance

# See disk usage breakdown (images/containers/volumes/build cache)
docker system df                                              # MEDIUM

# Remove all stopped containers, unused networks, dangling images, build cache
docker system prune                                             # MEDIUM
docker system prune -a                                            # MEDIUM (also removes unused, non-dangling images — careful)

# Remove only unused images / stopped containers / unused volumes individually
docker image prune -a                                              # MEDIUM
docker container prune                                               # MEDIUM
docker volume prune                                                    # MEDIUM (destructive — removes unattached volumes/data)

8. Registry / Private Artifactory Auth

# Log in to a private registry (e.g. Kyndryl's JFrog Artifactory Docker registry)
docker login kyndryl.jfrog.io -u $INIT_USER -p $ARTIFACTORY_PASS   # DAILY (in CI), MEDIUM (locally)

docker logout kyndryl.jfrog.io                                       # MEDIUM

# Pull/push against that registry once authenticated
docker pull kyndryl.jfrog.io/ibmcb-docker-local/dash/portal_jfrog_app:1.2.0   # DAILY
docker push kyndryl.jfrog.io/ibmcb-docker-local/dash/portal_jfrog_app:1.2.0     # DAILY
Never pass -p with a plaintext password in shared shell history/scripts in real usage — prefer --password-stdin (echo "$ARTIFACTORY_PASS" | docker login kyndryl.jfrog.io -u "$INIT_USER" --password-stdin) so the secret doesn't linger in process listings or shell history.

9. Medium/Advanced Daily Use

# Limit resources at run time (mirrors K8s requests/limits, useful for local repro of OOM issues)
docker run -d --memory=512m --cpus=0.5 --name jfrog-svc di-jfrog:local    # MEDIUM

# Override the healthcheck defined in the image, or disable it
docker run --health-cmd="curl -f http://localhost:8080/health || exit 1" \
  --health-interval=30s di-jfrog:local                                     # MEDIUM
docker run --no-healthcheck di-jfrog:local                                   # MEDIUM

# Attach vs exec — attach connects to the MAIN process's stdin/stdout (risky: Ctrl+C can kill it);
# exec starts a NEW process inside the container (safe for poking around)
docker attach jfrog-svc                                                       # MEDIUM (use with care)
docker exec -it jfrog-svc sh                                                     # DAILY (preferred for debugging)

# Export a container's filesystem as a tarball (different from `save`, which exports an IMAGE)
docker export jfrog-svc -o container-fs.tar                                       # MEDIUM

# Multi-architecture build (e.g. amd64 + arm64) via buildx — relevant on Apple Silicon dev machines
docker buildx build --platform linux/amd64,linux/arm64 -t di-jfrog:multi .           # MEDIUM

# Show real-time events from the Docker daemon (container start/stop/die, useful while debugging)
docker events --filter container=jfrog-svc                                            # MEDIUM
