🖼️ Images
docker build -t app:dev .
build from Dockerfile in cwd
docker images
list local images
docker tag app:dev repo/app:1.0
docker push / pull repo/app:1.0
docker rmi app:dev
docker history app:dev
layer-by-layer size
📦 Containers
docker run -d -p 8080:8080 --name app img
docker ps / docker ps -a
docker start / stop / restart app
docker rm app (-f to force)
docker rename old new
docker pause / unpause app
🔍 Inspect & Debug
docker logs -f app
docker exec -it app sh
docker inspect app
docker stats live CPU/mem
docker top app
docker diff app fs changes vs image
💾 Volumes
docker volume create data
docker volume ls / inspect / rm
docker run -v data:/data/db mongo:7
docker cp app:/app/x.json ./x.json
docker cp ./x.json app:/app/x.json
bind mount: -v $(pwd):/app
🌐 Networking
docker network create di-net
docker run --network di-net --name mongo mongo:7
docker network ls / inspect di-net
-p 8080:80 host:container
default bridge = no name DNS; user-defined bridge / Compose = name DNS works
🧩 Compose
docker compose up -d --build
docker compose logs -f svc
docker compose ps
docker compose exec svc sh
docker compose down -v -v also wipes volumes
docker compose config print resolved config
🧹 Cleanup
docker system df disk usage breakdown
docker system prune
docker system prune -a ⚠ removes unused images too
docker container prune
docker volume prune ⚠ destroys unattached data
🔐 Registry (JFrog Artifactory)
echo "$PASS" | docker login kyndryl.jfrog.io \
  -u "$USER" --password-stdin
docker logout kyndryl.jfrog.io
docker pull kyndryl.jfrog.io/ibmcb-docker-local/...
⚠ never pass -p plaintext — always --password-stdin

📜 Dockerfile Instructions — Quick Reference

FROM base image WORKDIR sets cwd COPY host → image RUN build-time exec ENV baked var (build+run) ARG build-time only var EXPOSE docs the port (no publish) USER drop root CMD default, overridable ENTRYPOINT fixed executable HEALTHCHECK liveness probe
