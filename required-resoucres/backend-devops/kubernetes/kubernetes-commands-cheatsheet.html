1. Cluster & Context

# Basic cluster health / info
kubectl cluster-info                                     # MEDIUM
kubectl get nodes                                          # DAILY
kubectl get nodes -o wide                                    # MEDIUM

# See/switch which cluster+namespace kubectl is currently pointed at
kubectl config get-contexts                                   # DAILY
kubectl config use-context my-dev-cluster                       # DAILY
kubectl config view --minify                                       # MEDIUM

# Set a default namespace for the current context (avoids typing -n every time)
kubectl config set-context --current --namespace=dash                # MEDIUM

# API resource discovery — what fields/kinds are available (great for writing YAML)
kubectl explain deployment.spec.template.spec.containers               # MEDIUM
kubectl api-resources                                                    # MEDIUM

2. Pods

# List pods (current namespace / all namespaces / with more detail)
kubectl get pods                                            # DAILY
kubectl get pods -n dash                                      # DAILY
kubectl get pods -A                                             # MEDIUM
kubectl get pods -o wide                                          # DAILY (shows node, IP)

# Full details + recent Events (the #1 debugging command)
kubectl describe pod di-jfrog-abc123                              # DAILY

# Logs — current, follow, previous (crashed) instance, specific container in multi-container pod
kubectl logs di-jfrog-abc123                                        # DAILY
kubectl logs -f di-jfrog-abc123                                       # DAILY
kubectl logs di-jfrog-abc123 --previous                                 # DAILY (for CrashLoopBackOff)
kubectl logs di-jfrog-abc123 -c sidecar-name                              # MEDIUM

# Shell into a running pod
kubectl exec -it di-jfrog-abc123 -- sh                                     # DAILY
kubectl exec -it di-jfrog-abc123 -c sidecar-name -- sh                       # MEDIUM

# Run a one-off command without a shell
kubectl exec di-jfrog-abc123 -- env                                           # MEDIUM
kubectl exec di-jfrog-abc123 -- cat /etc/config/app.json                       # MEDIUM

# Delete a pod (ReplicaSet recreates it automatically if managed)
kubectl delete pod di-jfrog-abc123                                              # DAILY

# Live CPU/memory usage per pod (requires metrics-server)
kubectl top pod                                                                  # MEDIUM
kubectl top pod di-jfrog-abc123                                                    # MEDIUM

# Copy a file to/from a pod
kubectl cp di-jfrog-abc123:/app/config.json ./local-config.json                     # MEDIUM
kubectl cp ./local-config.json di-jfrog-abc123:/app/config.json                       # MEDIUM

3. Deployments & Rollouts

# List / describe deployments
kubectl get deployments                                     # DAILY
kubectl describe deployment di-jfrog                            # DAILY

# Scale manually
kubectl scale deployment di-jfrog --replicas=5                    # MEDIUM

# Trigger a new rollout by updating the image
kubectl set image deployment/di-jfrog di-jfrog=kyndryl.jfrog.io/ibmcb-docker-local/dash/portal_jfrog_app:1.2.0   # DAILY

# Watch a rollout in progress
kubectl rollout status deployment/di-jfrog                          # DAILY

# Rollback to the previous revision, or a specific one
kubectl rollout undo deployment/di-jfrog                               # DAILY (when a release is broken)
kubectl rollout undo deployment/di-jfrog --to-revision=3                 # MEDIUM

# See revision history
kubectl rollout history deployment/di-jfrog                                # MEDIUM

# Force a restart of all pods without changing the image (e.g. to pick up a ConfigMap change)
kubectl rollout restart deployment/di-jfrog                                  # MEDIUM

# Pause/resume a rollout (useful mid-rollout to inspect canary pods before continuing)
kubectl rollout pause deployment/di-jfrog                                      # MEDIUM
kubectl rollout resume deployment/di-jfrog                                       # MEDIUM

4. Services & Networking

# List / describe services
kubectl get svc                                              # DAILY
kubectl describe svc di-jfrog-svc                                # DAILY

# Check which pods a Service is actually routing to (empty = selector mismatch)
kubectl get endpoints di-jfrog-svc                                 # MEDIUM

# Tunnel a cluster Service (or Pod) to your local machine — great for local debugging against a real cluster
kubectl port-forward svc/di-jfrog-svc 8080:80                        # DAILY
kubectl port-forward pod/di-jfrog-abc123 8080:8080                     # MEDIUM

# Quickly expose an existing deployment as a new Service (ad-hoc, mostly for local/dev clusters)
kubectl expose deployment di-jfrog --port=80 --target-port=8080           # MEDIUM

# Ingress
kubectl get ingress                                                          # DAILY
kubectl describe ingress dash-ingress                                          # MEDIUM

5. ConfigMaps & Secrets

# Create a ConfigMap from literals or a file
kubectl create configmap di-jfrog-config --from-literal=LOG_LEVEL=debug     # MEDIUM
kubectl create configmap di-jfrog-config --from-file=config.json               # MEDIUM

# Create a Secret (values are base64-encoded automatically, NOT encrypted)
kubectl create secret generic di-jfrog-secret --from-literal=DB_PASS=xxxx        # MEDIUM

# View (Secrets show base64; decode manually if needed)
kubectl get configmap di-jfrog-config -o yaml                                       # DAILY
kubectl get secret di-jfrog-secret -o jsonpath='{.data.DB_PASS}' | base64 -d           # MEDIUM

# Edit live (opens in your $EDITOR, applies on save — use sparingly, prefer Helm values + apply)
kubectl edit configmap di-jfrog-config                                                   # MEDIUM

6. Debugging & Troubleshooting

# The single most useful debugging command — status + recent Events
kubectl describe pod di-jfrog-abc123                          # DAILY

# Cluster-wide recent events, newest last (great for spotting scheduling/OOM issues fast)
kubectl get events --sort-by='.lastTimestamp'                    # DAILY
kubectl get events -n dash --sort-by='.lastTimestamp'               # DAILY

# Why is a pod Pending? Why did a rollout stall? Check both pod AND deployment describe
kubectl describe deployment di-jfrog                                  # MEDIUM

# Attach an ephemeral debug container to a running pod without restarting it (needs a distro-with-tools image)
kubectl debug -it di-jfrog-abc123 --image=busybox:1.36 --target=di-jfrog   # MEDIUM

# Get a pod's full resolved YAML (spec + status) — useful to see actually-applied values
kubectl get pod di-jfrog-abc123 -o yaml                                       # MEDIUM

# Check readiness/liveness probe results directly from the Conditions section
kubectl get pod di-jfrog-abc123 -o jsonpath='{.status.conditions}'               # MEDIUM

7. Namespaces & Labels

# List namespaces; switch working namespace for a single command
kubectl get namespaces                                       # DAILY
kubectl get pods -n dash                                        # DAILY

# Add/remove a label on the fly (e.g. temporarily tagging a pod for a query)
kubectl label pod di-jfrog-abc123 debug=true                       # MEDIUM
kubectl label pod di-jfrog-abc123 debug-                             # MEDIUM (trailing - removes it)

# Filter by label selector — the same mechanism Services/ReplicaSets use internally
kubectl get pods -l app=di-jfrog                                       # DAILY
kubectl get pods -l 'app in (di-jfrog, di-authorization)'                 # MEDIUM

# Add an annotation (metadata, not used for selection)
kubectl annotate pod di-jfrog-abc123 owner=squad-di                          # MEDIUM

8. Apply / Diff / Dry-Run

# Create or update from a manifest — the standard declarative workflow
kubectl apply -f deployment.yaml                              # DAILY

# See what WOULD change before actually applying it
kubectl diff -f deployment.yaml                                  # MEDIUM

# Render what apply would create without sending it (client-side, no cluster call)
kubectl apply -f deployment.yaml --dry-run=client -o yaml           # MEDIUM

# Server-side dry-run (validates against the actual API server, catches more errors)
kubectl apply -f deployment.yaml --dry-run=server                     # MEDIUM

# Delete everything defined in a manifest file
kubectl delete -f deployment.yaml                                        # DAILY

# Get ALL common resource types in a namespace at once (quick overview)
kubectl get all -n dash                                                    # DAILY

9. Helm

# Install / upgrade a release (upgrade --install is idempotent — safe for both first-time and repeat)
helm upgrade --install dash ./dash-chart -f env-config/shared-prod/values.yaml   # DAILY

# See what values are actually in effect for a running release
helm get values dash                                              # DAILY
helm get values dash -a                                              # MEDIUM (includes computed/default values)

# Render final manifests locally WITHOUT touching the cluster — best first debugging step
helm template ./dash-chart -f env-config/shared-prod/values.yaml      # DAILY

# List releases / see revision history / rollback
helm list                                                                # DAILY
helm history dash                                                          # MEDIUM
helm rollback dash 3                                                         # MEDIUM

# Lint a chart for template/schema errors before installing
helm lint ./dash-chart                                                         # MEDIUM

# Uninstall a release
helm uninstall dash                                                              # MEDIUM

10. Medium/Advanced Daily Use

# Watch a resource live update in your terminal (like `watch kubectl get pods`)
kubectl get pods --watch                                       # MEDIUM

# Jsonpath output for scripting (e.g. grab just the image tag currently deployed)
kubectl get deployment di-jfrog -o jsonpath='{.spec.template.spec.containers[0].image}'   # MEDIUM

# Field/label selectors together to narrow down a busy namespace
kubectl get pods --field-selector=status.phase=Running -l app=di-jfrog   # MEDIUM

# See which node a pod landed on, and that node's allocatable resources
kubectl get pod di-jfrog-abc123 -o wide                                   # DAILY
kubectl describe node <node-name>                                          # MEDIUM

# Restart just the Pods without deleting the Deployment (rollout restart is the safe way; direct
# pod delete also works since the ReplicaSet recreates it, but rollout restart preserves history)
kubectl delete pod -l app=di-jfrog                                            # MEDIUM

# Quick sanity check — is a Service resolvable/reachable from inside the cluster?
kubectl run tmp-debug --rm -it --image=busybox:1.36 -- sh -c "nslookup di-jfrog-svc"   # MEDIUM
