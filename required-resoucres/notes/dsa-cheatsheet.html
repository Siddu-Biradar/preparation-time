02 Fast I/O Template

fmt.Scan is SLOW for big inputs. Always use bufio on competitive sites.

package main

import (
    "bufio"
    "fmt"
    "os"
    "strconv"
)

var (
    reader = bufio.NewReader(os.Stdin)
    writer = bufio.NewWriter(os.Stdout)
)

func readInt() int {
    var n int
    fmt.Fscan(reader, &n)
    return n
}

func readLine() string {
    line, _ := reader.ReadString('\n')
    return line
}

func main() {
    defer writer.Flush()  // CRITICAL — without this, no output!

    n := readInt()
    for i := 0; i < n; i++ {
        x := readInt()
        fmt.Fprintln(writer, x*2)
    }
}
Forgot writer.Flush()? Output won't appear. Use defer writer.Flush() as the FIRST line of main.
For huge inputs (1e6+ lines), use bufio.Scanner with custom buffer: 
sc := bufio.NewScanner(os.Stdin)
sc.Buffer(make([]byte, 1024*1024), 1024*1024*64)
sc.Split(bufio.ScanWords)
sc.Scan(); n, _ := strconv.Atoi(sc.Text())

03 Numeric Tricks

Min / Max / Abs (Go 1.21+)

// Built-in (Go 1.21+) — works for any ordered type
a := min(3, 5, 1)        // 1 — variadic!
b := max(3.14, 2.71)      // 3.14

// Older Go: math.Min/Max are FLOAT64 only — be careful
math.Min(3.0, 5.0)         // returns float64

// Abs for ints (no built-in!)
func abs(x int) int { if x < 0 { return -x }; return x }

Integer Limits

import "math"

math.MaxInt      // platform-dependent (usually 1<<63 - 1)
math.MinInt      // -1<<63
math.MaxInt32    // 2147483647
math.MaxInt64    // 9223372036854775807
math.Inf(1)      // +∞ (float64)
math.Inf(-1)     // -∞

Bit Operations

x & y          // AND
x | y          // OR
x ^ y          // XOR (also: NOT when used as ^x — unary)
x &^ y         // AND NOT (clear bits) — Go-SPECIFIC, super useful!
x << k         // shift left
x >> k         // shift right (signed for int, unsigned for uint)

// math/bits package — fast intrinsics
import "math/bits"
bits.OnesCount(0b1011)         // 3 (popcount)
bits.LeadingZeros(x)           // for log2-like ops
bits.TrailingZeros(x)
bits.Len(x)                     // bit length (1+floor(log2(x)))
bits.Reverse(x)
Overflow gotcha: Go does NOT check overflow. int(1e18) * 10 silently overflows. For very large numbers use math/big: 
import "math/big"
a := new(big.Int).SetInt64(100)
a.Mul(a, a)        // in-place
a.Mod(a, mod)
fmt.Println(a.String())

Power, GCD, LCM

// gcd (recursive)
func gcd(a, b int) int { if b == 0 { return a }; return gcd(b, a%b) }

// lcm
func lcm(a, b int) int { return a / gcd(a, b) * b }  // divide first to avoid overflow

// Fast power (mod)
func powMod(base, exp, mod int) int {
    res := 1
    base %= mod
    for exp > 0 {
        if exp&1 == 1 { res = res * base % mod }
        base = base * base % mod
        exp >>= 1
    }
    return res
}

04 Strings & Runes

Strings in Go are immutable bytes. Indexing returns a byte, not a character.

The CRITICAL distinction

s := "héllo"           // 6 bytes (é is 2 bytes UTF-8)
len(s)                  // 6 — BYTES, not characters!
s[1]                    // byte (0xC3) — not 'é'!

// To iterate by CHARACTER (rune):
for i, r := range s {  // i = byte index, r = rune (int32)
    fmt.Printf("%d: %c\n", i, r)
}

// Convert to []rune for index-based access
runes := []rune(s)      // allocates! O(n)
runes[1]               // 'é'

// Convert to []byte (mutable copy)
bytes := []byte(s)

Building Strings — use strings.Builder

// ❌ DON'T — quadratic time!
result := ""
for _, c := range input {
    result += string(c)   // allocates each time
}

// ✅ DO — O(n) total
var sb strings.Builder
sb.Grow(len(input))    // pre-size for performance
for _, c := range input {
    sb.WriteRune(c)        // or WriteByte / WriteString
}
result := sb.String()

Common strings functions

strings.Contains(s, sub)substring check
strings.Index(s, sub)first index, -1 if absent
strings.Split(s, sep)→ []string
strings.Join(parts, sep)→ string
strings.Replace(s, old, new, n)n=-1 for all
strings.ReplaceAll(s, old, new)cleaner alias
strings.ToLower / ToUppercase
strings.TrimSpace(s)strip whitespace
strings.Repeat("ab", 3)"ababab"
strings.HasPrefix / HasSuffixprefix/suffix check
strconv.Itoa(n)int → string
strconv.Atoi(s)string → int (returns err)
strconv.FormatFloat(f, 'f', 6, 64)float formatting
fmt.Sprintf("%d-%s", n, s)generic formatting

Char checks (instead of isalpha/isdigit)

import "unicode"
unicode.IsLetter(r)
unicode.IsDigit(r)
unicode.IsSpace(r)
unicode.IsUpper(r)
unicode.ToLower(r)

// For ASCII speed-hack:
isDigit := c >= '0' && c <= '9'
isLower := c >= 'a' && c <= 'z'
toLower := c | 0x20     // works for ASCII letters only!

05 Slices — the Swiss Army Knife

Initialize

var s []int                        // nil slice (len 0, cap 0)
s := []int{}                       // empty (not nil)
s := []int{1, 2, 3}                // literal
s := make([]int, 10)              // len=10, all 0
s := make([]int, 0, 100)            // len=0, cap=100 (pre-sized)

// 2D slice (matrix)
n, m := 3, 4
grid := make([][]int, n)
for i := range grid {
    grid[i] = make([]int, m)
}

Common operations

// Append
s = append(s, x)
s = append(s, a, b, c)
s = append(s, other...)         // spread!

// Slice (a[low:high:max])
sub := s[2:5]                  // [2,3,4]
sub := s[:3]                    // first 3
sub := s[3:]                    // from 3 to end
copySafe := append([]int{}, s...)  // independent copy

// Length / capacity
len(s)   // elements present
cap(s)   // total allocated

// Remove element at index i (order doesn't matter)
s[i] = s[len(s)-1]
s = s[:len(s)-1]

// Remove preserving order
s = append(s[:i], s[i+1:]...)

// Insert at index i
s = append(s[:i], append([]int{x}, s[i:]...)...)

// Reverse in-place
for i, j := 0, len(s)-1; i < j; i, j = i+1, j-1 {
    s[i], s[j] = s[j], s[i]
}

// slices package (Go 1.21+) — USE THESE!
import "slices"
slices.Contains(s, x)
slices.Index(s, x)             // -1 if not found
slices.Reverse(s)               // in-place
slices.Sort(s)
slices.Min(s)                   // panics on empty
slices.Max(s)
slices.Equal(a, b)
slices.Clone(s)
slices.Concat(a, b, c)
The #1 slice gotcha: Slicing shares memory with the parent array. Modifying sub changes s! 
s := []int{1, 2, 3, 4, 5}
sub := s[1:3]
sub[0] = 99          // s is now [1, 99, 3, 4, 5] !!

// Make independent copy:
sub = append([]int{}, s[1:3]...)
// or:
sub = slices.Clone(s[1:3])
Append in DFS/backtracking: When passing slices to recursive calls, append may or may not share underlying array. ALWAYS clone before adding to result: 
result = append(result, append([]int{}, path...))

06 Maps & Sets

Map basics

m := make(map[string]int)
m := map[string]int{"a": 1, "b": 2}
m := make(map[string]int, 100)  // pre-sized

m["key"] = 42

// Existence check (the comma-ok idiom — MEMORIZE THIS)
v, ok := m["key"]
if ok { // key exists }

// Read-only — missing key returns ZERO VALUE (no error!)
count := m["missing"]   // 0 — gotcha if you confuse "missing" with "0"

delete(m, "key")         // no-op if absent
len(m)                  // number of keys

// Iterate — ORDER IS RANDOMIZED (Go does this on purpose!)
for k, v := range m {
    // ...
}

Set pattern (Go has no built-in set!)

// ✅ Idiomatic: map[T]struct{} — uses 0 bytes per value
seen := make(map[int]struct{})
seen[42] = struct{}{}
_, exists := seen[42]

// Lazier: map[T]bool (uses 1 byte per value, but cleaner code)
seen := make(map[int]bool)
seen[42] = true
if seen[42] { // works because zero value is false }

Counter / frequency map

freq := make(map[rune]int)
for _, c := range s {
    freq[c]++          // missing key starts at 0, ++ works!
}

Map of slices (default value pattern)

graph := make(map[int][]int)
graph[1] = append(graph[1], 2)   // works! nil slice + append = new slice
graph[1] = append(graph[1], 3)
Cannot take address of map value: 
m := map[int]Point{1: {x: 5}}
// m[1].x = 10  // ❌ compile error!
p := m[1]; p.x = 10; m[1] = p   // ✅ read-modify-write
// OR use map[int]*Point
Map iteration order is RANDOMIZED. Don't rely on order. To iterate in sorted key order: 
keys := make([]int, 0, len(m))
for k := range m { keys = append(keys, k) }
sort.Ints(keys)
for _, k := range keys { /* use m[k] */ }

07 Sorting & Custom Comparators

Built-in sorting

import "sort"

sort.Ints(nums)                     // []int ascending
sort.Strings(words)
sort.Float64s(vals)
sort.Sort(sort.Reverse(sort.IntSlice(nums)))  // descending

// Custom — use sort.Slice (most common)
sort.Slice(people, func(i, j int) bool {
    return people[i].age < people[j].age   // ascending by age
})

// Stable sort (preserves order of equal elements)
sort.SliceStable(s, func(i, j int) bool { /* ... */ })

// slices package (Go 1.21+)
slices.Sort(nums)                    // generic, faster than sort.Ints!
slices.SortFunc(people, func(a, b Person) int {
    return a.age - b.age              // negative=less, 0=equal, positive=greater
})
slices.SortStableFunc(...)

Multi-key sort (chained comparator)

sort.Slice(items, func(i, j int) bool {
    if items[i].priority != items[j].priority {
        return items[i].priority > items[j].priority   // 1st key DESC
    }
    if items[i].name != items[j].name {
        return items[i].name < items[j].name           // 2nd key ASC
    }
    return items[i].id < items[j].id                   // tie-breaker
})

Binary search

idx := sort.SearchInts(sortedNums, target)
// Returns the smallest index i such that sortedNums[i] >= target.
// If target is bigger than all, returns len(sortedNums).

// Generic version (Go 1.21+)
idx, found := slices.BinarySearch(sortedNums, target)

// Custom predicate
idx := sort.Search(n, func(i int) bool {
    return isFeasible(i)   // returns smallest i where this is true
})
Pattern: Use sort.Search for "binary search the answer" problems (e.g. min capacity to ship in D days).

08 Stack, Queue & Deque

Go has no built-in stack/queue/deque. Slices do everything.

Stack — slice with append/pop

var stack []int

stack = append(stack, x)         // push
top := stack[len(stack)-1]      // peek
stack = stack[:len(stack)-1]    // pop
isEmpty := len(stack) == 0

Queue — use slice as ring buffer OR just slice (BFS)

// Simple FIFO via slice (fine for BFS — total work is O(n))
queue := []int{start}
for len(queue) > 0 {
    node := queue[0]
    queue = queue[1:]            // dequeue (O(1) amortized — slice header moves)
    // process node...
    queue = append(queue, neighbor)  // enqueue
}
The queue[1:] trick is fine for BFS but the underlying array grows unboundedly if you push and pop a LOT (memory leak). For long-running queues use container/list or a ring buffer.

Deque — double-ended queue (sliding window max etc.)

var dq []int

// Push back / pop back (cheap)
dq = append(dq, x)
back := dq[len(dq)-1]; dq = dq[:len(dq)-1]

// Push front / pop front
dq = append([]int{x}, dq...)   // push front (O(n) — avoid hot loops)
front := dq[0]; dq = dq[1:]    // pop front (O(1) amortized)

container/list — doubly linked list (rarely needed)

import "container/list"

l := list.New()
l.PushBack(1)
l.PushFront(2)
front := l.Front().Value.(int)   // type assert!
l.Remove(l.Front())
In practice plain slices beat container/list for almost everything due to cache locality. Only use list when you need O(1) middle removal.

09 Heap / Priority Queue

Go's container/heap is a INTERFACE — you must implement 5 methods. Verbose but you'll memorize it.

Min-heap of ints (template)

import "container/heap"

// 1. Define a slice TYPE that satisfies heap.Interface
type IntHeap []int

func (h IntHeap) Len() int           { return len(h) }
func (h IntHeap) Less(i, j int) bool { return h[i] < h[j] }  // MIN heap (use > for max)
func (h IntHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *IntHeap) Push(x any)        { *h = append(*h, x.(int)) }
func (h *IntHeap) Pop() any {
    old := *h
    n := len(old)
    x := old[n-1]
    *h = old[:n-1]
    return x
}

// 2. Use it
h := &IntHeap{3, 1, 4, 1, 5}
heap.Init(h)                    // heapify O(n)
heap.Push(h, 9)
top := (*h)[0]                 // peek (no Peek method!)
min := heap.Pop(h).(int)        // pop & type-assert

Max-heap → flip the Less sign

func (h IntHeap) Less(i, j int) bool { return h[i] > h[j] }

Priority queue with custom struct

type Item struct {
    val      int
    priority int
}
type PQ []*Item

func (q PQ) Len() int            { return len(q) }
func (q PQ) Less(i, j int) bool  { return q[i].priority < q[j].priority }
func (q PQ) Swap(i, j int)       { q[i], q[j] = q[j], q[i] }
func (q *PQ) Push(x any)         { *q = append(*q, x.(*Item)) }
func (q *PQ) Pop() any            { o := *q; n := len(o); x := o[n-1]; *q = o[:n-1]; return x }
Common mistakes:
  1. Calling h.Push(x) instead of heap.Push(h, x) — won't reheapify!
  2. Forgetting type assertion: heap.Pop(h).(int)
  3. Push/Pop must be on POINTER receiver (they modify the slice header).

10 Linked List

type ListNode struct {
    Val  int
    Next *ListNode
}

// Sentinel/dummy node — eliminates edge cases for head changes
dummy := &ListNode{}
dummy.Next = head
prev := dummy
for prev.Next != nil {
    if prev.Next.Val == target {
        prev.Next = prev.Next.Next     // remove
    } else {
        prev = prev.Next
    }
}
return dummy.Next

// Reverse
var prev *ListNode
curr := head
for curr != nil {
    next := curr.Next
    curr.Next = prev
    prev, curr = curr, next
}
return prev   // new head

// Cycle detection (Floyd's)
slow, fast := head, head
for fast != nil && fast.Next != nil {
    slow, fast = slow.Next, fast.Next.Next
    if slow == fast { return true }
}
return false

11 Trees & Recursion

type TreeNode struct {
    Val         int
    Left, Right *TreeNode
}

// DFS — recursive
var dfs func(node *TreeNode) int
dfs = func(node *TreeNode) int {
    if node == nil { return 0 }
    l := dfs(node.Left)
    r := dfs(node.Right)
    return 1 + max(l, r)
}
height := dfs(root)

// DFS — iterative with stack
stack := []*TreeNode{root}
for len(stack) > 0 {
    n := stack[len(stack)-1]
    stack = stack[:len(stack)-1]
    if n == nil { continue }
    // process n.Val
    stack = append(stack, n.Right, n.Left)   // push right first so left is popped first
}

// BFS — level order
queue := []*TreeNode{root}
for len(queue) > 0 {
    levelSize := len(queue)
    for i := 0; i < levelSize; i++ {
        n := queue[0]; queue = queue[1:]
        if n.Left != nil  { queue = append(queue, n.Left) }
        if n.Right != nil { queue = append(queue, n.Right) }
    }
    // `levelSize` = nodes on this level
}

// Closure trick for backtracking — captures `result` cleanly
func paths(root *TreeNode) [][]int {
    var result [][]int
    var path []int
    var dfs func(*TreeNode)
    dfs = func(node *TreeNode) {
        if node == nil { return }
        path = append(path, node.Val)
        if node.Left == nil && node.Right == nil {
            result = append(result, append([]int{}, path...))  // CLONE!
        }
        dfs(node.Left)
        dfs(node.Right)
        path = path[:len(path)-1]    // backtrack
    }
    dfs(root)
    return result
}
The closure trick is the cleanest way to do recursion in Go. Always clone slices before adding to result in backtracking.

12 Graphs (BFS / DFS / Dijkstra / Union-Find)

Adjacency list

// For n nodes labeled 0..n-1
graph := make([][]int, n)
for _, e := range edges {
    u, v := e[0], e[1]
    graph[u] = append(graph[u], v)
    graph[v] = append(graph[v], u)   // undirected
}

BFS shortest path (unweighted)

dist := make([]int, n)
for i := range dist { dist[i] = -1 }
dist[start] = 0
queue := []int{start}
for len(queue) > 0 {
    u := queue[0]; queue = queue[1:]
    for _, v := range graph[u] {
        if dist[v] == -1 {
            dist[v] = dist[u] + 1
            queue = append(queue, v)
        }
    }
}

Dijkstra (weighted, non-negative)

type Edge struct { to, w int }
type PQItem struct { node, dist int }
type PQ []PQItem

func (q PQ) Len() int           { return len(q) }
func (q PQ) Less(i, j int) bool { return q[i].dist < q[j].dist }
func (q PQ) Swap(i, j int)      { q[i], q[j] = q[j], q[i] }
func (q *PQ) Push(x any)        { *q = append(*q, x.(PQItem)) }
func (q *PQ) Pop() any           { o := *q; x := o[len(o)-1]; *q = o[:len(o)-1]; return x }

func dijkstra(graph [][]Edge, src int) []int {
    n := len(graph)
    dist := make([]int, n)
    for i := range dist { dist[i] = math.MaxInt }
    dist[src] = 0

    pq := &PQ{{src, 0}}
    heap.Init(pq)
    for pq.Len() > 0 {
        cur := heap.Pop(pq).(PQItem)
        if cur.dist > dist[cur.node] { continue }   // stale entry
        for _, e := range graph[cur.node] {
            if nd := cur.dist + e.w; nd < dist[e.to] {
                dist[e.to] = nd
                heap.Push(pq, PQItem{e.to, nd})
            }
        }
    }
    return dist
}

Union-Find (DSU)

type DSU struct {
    parent, rank []int
}

func NewDSU(n int) *DSU {
    p := make([]int, n)
    for i := range p { p[i] = i }
    return &DSU{parent: p, rank: make([]int, n)}
}

func (d *DSU) Find(x int) int {
    if d.parent[x] != x {
        d.parent[x] = d.Find(d.parent[x])   // path compression
    }
    return d.parent[x]
}

func (d *DSU) Union(x, y int) bool {
    rx, ry := d.Find(x), d.Find(y)
    if rx == ry { return false }
    if d.rank[rx] < d.rank[ry] { rx, ry = ry, rx }
    d.parent[ry] = rx
    if d.rank[rx] == d.rank[ry] { d.rank[rx]++ }
    return true
}

13 Dynamic Programming Templates

1D DP — climbing stairs / coin change

dp := make([]int, n+1)
dp[0] = 1
for i := 1; i <= n; i++ {
    for _, c := range coins {
        if i >= c { dp[i] += dp[i-c] }
    }
}

2D DP — LCS / edit distance

m, n := len(a), len(b)
dp := make([][]int, m+1)
for i := range dp { dp[i] = make([]int, n+1) }

for i := 1; i <= m; i++ {
    for j := 1; j <= n; j++ {
        if a[i-1] == b[j-1] {
            dp[i][j] = dp[i-1][j-1] + 1
        } else {
            dp[i][j] = max(dp[i-1][j], dp[i][j-1])
        }
    }
}

Memoization with map (top-down)

memo := make(map[int]int)            // or [2]int key for 2D
var solve func(int) int
solve = func(state int) int {
    if v, ok := memo[state]; ok { return v }
    if base { return 0 }
    res := /* recurrence */ 0
    memo[state] = res
    return res
}
For 2D memo with int keys, encode as i*MAX + j or use map[[2]int]int (struct/array keys are valid in Go).

Bitmask DP (TSP-style)

// dp[mask][i] = min cost to visit set `mask` ending at i
dp := make([][]int, 1<<n)
for i := range dp {
    dp[i] = make([]int, n)
    for j := range dp[i] { dp[i][j] = math.MaxInt }
}
dp[1][0] = 0   // start at node 0

for mask := 1; mask < 1<<n; mask++ {
    for u := 0; u < n; u++ {
        if mask&(1<<u) == 0 { continue }
        for v := 0; v < n; v++ {
            if mask&(1<<v) != 0 { continue }
            newMask := mask | (1<<v)
            if nd := dp[mask][u] + cost[u][v]; nd < dp[newMask][v] {
                dp[newMask][v] = nd
            }
        }
    }
}

14 Two Pointers / Sliding Window / Binary Search

Two pointers (sorted array, e.g. two-sum)

l, r := 0, len(nums)-1
for l < r {
    sum := nums[l] + nums[r]
    switch {
    case sum == target: return []int{l, r}
    case sum < target:  l++

Segment Tree (range sum, point update)

type SegTree struct {
    n    int
    data []int
}

func NewSeg(n int) *SegTree { return &SegTree{n: n, data: make([]int, 4*n)} }

func (s *SegTree) Update(node, l, r, idx, v int) {
    if l == r { s.data[node] = v; return }
    mid := (l + r) / 2
    if idx <= mid { s.Update(2*node, l, mid, idx, v) } else { s.Update(2*node+1, mid+1, r, idx, v) }
    s.data[node] = s.data[2*node] + s.data[2*node+1]
}

func (s *SegTree) Query(node, l, r, ql, qr int) int {
    if qr < l || r < ql { return 0 }
    if ql <= l && r <= qr { return s.data[node] }
    mid := (l + r) / 2
    return s.Query(2*node, l, mid, ql, qr) + s.Query(2*node+1, mid+1, r, ql, qr)
}

18 Top 15 Go DSA Gotchas

  1. String indexing returns byte, not character. Use []rune(s) for Unicode.
  2. len(s) for strings is bytes, not characters. Use utf8.RuneCountInString(s).
  3. Map iteration order is RANDOMIZED. Sort keys if order matters.
  4. Cannot modify struct field directly via map index: m[k].x = 5 ❌. Use pointer or read-modify-write.
  5. Slicing shares memory. Modifying sub := s[1:3] mutates s.
  6. Append in backtracking shares array. Always append([]int{}, path...) before adding to results.
  7. Integer division truncates toward zero, not floor! -7/2 == -3, not -4. For floor division of negatives, special-case.
  8. Go has no overflow protection. int is 64-bit on modern systems but doesn't panic on overflow.
  9. Heap requires heap.Push(h, x), not h.Push(x). The latter compiles but doesn't reheapify!
  10. Type assertion can panic: x.(int) panics if not int. Use x.(int) with comma-ok: v, ok := x.(int).
  11. Defer captures by value (vars) but slices/maps are references. defer fmt.Println(slice) prints CURRENT slice contents, not snapshot.
  12. Range loop variable is reused (pre-Go 1.22). Capturing in closure/goroutine needs v := v. Go 1.22+ fixes this.
  13. Comparing slices/maps with == is forbidden (compile error). Use reflect.DeepEqual or slices.Equal.
  14. Nil slice vs empty slice: var s []int (nil) and s := []int{} (empty) behave identically for len/append/range, but differ in JSON marshaling and == nil.
  15. fmt.Println is SLOW. For competitive programming, use bufio.Writer + defer w.Flush().

20 Monotonic Stack & Queue

A specialized stack/queue that maintains elements in increasing/decreasing order. Solves "next greater element" family in O(n).

When to reach for it

  • Next/previous greater (or smaller) element
  • Largest rectangle in histogram
  • Trapping rain water
  • Daily temperatures
  • Sliding window maximum (monotonic deque)

Next Greater Element (template)

func nextGreater(nums []int) []int {
    n := len(nums)
    res := make([]int, n)
    for i := range res { res[i] = -1 }

    stack := []int{}                 // holds INDICES, not values
    for i, v := range nums {
        for len(stack) > 0 && nums[stack[len(stack)-1]] < v {
            top := stack[len(stack)-1]
            stack = stack[:len(stack)-1]
            res[top] = v          // v is the next greater for index `top`
        }
        stack = append(stack, i)
    }
    return res
}

Sliding Window Maximum (monotonic deque)

func maxSlidingWindow(nums []int, k int) []int {
    res := []int{}
    dq := []int{}                     // indices, values DECREASING
    for i, v := range nums {
        // 1. drop indices that fall out of the window
        for len(dq) > 0 && dq[0] <= i-k {
            dq = dq[1:]
        }
        // 2. drop tail values smaller than current (they can never be max)
        for len(dq) > 0 && nums[dq[len(dq)-1]] < v {
            dq = dq[:len(dq)-1]
        }
        dq = append(dq, i)
        if i >= k-1 {
            res = append(res, nums[dq[0]])  // front is window max
        }
    }
    return res
}
Mental model: A monotonic stack maintains an invariant. When the invariant is broken by a new element, you POP — and the popped element's "answer" is the current element. That's why this runs O(n): each element is pushed once, popped once.

22 Topological Sort

Order DAG nodes so every edge u→v has u before v. Used in: course schedule, build systems, alien dictionary.

Kahn's algorithm (BFS-based, also detects cycles)

func topoSort(n int, edges [][]int) ([]int, bool) {
    graph := make([][]int, n)
    indeg := make([]int, n)
    for _, e := range edges {
        u, v := e[0], e[1]    // edge u → v
        graph[u] = append(graph[u], v)
        indeg[v]++
    }

    queue := []int{}
    for i, d := range indeg {
        if d == 0 { queue = append(queue, i) }
    }

    order := make([]int, 0, n)
    for len(queue) > 0 {
        u := queue[0]; queue = queue[1:]
        order = append(order, u)
        for _, v := range graph[u] {
            indeg[v]--
            if indeg[v] == 0 { queue = append(queue, v) }
        }
    }
    if len(order) != n {
        return nil, false     // cycle exists!
    }
    return order, true
}

DFS-based topo sort (with cycle detection)

const ( WHITE = iota; GRAY; BLACK )    // iota = 0,1,2 — Go enum idiom

func topoDFS(n int, graph [][]int) ([]int, bool) {
    color := make([]int, n)            // all WHITE initially
    order := []int{}
    var dfs func(int) bool
    dfs = func(u int) bool {
        if color[u] == GRAY  { return false }    // back edge = cycle
        if color[u] == BLACK { return true }
        color[u] = GRAY
        for _, v := range graph[u] {
            if !dfs(v) { return false }
        }
        color[u] = BLACK
        order = append(order, u)              // post-order
        return true
    }
    for i := 0; i < n; i++ {
        if !dfs(i) { return nil, false }
    }
    // reverse for forward topo order
    for i, j := 0, len(order)-1; i < j; i, j = i+1, j-1 {
        order[i], order[j] = order[j], order[i]
    }
    return order, true
}

23 More Graphs: MST, Bellman-Ford, Floyd-Warshall

Kruskal's MST (use the DSU from §12)

type Edge struct { u, v, w int }

func kruskal(n int, edges []Edge) int {
    sort.Slice(edges, func(i, j int) bool { return edges[i].w < edges[j].w })
    dsu := NewDSU(n)
    total, taken := 0, 0
    for _, e := range edges {
        if dsu.Union(e.u, e.v) {       // returns true if not already connected
            total += e.w
            taken++
            if taken == n-1 { break }
        }
    }
    return total
}

Prim's MST (use a heap)

func prim(graph [][]Edge) int {
    n := len(graph)
    inMST := make([]bool, n)
    pq := &PQ{{0, 0}}                  // {node, weight} — start at 0
    heap.Init(pq)
    total := 0
    for pq.Len() > 0 {
        cur := heap.Pop(pq).(PQItem)
        if inMST[cur.node] { continue }
        inMST[cur.node] = true
        total += cur.dist
        for _, e := range graph[cur.node] {
            if !inMST[e.to] {
                heap.Push(pq, PQItem{e.to, e.w})
            }
        }
    }
    return total
}

Bellman-Ford (handles negative edges, detects negative cycles)

func bellmanFord(n int, edges []Edge, src int) ([]int, bool) {
    dist := make([]int, n)
    for i := range dist { dist[i] = math.MaxInt }
    dist[src] = 0

    // Relax all edges (n-1) times
    for i := 0; i < n-1; i++ {
        for _, e := range edges {
            if dist[e.u] != math.MaxInt && dist[e.u]+e.w < dist[e.v] {
                dist[e.v] = dist[e.u] + e.w
            }
        }
    }
    // One more pass — any update means negative cycle
    for _, e := range edges {
        if dist[e.u] != math.MaxInt && dist[e.u]+e.w < dist[e.v] {
            return nil, false          // negative cycle
        }
    }
    return dist, true
}

Floyd-Warshall (all-pairs shortest path, O(n³))

dist := make([][]int, n)
for i := range dist {
    dist[i] = make([]int, n)
    for j := range dist[i] { dist[i][j] = math.MaxInt / 2 }   // /2 prevents overflow on add
    dist[i][i] = 0
}
for _, e := range edges { dist[e.u][e.v] = e.w }

for k := 0; k < n; k++ {
    for i := 0; i < n; i++ {
        for j := 0; j < n; j++ {
            if dist[i][k]+dist[k][j] < dist[i][j] {
                dist[i][j] = dist[i][k] + dist[k][j]
            }
        }
    }
}

24 String Matching

KMP (Knuth-Morris-Pratt) — O(n+m)

// Build LPS array (longest proper prefix that's also suffix)
func buildLPS(p string) []int {
    lps := make([]int, len(p))
    k := 0
    for i := 1; i < len(p); i++ {
        for k > 0 && p[k] != p[i] {
            k = lps[k-1]
        }
        if p[k] == p[i] { k++ }
        lps[i] = k
    }
    return lps
}

func kmpSearch(text, pattern string) []int {
    lps := buildLPS(pattern)
    matches := []int{}
    j := 0
    for i := 0; i < len(text); i++ {
        for j > 0 && text[i] != pattern[j] {
            j = lps[j-1]
        }
        if text[i] == pattern[j] { j++ }
        if j == len(pattern) {
            matches = append(matches, i-j+1)
            j = lps[j-1]
        }
    }
    return matches
}

Rolling hash (Rabin-Karp)

const (
    base = 31
    mod  = 1_000_000_007
)

func rabinKarp(text, pattern string) int {
    n, m := len(text), len(pattern)
    if m > n { return -1 }

    var patHash, curHash, power int = 0, 0, 1
    for i := 0; i < m; i++ {
        patHash = (patHash*base + int(pattern[i])) % mod
        curHash = (curHash*base + int(text[i]))    % mod
        if i < m-1 { power = power * base % mod }
    }
    if curHash == patHash && text[:m] == pattern { return 0 }

    for i := m; i < n; i++ {
        curHash = (curHash - int(text[i-m])*power%mod + mod*mod) % mod
        curHash = (curHash*base + int(text[i])) % mod
        if curHash == patHash && text[i-m+1:i+1] == pattern {
            return i - m + 1
        }
    }
    return -1
}
Quick win: for simple "find substring" problems Go's strings.Index(text, pattern) uses an optimized algorithm and is usually fast enough. Reach for KMP/RK only when you need multiple matches OR are doing rolling-hash tricks (e.g., longest duplicate substring).

Kadane's algorithm — max subarray

func maxSubarray(nums []int) int {
    best, cur := nums[0], nums[0]
    for i := 1; i < len(nums); i++ {
        cur = max(nums[i], cur+nums[i])    // extend or restart
        best = max(best, cur)

Longest Increasing Subsequence — O(n log n) with binary search

func lengthOfLIS(nums []int) int {
    tails := []int{}                         // tails[i] = smallest tail of LIS of length i+1
    for _, v := range nums {
        i := sort.SearchInts(tails, v)         // first idx where tails[idx] >= v
        if i == len(tails) {
            tails = append(tails, v)
        } else {
            tails[i] = v                       // replace to keep small
        }
    }
    return len(tails)                       // note: tails is NOT the LIS itself
}

0/1 Knapsack (space-optimized)

func knapsack(weights, values []int, capacity int) int {
    dp := make([]int, capacity+1)
    for i, w := range weights {
        // iterate BACKWARDS — each item used at most once
        for c := capacity; c >= w; c-- {
            dp[c] = max(dp[c], dp[c-w]+values[i])
        }
    }
    return dp[capacity]
}
// Unbounded knapsack: iterate FORWARD instead.

Tree DP — diameter / max path sum

func diameter(root *TreeNode) int {
    best := 0
    var depth func(*TreeNode) int
    depth = func(n *TreeNode) int {
        if n == nil { return 0 }
        l := depth(n.Left)
        r := depth(n.Right)
        best = max(best, l+r)              // path through this node
        return 1 + max(l, r)
    }
    depth(root)
    return best
}

26 Backtracking Patterns

Same skeleton works for permutations, combinations, subsets, N-queens, sudoku.

Permutations

func permute(nums []int) [][]int {
    var res [][]int
    used := make([]bool, len(nums))
    var path []int
    var dfs func()
    dfs = func() {
        if len(path) == len(nums) {
            res = append(res, append([]int{}, path...))
            return
        }
        for i, v := range nums {
            if used[i] { continue }
            used[i] = true
            path = append(path, v)
            dfs()
            path = path[:len(path)-1]    // undo choice
            used[i] = false
        }
    }
    dfs()
    return res
}

Combinations / Subsets

func subsets(nums []int) [][]int {
    var res [][]int
    var path []int
    var dfs func(int)
    dfs = func(start int) {
        res = append(res, append([]int{}, path...))
        for i := start; i < len(nums); i++ {
            path = append(path, nums[i])
            dfs(i + 1)                  // i+1 = no reuse
            path = path[:len(path)-1]
        }
    }
    dfs(0)
    return res
}
// For combinations of fixed size k, add: if len(path)==k { record; return }
// For "with reuse" (combination sum), recurse with `i` instead of `i+1`

N-Queens (board[row]=col representation)

func solveNQueens(n int) int {
    cols := make([]bool, n)
    diag1 := make([]bool, 2*n)        // row+col
    diag2 := make([]bool, 2*n)        // row-col+n
    count := 0
    var dfs func(int)
    dfs = func(row int) {
        if row == n { count++; return }
        for c := 0; c < n; c++ {
            if cols[c] || diag1[row+c] || diag2[row-c+n] { continue }
            cols[c], diag1[row+c], diag2[row-c+n] = true, true, true
            dfs(row + 1)
            cols[c], diag1[row+c], diag2[row-c+n] = false, false, false
        }
    }
    dfs(0)
    return count
}
Universal backtracking template: 1) check goal → record. 2) for each choice → make → recurse → undo. The "undo" is what makes it backtracking, not just DFS.

Sieve of Eratosthenes

func sieve(n int) []int {
    isComposite := make([]bool, n+1)
    primes := []int{}
    for i := 2; i <= n; i++ {
        if !isComposite[i] {
            primes = append(primes, i)
            for j := i * i; j <= n; j += i {
                isComposite[j] = true
            }
        }
    }
    return primes
}

var fact, invFact []int

func precompute(n int) {
    fact    = make([]int, n+1)
    invFact = make([]int, n+1)
    fact[0] = 1
    for i := 1; i <= n; i++ {
        fact[i] = fact[i-1] * i % mod
    }
    invFact[n] = modInverse(fact[n])
    for i := n - 1; i >= 0; i-- {
        invFact[i] = invFact[i+1] * (i + 1) % mod
    }
}

func nCr(n, r int) int {
    if r < 0 || r > n { return 0 }
    return fact[n] * invFact[r] % mod * invFact[n-r] % mod
}

// Slices/maps can't be map keys. But arrays and structs CAN!
visited := make(map[[2]int]bool)         // (row, col) key
visited[[2]int{r, c}] = true

type State struct { x, y, mask int }
seen := make(map[State]int)            // 3D BFS state
seen[State{r, c, m}] = dist

Fixed-size arrays beat slices for hot loops

// Counting alphabet — array is faster (no bounds checks, cache-friendly)
var count [26]int
for _, c := range s {
    count[c-'a']++
}
// Comparison: arrays support == directly!
var a, b [26]int
if a == b { /* anagram check in one line */ }

Loop variable in closures (Go < 1.22)

// ❌ Pre-Go-1.22: all goroutines see the SAME `i`!
for i := 0; i < 10; i++ {
    go func() { fmt.Println(i) }()    // usually prints 10, 10, 10, ...
}

// ✅ Capture by argument
for i := 0; i < 10; i++ {
    go func(i int) { fmt.Println(i) }(i)
}

// ✅ Or shadow
for i := 0; i < 10; i++ {
    i := i
    go func() { fmt.Println(i) }()
}
// In Go 1.22+ this is FIXED — each iteration has its own variable.

Compare slices/maps

// slices/maps cannot use ==
slices.Equal(a, b)             // Go 1.21+ generic, fast
maps.Equal(m1, m2)            // Go 1.21+
reflect.DeepEqual(a, b)       // works for anything but slow

new() vs make()

p := new(MyStruct)              // returns *MyStruct, all-zero
s := make([]int, 10)            // initialized slice — REQUIRED for slice/map/chan
m := make(map[int]int)
ch := make(chan int)
// Rule: use make() for slice/map/chan; new() for everything else (or just &Foo{}).
