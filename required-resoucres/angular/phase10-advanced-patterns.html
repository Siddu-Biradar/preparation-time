1. Smart/Container vs Dumb/Presentational Components

// SMART (Container) Component — knows about services, state, business logic
@Component({
  selector: 'app-user-list-page',
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    <app-search-bar (search)="onSearch($event)" />
    
    @if (loading$ | async) {
      <app-skeleton-list />
    } @else {
      <app-user-list 
        [users]="users$ | async" 
        [selectedId]="selectedId()"
        (selectUser)="onSelectUser($event)"
        (deleteUser)="onDeleteUser($event)"
      />
    }
    
    <app-pagination 
      [currentPage]="page()" 
      [total]="totalPages()"
      (pageChange)="onPageChange($event)" 
    />
  `
})
export class UserListPageComponent {
  private store = inject(UserStore);
  
  users$ = this.store.filteredUsers$;
  loading$ = this.store.loading$;
  selectedId = this.store.selectedId;
  page = this.store.page;
  totalPages = this.store.totalPages;

  onSearch(term: string) { this.store.setFilter(term); }
  onSelectUser(id: string) { this.store.select(id); }
  onDeleteUser(id: string) { this.store.delete(id); }
  onPageChange(page: number) { this.store.setPage(page); }
}

// DUMB (Presentational) Component — pure UI, no services
@Component({
  selector: 'app-user-list',
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    @for (user of users; track user.id) {
      <div 
        [class.selected]="user.id === selectedId"
        (click)="selectUser.emit(user.id)"
      >
        {{ user.name }}
        <button (click)="deleteUser.emit(user.id); $event.stopPropagation()">
          Delete
        </button>
      </div>
    }
  `
})
export class UserListComponent {
  @Input() users: User[] = [];
  @Input() selectedId: string | null = null;
  @Output() selectUser = new EventEmitter<string>();
  @Output() deleteUser = new EventEmitter<string>();
  // NO inject(), NO services, NO business logic
  // Just @Input → render, user action → @Output
}

2. Server-Side Rendering (SSR) & Hydration

// Angular Universal / @angular/ssr
// ng add @angular/ssr

// app.config.server.ts
import { mergeApplicationConfig } from '@angular/core';
import { provideServerRendering } from '@angular/platform-server';
import { appConfig } from './app.config';

const serverConfig: ApplicationConfig = {
  providers: [provideServerRendering()]
};

export const config = mergeApplicationConfig(appConfig, serverConfig);

// Handling browser-only APIs in SSR:
import { isPlatformBrowser, isPlatformServer } from '@angular/common';
import { PLATFORM_ID } from '@angular/core';

export class MyComponent {
  private platformId = inject(PLATFORM_ID);

  ngOnInit() {
    if (isPlatformBrowser(this.platformId)) {
      // Safe to use window, document, localStorage
      window.addEventListener('scroll', this.onScroll);
    }
  }
}

// OR use afterNextRender (Angular 16+) — runs only in browser
export class ChartComponent {
  constructor() {
    afterNextRender(() => {
      // Runs only in browser, after first render
      this.initD3Chart();
    });
  }
}

// Hydration (Angular 16+) — reuse server-rendered DOM
// Enabled by default in Angular 17+ SSR apps
// Angular 19: Incremental hydration
@defer (on viewport; hydrate on interaction) {
  <app-heavy-widget />
}
// Server renders HTML, but doesn't hydrate (make interactive) 
// until user interacts with it!

3. Angular Elements (Web Components)

// Convert Angular components to Custom Elements (Web Components)
// They work in ANY framework (React, Vue, plain HTML)

import { createCustomElement } from '@angular/elements';

@Component({
  selector: 'app-feedback-widget',
  standalone: true,
  template: `
    <div class="widget">
      <h3>{{ title }}</h3>
      <textarea [(ngModel)]="feedback"></textarea>
      <button (click)="submit()">Send</button>
    </div>
  `
})
export class FeedbackWidgetComponent {
  @Input() title = 'Feedback';
  @Output() submitted = new EventEmitter<string>();
  feedback = '';
  
  submit() { this.submitted.emit(this.feedback); }
}

// Register as custom element
export class AppModule {
  constructor(private injector: Injector) {
    const el = createCustomElement(FeedbackWidgetComponent, { injector });
    customElements.define('feedback-widget', el);
  }
  ngDoBootstrap() {}  // No bootstrap component
}

// Now use ANYWHERE:
// <feedback-widget title="How was your experience?"></feedback-widget>
// Works in React, Vue, WordPress, static HTML!

5. Security Best Practices

// 1. XSS Protection — Angular sanitizes by default
// SAFE: Angular escapes all interpolation
{{ userInput }}  // <script> becomes <script> (text, not executed)

// DANGEROUS: Bypassing sanitization
// Only use when you TRUST the source (e.g., your own CMS)
import { DomSanitizer, SafeHtml } from '@angular/platform-browser';

@Component({ ... })
export class RichContentComponent {
  private sanitizer = inject(DomSanitizer);
  
  trustedHtml: SafeHtml;
  
  constructor() {
    // Only for TRUSTED content (e.g., your own API, not user input!)
    this.trustedHtml = this.sanitizer.bypassSecurityTrustHtml(htmlFromCMS);
  }
}

// 2. CSRF Protection
// Angular's HttpClient automatically reads XSRF-TOKEN cookie 
// and sends it as X-XSRF-TOKEN header
provideHttpClient(withXsrfConfiguration({
  cookieName: 'XSRF-TOKEN',
  headerName: 'X-XSRF-TOKEN'
}))

// 3. Content Security Policy (CSP)
// In server headers or meta tag — prevents inline scripts
// Angular AOT is CSP-compatible (no eval())

// 4. Route Guard for authorization
// 5. HTTP Interceptors for token management
// 6. Avoid using innerHTML — use [innerHtml] with sanitization
// 7. Validate on BOTH client and server
// 8. Use HttpOnly cookies for tokens (not localStorage)

7. Error Handling & Logging Architecture

// Global error handler
@Injectable()
export class GlobalErrorHandler implements ErrorHandler {
  private notification = inject(NotificationService);
  private logger = inject(LoggingService);

  handleError(error: Error): void {
    // Log to external service (Sentry, DataDog, etc.)
    this.logger.logError(error);

    // Show user-friendly message
    if (error instanceof HttpErrorResponse) {
      this.handleHttpError(error);
    } else {
      this.notification.show('An unexpected error occurred', 'error');
    }

    console.error('Global Error:', error);
  }

  private handleHttpError(error: HttpErrorResponse): void {
    switch (error.status) {
      case 0: this.notification.show('No internet connection', 'warning'); break;
      case 401: /* handled by interceptor */ break;
      case 403: this.notification.show('Access denied', 'error'); break;
      case 404: this.notification.show('Resource not found', 'error'); break;
      case 500: this.notification.show('Server error', 'error'); break;
    }
  }
}

// Register
providers: [
  { provide: ErrorHandler, useClass: GlobalErrorHandler }
]
