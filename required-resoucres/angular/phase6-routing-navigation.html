2. Accessing Route Data

// Method 1: withComponentInputBinding() — MODERN (Angular 16+)
// Route params automatically bound to @Input with same name!
@Component({ ... })
export class UserDetailComponent {
  @Input() id!: string;  // Automatically bound from :id in URL
  // Works for route params, query params, data, and resolve
}

// Method 2: ActivatedRoute (traditional)
export class UserDetailComponent implements OnInit {
  private route = inject(ActivatedRoute);
  private router = inject(Router);

  ngOnInit() {
    // Route params (observable — reacts to URL changes)
    this.route.params.subscribe(params => {
      this.userId = params['id'];
    });

    // Snapshot (one-time, no reactivity)
    const id = this.route.snapshot.params['id'];

    // Query params: /users?page=2&sort=name
    this.route.queryParams.subscribe(params => {
      this.page = +params['page'] || 1;
      this.sort = params['sort'] || 'name';
    });

    // Fragment: /users#section1
    this.route.fragment.subscribe(f => this.fragment = f);

    // Route data (static data defined in route config)
    this.route.data.subscribe(data => {
      this.role = data['role'];
      this.resolvedUser = data['user'];  // From resolver
    });

    // URL segments
    this.route.url.subscribe(segments => { ... });
  }

  // Programmatic navigation
  goToUser(id: string) {
    this.router.navigate(['/users', id]);
    this.router.navigate(['/users', id], { queryParams: { tab: 'posts' } });
    this.router.navigate(['/users', id], { 
      queryParams: { page: 2 },
      queryParamsHandling: 'merge'  // Keep existing query params
    });
    this.router.navigateByUrl('/users/123?tab=posts');
  }
}

3. Route Guards (Functional — Modern)

// auth.guard.ts — canActivate
export const authGuard: CanActivateFn = (route, state) => {
  const authService = inject(AuthService);
  const router = inject(Router);

  if (authService.isAuthenticated()) {
    return true;
  }
  
  // Store attempted URL for redirect after login
  return router.createUrlTree(['/login'], {
    queryParams: { returnUrl: state.url }
  });
};

// role.guard.ts — role-based access
export const roleGuard: CanActivateFn = (route) => {
  const authService = inject(AuthService);
  const requiredRole = route.data['role'] as string;
  
  return authService.hasRole(requiredRole);
};

// unsaved-changes.guard.ts — canDeactivate
export interface HasUnsavedChanges {
  hasUnsavedChanges(): boolean;
}

export const unsavedChangesGuard: CanDeactivateFn<HasUnsavedChanges> = 
  (component) => {
    if (component.hasUnsavedChanges()) {
      return confirm('You have unsaved changes. Leave anyway?');
    }
    return true;
  };

// canMatch — prevent route from matching (useful for feature flags)
export const featureFlagGuard: CanMatchFn = (route) => {
  const featureService = inject(FeatureFlagService);
  return featureService.isEnabled(route.data?.['feature']);
};

// Usage in routes:
{
  path: 'admin',
  canActivate: [authGuard, roleGuard],
  canDeactivate: [unsavedChangesGuard],
  canMatch: [featureFlagGuard],
  data: { role: 'admin', feature: 'admin-panel' },
  loadComponent: () => import('./admin.component')
}

4. Resolvers

// user.resolver.ts — Fetch data BEFORE navigation
export const userResolver: ResolveFn<User> = (route) => {
  const userService = inject(UserService);
  const router = inject(Router);
  const id = route.params['id'];

  return userService.getUser(id).pipe(
    catchError(() => {
      router.navigate(['/not-found']);
      return EMPTY;
    })
  );
};

// Usage:
{ 
  path: 'users/:id', 
  component: UserDetailComponent,
  resolve: { user: userResolver }
}

// Component receives resolved data via route data or @Input
@Component({ ... })
export class UserDetailComponent {
  @Input() user!: User;  // With withComponentInputBinding()
  // OR
  user = inject(ActivatedRoute).snapshot.data['user'];
}

6. Router Events

export class AppComponent {
  private router = inject(Router);

  constructor() {
    this.router.events.pipe(
      filter(event => 
        event instanceof NavigationStart ||
        event instanceof NavigationEnd ||
        event instanceof NavigationError
      )
    ).subscribe(event => {
      if (event instanceof NavigationStart) {
        this.showLoading();
      }
      if (event instanceof NavigationEnd) {
        this.hideLoading();
        this.trackPageView(event.urlAfterRedirects);
      }
      if (event instanceof NavigationError) {
        this.hideLoading();
        this.handleError(event.error);
      }
    });
  }
}

// All router events:
// NavigationStart → RoutesRecognized → GuardsCheckStart → 
// ChildActivationStart → ActivationStart → GuardsCheckEnd →
// ResolveStart → ResolveEnd → ActivationEnd → 
// ChildActivationEnd → NavigationEnd/NavigationCancel/NavigationError

7. View Transitions API (withViewTransitions)

Angular 17+ integrates the browser's View Transitions API for smooth page-to-page animations during navigation — no manual animation code needed.

// Enable view transitions:
provideRouter(routes,
  withViewTransitions({
    skipInitialTransition: true,  // Don't animate first page load
    onViewTransitionCreated: (info) => {
      // info.transition: ViewTransition object
      // info.from: ActivatedRouteSnapshot
      // info.to: ActivatedRouteSnapshot
      
      // Custom logic: skip transition for back navigation
      if (info.from.url.toString() > info.to.url.toString()) {
        info.transition.skipTransition();
      }
    }
  })
)

// CSS for transitions (in global styles):
::view-transition-old(root) {
  animation: slide-out 0.3s ease-in;
}
::view-transition-new(root) {
  animation: slide-in 0.3s ease-out;
}

// Per-element transitions (shared element animation):
// Component A:
<img [src]="product.image" style="view-transition-name: product-hero" />
// Component B (same view-transition-name):
<img [src]="product.image" style="view-transition-name: product-hero" />
// → Browser animates the image smoothly between pages!

8. Advanced Preloading Strategies

// Built-in strategies:
provideRouter(routes,
  withPreloading(PreloadAllModules)  // Preload all lazy routes after initial load
)

// Custom strategy — preload based on route data:
@Injectable({ providedIn: 'root' })
class SelectivePreloadStrategy implements PreloadingStrategy {
  preload(route: Route, load: () => Observable<any>): Observable<any> {
    // Only preload routes marked with data.preload = true
    return route.data?.['preload'] ? load() : of(null);
  }
}

// Routes:
{ path: 'dashboard', loadComponent: ..., data: { preload: true } },  // ← preloaded
{ path: 'admin', loadComponent: ... }  // ← NOT preloaded (loads on navigate)

// Network-aware preloading (only on fast connections):
@Injectable({ providedIn: 'root' })
class NetworkAwarePreload implements PreloadingStrategy {
  preload(route: Route, load: () => Observable<any>): Observable<any> {
    const conn = (navigator as any).connection;
    const dominated = conn?.saveData || conn?.effectiveType === '2g';
    return dominated ? of(null) : load();
  }
}

provideRouter(routes, withPreloading(NetworkAwarePreload))

9. Other Functional Router Features

provideRouter(routes,
  withComponentInputBinding(),       // Bind route params to @Input / input()
  withViewTransitions(),             // Smooth page transitions
  withPreloading(PreloadAllModules), // Background preloading
  withNavigationErrorHandler((err) => {
    // Global handler for navigation errors (replaces NavigationError subscription)
    inject(Router).navigate(['/error'], { state: { error: err } });
  }),
  withHashLocation(),                // Use hash-based URLs (#/path)
  withRouterConfig({
    onSameUrlNavigation: 'reload',   // Allow re-triggering same route
    paramsInheritanceStrategy: 'always', // Children inherit parent params
    urlUpdateStrategy: 'eager'       // Update URL before guards resolve
  }),
  withInMemoryScrolling({
    scrollPositionRestoration: 'enabled',  // Restore scroll on back/forward
    anchorScrolling: 'enabled'             // Scroll to #anchors
  }),
  withDebugTracing()                 // Log all router events (dev only)
)
