1. NgRx Entity Adapter — CRUD at Scale

EntityAdapter provides a standard way to manage normalized collections. Instead of arrays, it uses a dictionary (Record) + id array for O(1) lookups.

Entity State Structure

// What EntityAdapter manages:
interface EntityState<T> {
  ids: string[] | number[];      // Ordered array of IDs (controls display order)
  entities: Record<string | number, T>;  // Dictionary for O(1) lookup
}

// Example — ProductState:
// {
//   ids: [1, 3, 5],
//   entities: {
//     1: { id: 1, name: 'Widget', price: 10 },
//     3: { id: 3, name: 'Gadget', price: 25 },
//     5: { id: 5, name: 'Tool', price: 15 }
//   }
// }

Complete Entity Setup

import { createEntityAdapter, EntityState } from '@ngrx/entity';

// 1. Define entity interface
interface Product {
  id: number;
  name: string;
  price: number;
  category: string;
}

// 2. Create adapter (optionally with sort)
const productAdapter = createEntityAdapter<Product>({
  selectId: (product) => product.id,        // Custom ID selector (default: entity.id)
  sortComparer: (a, b) => a.name.localeCompare(b.name), // Keep sorted by name
});

// 3. Define state (can extend with additional properties)
interface ProductState extends EntityState<Product> {
  loading: boolean;
  selectedId: number | null;
  error: string | null;
}

// 4. Initial state
const initialState: ProductState = productAdapter.getInitialState({
  loading: false,
  selectedId: null,
  error: null
});

// 5. Reducer using adapter methods
const productReducer = createReducer(
  initialState,

  on(ProductActions.loadSuccess, (state, { products }) =>
    productAdapter.setAll(products, { ...state, loading: false })
  ),

  on(ProductActions.addSuccess, (state, { product }) =>
    productAdapter.addOne(product, state)
  ),

  on(ProductActions.updateSuccess, (state, { update }) =>
    productAdapter.updateOne(update, state)
    // update = { id: 5, changes: { price: 20 } }
  ),

  on(ProductActions.deleteSuccess, (state, { id }) =>
    productAdapter.removeOne(id, state)
  ),

  on(ProductActions.upsertMany, (state, { products }) =>
    productAdapter.upsertMany(products, state)
    // Adds if doesn't exist, updates if exists
  ),
);

// 6. Selectors (adapter generates them)
const { selectAll, selectEntities, selectIds, selectTotal } =
  productAdapter.getSelectors();

// Feature selector
const selectProductState = createFeatureSelector<ProductState>('products');
const selectAllProducts = createSelector(selectProductState, selectAll);
const selectProductEntities = createSelector(selectProductState, selectEntities);
const selectProductCount = createSelector(selectProductState, selectTotal);
const selectSelectedProduct = createSelector(
  selectProductEntities,
  selectProductState,
  (entities, state) => state.selectedId ? entities[state.selectedId] : null
);

2. NgRx Router Store

Syncs Angular Router state into the NgRx store, making route params/data available via selectors (instead of injecting ActivatedRoute everywhere).

// Setup
import { provideRouterStore, routerReducer } from '@ngrx/router-store';

// app.config.ts
providers: [
  provideStore({ router: routerReducer }),
  provideRouterStore()  // Syncs router state to store
]

// Custom serializer (control what gets stored)
import { RouterStateSerializer } from '@ngrx/router-store';

interface AppRouterState {
  url: string;
  params: Record<string, string>;
  queryParams: Record<string, string>;
  data: Record<string, any>;
}

class CustomSerializer implements RouterStateSerializer<AppRouterState> {
  serialize(routerState: RouterStateSnapshot): AppRouterState {
    let route = routerState.root;
    while (route.firstChild) route = route.firstChild;
    return {
      url: routerState.url,
      params: route.params,
      queryParams: routerState.root.queryParams,
      data: route.data
    };
  }
}

// Register:
provideRouterStore({ serializer: CustomSerializer })

3. Meta-Reducers (Middleware Pattern)

Meta-reducers wrap ALL reducers. They receive a reducer and return a new reducer — like middleware. Every action passes through all meta-reducers before reaching the actual reducer.

// Meta-reducer signature:
type MetaReducer<T> = (reducer: ActionReducer<T>) => ActionReducer<T>;

// 1. Logger Meta-Reducer (most common interview question)
function logger(reducer: ActionReducer<any>): ActionReducer<any> {
  return (state, action) => {
    console.group(action.type);
    console.log('Previous State:', state);
    console.log('Action:', action);
    const nextState = reducer(state, action);  // Call actual reducer
    console.log('Next State:', nextState);
    console.groupEnd();
    return nextState;
  };
}

// 2. State Hydration (localStorage persistence)
function hydrationMetaReducer(reducer: ActionReducer<any>): ActionReducer<any> {
  return (state, action) => {
    if (action.type === '@ngrx/store/init') {
      const stored = localStorage.getItem('appState');
      if (stored) return JSON.parse(stored);
    }
    const nextState = reducer(state, action);
    localStorage.setItem('appState', JSON.stringify(nextState));
    return nextState;
  };
}

// 3. Reset State on Logout
function resetOnLogout(reducer: ActionReducer<any>): ActionReducer<any> {
  return (state, action) => {
    if (action.type === '[Auth] Logout') {
      state = undefined;  // Forces all reducers to return initial state
    }
    return reducer(state, action);
  };
}

// 4. Undo/Redo Meta-Reducer
function undoRedo(reducer: ActionReducer<any>): ActionReducer<any> {
  const history: any[] = [];
  let currentIndex = -1;

  return (state, action) => {
    switch (action.type) {
      case '[App] Undo':
        return currentIndex > 0 ? history[--currentIndex] : state;
      case '[App] Redo':
        return currentIndex < history.length - 1 ? history[++currentIndex] : state;
      default:
        const nextState = reducer(state, action);
        history.splice(++currentIndex);
        history.push(nextState);
        return nextState;
    }
  };
}

// Register meta-reducers (order matters — first wraps outermost):
provideStore(reducers, {
  metaReducers: [logger, hydrationMetaReducer, resetOnLogout]
})
// Execution: logger → hydration → resetOnLogout → actual reducer
🎯 Interview Questions:
  • What is a meta-reducer? → Higher-order function that wraps all reducers; like middleware for the store
  • In what order do meta-reducers execute? → First in array is outermost (wraps everything)
  • How would you implement undo/redo with NgRx? → Meta-reducer maintaining history stack
  • How do you persist NgRx state to localStorage? → Hydration meta-reducer that checks on init and saves on every action

5. NgRx SignalStore — Advanced Features

import { signalStore, withState, withComputed, withMethods, withHooks, patchState } from '@ngrx/signals';
import { withEntities, setAllEntities, addEntity, removeEntity } from '@ngrx/signals/entities';

// Advanced SignalStore with entities + custom features
export const ProductStore = signalStore(
  { providedIn: 'root' },  // Singleton

  // Entities
  withEntities<Product>(),

  // Additional state
  withState({
    loading: false,
    filter: '',
    selectedId: null as number | null
  }),

  // Computed (derived signals)
  withComputed(({ entities, filter, selectedId }) => ({
    filteredProducts: computed(() => {
      const f = filter().toLowerCase();
      return entities().filter(p => p.name.toLowerCase().includes(f));
    }),
    selectedProduct: computed(() =>
      entities().find(p => p.id === selectedId())
    ),
    totalPrice: computed(() =>
      entities().reduce((sum, p) => sum + p.price, 0)
    )
  })),

  // Methods
  withMethods((store, api = inject(ProductApi)) => ({
    async loadAll() {
      patchState(store, { loading: true });
      const products = await firstValueFrom(api.getAll());
      patchState(store, setAllEntities(products), { loading: false });
    },
    select(id: number) {
      patchState(store, { selectedId: id });
    },
    async delete(id: number) {
      patchState(store, removeEntity(id));  // Optimistic
      try {
        await firstValueFrom(api.delete(id));
      } catch {
        // Rollback: reload all
        this.loadAll();
      }
    },
    setFilter(filter: string) {
      patchState(store, { filter });
    }
  })),

  // Lifecycle hooks
  withHooks({
    onInit(store) {
      store.loadAll();  // Auto-load on first injection
    },
    onDestroy() {
      console.log('Store destroyed');
    }
  })
);

// Custom Store Feature (reusable across stores)
function withLoading() {
  return signalStoreFeature(
    withState({ loading: false, error: null as string | null }),
    withMethods((store) => ({
      setLoading() { patchState(store, { loading: true, error: null }); },
      setLoaded() { patchState(store, { loading: false }); },
      setError(error: string) { patchState(store, { loading: false, error }); }
    }))
  );
}

// Usage:
const MyStore = signalStore(
  withLoading(),  // Reusable feature
  withEntities<Item>(),
  withMethods(/* ... */)
);

6. Custom RxJS Operators

Building Pipeable Operators

import { Observable, OperatorFunction, MonoTypeOperatorFunction } from 'rxjs';

// Pattern: A function that takes config → returns OperatorFunction

// 1. Simple: Debug operator
function debug<T>(tag: string): MonoTypeOperatorFunction<T> {
  return (source: Observable<T>) =>
    source.pipe(
      tap({
        next: (val) => console.log(`[${tag}] next:`, val),
        error: (err) => console.error(`[${tag}] error:`, err),
        complete: () => console.log(`[${tag}] complete`)
      })
    );
}

// Usage: this.data$.pipe(debug('user-data'), map(...))

// 2. Transform: filterNil (remove null/undefined)
function filterNil<T>(): OperatorFunction<T | null | undefined, T> {
  return (source) => source.pipe(
    filter((value): value is T => value != null)
  );
}

// Usage: this.user$.pipe(filterNil(), map(u => u.name)) // u is guaranteed non-null

// 3. Stateful: distinctUntilKeyChanged with deep comparison
function distinctByKey<T, K extends keyof T>(key: K): MonoTypeOperatorFunction<T> {
  return (source) => source.pipe(
    distinctUntilChanged((prev, curr) =>
      JSON.stringify(prev[key]) === JSON.stringify(curr[key])
    )
  );
}

// 4. Complex: retryWithBackoff
function retryWithBackoff<T>(config: {
  maxRetries: number;
  initialDelay: number;
  retryWhen?: (error: any) => boolean;
}): MonoTypeOperatorFunction<T> {
  return (source) => source.pipe(
    retry({
      count: config.maxRetries,
      delay: (error, retryCount) => {
        if (config.retryWhen && !config.retryWhen(error)) {
          throw error; // Don't retry
        }
        const delay = config.initialDelay * Math.pow(2, retryCount - 1);
        console.log(`Retry #${retryCount} in ${delay}ms`);
        return timer(delay);
      }
    })
  );
}

// Usage:
this.http.get('/api/data').pipe(
  retryWithBackoff({
    maxRetries: 3,
    initialDelay: 1000,
    retryWhen: (err) => err.status >= 500  // Only retry server errors
  })
)

Higher-Order Custom Operator

// Operator that takes an observable as parameter
function takeUntilDestroyed<T>(destroyRef?: DestroyRef): MonoTypeOperatorFunction<T> {
  const ref = destroyRef ?? inject(DestroyRef);
  const destroy$ = new Observable<void>(subscriber => {
    ref.onDestroy(() => {
      subscriber.next();
      subscriber.complete();
    });
  });
  return (source) => source.pipe(takeUntil(destroy$));
}

// Operator that wraps inner observables with caching
function cacheResult<T>(duration: number): OperatorFunction<Observable<T>, T> {
  let cached: T | undefined;
  let lastFetch = 0;

  return (source: Observable<Observable<T>>) =>
    source.pipe(
      switchMap(inner$ => {
        if (cached && Date.now() - lastFetch < duration) {
          return of(cached);
        }
        return inner$.pipe(
          tap(val => { cached = val; lastFetch = Date.now(); })
        );
      })
    );
}

7. RxJS Schedulers Deep Dive

Schedulers control WHEN work is executed. They're rarely used directly in application code but are critical for testing and performance optimization.

SchedulerExecutionUse Case
queueSchedulerSynchronous, FIFO queue (current call stack)Recursive operations, prevent stack overflow
asapSchedulerMicrotask (Promise.resolve)After current sync code, before setTimeout
asyncSchedulerMacrotask (setTimeout)Default for time-based operators (delay, interval)
animationFrameSchedulerBefore next browser repaint (rAF)Smooth animations, DOM reads/writes
import { asyncScheduler, asapScheduler, queueScheduler, animationFrameScheduler,
         observeOn, subscribeOn, scheduled } from 'rxjs';

// 1. observeOn — control WHERE emissions are processed
source$.pipe(
  observeOn(animationFrameScheduler)  // Process each emission in rAF
).subscribe(val => {
  // This runs in requestAnimationFrame — smooth for DOM updates
  element.style.transform = `translateX(${val}px)`;
});

// 2. subscribeOn — control WHEN subscription happens
of(1, 2, 3).pipe(
  subscribeOn(asyncScheduler)  // Delay subscription to next macrotask
).subscribe(console.log);
console.log('sync');
// Output: "sync", 1, 2, 3 (subscription deferred)

// 3. scheduled — create observable ON a scheduler
scheduled([1, 2, 3], asapScheduler).subscribe(console.log);
// Emissions delivered via microtasks (between sync code and setTimeout)

// 4. animationFrameScheduler for smooth scrolling
interval(0, animationFrameScheduler).pipe(
  take(60),  // 60 frames
  map(frame => frame * 10)  // 10px per frame
).subscribe(offset => {
  window.scrollTo(0, offset);  // Smooth 60fps scroll
});

// 5. queueScheduler prevents stack overflow in recursion
const factorial = (n: number): Observable<number> =>
  n <= 1
    ? scheduled([1], queueScheduler)
    : scheduled([n], queueScheduler).pipe(
        mergeMap(x => factorial(x - 1).pipe(map(r => x * r)))
      );

Scheduler in Testing (TestScheduler)

// TestScheduler virtualizes time — tests run INSTANTLY regardless of delays
import { TestScheduler } from 'rxjs/testing';

const scheduler = new TestScheduler((actual, expected) => {
  expect(actual).toEqual(expected);
});

scheduler.run(({ cold, hot, expectObservable, expectSubscriptions }) => {
  // Virtual time — no actual waiting!
  const source$ = cold('--a--b--c|');
  const result$ = source$.pipe(delay(10));  // 10ms delay
  
  // In marble notation: 10ms = 10 frames
  expectObservable(result$).toBe('------------a--b--c|');
});

Cold vs Hot in Marble Testing

scheduler.run(({ cold, hot, expectObservable }) => {
  // COLD: Starts from beginning for each subscriber
  // Subscription starts at frame 0 of the marble string
  const cold$ = cold('--a--b--c|', { a: 1, b: 2, c: 3 });

  // HOT: Already "running" — subscriber joins mid-stream
  // ^ marks where subscription happens
  const hot$ = hot('--a--b--^--c--d|', { a: 1, b: 2, c: 3, d: 4 });
  // Subscriber only sees c and d (joined after ^)
});

Complete Testing Examples

describe('SearchService', () => {
  let scheduler: TestScheduler;

  beforeEach(() => {
    scheduler = new TestScheduler((actual, expected) => {
      expect(actual).toEqual(expected);
    });
  });

    scheduler.run(({ cold, expectObservable }) => {
      // Simulate user typing with 300ms debounce
      const input$  = cold('--a--b-----c--|');
      const apiResp = cold('--x|');  // API takes 2 frames

      const result$ = input$.pipe(
        debounceTime(5),   // 5ms debounce
        switchMap(() => apiResp)

      // a: emitted at 2, debounce waits 5 → triggers at 7, api responds at 9
      // b: emitted at 5, debounce waits 5 → triggers at 10, CANCELS a's request, api responds at 12
      // c: emitted at 11, debounce waits 5 → triggers at 16... wait that's after complete
      expectObservable(result$).toBe('---------x----x---|');
    });
  });

        attempt++;
        if (attempt < 3) return cold('#');  // Fail first 2 times
        return cold('--a|');                 // Succeed on 3rd
      });

      const result$ = source$.pipe(retry(3));
      expectObservable(result$).toBe('--a|');
    });
  });

  it('should test subscription timing', () => {

9. HTTP Request Deduplication

shareReplay Pattern

// PROBLEM: Multiple components call the same API simultaneously
// → Multiple network requests for the same data!

// SOLUTION 1: shareReplay in service
@Injectable({ providedIn: 'root' })
export class ConfigService {
  // Single shared request — all subscribers get same response
  config$ = this.http.get<Config>('/api/config').pipe(
    shareReplay({ bufferSize: 1, refCount: true })
    // bufferSize: 1 → cache latest value for new subscribers
    // refCount: true → resubscribe (refetch) after all unsubscribe
  );

  constructor(private http: HttpClient) {}
}

// ⚠️ GOTCHA: shareReplay WITHOUT refCount keeps the subscription alive forever
// (memory leak if the source can error or the service is recreated)

// SOLUTION 2: Deduplication Interceptor (advanced)
const inFlightRequests = new Map<string, Observable<HttpEvent<unknown>>>();

export const dedupeInterceptor: HttpInterceptorFn = (req, next) => {
  // Only dedupe GET requests
  if (req.method !== 'GET') return next(req);

  const key = req.urlWithParams;

  if (inFlightRequests.has(key)) {
    return inFlightRequests.get(key)!;  // Return existing in-flight request
  }

  const shared$ = next(req).pipe(
    finalize(() => inFlightRequests.delete(key)),  // Cleanup after complete/error
    shareReplay(1)
  );

  inFlightRequests.set(key, shared$);
  return shared$;
};

10. Enterprise RxJS Patterns

Token Refresh Queue (Serial Execution)

// Multiple 401 responses shouldn't trigger multiple refresh calls
@Injectable({ providedIn: 'root' })
class AuthTokenService {
  private refreshing = false;
  private refreshSubject = new BehaviorSubject<string | null>(null);

  refreshToken(): Observable<string> {
    if (this.refreshing) {
      // Already refreshing — wait for the result
      return this.refreshSubject.pipe(
        filter(token => token !== null),
        take(1)
      );
    }

    this.refreshing = true;
    this.refreshSubject.next(null);

    return this.http.post<{ token: string }>('/auth/refresh', {}).pipe(
      tap(({ token }) => {
        this.refreshing = false;
        this.refreshSubject.next(token);
      }),
      map(({ token }) => token),
      catchError(err => {
        this.refreshing = false;
        this.refreshSubject.error(err);
        this.refreshSubject = new BehaviorSubject<string | null>(null);
        return throwError(() => err);
      })
    );
  }
}

WebSocket with Auto-Reconnect

function createReconnectingWebSocket(url: string): Observable<MessageEvent> {
  return new Observable(subscriber => {
    let ws: WebSocket;
    let reconnectTimer: any;

    function connect() {
      ws = new WebSocket(url);
      ws.onmessage = (msg) => subscriber.next(msg);
      ws.onerror = (err) => console.warn('WS error:', err);
      ws.onclose = (event) => {
        if (!event.wasClean) {
          // Reconnect with exponential backoff
          const delay = Math.min(30000, 1000 * Math.pow(2, Math.random() * 3));
          reconnectTimer = setTimeout(connect, delay);
        }
      };
    }

    connect();

    return () => {
      clearTimeout(reconnectTimer);
      ws?.close(1000, 'Component destroyed');
    };
  }).pipe(
    share()  // Share single connection among subscribers
  );
}

// Usage:
messages$ = createReconnectingWebSocket('wss://api.example.com/ws').pipe(
  map(event => JSON.parse(event.data)),
  retry({ delay: 5000 })  // Additional retry layer
);

Conditional Polling with Visibility API

// Stop polling when tab is hidden, resume when visible
function smartPoll<T>(apiCall: () => Observable<T>, intervalMs: number): Observable<T> {
  const visibility$ = fromEvent(document, 'visibilitychange').pipe(
    map(() => document.visibilityState === 'visible'),
    startWith(true)
  );

  return visibility$.pipe(
    switchMap(visible =>
      visible
        ? timer(0, intervalMs).pipe(switchMap(() => apiCall()))
        : EMPTY  // Stop polling when hidden
    )
  );
}

// Usage:
notifications$ = smartPoll(() => this.http.get('/api/notifications'), 10_000);

Race Condition Prevention (exhaustMap + queue)

// Form submissions that must be sequential (not lost, not duplicated)
@Component({ ... })
class OrderComponent {
  private submitQueue = new Subject<Order>();

  orders$ = this.submitQueue.pipe(
    // concatMap queues requests — ensures order, no loss
    concatMap(order =>
      this.api.createOrder(order).pipe(
        catchError(err => {
          this.snackbar.open(`Failed: ${err.message}`);
          return EMPTY;  // Skip failed, continue queue
        })
      )
    )
  );

  submit(order: Order) {
    this.submitQueue.next(order);  // Just queue it
  }
}

Combining Multiple Async Sources with Loading State

// Dashboard that needs data from 3 APIs with unified loading
interface DashboardState {
  users: User[] | null;
  orders: Order[] | null;
  stats: Stats | null;
  loading: boolean;
  error: string | null;
}

dashboard$ = forkJoin({
  users: this.userApi.getAll(),
  orders: this.orderApi.getRecent(),
  stats: this.statsApi.getSummary()
}).pipe(
  map(({ users, orders, stats }): DashboardState => ({
    users, orders, stats, loading: false, error: null
  })),
  startWith({ users: null, orders: null, stats: null, loading: true, error: null }),
  catchError(err => of({
    users: null, orders: null, stats: null, loading: false, error: err.message
  }))
);

// Alternative with individual loading (show data as it arrives):
dashboard$ = combineLatest({
  users: this.userApi.getAll().pipe(startWith(null)),
  orders: this.orderApi.getRecent().pipe(startWith(null)),
  stats: this.statsApi.getSummary().pipe(startWith(null))
}).pipe(
  map(data => ({
    ...data,
    loading: !data.users || !data.orders || !data.stats
  }))
);
