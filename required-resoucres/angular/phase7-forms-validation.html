2. Reactive Forms — Complete Guide

Form Building

import { FormBuilder, FormGroup, FormControl, FormArray, Validators } from '@angular/forms';

@Component({
  imports: [ReactiveFormsModule],
  template: `
    <form [formGroup]="userForm" (ngSubmit)="onSubmit()">
      <!-- Simple controls -->
      <input formControlName="name" placeholder="Name">
      <input formControlName="email" type="email" placeholder="Email">
      
      <!-- Nested FormGroup -->
      <div formGroupName="address">
        <input formControlName="street">
        <input formControlName="city">
        <input formControlName="zip">
      </div>
      
      <!-- FormArray -->
      <div formArrayName="skills">
        @for (skill of skillControls.controls; track $index; let i = $index) {
          <div>
            <input [formControlName]="i">
            <button type="button" (click)="removeSkill(i)">Remove</button>
          </div>
        }
        <button type="button" (click)="addSkill()">Add Skill</button>
      </div>
      
      <button type="submit" [disabled]="userForm.invalid">Save</button>
    </form>
    
    <!-- Error display -->
    @if (userForm.get('email')?.hasError('email') && userForm.get('email')?.touched) {
      <span class="error">Invalid email format</span>
    }
  `
})
export class UserFormComponent {
  private fb = inject(FormBuilder);

  userForm = this.fb.group({
    name: ['', [Validators.required, Validators.minLength(2)]],
    email: ['', [Validators.required, Validators.email]],
    age: [null as number | null, [Validators.min(18), Validators.max(120)]],
    address: this.fb.group({
      street: ['', Validators.required],
      city: ['', Validators.required],
      zip: ['', [Validators.required, Validators.pattern(/^\d{5}(-\d{4})?$/)]]
    }),
    skills: this.fb.array([
      this.fb.control('', Validators.required)
    ])
  });

  get skillControls(): FormArray {
    return this.userForm.get('skills') as FormArray;
  }

  addSkill(): void {
    this.skillControls.push(this.fb.control('', Validators.required));
  }

  removeSkill(index: number): void {
    this.skillControls.removeAt(index);
  }

  onSubmit(): void {
    if (this.userForm.valid) {
      console.log(this.userForm.value);  // Typed!
      console.log(this.userForm.getRawValue());  // Includes disabled controls
    } else {
      this.userForm.markAllAsTouched();  // Show all errors
    }
  }
}

3. Typed Forms (Angular 14+)

// Strictly typed form — compiler catches errors!
interface UserFormValue {
  name: string;
  email: string;
  age: number | null;
  address: {
    street: string;
    city: string;
  };
}

// FormGroup is now generic
userForm = new FormGroup({
  name: new FormControl('', { nonNullable: true }),  // Type: FormControl<string>
  email: new FormControl('', { nonNullable: true }),
  age: new FormControl<number | null>(null),
  address: new FormGroup({
    street: new FormControl('', { nonNullable: true }),
    city: new FormControl('', { nonNullable: true })
  })
});

// Type-safe access:
const name: string = this.userForm.controls.name.value;  // ✓ Typed!
this.userForm.controls.nonexistent;  // ✗ Compile error!

// NonNullable forms (reset returns to initial value, not null)
const form = this.fb.nonNullable.group({
  name: ['John'],  // Always string, never null
  email: ['']
});
form.reset();  // Values reset to 'John' and '', NOT null

4. Custom Validators

Synchronous Validators

// Validator function signature: (control: AbstractControl) => ValidationErrors | null
export function noWhitespace(control: AbstractControl): ValidationErrors | null {
  if (control.value && control.value.trim().length === 0) {
    return { whitespace: true };  // Error key
  }
  return null;  // Valid
}

// Validator with parameters (factory function)
export function minAge(age: number): ValidatorFn {
  return (control: AbstractControl): ValidationErrors | null => {
    const value = control.value;
    if (value !== null && value < age) {
      return { minAge: { required: age, actual: value } };
    }
    return null;
  };
}

// Cross-field validator (applied to FormGroup)
export function passwordMatch(control: AbstractControl): ValidationErrors | null {
  const form = control as FormGroup;
  const password = form.get('password')?.value;
  const confirm = form.get('confirmPassword')?.value;

  if (password && confirm && password !== confirm) {
    return { passwordMismatch: true };
  }
  return null;
}

// Usage:
this.fb.group({
  name: ['', [Validators.required, noWhitespace]],
  age: [null, [minAge(18)]],
  password: ['', Validators.required],
  confirmPassword: ['', Validators.required]
}, { validators: [passwordMatch] })  // Group-level validator

Async Validators

// Check if username/email is taken (calls API)
export function uniqueEmail(userService: UserService): AsyncValidatorFn {
  return (control: AbstractControl): Observable<ValidationErrors | null> => {
    if (!control.value) return of(null);
    
    return userService.checkEmailExists(control.value).pipe(
      debounceTime(300),
      map(exists => exists ? { emailTaken: true } : null),
      catchError(() => of(null))  // Don't block form on API error
    );
  };
}

// Usage:
email: ['', {
  validators: [Validators.required, Validators.email],
  asyncValidators: [uniqueEmail(this.userService)],
  updateOn: 'blur'  // Only validate on blur (not every keystroke)
}]

// In template:
@if (emailControl.pending) {
  <span>Checking availability...</span>
}
@if (emailControl.hasError('emailTaken')) {
  <span class="error">This email is already registered</span>
}

5. Dynamic Forms

// Generate form from configuration
interface FieldConfig {
  key: string;
  label: string;
  type: 'text' | 'number' | 'select' | 'checkbox' | 'textarea';
  validators?: ValidatorFn[];
  options?: { value: string; label: string }[];  // For select
}

@Component({
  template: `
    <form [formGroup]="dynamicForm" (ngSubmit)="submit()">
      @for (field of fields; track field.key) {
        <div class="form-field">
          <label>{{ field.label }}</label>
          
          @switch (field.type) {
            @case ('text') {
              <input type="text" [formControlName]="field.key">
            }
            @case ('number') {
              <input type="number" [formControlName]="field.key">
            }
            @case ('select') {
              <select [formControlName]="field.key">
                @for (opt of field.options; track opt.value) {
                  <option [value]="opt.value">{{ opt.label }}</option>
                }
              </select>
            }
            @case ('textarea') {
              <textarea [formControlName]="field.key"></textarea>
            }
          }
          
          @if (dynamicForm.get(field.key)?.invalid && dynamicForm.get(field.key)?.touched) {
            <span class="error">{{ field.label }} is invalid</span>
          }
        </div>
      }
      <button type="submit">Submit</button>
    </form>
  `
})
export class DynamicFormComponent implements OnInit {
  @Input() fields: FieldConfig[] = [];
  dynamicForm!: FormGroup;

  private fb = inject(FormBuilder);

  ngOnInit() {
    const group: Record<string, FormControl> = {};
    this.fields.forEach(field => {
      group[field.key] = new FormControl('', field.validators || []);
    });
    this.dynamicForm = new FormGroup(group);
  }

  submit() {
    if (this.dynamicForm.valid) {
      console.log(this.dynamicForm.value);
    }
  }
}

6. Form Control States & CSS Classes

// Angular automatically adds CSS classes to form controls:
// .ng-valid / .ng-invalid
// .ng-pristine / .ng-dirty (user has changed value)
// .ng-untouched / .ng-touched (user has focused and left field)
// .ng-pending (async validation in progress)

// SCSS for form validation styling:
input.ng-invalid.ng-touched {
  border-color: red;
}

input.ng-valid.ng-touched {
  border-color: green;
}

// Programmatic state checks:
const ctrl = this.form.get('email');
ctrl.valid        // passes all validators
ctrl.invalid      // fails at least one validator
ctrl.pristine     // value not changed by user
ctrl.dirty        // value changed by user
ctrl.touched      // field has been focused and blurred
ctrl.untouched    // field never focused
ctrl.pending      // async validation running
ctrl.disabled     // control is disabled
ctrl.enabled      // control is enabled
ctrl.errors       // object of current errors or null
ctrl.value        // current value

// Control manipulation:
ctrl.setValue('new value');           // Set exact value (must match structure for groups)
ctrl.patchValue({ name: 'partial' }); // Partial update (groups only)
ctrl.reset();                         // Reset to initial value
ctrl.disable();                       // Disable (excluded from form.value)
ctrl.enable();
ctrl.markAsTouched();
ctrl.markAllAsTouched();             // Mark all children touched (show all errors)
ctrl.updateValueAndValidity();        // Re-run validators

7. ControlValueAccessor — Custom Form Controls

// Create a custom component that works with formControlName and ngModel
@Component({
  selector: 'app-star-rating',
  standalone: true,
  template: `
    @for (star of stars; track $index; let i = $index) {
      <span 
        [class.filled]="i < value"
        (click)="rate(i + 1)"
      >★</span>
    }
  `,
  providers: [{
    provide: NG_VALUE_ACCESSOR,
    useExisting: forwardRef(() => StarRatingComponent),
    multi: true
  }]
})
export class StarRatingComponent implements ControlValueAccessor {
  stars = [1, 2, 3, 4, 5];
  value = 0;
  disabled = false;

  private onChange: (value: number) => void = () => {};
  private onTouched: () => void = () => {};

  // Called by Angular to write value to component
  writeValue(value: number): void {
    this.value = value || 0;
  }

  // Register callback for when value changes
  registerOnChange(fn: (value: number) => void): void {
    this.onChange = fn;
  }

  // Register callback for when control is touched
  registerOnTouched(fn: () => void): void {
    this.onTouched = fn;
  }

  // Called when control is disabled/enabled
  setDisabledState(isDisabled: boolean): void {
    this.disabled = isDisabled;
  }

  rate(stars: number): void {
    if (this.disabled) return;
    this.value = stars;
    this.onChange(stars);  // Notify form of change
    this.onTouched();     // Mark as touched
  }
}

// Usage — works with reactive forms AND template-driven:
<app-star-rating formControlName="rating" />
<app-star-rating [(ngModel)]="myRating" />
