1. Signals Fundamentals

What are Signals?

Signals are reactive primitives that hold a value and notify consumers when that value changes. They enable fine-grained reactivity — Angular knows exactly which parts of the template need updating, without Zone.js.

import { signal, computed, effect } from '@angular/core';

// Create a writable signal
const count = signal(0);

// Read the value (call the signal like a function)
console.log(count());  // 0

// Update the value
count.set(5);          // Set to specific value
count.update(c => c + 1);  // Update based on current value

// computed — derived signal (read-only, auto-tracks dependencies)
const doubleCount = computed(() => count() * 2);
console.log(doubleCount());  // 12

// effect — side effect that re-runs when dependencies change
effect(() => {
  console.log(`Count changed to: ${count()}`);
  // Automatically tracks which signals are read inside
  // Re-runs whenever count() changes
});

// Signal is read-only externally via .asReadonly()
const readonlyCount = count.asReadonly();  // WritableSignal → Signal

Signal vs BehaviorSubject

FeatureSignalBehaviorSubject
Read valuemySignal() — synchronous.getValue() or .value
SubscribeAutomatic in templates & computedManual .subscribe()
UnsubscribeNot needed (automatic)Must unsubscribe (memory leaks)
Glitch-freeYes (consistent state)No (intermediate values possible)
Template integrationDirect: {{ count() }}Needs async pipe: {{ count$ | async }}

2. computed() — Derived State

@Component({
  template: `
    <input [value]="searchTerm()" (input)="searchTerm.set($any($event.target).value)">
    <select (change)="sortBy.set($any($event.target).value)">
      <option value="name">Name</option>
      <option value="date">Date</option>
    </select>
    
    <p>Showing {{ filteredUsers().length }} of {{ users().length }} users</p>
    
    @for (user of filteredUsers(); track user.id) {
      <app-user-card [user]="user" />
    }
  `
})
export class UserListComponent {
  users = signal<User[]>([]);
  searchTerm = signal('');
  sortBy = signal<'name' | 'date'>('name');

  // Automatically recomputes when users(), searchTerm(), or sortBy() changes
  filteredUsers = computed(() => {
    let result = this.users();
    
    // Filter
    const term = this.searchTerm().toLowerCase();
    if (term) {
      result = result.filter(u => u.name.toLowerCase().includes(term));
    }
    
    // Sort
    const sort = this.sortBy();
    result = [...result].sort((a, b) => {
      if (sort === 'name') return a.name.localeCompare(b.name);
      return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
    });
    
    return result;
  });

  // Computed are LAZY and MEMOIZED:
  // - Only recalculates when dependencies actually change
  // - Reading the same computed multiple times returns cached value
  // - If no one reads it, it doesn't compute at all
}

3. effect() — Side Effects

@Component({ ... })
export class SettingsComponent {
  theme = signal<'light' | 'dark'>('light');
  fontSize = signal(14);

  constructor() {
    // effect runs once immediately, then re-runs when tracked signals change
    effect(() => {
      // Auto-persists settings to localStorage
      const settings = { theme: this.theme(), fontSize: this.fontSize() };
      localStorage.setItem('settings', JSON.stringify(settings));
    });

    // effect with cleanup
    effect((onCleanup) => {
      const id = setInterval(() => this.poll(), 5000);
      
      onCleanup(() => {
        clearInterval(id);  // Runs before next execution or on destroy
      });
    });
  }

  // IMPORTANT: You generally should NOT set signals inside effects
  // Use computed() for derived state instead
  // If you must (e.g., calling an API), use allowSignalWrites:
  constructor() {
    effect(() => {
      const userId = this.selectedUserId();
      // Fetch user data when selection changes
      this.userService.getUser(userId).subscribe(user => {
        this.currentUser.set(user);  // Writing signal in effect
      });
    }, { allowSignalWrites: true });
  }
}

4. Signal Inputs (Angular 17.1+)

// Traditional @Input
@Input() name: string = '';
@Input({ required: true }) id!: string;

// Signal Input — reactive, no need for ngOnChanges!
@Component({
  template: `
    <h1>{{ fullName() }}</h1>
    <p>ID: {{ id() }}</p>
  `
})
export class UserCardComponent {
  // Required signal input
  id = input.required<string>();
  
  // Optional signal input with default
  name = input('Anonymous');
  
  // Input with alias
  avatar = input('', { alias: 'userAvatar' });
  
  // Input with transform
  disabled = input(false, { transform: booleanAttribute });
  size = input(0, { transform: numberAttribute });

  // Computed based on inputs — REPLACES ngOnChanges!
  fullName = computed(() => {
    return `User: ${this.name()} (${this.id()})`;
  });

  // React to input changes with effect (replaces ngOnChanges)
  constructor() {
    effect(() => {
      console.log('ID changed to:', this.id());
      // Automatically re-runs when id input changes
    });
  }
}

// Usage:
// <app-user-card [id]="userId" [name]="userName" />

5. Signal Outputs & Model Inputs

// output() function (Angular 17.3+)
import { output } from '@angular/core';

@Component({ ... })
export class CounterComponent {
  // Signal-based output
  countChanged = output<number>();
  
  // With alias
  deleted = output<string>({ alias: 'itemDeleted' });

  increment() {
    this.countChanged.emit(this.count() + 1);
  }
}

// model() — Two-way binding with signals (Angular 17.2+)
import { model } from '@angular/core';

@Component({
  selector: 'app-toggle',
  template: `
    <button (click)="checked.set(!checked())">
      {{ checked() ? 'ON' : 'OFF' }}
    </button>
  `
})
export class ToggleComponent {
  // model() creates a writable signal that supports two-way binding
  checked = model(false);  // ModelSignal<boolean>
  
  // Required model
  value = model.required<string>();
}

// Parent usage with two-way binding:
// <app-toggle [(checked)]="isEnabled" />
// Equivalent to: [checked]="isEnabled" (checkedChange)="isEnabled = $event"

6. linkedSignal (Angular 19)

// linkedSignal — writable signal that resets when a source changes
import { linkedSignal } from '@angular/core';

@Component({ ... })
export class ProductPageComponent {
  products = signal<Product[]>([]);
  
  // Selected product resets to first item when products list changes
  selectedProduct = linkedSignal(() => this.products()[0]);
  
  // linkedSignal with explicit source tracking
  selectedIndex = linkedSignal({
    source: this.products,
    computation: (products) => products.length > 0 ? 0 : -1
  });

  // User can manually change it:
  selectProduct(product: Product) {
    this.selectedProduct.set(product);  // It's writable!
  }
  // But when products() changes, it auto-resets to products()[0]
}

// Real-world: pagination that resets when filters change
export class DataGridComponent {
  filters = signal<Filters>({});
  
  // Page resets to 1 whenever filters change
  currentPage = linkedSignal({
    source: this.filters,
    computation: () => 1  // Always reset to page 1
  });
  
  nextPage() {
    this.currentPage.update(p => p + 1);  // User navigates
  }
}

7. Resource API (Angular 19)

import { resource, rxResource } from '@angular/core';

@Component({
  template: `
    @if (userResource.isLoading()) {
      <app-spinner />
    }
    @if (userResource.error()) {
      <p>Error: {{ userResource.error() }}</p>
    }
    @if (userResource.value(); as user) {
      <h1>{{ user.name }}</h1>
    }
  `
})
export class UserProfileComponent {
  userId = input.required<string>();

  // resource() — async data fetching tied to signals
  userResource = resource({
    request: () => ({ id: this.userId() }),  // Reactive request
    loader: async ({ request, abortSignal }) => {
      const response = await fetch(`/api/users/${request.id}`, { signal: abortSignal });
      if (!response.ok) throw new Error('Failed to load');
      return response.json() as Promise<User>;
    }
  });

  // rxResource — Observable-based (for existing Angular services)
  userResource2 = rxResource({
    request: () => this.userId(),
    loader: ({ request: id }) => this.userService.getUser(id)
  });

  // When userId() changes:
  // 1. Previous request is automatically CANCELLED (abort signal)
  // 2. isLoading() becomes true
  // 3. New request fires
  // 4. value() updates when complete

  refresh() {
    this.userResource.reload();  // Force re-fetch
  }
}

8. RxJS ↔ Signals Interop

import { toSignal, toObservable } from '@angular/core/rxjs-interop';

// Observable → Signal
export class SearchComponent {
  private route = inject(ActivatedRoute);
  
  // Convert route params observable to signal
  userId = toSignal(
    this.route.params.pipe(map(p => p['id'])),
    { initialValue: '' }
  );
  
  // Convert HTTP observable to signal
  private userService = inject(UserService);
  users = toSignal(this.userService.getAll(), { initialValue: [] });
  
  // With requireSync (when observable is guaranteed to emit synchronously)
  // e.g., BehaviorSubject
  currentUser = toSignal(this.authService.currentUser$, { requireSync: true });
}

// Signal → Observable
export class FilterComponent {
  searchTerm = signal('');
  
  // Convert signal to observable (for existing RxJS-based services)
  searchTerm$ = toObservable(this.searchTerm);
  
  results$ = this.searchTerm$.pipe(
    debounceTime(300),
    distinctUntilChanged(),
    switchMap(term => this.searchService.search(term))
  );
}

9. Migrating from Observables to Signals

// BEFORE: Observable-based component
@Component({
  template: `
    @if (user$ | async; as user) { <h1>{{ user.name }}</h1> }
    <p>Items: {{ (items$ | async)?.length }}</p>
  `
})
export class BeforeComponent implements OnInit, OnDestroy {
  private destroy$ = new Subject<void>();
  user$!: Observable<User>;
  items$!: Observable<Item[]>;

  ngOnInit() {
    this.user$ = this.userService.getUser(this.userId);
    this.items$ = this.itemService.getItems().pipe(takeUntil(this.destroy$));
  }
  ngOnDestroy() { this.destroy$.next(); this.destroy$.complete(); }
}

// AFTER: Signal-based component
@Component({
  template: `
    @if (user(); as user) { <h1>{{ user.name }}</h1> }
    <p>Items: {{ items().length }}</p>
  `
})
export class AfterComponent {
  userId = input.required<string>();
  
  userResource = rxResource({
    request: () => this.userId(),
    loader: ({ request: id }) => this.userService.getUser(id)
  });
  
  user = computed(() => this.userResource.value());
  items = toSignal(this.itemService.getItems(), { initialValue: [] });
  // No OnDestroy! No async pipe! No subscription management!
}
