Challenge 1: Autocomplete Search Component

MEDIUM — 30 minutes

Requirements:

  • Input field with debounced API search
  • Show dropdown with results
  • Keyboard navigation (arrow keys + Enter)
  • Click outside to close
  • Loading & empty states
  • Cancel previous requests on new input
@Component({
  selector: 'app-autocomplete',
  standalone: true,
  imports: [ReactiveFormsModule],
  template: `
    <div class="autocomplete" (clickOutside)="close()">
      <input 
        [formControl]="searchControl"
        (keydown)="onKeyDown($event)"
        (focus)="open()"
        placeholder="Search..."
        [attr.aria-expanded]="isOpen()"
        role="combobox"
      >
      
      @if (isOpen()) {
        <ul class="dropdown" role="listbox">
          @if (loading()) {
            <li class="loading">Searching...</li>
          } @else if (results().length === 0) {
            <li class="empty">No results found</li>
          } @else {
            @for (item of results(); track item.id; let i = $index) {
              <li 
                [class.active]="i === activeIndex()"
                (click)="select(item)"
                (mouseenter)="activeIndex.set(i)"
                role="option"
              >
                {{ item.label }}
              </li>
            }
          }
        </ul>
      }
    </div>
  `
})
export class AutocompleteComponent {
  searchControl = new FormControl('');
  results = signal<SearchResult[]>([]);
  loading = signal(false);
  isOpen = signal(false);
  activeIndex = signal(-1);

  selected = output<SearchResult>();

  private searchService = inject(SearchService);
  private destroyRef = inject(DestroyRef);

  constructor() {
    this.searchControl.valueChanges.pipe(
      debounceTime(300),
      distinctUntilChanged(),
      tap(() => { this.loading.set(true); this.activeIndex.set(-1); }),
      switchMap(term => {
        if (!term || term.length < 2) return of([]);
        return this.searchService.search(term).pipe(
          catchError(() => of([]))
        );
      }),
      takeUntilDestroyed(this.destroyRef)
    ).subscribe(results => {
      this.results.set(results);
      this.loading.set(false);
      this.isOpen.set(true);
    });
  }

  onKeyDown(event: KeyboardEvent): void {
    const len = this.results().length;
    switch (event.key) {
      case 'ArrowDown':
        event.preventDefault();
        this.activeIndex.update(i => (i + 1) % len);
        break;
      case 'ArrowUp':
        event.preventDefault();
        this.activeIndex.update(i => (i - 1 + len) % len);
        break;
      case 'Enter':
        if (this.activeIndex() >= 0) {
          this.select(this.results()[this.activeIndex()]);
        }
        break;
      case 'Escape':
        this.close();
        break;
    }
  }

  select(item: SearchResult): void {
    this.searchControl.setValue(item.label, { emitEvent: false });
    this.selected.emit(item);
    this.close();
  }

  open() { this.isOpen.set(true); }
  close() { this.isOpen.set(false); this.activeIndex.set(-1); }
}

Challenge 2: Data Table with Sort, Filter, Paginate

HARD — 45 minutes

Requirements:

  • Column sorting (asc/desc toggle)
  • Global text filter
  • Client-side pagination
  • Configurable columns
  • Row selection (single/multi)
  • OnPush change detection
interface Column<T> {
  key: keyof T;
  label: string;
  sortable?: boolean;
  template?: TemplateRef<any>;
}

@Component({
  selector: 'app-data-table',
  standalone: true,
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    <div class="table-controls">
      <input placeholder="Search..." (input)="filterTerm.set($any($event.target).value)">
      <span>{{ paginatedData().length }} of {{ filteredData().length }} items</span>
    </div>
    
    <table>
      <thead>
        <tr>
          @if (selectable()) {
            <th><input type="checkbox" (change)="toggleAll($event)" [checked]="allSelected()"></th>
          }
          @for (col of columns(); track col.key) {
            <th (click)="col.sortable !== false && toggleSort(col.key)"
                [class.sortable]="col.sortable !== false">
              {{ col.label }}
              @if (sortKey() === col.key) {
                <span>{{ sortDirection() === 'asc' ? '▲' : '▼' }}</span>
              }
            </th>
          }
        </tr>
      </thead>
      <tbody>
        @for (row of paginatedData(); track trackFn()(row)) {
          <tr [class.selected]="isSelected(row)">
            @if (selectable()) {
              <td><input type="checkbox" [checked]="isSelected(row)" (change)="toggleRow(row)"></td>
            }
            @for (col of columns(); track col.key) {
              <td>{{ row[col.key] }}</td>
            }
          </tr>
        }
      </tbody>
    </table>
    
    <div class="pagination">
      <button (click)="page.update(p => p - 1)" [disabled]="page() === 1">Prev</button>
      <span>Page {{ page() }} of {{ totalPages() }}</span>
      <button (click)="page.update(p => p + 1)" [disabled]="page() === totalPages()">Next</button>
    </div>
  `
})
export class DataTableComponent<T extends Record<string, any>> {
  data = input.required<T[]>();
  columns = input.required<Column<T>[]>();
  pageSize = input(10);
  selectable = input(false);
  trackFn = input<(item: T) => any>(() => (item: any) => item.id);

  // State
  filterTerm = signal('');
  sortKey = signal<keyof T | null>(null);
  sortDirection = signal<'asc' | 'desc'>('asc');
  page = signal(1);
  selectedRows = signal<Set<T>>(new Set());

  // Derived
  filteredData = computed(() => {
    let result = this.data();
    const term = this.filterTerm().toLowerCase();
    if (term) {
      result = result.filter(row =>
        Object.values(row).some(val => 
          String(val).toLowerCase().includes(term)
        )
      );
    }
    const key = this.sortKey();
    if (key) {
      const dir = this.sortDirection() === 'asc' ? 1 : -1;
      result = [...result].sort((a, b) => {
        if (a[key] < b[key]) return -1 * dir;
        if (a[key] > b[key]) return 1 * dir;
        return 0;
      });
    }
    return result;
  });

  totalPages = computed(() => Math.ceil(this.filteredData().length / this.pageSize()));

  paginatedData = computed(() => {
    const start = (this.page() - 1) * this.pageSize();
    return this.filteredData().slice(start, start + this.pageSize());
  });

  allSelected = computed(() => {
    const current = this.paginatedData();
    return current.length > 0 && current.every(r => this.selectedRows().has(r));
  });

  toggleSort(key: keyof T): void {
    if (this.sortKey() === key) {
      this.sortDirection.update(d => d === 'asc' ? 'desc' : 'asc');
    } else {
      this.sortKey.set(key);
      this.sortDirection.set('asc');
    }
    this.page.set(1);
  }

  toggleRow(row: T): void {
    this.selectedRows.update(set => {
      const newSet = new Set(set);
      newSet.has(row) ? newSet.delete(row) : newSet.add(row);
      return newSet;
    });
  }

  toggleAll(event: Event): void {
    const checked = (event.target as HTMLInputElement).checked;
    this.selectedRows.update(set => {
      const newSet = new Set(set);
      this.paginatedData().forEach(row => checked ? newSet.add(row) : newSet.delete(row));
      return newSet;
    });
  }

  isSelected(row: T): boolean {
    return this.selectedRows().has(row);
  }
}

Challenge 3: Multi-Step Form Wizard

HARD — 45 minutes

@Component({
  selector: 'app-form-wizard',
  template: `
    <!-- Progress bar -->
    <div class="steps">
      @for (step of steps(); track $index; let i = $index) {
        <div [class.active]="i === currentStep()" [class.completed]="i < currentStep()">
          {{ i + 1 }}. {{ step.title }}
        </div>
      }
    </div>

    <!-- Current step content -->
    <form [formGroup]="currentFormGroup()">
      @switch (currentStep()) {
        @case (0) {
          <h3>Personal Info</h3>
          <input formControlName="firstName" placeholder="First Name">
          <input formControlName="lastName" placeholder="Last Name">
          <input formControlName="email" type="email" placeholder="Email">
        }
        @case (1) {
          <h3>Address</h3>
          <input formControlName="street" placeholder="Street">
          <input formControlName="city" placeholder="City">
          <input formControlName="zip" placeholder="ZIP">
        }
        @case (2) {
          <h3>Review & Submit</h3>
          <pre>{{ formData() | json }}</pre>
        }
      }
    </form>

    <!-- Navigation -->
    <div class="actions">
      <button (click)="prev()" [disabled]="currentStep() === 0">Back</button>
      @if (currentStep() < steps().length - 1) {
        <button (click)="next()" [disabled]="currentFormGroup().invalid">Next</button>
      } @else {
        <button (click)="submit()" [disabled]="!allValid()">Submit</button>
      }
    </div>
  `
})
export class FormWizardComponent {
  private fb = inject(FormBuilder);
  
  currentStep = signal(0);

  steps = signal([
    { title: 'Personal', form: this.fb.group({
      firstName: ['', Validators.required],
      lastName: ['', Validators.required],
      email: ['', [Validators.required, Validators.email]]
    })},
    { title: 'Address', form: this.fb.group({
      street: ['', Validators.required],
      city: ['', Validators.required],
      zip: ['', [Validators.required, Validators.pattern(/^\d{5}$/)]]
    })},
    { title: 'Review', form: this.fb.group({}) }
  ]);

  currentFormGroup = computed(() => this.steps()[this.currentStep()].form);

  formData = computed(() => {
    const data: any = {};
    this.steps().forEach(step => Object.assign(data, step.form.value));
    return data;
  });

  allValid = computed(() => 
    this.steps().every(step => step.form.valid)
  );

  next(): void {
    if (this.currentFormGroup().valid) {
      this.currentStep.update(s => s + 1);
    } else {
      this.currentFormGroup().markAllAsTouched();
    }
  }

  prev(): void {
    this.currentStep.update(s => s - 1);
  }

  submit(): void {
    if (this.allValid()) {
      console.log('Submitting:', this.formData());
    }
  }
}
