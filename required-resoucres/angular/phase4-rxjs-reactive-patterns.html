1. Observables — The Foundation

Observable vs Promise

FeaturePromiseObservable
ValuesSingle valueMultiple values over time
ExecutionEager (executes immediately)Lazy (only when subscribed)
CancellationNot cancellableCancellable (unsubscribe)
Operators.then(), .catch(), .finally()100+ pipeable operators
MulticastAlways sharedUnicast by default (each subscriber gets own execution)

Creating Observables

import { Observable } from 'rxjs';

// Custom Observable
const custom$ = new Observable<number>(subscriber => {
  subscriber.next(1);
  subscriber.next(2);
  subscriber.next(3);

  // Async
  setTimeout(() => {
    subscriber.next(4);
    subscriber.complete();  // Signal completion
  }, 1000);

  // Teardown logic (cleanup when unsubscribed)
  return () => {
    console.log('Unsubscribed! Cleanup here.');
  };
});

// Subscribe
const subscription = custom$.subscribe({
  next: value => console.log('Value:', value),
  error: err => console.error('Error:', err),
  complete: () => console.log('Done!')
});

// Unsubscribe (triggers teardown)
subscription.unsubscribe();

Hot vs Cold Observables

// COLD Observable — creates new execution for each subscriber
// Like a YouTube video — each viewer starts from the beginning
const cold$ = new Observable(sub => {
  sub.next(Math.random());  // Each subscriber gets different random number
});

cold$.subscribe(v => console.log('A:', v));  // A: 0.123
cold$.subscribe(v => console.log('B:', v));  // B: 0.789 (different!)

// HOT Observable — shares execution among all subscribers
// Like a live stream — everyone sees the same thing
const subject = new Subject<number>();
subject.subscribe(v => console.log('A:', v));
subject.subscribe(v => console.log('B:', v));
subject.next(42);  // A: 42, B: 42 (same!)

2. Subjects — All Types Explained

Subject TypeBehaviorUse Case
SubjectNo initial value, no replayEvent bus, actions
BehaviorSubjectRequires initial value, replays LAST value to new subscribersCurrent state, "current user"
ReplaySubjectReplays N values to new subscribersCache last N events
AsyncSubjectOnly emits LAST value, only on completeHTTP-like (one response)
// BehaviorSubject — MOST USED in Angular services
@Injectable({ providedIn: 'root' })
export class AuthService {
  // Must have initial value (null = not logged in)
  private currentUserSubject = new BehaviorSubject<User | null>(null);

  // Public observable (consumers can't call .next())
  currentUser$ = this.currentUserSubject.asObservable();

  // Synchronous access to current value
  get currentUser(): User | null {
    return this.currentUserSubject.getValue();
  }

  login(credentials: LoginCredentials): Observable<User> {
    return this.http.post<User>('/api/login', credentials).pipe(
      tap(user => this.currentUserSubject.next(user))
    );
  }

  logout(): void {
    this.currentUserSubject.next(null);
  }
}

// ReplaySubject — replay last N emissions
const replay$ = new ReplaySubject<string>(3);  // Buffer size = 3
replay$.next('A');
replay$.next('B');
replay$.next('C');
replay$.next('D');

replay$.subscribe(v => console.log(v));
// Outputs: B, C, D (last 3 values replayed)

// ReplaySubject with time window
const timed$ = new ReplaySubject<string>(Infinity, 5000);  // Replay all within last 5 seconds

3. Creation Operators

import { of, from, interval, timer, fromEvent, EMPTY, NEVER, throwError, defer } from 'rxjs';

// of — emit fixed values synchronously
of(1, 2, 3).subscribe(v => console.log(v));  // 1, 2, 3, complete

// from — convert array/promise/iterable to observable
from([10, 20, 30]).subscribe();
from(fetch('/api/data')).subscribe();  // Promise → Observable
from('hello').subscribe(v => console.log(v));  // h, e, l, l, o

// interval — emit incrementing numbers at interval
interval(1000).subscribe(n => console.log(n));  // 0, 1, 2, ... every second

// timer — wait, then emit (or start interval after delay)
timer(3000).subscribe(() => console.log('3 seconds passed'));
timer(2000, 1000).subscribe(n => console.log(n));  // Wait 2s, then emit every 1s

// fromEvent — DOM events as observable
fromEvent(document, 'click').subscribe(event => console.log(event));
fromEvent(inputElement, 'input').pipe(
  debounceTime(300),
  map((e: Event) => (e.target as HTMLInputElement).value)
).subscribe(value => this.search(value));

// defer — create observable lazily (new observable per subscriber)
const deferred$ = defer(() => {
  const random = Math.random();
  return of(random);  // Each subscriber gets fresh random
});

// EMPTY — completes immediately with no values
// NEVER — never emits, never completes
// throwError — immediately errors
const error$ = throwError(() => new Error('Something went wrong'));

4. Transformation Operators

import { map, pluck, scan, reduce, buffer, bufferTime, toArray, groupBy } from 'rxjs/operators';

// map — transform each value
source$.pipe(
  map(user => user.name.toUpperCase())
);

// scan — running accumulator (like reduce but emits each step)
clicks$.pipe(
  scan((count, click) => count + 1, 0)  // Running click counter: 1, 2, 3...
);

// reduce — accumulate all values, emit on complete
of(1, 2, 3, 4, 5).pipe(
  reduce((sum, val) => sum + val, 0)
).subscribe(total => console.log(total));  // 15

// buffer / bufferTime — collect emissions into arrays
source$.pipe(
  bufferTime(2000)  // Emit array of all values received in last 2 seconds
);

// toArray — collect all emissions into single array on complete
source$.pipe(toArray()).subscribe(arr => console.log(arr));

// groupBy — split into grouped observables
users$.pipe(
  groupBy(user => user.department),
  mergeMap(group$ => group$.pipe(
    toArray(),
    map(users => ({ department: group$.key, users }))
  ))
);

5. Filtering Operators

import { filter, first, last, take, takeUntil, takeWhile, skip, 
         distinctUntilChanged, debounceTime, throttleTime, auditTime } from 'rxjs/operators';

// filter — only pass values meeting condition
source$.pipe(filter(x => x > 10));

// distinctUntilChanged — only emit when value CHANGES
source$.pipe(distinctUntilChanged());  // [1,1,2,2,3,1] → [1,2,3,1]

// With comparison function (for objects)
users$.pipe(
  distinctUntilChanged((prev, curr) => prev.id === curr.id)
);

// debounceTime — wait for silence, then emit last value
searchInput$.pipe(
  debounceTime(300),           // Wait 300ms of no typing
  distinctUntilChanged(),       // Don't search same term twice
  filter(term => term.length >= 2)  // Min 2 chars
);

// throttleTime — emit first value, then ignore for duration
scroll$.pipe(throttleTime(100));  // At most 1 scroll event per 100ms

// auditTime — like throttle but emits LAST value in window
resize$.pipe(auditTime(200));

// take — take first N values then complete
source$.pipe(take(5));  // Only first 5 emissions

// takeUntil — take values until notifier emits (CRITICAL for cleanup)
source$.pipe(takeUntil(this.destroy$));

// first / last — take first/last matching value
users$.pipe(first(u => u.role === 'admin'));  // First admin, then complete

// skip — skip first N values
source$.pipe(skip(3));  // Ignore first 3, then pass through

6. Combination Operators

import { combineLatest, forkJoin, merge, concat, zip, race, withLatestFrom } from 'rxjs';

// combineLatest — emit when ANY source emits (after all have emitted at least once)
combineLatest([filters$, page$, sort$]).pipe(
  map(([filters, page, sort]) => ({ filters, page, sort }))
).subscribe(params => this.loadData(params));

// forkJoin — wait for ALL observables to COMPLETE, emit last values as array
// Like Promise.all() for observables
forkJoin({
  users: this.http.get<User[]>('/api/users'),
  roles: this.http.get<Role[]>('/api/roles'),
  settings: this.http.get<Settings>('/api/settings')
}).subscribe(({ users, roles, settings }) => {
  // All three requests completed
});

// merge — combine multiple observables, emit as they come (interleaved)
merge(
  fromEvent(saveBtn, 'click').pipe(map(() => 'save')),
  fromEvent(deleteBtn, 'click').pipe(map(() => 'delete')),
  fromEvent(document, 'keydown').pipe(
    filter((e: KeyboardEvent) => e.ctrlKey && e.key === 's'),
    map(() => 'save')
  )
).subscribe(action => this.handleAction(action));

// concat — subscribe to observables IN ORDER (next starts when previous completes)
concat(
  this.validateForm(),
  this.uploadFile(),
  this.saveRecord()
).subscribe();

// withLatestFrom — combine with latest value from another observable
saveClicks$.pipe(
  withLatestFrom(formValue$, user$),
  map(([click, formData, user]) => ({ ...formData, updatedBy: user.id }))
).subscribe(data => this.save(data));

// zip — pair emissions by index (1st with 1st, 2nd with 2nd)
zip(request$, timer(1000)).pipe(
  map(([response, _]) => response)  // Ensure minimum 1s loading time
);

7. Higher-Order Mapping Operators

THE most important RxJS concept for interviews. These operators handle "Observable of Observables" — when an outer observable triggers an inner observable (e.g., click → HTTP request).
OperatorBehaviorUse Case
switchMapCancel previous inner, subscribe to newSearch typeahead, route params → data load
mergeMapRun all inner observables concurrentlyParallel file uploads, likes/favorites
concatMapQueue: wait for previous to completeSequential writes, ordered operations
exhaustMapIgnore new until current completesLogin button (prevent double-submit)
// switchMap — MOST USED: Cancel previous, use latest
// Perfect for: search, autocomplete, route-based data loading
this.searchControl.valueChanges.pipe(
  debounceTime(300),
  distinctUntilChanged(),
  switchMap(term => this.searchService.search(term))
  // If user types "ang" then quickly "angular":
  // Request for "ang" is CANCELLED, only "angular" request runs
).subscribe(results => this.results = results);

// Route params → load data
this.route.params.pipe(
  map(params => params['id']),
  switchMap(id => this.userService.getUser(id))
  // When URL changes from /users/1 to /users/2:
  // Request for user 1 cancelled, request for user 2 starts
).subscribe(user => this.user = user);

// mergeMap — Run ALL concurrently (order not guaranteed)
// Perfect for: parallel uploads, independent actions
this.filesToUpload$.pipe(
  mergeMap(file => this.uploadService.upload(file), 3)  // Max 3 concurrent
).subscribe(response => this.uploadedFiles.push(response));

// concatMap — Queue: execute one at a time, in ORDER
// Perfect for: sequential API calls, ordered writes
this.saveActions$.pipe(
  concatMap(action => this.api.save(action))
  // Actions queued: save1 → wait → save2 → wait → save3
).subscribe();

// exhaustMap — Ignore new emissions while current is running
// Perfect for: login button, refresh button (prevent spam)
this.loginButton$.pipe(
  exhaustMap(() => this.authService.login(this.credentials))
  // User clicks 5 times rapidly: only FIRST click triggers login
  // Subsequent clicks ignored until login request completes
).subscribe(user => this.router.navigate(['/dashboard']));

Visual Comparison

// Source emits: --A-----B-----C--|
// Inner takes 3 units: ===|

// switchMap:  --===      ===  (A cancelled by B, B cancelled by C, only C completes)
//            --A-----B--C--|  → gets C's result

// mergeMap:   --===---===---===  (all run, results interleaved)
//            --A-----B-----C--| → gets A,B,C results (any order)

// concatMap:  --===---===---===  (queued, sequential)
//            --A-----B-----C--| → gets A,B,C results (guaranteed order)

// exhaustMap: --===         ===  (B ignored because A still running)
//            --A-----B-----C--| → gets A,C results (B skipped)

8. Error Handling Strategies

import { catchError, retry, retryWhen, throwError, EMPTY, of } from 'rxjs';

// catchError — handle error, return fallback observable
this.http.get<User[]>('/api/users').pipe(
  catchError(error => {
    console.error('Failed to load users:', error);
    return of([]);  // Return empty array as fallback
  })
);

// retry — retry N times on error
this.http.get('/api/data').pipe(
  retry({ count: 3, delay: 1000 }),  // Retry 3 times with 1s delay
  catchError(err => {
    this.notification.show('Failed after 3 retries');
    return EMPTY;
  })
);

// Exponential backoff retry
this.http.get('/api/data').pipe(
  retry({
    count: 3,
    delay: (error, retryCount) => timer(Math.pow(2, retryCount) * 1000)
    // 2s, 4s, 8s delays
  })
);

// Error handling in switchMap (CRITICAL: prevent stream death)
this.searchControl.valueChanges.pipe(
  debounceTime(300),
  switchMap(term => 
    this.searchService.search(term).pipe(
      catchError(err => {
        // Handle error INSIDE switchMap so outer stream survives
        this.showError('Search failed');
        return of([]);  // Fallback value
      })
    )
  )
).subscribe(results => this.results = results);
// If catchError was OUTSIDE switchMap, one error would kill the whole stream!
Critical Rule: Always put catchError INSIDE the inner observable (inside switchMap/mergeMap) to prevent the outer stream from completing on error. If the outer stream errors, the subscription is dead — no more emissions!

9. Unsubscription Patterns

Why Unsubscribe?

Failing to unsubscribe from observables that don't complete (intervals, subjects, event listeners) causes memory leaks — the component is destroyed but the subscription keeps running, holding references to the destroyed component.

Pattern 1: takeUntilDestroyed (Angular 16+ — BEST)

import { takeUntilDestroyed } from '@angular/core/rxjs-interop';

export class MyComponent {
  private data$ = inject(DataService).getData().pipe(
    takeUntilDestroyed()  // Automatically unsubscribes when component destroyed
  );

  constructor() {
    // Must be called in injection context (constructor or field initializer)
    this.data$.subscribe(data => this.processData(data));
  }
}

Pattern 2: DestroyRef (Angular 16+)

export class MyComponent {
  private destroyRef = inject(DestroyRef);

  ngOnInit() {
    this.someService.data$.pipe(
      takeUntilDestroyed(this.destroyRef)  // Pass DestroyRef explicitly
    ).subscribe();
  }
}

Pattern 3: takeUntil + Subject (Pre-Angular 16)

export class MyComponent implements OnDestroy {
  private destroy$ = new Subject<void>();

  ngOnInit() {
    this.service.data$.pipe(
      takeUntil(this.destroy$)
    ).subscribe(data => this.data = data);

    this.anotherService.events$.pipe(
      takeUntil(this.destroy$)
    ).subscribe(event => this.handleEvent(event));
  }

  ngOnDestroy() {
    this.destroy$.next();
    this.destroy$.complete();
  }
}

Pattern 4: async pipe (BEST for templates)

// The async pipe automatically subscribes AND unsubscribes
@Component({
  template: `
    @if (user$ | async; as user) {
      <h1>{{ user.name }}</h1>
    }
    
    @for (item of items$ | async; track item.id) {
      <app-item [data]="item" />
    }
  `
})
export class MyComponent {
  user$ = this.userService.currentUser$;
  items$ = this.itemService.getItems();
  // NO manual subscription! NO manual unsubscription! 
}

When You DON'T Need to Unsubscribe

  • HTTP requests — Complete after single response (but cancel if you switchMap)
  • Finite observables — of(), from([]), take(5) — complete on their own
  • async pipe — Handles subscription lifecycle
  • Router observables — Cleaned up by Angular automatically in routed components

10. Real-World Angular RxJS Patterns

Pattern: Smart Autocomplete Search

export class SearchComponent {
  searchControl = new FormControl('');
  results$ = this.searchControl.valueChanges.pipe(
    debounceTime(300),
    distinctUntilChanged(),
    filter(term => term.length >= 2),
    switchMap(term => this.searchService.search(term).pipe(
      startWith(null),  // Show loading state
      catchError(() => of([]))
    )),
    shareReplay(1)
  );
  
  isLoading$ = this.results$.pipe(map(r => r === null));
}

Pattern: Polling with Pause/Resume

export class DashboardComponent {
  private pause$ = new BehaviorSubject<boolean>(false);

  data$ = this.pause$.pipe(
    switchMap(paused => paused ? EMPTY : timer(0, 5000)),
    switchMap(() => this.api.getData()),
    shareReplay(1)
  );

  pause() { this.pause$.next(true); }
  resume() { this.pause$.next(false); }
}

Pattern: Combine Multiple API Calls for Page Data

export class ProfilePageComponent {
  private userId$ = this.route.params.pipe(map(p => p['id']));

  vm$ = this.userId$.pipe(
    switchMap(id => forkJoin({
      user: this.userService.getUser(id),
      posts: this.postService.getUserPosts(id),
      followers: this.socialService.getFollowers(id)
    })),
    shareReplay(1)
  );
}

Pattern: Optimistic UI Update with Rollback

toggleLike(postId: string) {
  // Immediately update UI
  this.post.liked = !this.post.liked;
  this.post.likeCount += this.post.liked ? 1 : -1;

  this.api.toggleLike(postId).pipe(
    catchError(err => {
      // Rollback on failure
      this.post.liked = !this.post.liked;
      this.post.likeCount += this.post.liked ? 1 : -1;
      this.notification.show('Failed to update');
      return EMPTY;
    })
  ).subscribe();
}
