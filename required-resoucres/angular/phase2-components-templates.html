1. Data Binding — All 4 Types Explained

The Four Binding Types

TypeSyntaxDirectionExample
Interpolation{{ expression }}Component → DOM{{ user.name }}
Property Binding[property]="expr"Component → DOM[disabled]="isLoading"
Event Binding(event)="handler()"DOM → Component(click)="save()"
Two-way Binding[(ngModel)]="prop"Both directions[(ngModel)]="name"

Deep Dive: Property Binding vs Interpolation

<!-- These are equivalent for strings: -->
<img src="{{ imageUrl }}">
<img [src]="imageUrl">

<!-- BUT only property binding works for non-string values: -->
<button [disabled]="isLoading">Save</button>  ✓
<button disabled="{{ isLoading }}">Save</button>  ✗ (always disabled — string "false" is truthy!)

<!-- Property binding sets the DOM PROPERTY, not HTML attribute -->
<!-- For HTML attributes that have no DOM property: -->
<td [attr.colspan]="columnSpan"></td>
<svg><circle [attr.cx]="x" [attr.cy]="y" /></svg>

Two-Way Binding — How It Actually Works

<!-- [(ngModel)]="name" is syntactic sugar for: -->
<input [ngModel]="name" (ngModelChange)="name = $event">

<!-- Custom two-way binding in your components: -->
@Component({
  selector: 'app-counter',
  template: `
    <button (click)="decrement()">-</button>
    <span>{{ count }}</span>
    <button (click)="increment()">+</button>
  `
})
export class CounterComponent {
  @Input() count = 0;
  @Output() countChange = new EventEmitter<number>();  // Must be: inputName + "Change"

  increment() {
    this.count++;
    this.countChange.emit(this.count);
  }

  decrement() {
    this.count--;
    this.countChange.emit(this.count);
  }
}

// Parent uses:
<app-counter [(count)]="totalItems"></app-counter>

Structural Directives — The * Syntax Explained

<!-- *ngIf is syntactic sugar. These are equivalent: -->
<div *ngIf="isVisible">Content</div>

<!-- Desugared: -->
<ng-template [ngIf]="isVisible">
  <div>Content</div>
</ng-template>

<!-- *ngFor with all available variables: -->
<li *ngFor="let item of items; let i = index; let first = first; 
    let last = last; let even = even; let odd = odd; trackBy: trackById">
  {{ i }}: {{ item.name }}
</li>

<!-- *ngIf with else and then: -->
<div *ngIf="user$ | async as user; else loading">
  Welcome, {{ user.name }}!
</div>
<ng-template #loading>
  <app-spinner></app-spinner>
</ng-template>

<!-- ngSwitch: -->
<div [ngSwitch]="role">
  <span *ngSwitchCase="'admin'">Admin Panel</span>
  <span *ngSwitchCase="'user'">User Dashboard</span>
  <span *ngSwitchDefault>Guest View</span>
</div>

Custom Structural Directive

@Directive({
  selector: '[appRepeat]',
  standalone: true
})
export class RepeatDirective {
  constructor(
    private templateRef: TemplateRef<any>,
    private viewContainer: ViewContainerRef
  ) {}

  @Input() set appRepeat(times: number) {
    this.viewContainer.clear();
    for (let i = 0; i < times; i++) {
      this.viewContainer.createEmbeddedView(this.templateRef, {
        $implicit: i,    // Available as "let item" 
        index: i         // Available as "let i = index"
      });
    }
  }
}

// Usage:
<div *appRepeat="5; let i">Item {{ i }}</div>

Custom Attribute Directive

@Directive({
  selector: '[appHighlight]',
  standalone: true
})
export class HighlightDirective {
  @Input() appHighlight = 'yellow';
  @Input() appHighlightOpacity = 0.3;

  @HostBinding('style.backgroundColor') bgColor = '';

  @HostListener('mouseenter')
  onMouseEnter() {
    this.bgColor = this.appHighlight || 'yellow';
  }

  @HostListener('mouseleave')
  onMouseLeave() {
    this.bgColor = '';
  }
}

// Usage:
<p [appHighlight]="'lightblue'">Hover me!</p>

3. New Control Flow Syntax (@if, @for, @switch)

Angular 17+ Feature: The new built-in control flow replaces structural directives with better performance (no need for CommonModule import) and cleaner syntax. 
<!-- @if (replaces *ngIf) -->
@if (user) {
  <h1>Welcome, {{ user.name }}!</h1>
} @else if (isLoading) {
  <app-spinner />
} @else {
  <p>Please log in</p>
}

<!-- @for (replaces *ngFor) — REQUIRES track expression -->
@for (item of items; track item.id) {
  <app-card [data]="item" />
} @empty {
  <p>No items found.</p>
}

<!-- @for with index and other context variables -->
@for (user of users; track user.id; let i = $index; let isFirst = $first; let isLast = $last) {
  <div [class.first]="isFirst" [class.last]="isLast">
    {{ i + 1 }}. {{ user.name }}
  </div>
}

<!-- @switch (replaces ngSwitch) -->
@switch (status) {
  @case ('loading') {
    <app-spinner />
  }
  @case ('error') {
    <app-error [message]="errorMsg" />
  }
  @case ('success') {
    <app-content [data]="data" />
  }
  @default {
    <p>Unknown state</p>
  }
}

Why @for requires `track`

Unlike *ngFor where trackBy was optional (defaulting to object identity), @for REQUIRES a track expression. This ensures Angular can efficiently reconcile the DOM when the list changes — preventing unnecessary re-creation of DOM elements.

// What track does internally:
// Angular uses the track value to map each item to a DOM node.
// When the array changes, it compares track values to decide:
// - Same track value → reuse DOM node (update bindings only)
// - New track value → create new DOM node
// - Missing track value → destroy DOM node

// Good: unique identifier
@for (user of users; track user.id) { ... }

// OK for static lists:
@for (item of items; track $index) { ... }

// BAD: object reference (defeats the purpose)
@for (item of items; track item) { ... }

4. Pipes — Built-in & Custom

Built-in Pipes

<!-- Date -->
{{ today | date:'short' }}          <!-- 6/11/26, 10:30 AM -->
{{ today | date:'fullDate' }}       <!-- Thursday, June 11, 2026 -->
{{ today | date:'yyyy-MM-dd' }}     <!-- 2026-06-11 -->

<!-- Number -->
{{ 3.14159 | number:'1.2-2' }}      <!-- 3.14 (min 1 int digit, 2-2 decimal) -->
{{ 1234567 | number }}              <!-- 1,234,567 -->

<!-- Currency -->
{{ 99.99 | currency }}              <!-- $99.99 -->
{{ 99.99 | currency:'EUR':'symbol' }} <!-- €99.99 -->

<!-- Percent -->
{{ 0.85 | percent }}                <!-- 85% -->

<!-- String -->
{{ 'hello world' | uppercase }}     <!-- HELLO WORLD -->
{{ 'HELLO' | lowercase }}           <!-- hello -->
{{ 'hello world' | titlecase }}     <!-- Hello World -->

<!-- JSON (debugging) -->
<pre>{{ object | json }}</pre>

<!-- Slice (arrays & strings) -->
{{ [1,2,3,4,5] | slice:1:3 }}       <!-- [2,3] -->

<!-- KeyValue (iterate objects) -->
@for (entry of myObj | keyvalue; track entry.key) {
  {{ entry.key }}: {{ entry.value }}
}

<!-- Async Pipe (CRITICAL for observables) -->
{{ user$ | async }}
@if (data$ | async; as data) {
  {{ data.title }}
}

Custom Pipe

@Pipe({
  name: 'truncate',
  standalone: true
})
export class TruncatePipe implements PipeTransform {
  transform(value: string, limit = 50, trail = '...'): string {
    if (!value) return '';
    return value.length > limit 
      ? value.substring(0, limit) + trail 
      : value;
  }
}

// Usage:
{{ longText | truncate:100:'…' }}

Pure vs Impure Pipes

// PURE pipe (default) — only recalculates when INPUT REFERENCE changes
@Pipe({ name: 'filter', pure: true })  // Won't detect array mutations (push/splice)
export class FilterPipe implements PipeTransform {
  transform(items: any[], field: string, value: string): any[] {
    return items.filter(item => item[field] === value);
  }
}

// IMPURE pipe — recalculates on EVERY change detection cycle
@Pipe({ name: 'filter', pure: false })  // Detects mutations but EXPENSIVE
export class FilterPipe implements PipeTransform { ... }

// Better approach: use a new array reference instead of impure pipe
this.items = [...this.items, newItem];  // Triggers pure pipe
Performance Warning: Impure pipes run on EVERY change detection cycle (could be hundreds of times per second). Prefer pure pipes + immutable data patterns. If you must filter/sort in templates, memoize with signals or computed properties.

5. Component Communication Patterns

Overview of All Patterns

PatternDirectionWhen to Use
@Input / @OutputParent ↔ ChildDirect parent-child relationship
Service + SubjectAny ↔ AnyUnrelated components, cross-feature communication
ViewChild / ContentChildParent → ChildParent needs to call child methods directly
Template VariablesTemplate-scopedSibling access within same template
SignalsAny ↔ AnyModern reactive state (Angular 16+)
NgRx / StoreGlobalComplex app-wide state management

Pattern 1: @Input / @Output

// Parent component
@Component({
  template: `
    <app-child 
      [items]="parentItems"
      [config]="{ showHeader: true, pageSize: 10 }"
      (itemSelected)="onItemSelected($event)"
      (itemDeleted)="onItemDeleted($event)"
    />
  `
})
export class ParentComponent {
  parentItems: Item[] = [];

  onItemSelected(item: Item): void {
    console.log('Selected:', item);
  }

  onItemDeleted(id: string): void {
    this.parentItems = this.parentItems.filter(i => i.id !== id);
  }
}

// Child component
@Component({
  selector: 'app-child',
  template: `
    @for (item of items; track item.id) {
      <div (click)="select(item)">
        {{ item.name }}
        <button (click)="delete(item.id); $event.stopPropagation()">×</button>
      </div>
    }
  `
})
export class ChildComponent {
  @Input({ required: true }) items!: Item[];
  @Input() config: ListConfig = { showHeader: false, pageSize: 20 };

  @Output() itemSelected = new EventEmitter<Item>();
  @Output() itemDeleted = new EventEmitter<string>();

  select(item: Item) { this.itemSelected.emit(item); }
  delete(id: string) { this.itemDeleted.emit(id); }
}

Pattern 2: Service with Subject (Sibling/Unrelated Communication)

// Shared service
@Injectable({ providedIn: 'root' })
export class NotificationService {
  private notificationSubject = new Subject<Notification>();
  notifications$ = this.notificationSubject.asObservable();

  show(message: string, type: 'success' | 'error' | 'info'): void {
    this.notificationSubject.next({ message, type, id: Date.now() });
  }
}

// Component A — sends notification
export class OrderComponent {
  private notify = inject(NotificationService);
  
  placeOrder() {
    // ... order logic
    this.notify.show('Order placed successfully!', 'success');
  }
}

// Component B — displays notification (completely unrelated)
export class ToastComponent {
  private notify = inject(NotificationService);
  notifications$ = this.notify.notifications$;
}

6. Content Projection (ng-content)

Single-slot Projection

// card.component.ts
@Component({
  selector: 'app-card',
  template: `
    <div class="card">
      <div class="card-header">{{ title }}</div>
      <div class="card-body">
        <ng-content></ng-content>  <!-- Projected content goes here -->
      </div>
    </div>
  `
})
export class CardComponent {
  @Input() title = '';
}

// Usage:
<app-card title="User Profile">
  <p>This content is PROJECTED into the card body</p>
  <app-user-avatar [user]="user" />
</app-card>

Multi-slot Projection (with select)

// modal.component.ts
@Component({
  selector: 'app-modal',
  template: `
    <div class="modal-overlay" (click)="close()">
      <div class="modal" (click)="$event.stopPropagation()">
        <header>
          <ng-content select="[modal-header]"></ng-content>
        </header>
        <main>
          <ng-content select="[modal-body]"></ng-content>
        </main>
        <footer>
          <ng-content select="[modal-footer]"></ng-content>
        </footer>
        <!-- Catch-all for unmatched content -->
        <ng-content></ng-content>
      </div>
    </div>
  `
})
export class ModalComponent {
  @Output() closed = new EventEmitter<void>();
  close() { this.closed.emit(); }
}

// Usage:
<app-modal (closed)="showModal = false">
  <h2 modal-header>Confirm Delete</h2>
  <p modal-body>Are you sure you want to delete this item?</p>
  <div modal-footer>
    <button (click)="showModal = false">Cancel</button>
    <button (click)="confirmDelete()">Delete</button>
  </div>
</app-modal>

Conditional Content Projection with ng-template

// data-table.component.ts
@Component({
  selector: 'app-data-table',
  template: `
    <table>
      <thead>
        <tr>
          @for (col of columns; track col.key) {
            <th>{{ col.label }}</th>
          }
        </tr>
      </thead>
      <tbody>
        @for (row of data; track row.id) {
          <tr>
            @for (col of columns; track col.key) {
              <td>
                <!-- Use custom template if provided, else default -->
                @if (cellTemplates[col.key]) {
                  <ng-container 
                    [ngTemplateOutlet]="cellTemplates[col.key]"
                    [ngTemplateOutletContext]="{ $implicit: row, column: col }"
                  />
                } @else {
                  {{ row[col.key] }}
                }
              </td>
            }
          </tr>
        }
      </tbody>
    </table>
  `
})
export class DataTableComponent {
  @Input() columns: Column[] = [];
  @Input() data: any[] = [];
  @ContentChildren(CellTemplateDirective) templates!: QueryList<CellTemplateDirective>;

  cellTemplates: Record<string, TemplateRef<any>> = {};

  ngAfterContentInit() {
    this.templates.forEach(t => {
      this.cellTemplates[t.columnKey] = t.templateRef;
    });
  }
}

7. ViewChild, ViewChildren, ContentChild

@Component({
  template: `
    <input #searchInput>
    <app-chart #chart [data]="chartData" />
    <app-item *ngFor="let item of items" [data]="item" />
  `
})
export class DashboardComponent implements AfterViewInit {
  // Single element/component reference
  @ViewChild('searchInput') searchInput!: ElementRef<HTMLInputElement>;
  @ViewChild('chart') chartComponent!: ChartComponent;

  // Multiple references (QueryList)
  @ViewChildren(ItemComponent) itemComponents!: QueryList<ItemComponent>;

  ngAfterViewInit() {
    // Access DOM element
    this.searchInput.nativeElement.focus();

    // Call child component method
    this.chartComponent.refresh();

    // Iterate children
    this.itemComponents.forEach(item => item.highlight());

    // React to dynamic changes in children
    this.itemComponents.changes.subscribe((items: QueryList<ItemComponent>) => {
      console.log('Items changed, new count:', items.length);
    });
  }
}

// ContentChild — access projected content
@Component({
  selector: 'app-tab-group',
  template: `<ng-content></ng-content>`
})
export class TabGroupComponent implements AfterContentInit {
  @ContentChildren(TabComponent) tabs!: QueryList<TabComponent>;

  ngAfterContentInit() {
    // Access projected tab components
    const firstTab = this.tabs.first;
    firstTab.activate();
  }
}

8. Dynamic Components

Modern Approach (Angular 13+)

@Component({
  template: `
    <ng-container #dynamicHost></ng-container>
    <button (click)="loadChart()">Load Chart</button>
    <button (click)="loadTable()">Load Table</button>
  `
})
export class DynamicLoaderComponent {
  @ViewChild('dynamicHost', { read: ViewContainerRef }) container!: ViewContainerRef;

  loadChart() {
    this.container.clear();
    const componentRef = this.container.createComponent(ChartComponent);
    componentRef.instance.data = this.chartData;  // Set inputs
    componentRef.instance.clicked.subscribe(e => this.handleClick(e));  // Subscribe to outputs
  }

  loadTable() {
    this.container.clear();
    const componentRef = this.container.createComponent(DataTableComponent);
    componentRef.instance.columns = this.columns;
    componentRef.instance.rows = this.rows;
  }
}

// Lazy-loaded dynamic component
async loadLazyComponent() {
  const { HeavyChartComponent } = await import('./heavy-chart/heavy-chart.component');
  this.container.clear();
  this.container.createComponent(HeavyChartComponent);
}

9. ng-template, ng-container, ng-content

ng-template — Invisible template (not rendered until used)

<!-- Define a reusable template -->
<ng-template #userCard let-user let-showAvatar="showAvatar">
  <div class="card">
    @if (showAvatar) {
      <img [src]="user.avatar">
    }
    <h3>{{ user.name }}</h3>
  </div>
</ng-template>

<!-- Render it with context -->
<ng-container 
  [ngTemplateOutlet]="userCard"
  [ngTemplateOutletContext]="{ $implicit: currentUser, showAvatar: true }"
/>

ng-container — Grouping without extra DOM element

<!-- Problem: you need *ngIf and *ngFor on same element (can't!) -->

<!-- Solution: ng-container adds no DOM element -->
<ng-container *ngIf="showList">
  <li *ngFor="let item of items">{{ item.name }}</li>
</ng-container>

<!-- With new control flow, less needed: -->
@if (showList) {
  @for (item of items; track item.id) {
    <li>{{ item.name }}</li>
  }
}

10. HostBinding & HostListener

@Directive({
  selector: '[appDropdown]',
  standalone: true
})
export class DropdownDirective {
  @HostBinding('class.open') isOpen = false;
  @HostBinding('attr.aria-expanded') get ariaExpanded() { return this.isOpen; }

  @HostListener('click') toggle() {
    this.isOpen = !this.isOpen;
  }

  @HostListener('document:click', ['$event.target'])
  onDocumentClick(target: HTMLElement) {
    // Close when clicking outside
    if (!this.elementRef.nativeElement.contains(target)) {
      this.isOpen = false;
    }
  }

  constructor(private elementRef: ElementRef) {}
}

// Modern alternative using host in @Component/@Directive metadata:
@Directive({
  selector: '[appTooltip]',
  standalone: true,
  host: {
    '(mouseenter)': 'show()',
    '(mouseleave)': 'hide()',
    '[class.tooltip-active]': 'isVisible',
    'role': 'tooltip'
  }
})
export class TooltipDirective {
  isVisible = false;
  show() { this.isVisible = true; }
  hide() { this.isVisible = false; }
}
