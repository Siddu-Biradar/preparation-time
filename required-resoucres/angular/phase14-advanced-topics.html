1. Angular Animations — Complete Guide

Angular's animation system (@angular/animations) builds on the Web Animations API. It allows declaring complex state-based transitions that Angular manages during component lifecycle and route changes.

Core Concepts

ConceptPurposeFunction
TriggerNames the animation, attached to element via @triggerNametrigger()
StateDefines styles for a named statestate()
TransitionDefines animation between statestransition()
AnimateSpecifies duration, delay, easinganimate()
KeyframesMulti-step animationskeyframes()
GroupRun animations in parallelgroup()
SequenceRun animations sequentiallysequence()
QueryTarget child elementsquery()
StaggerDelay between queried element animationsstagger()

Basic State-based Animation

import { trigger, state, transition, animate, style } from '@angular/animations';

@Component({
  selector: 'app-toggle-panel',
  template: `
    <button (click)="isOpen = !isOpen">Toggle</button>
    <div [@expandCollapse]="isOpen ? 'expanded' : 'collapsed'">
      Panel content here
    </div>
  `,
  animations: [
    trigger('expandCollapse', [
      state('collapsed', style({
        height: '0px',
        opacity: 0,
        overflow: 'hidden'
      })),
      state('expanded', style({
        height: '*',        // '*' = auto (natural height)
        opacity: 1,
        overflow: 'visible'
      })),
      transition('collapsed => expanded', [
        animate('300ms ease-out')
      ]),
      transition('expanded => collapsed', [
        animate('250ms ease-in')
      ]),
    ])
  ]
})
export class TogglePanelComponent {
  isOpen = false;
}

Overlay — Building a Custom Dropdown

import { Overlay, OverlayRef, OverlayConfig } from '@angular/cdk/overlay';
import { ComponentPortal } from '@angular/cdk/portal';

@Directive({ selector: '[appDropdown]', standalone: true })
export class DropdownDirective {
  private overlay = inject(Overlay);
  private elementRef = inject(ElementRef);
  private overlayRef: OverlayRef | null = null;

  @HostListener('click')
  toggle() {
    this.overlayRef ? this.close() : this.open();
  }

  private open() {
    const config: OverlayConfig = {
      hasBackdrop: true,
      backdropClass: 'cdk-overlay-transparent-backdrop',
      positionStrategy: this.overlay
        .position()
        .flexibleConnectedTo(this.elementRef)
        .withPositions([
          { originX: 'start', originY: 'bottom', overlayX: 'start', overlayY: 'top' },
          { originX: 'start', originY: 'top', overlayX: 'start', overlayY: 'bottom' }, // fallback
        ]),
      scrollStrategy: this.overlay.scrollStrategies.reposition(),
    };

    this.overlayRef = this.overlay.create(config);
    const portal = new ComponentPortal(DropdownMenuComponent);
    this.overlayRef.attach(portal);

    this.overlayRef.backdropClick().subscribe(() => this.close());
  }

  private close() {
    this.overlayRef?.dispose();
    this.overlayRef = null;
  }
}

4. Internationalization (i18n)

Angular's Built-in i18n (Compile-time)

Angular's native i18n is compile-time: it generates a separate build per locale. Pros: zero runtime cost, full AOT optimization. Cons: separate deployments per language.

// Step 1: Mark text for translation
<h1 i18n="site header|Main page greeting">Welcome to our app</h1>
<p i18n="@@introMessage">This app helps you learn Angular.</p>

// With ICU expressions for plurals/select
<span i18n>{count, plural,
  =0 {No items}
  =1 {One item}
  other {{{count}} items}
}</span>

<span i18n>{gender, select,
  male {He}
  female {She}
  other {They}
} logged in.</span>

// Attributes
<img [src]="logo" i18n-alt alt="Company logo" />

// In TypeScript
import { $localize } from '@angular/localize';
const msg = $localize`Hello ${name}:name:, welcome!`;
// Step 2: Extract messages
ng extract-i18n --output-path src/locale --format xlf2

// Generates messages.xlf → translate to messages.fr.xlf, messages.de.xlf

// Step 3: Build per locale (angular.json)
"i18n": {
  "sourceLocale": "en-US",
  "locales": {
    "fr": "src/locale/messages.fr.xlf",
    "de": "src/locale/messages.de.xlf"
  }
},
"architect": {
  "build": {
    "options": { "localize": true }  // builds ALL locales
  }
}

import { SwUpdate, VersionReadyEvent } from '@angular/service-worker';

@Component({ ... })
export class AppComponent {
  private swUpdate = inject(SwUpdate);

  constructor() {
    if (this.swUpdate.isEnabled) {
      // Check for updates periodically
      interval(6 * 60 * 60 * 1000).subscribe(() => this.swUpdate.checkForUpdate());

      // Notify user when update is available
      this.swUpdate.versionUpdates

          if (confirm('New version available. Update now?')) {
            window.location.reload();
          }
        });
    }
  }
}

6. HttpClient Advanced Patterns

Request Caching with Interceptor

// cache.interceptor.ts
const cache = new Map<string, { response: HttpResponse<unknown>; expiry: number }>();

export const cacheInterceptor: HttpInterceptorFn = (req, next) => {
  // Only cache GET requests
  if (req.method !== 'GET') {
    return next(req);
  }

  const cached = cache.get(req.urlWithParams);
  if (cached && cached.expiry > Date.now()) {
    return of(cached.response.clone());
  }

  return next(req).pipe(
    filter(event => event instanceof HttpResponse),
    tap(response => {
      if (response instanceof HttpResponse && response.ok) {
        cache.set(req.urlWithParams, {
          response: response.clone(),
          expiry: Date.now() + 5 * 60 * 1000 // 5 min TTL
        });
      }
    })
  );
};

Retry with Exponential Backoff

import { retry, timer } from 'rxjs';

export const retryInterceptor: HttpInterceptorFn = (req, next) => {
  return next(req).pipe(
    retry({
      count: 3,
      delay: (error, retryCount) => {
        // Only retry 5xx and network errors
        if (error.status >= 400 && error.status < 500) {
          throw error; // Don't retry client errors
        }
        const delayMs = Math.pow(2, retryCount) * 1000; // 2s, 4s, 8s
        console.log(`Retry #${retryCount} in ${delayMs}ms`);
        return timer(delayMs);
      }
    })
  );
};

Transfer State (SSR → Client Hydration)

// Avoid duplicate HTTP requests during SSR hydration
// Angular 17+ does this automatically with provideClientHydration(withHttpTransferCacheOptions(...))

import { provideClientHydration, withHttpTransferCacheOptions } from '@angular/platform-browser';

// app.config.ts
providers: [
  provideClientHydration(
    withHttpTransferCacheOptions({
      includePostRequests: true,          // Cache POST responses too
      filter: (req) => !req.url.includes('/auth'), // Skip auth endpoints
    })
  )
]

// How it works:
// 1. SSR makes HTTP requests, stores responses in TransferState (embedded in HTML)
// 2. Client hydration reads from TransferState instead of making duplicate requests
// 3. After hydration, requests go to network normally

Progress Events & Upload Tracking

upload(file: File): Observable<number | string> {
  const formData = new FormData();
  formData.append('file', file);

  return this.http.post('/api/upload', formData, {
    reportProgress: true,
    observe: 'events',
  }).pipe(
    map(event => {
      switch (event.type) {
        case HttpEventType.UploadProgress:
          return Math.round((100 * (event.loaded ?? 0)) / (event.total ?? 1));
        case HttpEventType.Response:
          return event.body as string;
        default:
          return 0;
      }
    }),
    distinctUntilChanged()
  );
}

Interceptor Ordering & Chaining

// Interceptors execute in ORDER for requests, REVERSE order for responses
provideHttpClient(
  withInterceptors([
    loggingInterceptor,    // 1st for request, last for response
    cacheInterceptor,      // 2nd for request
    authInterceptor,       // 3rd for request
    retryInterceptor,      // 4th for request, 1st for response (wraps retry around auth errors)
  ])
)

// To conditionally skip interceptors:
export const skipCache = new HttpContextToken<boolean>(() => false);

// In service:
this.http.get('/api/data', { context: new HttpContext().set(skipCache, true) });

// In interceptor:
if (req.context.get(skipCache)) return next(req);

    cy.get('[data-testid="error-message"]').should('contain', 'offline');
  });
});

E2E Best Practices

  • Use data-testid attributes for selectors (not CSS classes or text that changes)
  • Use getByRole / getByLabel in Playwright for accessible selectors
  • Mock API responses (cy.intercept / page.route) for deterministic tests
  • Use Page Object Model for reusable navigation/interaction patterns
  • Run in CI with headless browsers
  • Test critical user journeys, not individual components (use unit tests for that)

  • How do you test an Angular app with E2E? → Serve app → hit real browser → interact as user → assert DOM state + URL

10. Web Workers in Angular

When to Use Web Workers

  • CPU-intensive calculations (encryption, parsing large JSON, image processing)
  • Operations that would block the main thread and cause jank
  • Background data processing while UI stays responsive

Creating a Web Worker

// Generate:
ng generate web-worker heavy-computation

// heavy-computation.worker.ts
/// <reference lib="webworker" />
addEventListener('message', ({ data }) => {
  const result = performHeavyCalculation(data);
  postMessage(result);
});

function performHeavyCalculation(input: number[]): number {
  // Simulate expensive work
  return input.reduce((sum, n) => sum + fibonacci(n), 0);
}

function fibonacci(n: number): number {
  if (n <= 1) return n;
  return fibonacci(n - 1) + fibonacci(n - 2);
}

// component using the worker:
@Component({ ... })
export class CalculationComponent {
  result = signal<number | null>(null);
  loading = signal(false);

  calculate(data: number[]) {
    this.loading.set(true);

    if (typeof Worker !== 'undefined') {
      const worker = new Worker(new URL('./heavy-computation.worker', import.meta.url));
      worker.onmessage = ({ data: result }) => {
        this.result.set(result);
        this.loading.set(false);
        worker.terminate(); // Clean up
      };
      worker.postMessage(data);
    } else {
      // Fallback for environments without Worker support
      this.result.set(performHeavyCalculation(data));
      this.loading.set(false);
    }
  }
}

Comlink for Ergonomic Worker Communication

// Using Comlink library for cleaner API
// worker.ts
import { expose } from 'comlink';

const api = {
  async processData(items: Item[]): Promise<ProcessedItem[]> {

  async sortLargeDataset(data: any[], key: string): Promise<any[]> {

Dependency Injection — Resolution Order (Trick Question)

// Q: In which order does Angular search for a provider?

// Answer: TWO parallel hierarchies searched in order:

// 1. Element Injector Hierarchy (component tree):
//    Current component → parent component → ... → root component

// 2. Module Injector Hierarchy (only if not found above):
//    Lazy-loaded module injector → root module injector → platform injector → null injector

// The NULL injector throws "NullInjectorError: No provider for X"
// UNLESS you use @Optional() → returns null

// Resolution Modifiers change this:
// @Self()      → Only check current element injector
// @SkipSelf()  → Skip current, start from parent
// @Host()      → Stop at host component boundary
// @Optional()  → Return null instead of error

Signals — Gotchas & Edge Cases

// 1. effect() doesn't track signals called conditionally
const a = signal(1);
const b = signal(2);
const showB = signal(false);

effect(() => {
  console.log(a()); // Always tracked
  if (showB()) {
    console.log(b()); // Only tracked when showB is true!
  }
});

// 2. Equality checking — signals use Object.is by default
const obj = signal({ name: 'A' });
obj.set({ name: 'A' }); // DOES trigger (different reference!)

// Custom equality:
const obj2 = signal({ name: 'A' }, { equal: (a, b) => a.name === b.name });
obj2.set({ name: 'A' }); // Does NOT trigger (custom equality says equal)

// 3. computed() is lazy — only evaluates when read
const expensive = computed(() => heavyWork(source())); // Not called until expensive() is read

// 4. untracked() — read signal without creating dependency
effect(() => {
  console.log(tracked());          // dependency
  console.log(untracked(notTracked)); // NOT a dependency
});

NgZone Escape Patterns

// Running outside Angular's zone for performance
@Component({ ... })
export class CanvasComponent {
  private ngZone = inject(NgZone);

  ngOnInit() {
    // Animation loop — don't trigger CD on every frame!
    this.ngZone.runOutsideAngular(() => {
      this.animate();
    });
  }

  private animate() {
    // This runs 60fps WITHOUT triggering change detection
    this.drawFrame();
    requestAnimationFrame(() => this.animate());
  }

  // When you DO need to update the UI:
  onSignificantChange(data: any) {
    this.ngZone.run(() => {
      this.displayData.set(data); // This triggers CD
    });
  }
}
