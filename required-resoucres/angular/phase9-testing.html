1. Testing Pyramid in Angular

// Testing pyramid:
//          /\
//         /  \  E2E (Cypress/Playwright) — few, slow, expensive
//        /    \
//       /------\  Integration (TestBed + DOM) — moderate
//      /        \
//     /----------\  Unit Tests (isolated, no DOM) — many, fast, cheap
//    /____________\

// Tools:
// - Jasmine: Test framework (describe, it, expect)
// - Karma: Test runner (executes in browser) — being replaced by:
// - Jest: Alternative test runner (faster, jsdom)
// - Angular Testing Library: Component testing utilities
// - Cypress/Playwright: E2E testing

// Service with dependencies — mock the deps
describe('UserService', () => {
  let service: UserService;
  let httpMock: jest.Mocked<HttpClient>;

  beforeEach(() => {
    httpMock = {
      get: jest.fn(),
      post: jest.fn(),
      put: jest.fn(),
      delete: jest.fn()
    } as any;

    TestBed.configureTestingModule({
      providers: [
        UserService,
        { provide: HttpClient, useValue: httpMock }
      ]
    });
    service = TestBed.inject(UserService);
  });

  it('should fetch users', () => {
    const mockUsers: User[] = [
      { id: '1', name: 'Alice', email: 'alice@test.com' }
    ];
    httpMock.get.mockReturnValue(of(mockUsers));

    service.getUsers().subscribe(users => {
      expect(users).toEqual(mockUsers);
      expect(users.length).toBe(1);
    });

    expect(httpMock.get).toHaveBeenCalledWith('/api/users');
  });

  it('should handle errors', () => {
    httpMock.get.mockReturnValue(throwError(() => new Error('Server error')));

    service.getUsers().subscribe({
      error: err => expect(err.message).toBe('Server error')
    });
  });
});

// Testing with HttpClientTestingModule (Angular's built-in)
describe('UserService with HttpTestingController', () => {
  let service: UserService;
  let httpTestingController: HttpTestingController;

  beforeEach(() => {
    TestBed.configureTestingModule({
      providers: [
        UserService,
        provideHttpClient(),
        provideHttpClientTesting()
      ]
    });
    service = TestBed.inject(UserService);
    httpTestingController = TestBed.inject(HttpTestingController);
  });

  afterEach(() => {
    httpTestingController.verify();  // Ensure no outstanding requests
  });

  it('should GET users', () => {
    const mockUsers = [{ id: '1', name: 'Test' }];

    service.getUsers().subscribe(users => {
      expect(users).toEqual(mockUsers);
    });

    const req = httpTestingController.expectOne('/api/users');
    expect(req.request.method).toBe('GET');
    req.flush(mockUsers);  // Respond with mock data
  });

  it('should POST new user', () => {
    const newUser = { name: 'Alice', email: 'alice@test.com' };

    service.createUser(newUser).subscribe(user => {
      expect(user.id).toBeDefined();
    });

    const req = httpTestingController.expectOne('/api/users');
    expect(req.request.method).toBe('POST');
    expect(req.request.body).toEqual(newUser);
    req.flush({ ...newUser, id: '123' });
  });
});

4. Testing Observables & Async

// fakeAsync + tick — control time in tests
it('should debounce search', fakeAsync(() => {
  const searchSpy = jest.spyOn(component, 'performSearch');
  
  component.searchControl.setValue('ang');
  tick(100);  // Only 100ms passed
  expect(searchSpy).not.toHaveBeenCalled();

  tick(200);  // Now 300ms total (debounce complete)
  expect(searchSpy).toHaveBeenCalledWith('ang');
}));

// waitForAsync — for real async (Promises)
it('should load data on init', waitForAsync(() => {
  fixture.detectChanges();  // Triggers ngOnInit

  fixture.whenStable().then(() => {
    fixture.detectChanges();
    expect(component.data.length).toBeGreaterThan(0);
  });
}));

// Testing observables directly
it('should emit filtered results', (done) => {
  service.search('angular').subscribe(results => {
    expect(results.length).toBe(3);
    expect(results[0].title).toContain('Angular');
    done();
  });
});

// Testing with marbles (for complex observable chains)
import { TestScheduler } from 'rxjs/testing';

it('should debounce and switchMap', () => {
  const scheduler = new TestScheduler((actual, expected) => {
    expect(actual).toEqual(expected);
  });

  scheduler.run(({ cold, expectObservable }) => {
    const input = cold('  a-b-c---|');
    const expected = '    ------c-|';
    const result = input.pipe(debounceTime(3, scheduler));
    expectObservable(result).toBe(expected);
  });
});

6. Marble Testing — Deep Dive

Marble testing allows testing time-based RxJS logic without actual waiting. The TestScheduler virtualizes time.

Marble Syntax Reference

SymbolMeaning
-1 frame of virtual time (1ms in TestScheduler.run)
a-z, 0-9Emission (value from values map)
|Observable completes
#Observable errors
(abc)Synchronous emissions (same frame)
^Subscription point (hot observables)
!Unsubscription point
10ms aTime progression (10ms then emit a)

Testing a debounced search service

import { TestScheduler } from 'rxjs/testing';

describe('SearchService', () => {
  let scheduler: TestScheduler;

  beforeEach(() => {
    scheduler = new TestScheduler((actual, expected) => {
      expect(actual).toEqual(expected);
    });
  });

  it('should debounce search input by 300ms', () => {
    scheduler.run(({ cold, expectObservable }) => {
      const input$  = cold('a-b-c---------|');
      const values = { a: 'A', b: 'An', c: 'Ang' };
      
      const result$ = input$.pipe(
        debounceTime(3)  // 3 frames in test
      );

      // a at 0, debounce 3: would emit at 3 but b arrives at 2
      // b at 2, debounce 3: would emit at 5 but c arrives at 4
      // c at 4, debounce 3: emits at 7
      expectObservable(result$).toBe('-------c------|', values);
    });
  });

  it('should cancel in-flight search with switchMap', () => {
    scheduler.run(({ cold, hot, expectObservable }) => {
      const search$ = hot('--a------b----|');
      const apiA    = cold('  ---x|');
      const apiB    = cold('         --y|');
      
      let callCount = 0;
      const result$ = search$.pipe(
        switchMap(term => {
          callCount++;
          return callCount === 1 ? apiA : apiB;
        })
      );

      expectObservable(result$).toBe('-----------y--|');
    });
  });
});

7. Component Testing — Advanced Patterns

Testing OnPush Components

it('should update when input reference changes (OnPush)', () => {
  const fixture = TestBed.createComponent(OnPushChildComponent);
  
  fixture.componentRef.setInput('items', [{ id: 1, name: 'A' }]);
  fixture.detectChanges();
  expect(fixture.nativeElement.textContent).toContain('A');
  
  // Mutation WON'T update (OnPush!):
  fixture.componentInstance.items.push({ id: 2, name: 'B' });
  fixture.detectChanges();
  expect(fixture.nativeElement.textContent).not.toContain('B');
  
  // New reference WILL update:
  fixture.componentRef.setInput('items', [...fixture.componentInstance.items]);
  fixture.detectChanges();
  expect(fixture.nativeElement.textContent).toContain('B');
});

// Testing signal-based components:
it('should react to signal changes', () => {
  const fixture = TestBed.createComponent(SignalComponent);
  fixture.detectChanges();
  
  fixture.componentInstance.count.set(5);
  fixture.detectChanges();
  
  expect(fixture.nativeElement.querySelector('.count').textContent).toBe('5');
});

8. Testing Best Practices

Key Principles:
  • Test behavior, not implementation — Test what the component does, not how
  • Arrange-Act-Assert — Setup → Execute → Verify
  • One assertion per test (ideally) — Easy to identify failures
  • Mock external dependencies — Don't test Angular/RxJS, test YOUR code
  • Use fixture.detectChanges() explicitly — Control when CD runs
  • Test the DOM for components — Users interact with DOM, not properties
  • Prefer testing-library or spectator — Less boilerplate than raw TestBed
  • Use marble testing for time-dependent logic — Debounce, retry, polling
