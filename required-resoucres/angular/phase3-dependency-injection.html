1. What is Dependency Injection?

The Problem DI Solves

// WITHOUT DI — tightly coupled, untestable
class UserComponent {
  private userService: UserService;
  private http: HttpClient;
  
  constructor() {
    this.http = new HttpClient(/* what params?? */);  // How to create?
    this.userService = new UserService(this.http);    // Creates its own dependency!
  }
}
// Problems:
// 1. Component decides HOW to create dependencies
// 2. Can't swap implementations (testing, mocking)
// 3. If UserService constructor changes, ALL consumers break
// 4. No singleton sharing — each component creates new instances

// WITH DI — loosely coupled, testable
class UserComponent {
  // Angular PROVIDES the dependency — component just declares what it needs
  constructor(private userService: UserService) {}
  // OR modern approach:
  private userService = inject(UserService);
}

DI Terminology

TermMeaningExample
ProviderRecipe for creating a dependency{ provide: UserService, useClass: UserService }
TokenKey used to look up the providerUserService (class), InjectionToken
InjectorContainer that holds providers and creates instancesRoot injector, component injector
ConsumerClass that receives the dependencyComponent, Directive, Service, Pipe

2. How Angular's DI Works Under the Hood

The Resolution Algorithm

When Angular needs to inject a dependency, it follows this process:

  1. Look at the injection token — What class/token is requested?
  2. Check component's own injector — Was it provided in this component's providers?
  3. Walk UP the injector tree — Check parent component → grandparent → ... → root
  4. Check Element Injector hierarchy first, then Module Injector hierarchy
  5. If found — Return existing instance (or create if first request)
  6. If NOT found — Throw NullInjectorError (unless @Optional())

Two Injector Hierarchies

// Angular maintains TWO parallel injector trees:

// 1. Element Injector Tree (component/directive providers)
// Follows the DOM/component tree structure
AppComponent
  └── HeaderComponent (providers: [HeaderService])
        └── NavComponent
              └── NavItemComponent

// 2. Module Injector Tree (NgModule providers or providedIn)
// Flat (mostly) — root → lazy-loaded module injectors
PlatformInjector (built-in: DomSanitizer, etc.)
  └── RootInjector (providedIn: 'root' services)
        └── LazyModuleInjector (services in lazy-loaded modules)

// Resolution order:
// Component's Element Injector → Parent Element → ... → Root Element
// THEN: Component's Module Injector → Parent Module → Root Module → Platform

3. Providers — All Registration Methods

Where to Register Providers

// 1. providedIn: 'root' — RECOMMENDED for most services (singleton, tree-shakable)
@Injectable({ providedIn: 'root' })
export class AuthService { }

// 2. providedIn: 'platform' — shared across multiple Angular apps on same page
@Injectable({ providedIn: 'platform' })
export class AnalyticsService { }

// 3. Component-level providers — NEW instance per component
@Component({
  providers: [UserService]  // Each component instance gets its own UserService
})
export class UserListComponent { }

// 4. Route-level providers (Angular 14+)
export const routes: Routes = [
  {
    path: 'admin',
    providers: [AdminService],  // Only available in this route subtree
    loadComponent: () => import('./admin.component')
  }
];

// 5. bootstrapApplication providers
bootstrapApplication(AppComponent, {
  providers: [
    provideRouter(routes),
    provideHttpClient(withInterceptors([authInterceptor])),
    { provide: API_URL, useValue: 'https://api.example.com' }
  ]
});

Provider Types

// useClass — Create instance of a different class
{ provide: LoggerService, useClass: AdvancedLoggerService }

// useClass with deps
{ provide: LoggerService, useClass: AdvancedLoggerService, deps: [HttpClient] }

// useValue — Provide a static value
{ provide: API_URL, useValue: 'https://api.example.com' }
{ provide: APP_CONFIG, useValue: { production: true, apiVersion: 'v2' } }

// useFactory — Create dependency using a factory function
{
  provide: DataService,
  useFactory: (http: HttpClient, config: AppConfig) => {
    return config.useMock 
      ? new MockDataService() 
      : new RealDataService(http);
  },
  deps: [HttpClient, APP_CONFIG]
}

// useExisting — Alias to another token (both resolve to same instance)
{ provide: AbstractLogger, useExisting: ConsoleLoggerService }

// Example: implementing multiple interfaces with one service
@Injectable({ providedIn: 'root' })
class FullLoggerService implements Logger, ErrorHandler { }

providers: [
  FullLoggerService,
  { provide: Logger, useExisting: FullLoggerService },
  { provide: ErrorHandler, useExisting: FullLoggerService }
]

4. Hierarchical Injectors

Component-Level Providers Create New Instances

// Scenario: Shopping cart for different users in an admin panel
@Injectable()  // NOT providedIn: 'root' — we want multiple instances
export class CartService {
  items: CartItem[] = [];
  
  addItem(item: CartItem) { this.items.push(item); }
  getTotal() { return this.items.reduce((sum, i) => sum + i.price, 0); }
  clear() { this.items = []; }
}

// Each UserCartComponent gets its OWN CartService instance
@Component({
  selector: 'app-user-cart',
  providers: [CartService],  // New instance for each component
  template: `
    <h3>{{ userName }}'s Cart ({{ cart.items.length }} items)</h3>
    <!-- child components also get THIS CartService instance -->
    <app-cart-items />
    <app-cart-summary />
  `
})
export class UserCartComponent {
  @Input() userName = '';
  cart = inject(CartService);
}

viewProviders vs providers

@Component({
  selector: 'app-parent',
  providers: [ServiceA],       // Available to this component + ALL children (including projected)
  viewProviders: [ServiceB],   // Available to this component + VIEW children ONLY (not projected content)
  template: `
    <app-view-child />          <!-- Can access ServiceA AND ServiceB -->
    <ng-content></ng-content>  <!-- Can access ServiceA but NOT ServiceB -->
  `
})
export class ParentComponent { }

// Usage:
<app-parent>
  <app-projected-child />  <!-- This is projected content — NO access to ServiceB -->
</app-parent>

5. InjectionToken & Custom Tokens

Why InjectionToken?

You can't use interfaces or primitive values as DI tokens because they don't exist at runtime (TypeScript interfaces are erased during compilation). InjectionToken creates a unique runtime token.

// Create tokens for non-class dependencies
export const API_URL = new InjectionToken<string>('API_URL');
export const APP_CONFIG = new InjectionToken<AppConfig>('APP_CONFIG');
export const IS_PRODUCTION = new InjectionToken<boolean>('IS_PRODUCTION');
export const WINDOW = new InjectionToken<Window>('WINDOW');

// With factory (tree-shakable default)
export const LOGGER = new InjectionToken<Logger>('LOGGER', {
  providedIn: 'root',
  factory: () => {
    const isProd = inject(IS_PRODUCTION);
    return isProd ? new ProductionLogger() : new ConsoleLogger();
  }
});

// Provide the tokens
bootstrapApplication(AppComponent, {
  providers: [
    { provide: API_URL, useValue: environment.apiUrl },
    { provide: IS_PRODUCTION, useValue: environment.production },
    { provide: WINDOW, useFactory: () => window },
    { provide: APP_CONFIG, useValue: { theme: 'dark', locale: 'en-US' } }
  ]
});

// Inject the tokens
export class ApiService {
  private apiUrl = inject(API_URL);
  private config = inject(APP_CONFIG);
  private window = inject(WINDOW);
}

6. inject() Function vs Constructor Injection

Modern inject() Function (Angular 14+)

// Constructor injection (traditional)
@Component({...})
export class OldStyleComponent {
  constructor(
    private userService: UserService,
    private router: Router,
    @Inject(API_URL) private apiUrl: string,
    @Optional() private analytics?: AnalyticsService
  ) {}
}

// inject() function (modern — preferred)
@Component({...})
export class ModernComponent {
  private userService = inject(UserService);
  private router = inject(Router);
  private apiUrl = inject(API_URL);
  private analytics = inject(AnalyticsService, { optional: true });
}

// Advantages of inject():
// 1. Less boilerplate (no constructor parameter list)
// 2. Works in functions (not just classes)
// 3. Easier refactoring (no constructor signature changes)
// 4. Can be used in inheritance without super() headaches

inject() in Functions (Powerful Pattern)

// Create reusable injection-based utilities
export function injectDestroy() {
  const destroy$ = new Subject<void>();
  const destroyRef = inject(DestroyRef);
  destroyRef.onDestroy(() => {
    destroy$.next();
    destroy$.complete();
  });
  return destroy$;
}

// Usage in component:
export class MyComponent {
  private destroy$ = injectDestroy();

  ngOnInit() {
    this.someService.data$
      .pipe(takeUntil(this.destroy$))
      .subscribe(data => { ... });
  }
}

// Another pattern: custom inject helper
export function injectParams<T>(key: string): Signal<T> {
  const route = inject(ActivatedRoute);
  return toSignal(
    route.params.pipe(map(p => p[key] as T)),
    { requireSync: true }
  );
}

Resolution Modifiers

// @Optional / { optional: true } — don't throw if not found
private analytics = inject(AnalyticsService, { optional: true });

// @Self / { self: true } — only look in THIS component's injector
private service = inject(MyService, { self: true });

// @SkipSelf / { skipSelf: true } — skip THIS injector, start from parent
private parentService = inject(MyService, { skipSelf: true });

// @Host / { host: true } — search up to host component only
private hostService = inject(MyService, { host: true });

// Combine:
private service = inject(MyService, { optional: true, skipSelf: true });

7. Multi Providers

// Multi providers — multiple values registered under the same token
// All values are injected as an array

export const HTTP_INTERCEPTORS = new InjectionToken<HttpInterceptor[]>('HTTP_INTERCEPTORS');

// Register multiple interceptors
providers: [
  { provide: HTTP_INTERCEPTORS, useClass: AuthInterceptor, multi: true },
  { provide: HTTP_INTERCEPTORS, useClass: LoggingInterceptor, multi: true },
  { provide: HTTP_INTERCEPTORS, useClass: ErrorInterceptor, multi: true },
]

// Custom multi-provider example: Validators
export const FORM_VALIDATORS = new InjectionToken<Validator[]>('FORM_VALIDATORS');

providers: [
  { provide: FORM_VALIDATORS, useClass: RequiredValidator, multi: true },
  { provide: FORM_VALIDATORS, useClass: EmailValidator, multi: true },
  { provide: FORM_VALIDATORS, useClass: MinLengthValidator, multi: true },
]

// Inject as array:
export class FormComponent {
  private validators = inject(FORM_VALIDATORS);  // Validator[] — all three!
}

8. Tree-shakable Providers

// NOT tree-shakable (old way) — service always in bundle even if unused
// Module registers the service
@NgModule({
  providers: [NeverUsedService]  // Always bundled!
})

// TREE-SHAKABLE (modern way) — service only in bundle if actually injected
@Injectable({ providedIn: 'root' })
export class OnlyIfUsedService { }
// If no component/service injects this, it's removed from bundle!

// Tree-shakable with factory
export const FEATURE_FLAG = new InjectionToken<boolean>('FEATURE_FLAG', {
  providedIn: 'root',
  factory: () => inject(EnvironmentService).getFlag('newDashboard')
});

// Why this matters for bundle size:
// A library can export 50 services all with providedIn: 'root'
// But if your app only uses 3 of them, only those 3 are bundled!

9. HTTP Interceptors (DI in Action)

Functional Interceptors (Angular 15+ — Modern)

// auth.interceptor.ts
export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const authService = inject(AuthService);
  const token = authService.getToken();

  if (token) {
    const clonedReq = req.clone({
      setHeaders: { Authorization: `Bearer ${token}` }
    });
    return next(clonedReq);
  }
  return next(req);
};

// error.interceptor.ts
export const errorInterceptor: HttpInterceptorFn = (req, next) => {
  const router = inject(Router);
  const notification = inject(NotificationService);

  return next(req).pipe(
    retry({ count: 2, delay: 1000 }),
    catchError((error: HttpErrorResponse) => {
      switch (error.status) {
        case 401:
          router.navigate(['/login']);
          break;
        case 403:
          notification.show('Access denied', 'error');
          break;
        case 500:
          notification.show('Server error. Please try again.', 'error');
          break;
      }
      return throwError(() => error);
    })
  );
};

// loading.interceptor.ts
export const loadingInterceptor: HttpInterceptorFn = (req, next) => {
  const loading = inject(LoadingService);
  loading.show();

  return next(req).pipe(
    finalize(() => loading.hide())
  );
};

// Register in app.config.ts
export const appConfig: ApplicationConfig = {
  providers: [
    provideHttpClient(
      withInterceptors([authInterceptor, loadingInterceptor, errorInterceptor])
    )
  ]
};

Class-based Interceptors (Legacy but still in production)

@Injectable()
export class CacheInterceptor implements HttpInterceptor {
  private cache = new Map<string, HttpResponse<any>>();

  intercept(req: HttpRequest<any>, next: HttpHandler): Observable<HttpEvent<any>> {
    // Only cache GET requests
    if (req.method !== 'GET') {
      return next.handle(req);
    }

    const cached = this.cache.get(req.urlWithParams);
    if (cached) {
      return of(cached.clone());
    }

    return next.handle(req).pipe(
      tap(event => {
        if (event instanceof HttpResponse) {
          this.cache.set(req.urlWithParams, event.clone());
        }
      })
    );
  }
}

10. Advanced DI Patterns

Strategy Pattern with DI

// Define strategy interface
export interface PaymentStrategy {
  pay(amount: number): Observable<PaymentResult>;
}

export const PAYMENT_STRATEGY = new InjectionToken<PaymentStrategy>('PAYMENT_STRATEGY');

// Implementations
@Injectable()
export class StripePaymentStrategy implements PaymentStrategy {
  pay(amount: number) { /* Stripe logic */ }
}

@Injectable()
export class PayPalPaymentStrategy implements PaymentStrategy {
  pay(amount: number) { /* PayPal logic */ }
}

// Provide based on config
{
  provide: PAYMENT_STRATEGY,
  useFactory: (config: AppConfig) => {
    switch (config.paymentProvider) {
      case 'stripe': return inject(StripePaymentStrategy);
      case 'paypal': return inject(PayPalPaymentStrategy);
    }
  },
  deps: [APP_CONFIG]
}

Abstract Class as Token

// Abstract class works as both interface AND DI token (interfaces can't)
export abstract class DataSource<T> {
  abstract getAll(): Observable<T[]>;
  abstract getById(id: string): Observable<T>;
  abstract create(item: T): Observable<T>;
  abstract update(id: string, item: Partial<T>): Observable<T>;
  abstract delete(id: string): Observable<void>;
}

// Concrete implementation
@Injectable({ providedIn: 'root' })
export class HttpDataSource<T> extends DataSource<T> {
  private http = inject(HttpClient);
  private apiUrl = inject(API_URL);

  getAll(): Observable<T[]> {
    return this.http.get<T[]>(this.apiUrl);
  }
  // ... other methods
}

// For testing
@Injectable()
export class MockDataSource<T> extends DataSource<T> {
  private data: T[] = [];
  getAll(): Observable<T[]> { return of(this.data); }
  // ... other methods
}

// Provide
{ provide: DataSource, useClass: HttpDataSource }
